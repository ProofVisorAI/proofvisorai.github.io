1//! The embedded file-backed broker — the Bridge reference backend.
2//!
3//! Layout: `<data_dir>/manifest.yaml` + per partition under
4//! `<data_dir>/topics/<topic>/`: `p<N>.jsonl` (the JSONL segment),
5//! `p<N>.next-offset` (durable high-water mark), and `p<N>.event-uids.jsonl`
6//! (idempotency sidecar). Offsets are stable logical positions: retention
7//! rewrites keep surviving records under their original offsets, so after
8//! expiry the first file line can carry an offset > 0. Segment appends are
9//! serialized per partition; a crash can leave at most one torn trailing
10//! line, which recovery detects, truncates, and *counts* (never silently
11//! absorbs — D13.16). A torn sidecar tail is truncated without counting and
12//! rebuilt from the recovered segment.
13//!
14//! Scope: single-process access (the harness embeds the broker or fronts it).
15//! Cross-process multi-writer setups use the NATS/Kafka connectors.
16

63    /// Provision a fresh bridge in `data_dir` from `manifest` alone (R12).
64    /// Fails if the directory already contains a bridge.
65    pub fn provision(data_dir: &Path, manifest: &BridgeManifest) -> Result<Self, BusError> {
66        manifest
67            .validate()
68            .map_err(|e| BusError::Backend(e.to_string()))?;
69        // Reject a scheme-URI cold_uri BEFORE
70        // any filesystem side effect. The prior check lived in
71        // `open()` (after provision had already written manifest.yaml,
72        // per-topic dirs, and schema copies to disk); when
73        // provision-then-open failed here, the operator saw a "bridge
74        // already provisioned" error on retry after fixing the
75        // manifest, because the botched files still sat in data_dir.
76        // Move the check to the top of provision so a failure leaves
77        // data_dir untouched and retry-clean.
78        #[cfg(not(feature = "cold-store"))]
79        reject_cold_uri_without_feature(manifest)?;
80        // Resolve and schema-validate every referenced schema BEFORE the
81        // single-winner claim below: `manifest.validate()` does not check
82        // schema refs, so an unresolvable/invalid ref must fail here —
83        // side-effect-free and retry-clean — not after the claim has
84        // already been taken (see the rollback note further down).
85        let resolved_schemas = resolve_referenced_schemas(manifest)?;
86        let manifest_path = data_dir.join("manifest.yaml");
87        pv_core::fsutil::create_dir_all_synced(data_dir)?;
88        let yaml = manifest.to_yaml().map_err(|e| BusError::Backend(e.to_string()))?;
89        // Atomic single-winner claim on a fresh provision: the
90        // `hard_link(2)` below fails with EEXIST if the target already
91        // exists — an OS-level
92        // exclusion primitive, so N concurrent provisions race
93        // on the same directory and exactly one wins. The old shape
94        // (`exists()` check → separate write) had a TOCTOU: two racers
95        // could both see "does not exist", both write, both succeed.
96        //
97        // Any error path — write_all, sync_all, hard_link, os signal —
98        // must clean up the tmp file. TmpGuard's Drop covers the
99        // early-return paths (ENOSPC, EIO on write, permission errors
100        // on sync); the winner path explicitly disarms the guard just
101        // before returning Ok so its finalised link is not deleted.
102        use std::io::Write as _;
103        let tmp = data_dir.join(format!("manifest.yaml.{}.tmp", pv_core::new_event_uid()));
104        struct TmpGuard {
105            path: Option<std::path::PathBuf>,
106        }
107        impl TmpGuard {
108            fn disarm(&mut self) {
109                self.path = None;
110            }
111        }
112        impl Drop for TmpGuard {
113            fn drop(&mut self) {
114                if let Some(path) = self.path.take() {
115                    let _ = fs::remove_file(&path);
116                }
117            }
118        }
119        let mut guard = TmpGuard {
120            path: Some(tmp.clone()),
121        };
122        {
123            // Explicit 0o600 mode so the manifest doesn't inherit the
124            // ambient umask. `hard_link` creates a second entry with
125            // the SAME inode (and same mode), so the final manifest
126            // path is 0o600 too. Bridge topology, schema refs, and
127            // `cold_uri` are not per-session data, but the same
128            // uniformity discipline the harness spool applies —
129            // durability-critical control files should not ride on
130            // the operator's umask.
131            let mut tmp_options = fs::OpenOptions::new();
132            tmp_options.write(true).create_new(true);
133            #[cfg(unix)]
134            {
135                use std::os::unix::fs::OpenOptionsExt as _;
136                tmp_options.mode(0o600);
137            }
138            let mut file = tmp_options.open(&tmp)?;
139            file.write_all(yaml.as_bytes())?;
140            file.sync_all()?;
141        }
142        match fs::hard_link(&tmp, &manifest_path) {
143            Ok(_) => {
144                // The tmp file was successfully hard-linked to the
145                // final path; the link itself is now the manifest.
146                // Delete the tmp path entry (the inode stays alive
147                // through the second link) and disarm the guard.
148                let _ = fs::remove_file(&tmp);
149                guard.disarm();
150            }
151            Err(error) => {
152                if error.kind() == std::io::ErrorKind::AlreadyExists {
153                    // Loser arm: tmp is auto-removed by TmpGuard on Err.
154                    // Use the basename to avoid leaking
155                    // the absolute deployment dir if this BusError
156                    // ever flows through a tracing::warn!(%error)
157                    // path.
158                    return Err(BusError::Backend(format!(
159                        "bridge already provisioned at {}",
160                        pv_core::fsutil::basename(data_dir)
161                    )));
162                }
163                return Err(BusError::Io(error));
164            }
165        }
166        pv_core::fsutil::sync_directory(data_dir).map_err(BusError::Io)?;
167        // Schema copies and topic directories are written only AFTER the
168        // hard-link claim above declared this provision the single
169        // winner. Writing them before the claim (the old order) meant a
170        // provision attempt against an ALREADY-provisioned bridge
171        // overwrote the live bridge's schema files under
172        // `data_dir/<schema_ref>` before erroring "bridge already
173        // provisioned" — and `load_validators` falls back to those
174        // copies for non-builtin refs, so the next open() silently
175        // validated against the loser's schemas. It also let two racing
176        // provisions cross-pollute each other's directories.
177        //
178        // The refs themselves were resolved and validated pre-claim, so
179        // only environmental I/O errors (ENOSPC, permissions) can fail
180        // here — and those must roll the claim back, or every retry
181        // fails "bridge already provisioned" against a half-provisioned
182        // directory (the retry-clean invariant).
183        let post_claim = write_referenced_schemas(data_dir, &resolved_schemas).and_then(|()| {
184            for t in &manifest.topics {
185                pv_core::fsutil::create_dir_all_synced(&data_dir.join("topics").join(&t.name))?;
186            }
187            Ok(())
188        });
189        if let Err(error) = post_claim {
190            let _ = fs::remove_file(&manifest_path);
191            let _ = pv_core::fsutil::sync_directory(data_dir);
192            return Err(error);
193        }
194        Self::open(data_dir)
195    }

197    /// Add to an already-provisioned bridge every topic of `manifest` it
198    /// does not have yet, and return the names added. Existing topics,
199    /// their partitions and segments are never touched; a topic present in
200    /// both keeps its provisioned spec.
201    ///
202    /// The topic directories are created first and the manifest is
203    /// replaced atomically last, so a crash leaves either the old manifest
204    /// (plus harmless empty directories, picked up by a retry) or the new
205    /// one. Call it before [`Self::open`], from the only process writing to
206    /// `data_dir` (the daemon holds its spool lock).
207    pub fn add_missing_topics(data_dir: &Path, manifest: &BridgeManifest) -> Result<Vec<String>, BusError> {
208        let manifest_path = data_dir.join("manifest.yaml");
209        let yaml = pv_core::fsutil::read_capped_string(&manifest_path, pv_core::fsutil::MAX_CONTROL_BYTES)?;
210        let mut current = BridgeManifest::from_yaml(&yaml).map_err(|e| BusError::Backend(e.to_string()))?;
211        let missing: Vec<crate::manifest::TopicSpec> = manifest
212            .topics
213            .iter()
214            .filter(|topic| !current.topics.iter().any(|existing| existing.name == topic.name))
215            .cloned()
216            .collect();
217        if missing.is_empty() {
218            return Ok(Vec::new());
219        }
220        current.topics.extend(missing.iter().cloned());
221        current.validate().map_err(|e| BusError::Backend(e.to_string()))?;
222        let additions = BridgeManifest {
223            topics: missing.clone(),
224            ..current.clone()
225        };
226        #[cfg(not(feature = "cold-store"))]
227        reject_cold_uri_without_feature(&additions)?;
228        let resolved = resolve_referenced_schemas(&additions)?;
229        write_referenced_schemas(data_dir, &resolved)?;
230        for topic in &missing {
231            pv_core::fsutil::create_dir_all_synced(&data_dir.join("topics").join(&topic.name))?;
232        }
233        let updated = current.to_yaml().map_err(|e| BusError::Backend(e.to_string()))?;
234        pv_core::fsutil::write_atomic(&manifest_path, updated.as_bytes())?;
235        Ok(missing.into_iter().map(|topic| topic.name).collect())
236    }

238    /// Open an existing bridge, recovering offsets (and truncating at most one
239    /// torn trailing line per partition) from the segment files.
240    pub fn open(data_dir: &Path) -> Result<Self, BusError> {
241        // Cap the bridge manifest read. A hostile plant of a
242        // multi-GiB manifest.yaml would OOM the broker at startup before
243        // the YAML parser could complain.
244        let manifest_yaml = pv_core::fsutil::read_capped_string(
245            &data_dir.join("manifest.yaml"),
246            pv_core::fsutil::MAX_CONTROL_BYTES,
247        )?;
248        let manifest =
249            BridgeManifest::from_yaml(&manifest_yaml).map_err(|e| BusError::Backend(e.to_string()))?;
250        // NOTE: no orphaned-temp sweep here — the daemon performs the
251        // sweep itself at boot (main.rs), where it is the single
252        // writer. `open()` mutates on-disk state during recovery
253        // (torn-tail truncate, sidecar rewrite, append handles), so it
254        // must only ever run in the single-writer daemon; read-side
255        // tooling running beside a live daemon uses
256        // [`Self::fetch_read_only`] instead.
257        let mut partitions = HashMap::new();
258        let mut torn_total = 0u64;
259        for t in &manifest.topics {
260            let dir = data_dir.join("topics").join(&t.name);
261            pv_core::fsutil::create_dir_all_synced(&dir)?;
262            let mut parts = Vec::with_capacity(t.partitions as usize);
263            for p in 0..t.partitions {
264                let path = dir.join(format!("p{p}.jsonl"));
265                let watermark_path = dir.join(format!("p{p}.next-offset"));
266                let idempotency_path = dir.join(format!("p{p}.event-uids.jsonl"));
267                let persisted_offset = read_high_water(&watermark_path)?;
268                let (segment_offset, torn) = recover_segment(&path, persisted_offset)?;
269                let mut seen_event_uids = recover_event_uids(&idempotency_path)?;
270                recover_segment_event_uids(&path, &mut seen_event_uids)?;
271                // Post-reconciliation drop: any UID whose offset does
272                // not correspond to a record still present in the
273                // segment (e.g., record purged by retention but the
274                // sidecar rewrite lost the corresponding delete in a
275                // crash between segment rename and sidecar rewrite)
276                // must be evicted. Otherwise `publish_idempotent`
277                // short-circuits to a stale offset and callers fetch
278                // whatever event lives at that offset today, silently
279                // returning the wrong record.
280                //
281                // Mirror `enforce_retention`'s policy for
282                // unparseable-but-kept lines: when unparseable lines
283                // exist, any UID whose offset falls in the surviving
284                // offset range is kept, because it may correspond to an
285                // unparseable-but-authentic line at that offset.
286                // Without this parity, an unparseable segment record
287                // after a crash would drop its sidecar entry, letting
288                // the next publish_idempotent re-append a duplicate
289                // that the following retention pass would choke on.
290                // When the segment is FULLY parseable, `live_uids` is
291                // authoritative — `publish_with_uid` refuses idempotent
292                // publishes whose value lacks `metadata.uid`, so every
293                // surviving record's UID is re-derivable — and the
294                // range must not be consulted at all: keeping a
295                // non-live middle-offset UID (retention crash between
296                // segment rename and sidecar rewrite, e.g. after a
297                // wall-clock regression expired a middle record)
298                // resurrects exactly the stale-ack → wrong-record-fetch
299                // mapping the retention path abandoned this range
300                // heuristic for.
301                if !seen_event_uids.is_empty() {
302                    let mut live_uids =
303                        std::collections::HashSet::<String>::with_capacity(seen_event_uids.len());
304                    let mut min_offset = u64::MAX;
305                    let mut max_offset = 0u64;
306                    let mut have_parseable_line = false;
307                    let mut unparseable_lines = 0u64;
308                    if path.exists() {
309                        for line in BufReader::new(fs::File::open(&path)?).lines() {
310                            let Ok(line) = line else {
311                                continue;
312                            };
313                            if line.is_empty() {
314                                continue;
315                            }
316                            if let Ok(event) = serde_json::from_str::<StoredEvent>(&line) {
317                                if let Some(uid) = event_uid_from_value(&event.value) {
318                                    live_uids.insert(uid.to_owned());
319                                }
320                                min_offset = min_offset.min(event.offset);
321                                max_offset = max_offset.max(event.offset);
322                                have_parseable_line = true;
323                            } else {
324                                unparseable_lines = unparseable_lines.saturating_add(1);
325                            }
326                        }
327                    }
328                    // A TRAILING unparseable
329                    // complete line occupies an offset ABOVE the max
330                    // parseable one, so the previous [min, max]-of-
331                    // parseable range dropped its UID — the exact
332                    // duplicate-re-append case the range heuristic was
333                    // added for, just at the other edge. Extend the
334                    // upper bound by the count of unparseable lines
335                    // (each occupies exactly one offset). When the
336                    // segment has ONLY unparseable lines we have no
337                    // offset information at all — keep every sidecar
338                    // entry (conservative: at worst a stale ack that
339                    // the fetch-side digest checks catch) rather than
340                    // dropping evidence. And when there are NO
341                    // unparseable lines the range is not consulted:
342                    // a fully-parseable segment makes `live_uids`
343                    // authoritative, so every non-live UID is
344                    // definitively stale and must drop.
345                    let offset_range = if unparseable_lines == 0 {
346                        None
347                    } else if have_parseable_line {
348                        Some((min_offset, max_offset.saturating_add(unparseable_lines)))
349                    } else {
350                        tracing::warn!(
351                            topic = %t.name,
352                            partition = p,
353                            unparseable_lines,
354                            "segment has only unparseable lines; keeping all sidecar UID entries"
355                        );
356                        Some((0, u64::MAX))
357                    };
358                    let before = seen_event_uids.len();
359                    seen_event_uids.retain(|uid, offset| {
360                        if live_uids.contains(uid) {
361                            return true;
362                        }
363                        match offset_range {
364                            Some((lo, hi)) => *offset >= lo && *offset <= hi,
365                            None => false,
366                        }
367                    });
368                    if seen_event_uids.len() != before {
369                        tracing::warn!(
370                            topic = %t.name,
371                            partition = p,
372                            dropped = before - seen_event_uids.len(),
373                            "sidecar UID→offset entries dropped: corresponding segment records \
374                             absent (likely retention crash between segment rewrite and sidecar rewrite)"
375                        );
376                    }
377                }
378                torn_total += torn;
379                // Track first-time creation so we can fsync the directory
380                // after the file is materialised — `sync_data()` on the
381                // append handle flushes bytes and size, but the directory
382                // entry that names the inode is only durable after a
383                // `sync_directory(parent)`. Without this, N successful
384                // publishes → acked → power loss → boot back with the
385                // segment file absent and every acked event lost until
386                // another directory-fsyncing path runs (the retention
387                // rewrite, or the sidecar-creation fsync in
388                // `publish_with_uid`).
389                let segment_created = std::fs::symlink_metadata(&path).is_err();
390                // Same 0o600 + O_NOFOLLOW discipline the harness spool
391                // applies. Segment JSONL carries full `StoredEvent`
392                // records (instance_uid + event payload + tool/prompt
393                // bytes + cost), which is the same "session-derived
394                // data" class the harness-side R18/R19 fix protected.
395                let mut open_options = fs::OpenOptions::new();
396                open_options.create(true).append(true);
397                #[cfg(unix)]
398                {
399                    use std::os::unix::fs::OpenOptionsExt as _;
400                    open_options
401                        .custom_flags(pv_core::fsutil::unix_o_nofollow())
402                        .mode(0o600);
403                }
404                let writer = open_options.open(&path)?;
405                if segment_created {
406                    pv_core::fsutil::sync_directory(&dir).map_err(BusError::Io)?;
407                }
408                parts.push(Mutex::new(Partition {
409                    path,
410                    watermark_path,
411                    idempotency_path,
412                    seen_event_uids,
413                    next_offset: segment_offset.max(persisted_offset),
414                    writer,
415                    poisoned: None,
416                }));
417            }
418            partitions.insert(t.name.clone(), parts);
419        }
420        let validators = load_validators(data_dir, &manifest)?;
421        #[cfg(feature = "cold-store")]
422        let cold_archive = crate::cold_store::ColdArchive::from_manifest_with_pending_default(
423            &manifest,
424            Some(data_dir.join("cold-outbox")),
425        )?;
426        // Fail fast at boot on a scheme-URI `cold_uri` in a build without
427        // the `cold-store` feature. Previously the error only surfaced
428        // when the first record actually expired — up to `hot_hours`
429        // (default 168 h) after boot — and, because `enforce_retention`
430        // `?`-propagates out of the per-topic loop, halted retention for
431        // every topic ordered after the misconfigured one. Boot-time
432        // validation matches the fail-fast policy of every other
433        // feature-gated backend (kafka/nats/redis/onnx/qdrant).
434        #[cfg(not(feature = "cold-store"))]
435        reject_cold_uri_without_feature(&manifest)?;
436        Ok(Self {
437            data_dir: data_dir.to_owned(),
438            manifest,
439            #[cfg(feature = "cold-store")]
440            cold_archive,
441            partitions,
442            validators,
443            recovered_torn_lines: torn_total,
444        })
445    }

457    /// Read-only event fetch for tooling running BESIDE a live daemon
458    /// (`pvctl event-tail`). [`Self::open`] is NOT safe for that use:
459    /// its recovery "repairs" a torn segment tail via `set_len` and
460    /// atomically rewrites the idempotency sidecar — racing a daemon
461    /// that is mid-append (between `write_all` of the record bytes and
462    /// the trailing newline), a second process's repair truncates
463    /// bytes the daemon goes on to ack as durable: a silently lost or
464    /// corrupted acked audit event. There is no cross-process lock, so
465    /// the only safe concurrent mode is one that mutates nothing.
466    ///
467    /// This path only reads: the manifest (capped, to validate the
468    /// topic/partition and locate the segment), then a bounded
469    /// streaming scan of the one segment file. An unparseable line is
470    /// skipped (same policy as [`EventBus::fetch`]); an unterminated
471    /// trailing line — an in-flight append or crash-torn tail — is
472    /// ignored, never truncated; a record above the per-line bound
473    /// stops the scan at that point (fail-stop, mutation-free).
474    pub fn fetch_read_only(
475        data_dir: &Path,
476        topic: &str,
477        partition: u32,
478        offset: u64,
479        max: usize,
480    ) -> Result<Vec<StoredEvent>, BusError> {
481        /// Same generous per-record bound as `recover_segment`.
482        const MAX_SEGMENT_LINE_BYTES: u64 = 16 * 1024 * 1024;
483        let manifest_yaml = pv_core::fsutil::read_capped_string(
484            &data_dir.join("manifest.yaml"),
485            pv_core::fsutil::MAX_CONTROL_BYTES,
486        )?;
487        let manifest =
488            BridgeManifest::from_yaml(&manifest_yaml).map_err(|e| BusError::Backend(e.to_string()))?;
489        let topic_spec = manifest
490            .topics
491            .iter()
492            .find(|t| t.name == topic)
493            .ok_or_else(|| BusError::UnknownTopic(topic.to_owned()))?;
494        if partition >= topic_spec.partitions {
495            return Err(BusError::Backend(format!("partition {partition} out of range")));
496        }
497        let path = data_dir
498            .join("topics")
499            .join(topic)
500            .join(format!("p{partition}.jsonl"));
501        let file = match fs::File::open(&path) {
502            Ok(file) => file,
503            // Provisioned topic whose segment has no events yet (the
504            // daemon materialises it lazily at open/publish).
505            Err(error) if error.kind() == std::io::ErrorKind::NotFound => return Ok(Vec::new()),
506            Err(error) => return Err(BusError::Io(error)),
507        };
508        let mut reader = std::io::BufReader::new(file);
509        let mut out = Vec::with_capacity(max.min(1024));
510        let mut buf: Vec<u8> = Vec::new();
511        while out.len() < max {
512            buf.clear();
513            let read = {
514                use std::io::Read as _;
515                let mut limited = (&mut reader).take(MAX_SEGMENT_LINE_BYTES);
516                use std::io::BufRead as _;
517                limited.read_until(b'\n', &mut buf)?
518            };
519            if read == 0 {
520                break;
521            }
522            if buf.last() != Some(&b'\n') {
523                // In-flight append, crash-torn tail, or a record above
524                // the line cap. A writer-side open() decides what to do
525                // about those; a read-only tail just stops before them.
526                break;
527            }
528            let line = buf.get(..buf.len().saturating_sub(1)).unwrap_or_default();
529            if line.is_empty() {
530                continue;
531            }
532            let ev: StoredEvent = match serde_json::from_slice(line) {
533                Ok(ev) => ev,
534                Err(error) => {
535                    tracing::warn!(
536                        %error,
537                        path = %pv_core::fsutil::basename(&path),
538                        "skipping unparseable segment record during read-only fetch",
539                    );
540                    continue;
541                }
542            };
543            if ev.offset < offset {
544                continue;
545            }
546            out.push(ev);
547        }
548        Ok(out)
549    }

552/// Reject a manifest that names a scheme-URI `cold_uri` when this build
553/// lacks the `cold-store` feature: retention would silently accept the
554/// value at boot and only fail on the first tick that produced expired
555/// records (up to `hot_hours` later), and — because `enforce_retention`
556/// `?`-propagates out of the per-topic loop — halt retention for every
557/// topic ordered after it. Boot-time refusal matches the fail-fast
558/// policy of every other feature-gated backend.
559#[cfg(not(feature = "cold-store"))]
560fn reject_cold_uri_without_feature(manifest: &BridgeManifest) -> Result<(), BusError> {
561    for topic in &manifest.topics {
562        if let Some(cold) = topic.retention.cold_uri.as_deref() {
563            if cold.contains("://") {
564                return Err(BusError::Backend(format!(
565                    "topic {:?} configures cold_uri {cold:?} which requires feature cold-store; \
566                     rebuild pv-bridge with `--features cold-store` (or `cold-store-aws`) or clear \
567                     the field",
568                    topic.name,
569                )));
570            }
571        }
572    }
573    Ok(())
574}

576impl EmbeddedBroker {
577    /// Enforce per-topic hot retention at time `now_ms`: when
578    /// `retention.cold_uri` is set, each expired record
579    /// is first exported to the cold tier as its own write-once object (via the
580    /// authenticated `ColdArchive` for `scheme://` URIs, or
581    /// `write_cold_event_once` for local directory paths) before being
582    /// removed from the hot segment via atomic rewrite; with `cold_uri`
583    /// unset, expired records are dropped from the hot segment without
584    /// export. Returns the number of
585    /// records expired.
586    pub fn enforce_retention(&self, now_ms: u64) -> Result<u64, BusError> {
587        let mut expired_total = 0u64;
588        // Contain per-partition failures. A
589        // persistent error class (divergent cold intent, cold-dir
590        // permission error, content-mismatch) used to `?`-propagate out
591        // of the whole two-level loop — starving retention for that
592        // topic AND every topic after it in manifest order, forever,
593        // until disk filled. Process every partition, then surface the
594        // first error as an aggregate so the maintenance counter still
595        // fires.
596        let mut first_error: Option<(String, BusError)> = None;
597        for t in &self.manifest.topics {
598            let cutoff =
599                now_ms.saturating_sub(u64::from(t.retention.hot_hours) * pv_core::units::MS_PER_HOUR);
600            let Some(parts) = self.partitions.get(&t.name) else {
601                continue;
602            };
603            for p in parts {
604                let mut part = p.lock();
605                // A poisoned partition's segment may carry torn bytes from
606                // an unrepaired append failure. Rewriting it here would
607                // preserve the partial line WITH a trailing newline —
608                // converting a restart-repairable torn tail (recover_segment
609                // truncates a no-newline tail) into a permanent corrupt
610                // line that breaks `fetch` forever. Skip until a restart
611                // re-establishes a consistent tail.
612                if let Some(reason) = &part.poisoned {
613                    tracing::warn!(
614                        topic = %t.name,
615                        reason = %reason,
616                        "skipping retention on a poisoned partition (restart to recover)"
617                    );
618                    continue;
619                }
620                let outcome: Result<u64, BusError> = (|| {
621                    // Streaming pass 1: bail with zero allocation when
622                    // nothing expired (the common hourly case). Offsets are
623                    // kept for the post-rename idempotency-map prune.
624                    let expired_offsets = expired_offsets(&part.path, cutoff)?;
625                    if expired_offsets.is_empty() {
626                        return Ok(0);
627                    }
628                    let expired_count = expired_offsets.len() as u64;
629                    // Cold export first (never destroy before the copy lands).
630                    // Streaming pass 2: export each expired line as it is
631                    // read — no whole-segment buffering.
632                    if let Some(cold) = &t.retention.cold_uri {
633                        if cold.contains("://") {
634                            #[cfg(feature = "cold-store")]
635                            {
636                                let archive = self.cold_archive.as_ref().ok_or_else(|| {
637                                    BusError::Backend(format!("cold archive for {:?} is unavailable", t.name))
638                                })?;
639                                for_each_segment_line(&part.path, cutoff, |line, is_expired| {
640                                    if is_expired {
641                                        let event: StoredEvent = serde_json::from_str(line)?;
642                                        archive.put(&t.name, &event)?;
643                                    }
644                                    Ok(())
645                                })?;
646                            }
647                            #[cfg(not(feature = "cold-store"))]
648                            return Err(BusError::Backend(format!(
649                                "cold_uri {cold:?} requires feature cold-store"
650                            )));
651                        } else {
652                            let partition = part
653                                .path
654                                .file_stem()
655                                .and_then(std::ffi::OsStr::to_str)
656                                .unwrap_or("partition");
657                            let cold_dir = Path::new(cold).join(&t.name).join(partition);
658                            // Sync newly created ancestors too: the hot rewrite
659                            // below destroys the only other copy, so the cold
660                            // subtree's dirents must be durable before it runs.
661                            pv_core::fsutil::create_dir_all_synced(&cold_dir)?;
662                            for_each_segment_line(&part.path, cutoff, |line, is_expired| {
663                                if is_expired {
664                                    let event: StoredEvent = serde_json::from_str(line)?;
665                                    write_cold_event_once(&cold_dir, &event)?;
666                                }
667                                Ok(())
668                            })?;
669                            pv_core::fsutil::sync_directory(&cold_dir)?;
670                        }
671                    }
672                    persist_high_water(&part.watermark_path, part.next_offset)?;
673                    // Atomic hot rewrite. Use a UUID-suffixed tmp name so a
674                    // stale tmp from a prior crashed pass isn't reused (and
675                    // an external backup/rsync tool can't grab an
676                    // in-progress file thinking it's stable data).
677                    let tmp = part
678                        .path
679                        .with_extension(format!("jsonl.{}.tmp", pv_core::new_event_uid()));
680                    // RAII guard cleans up the tmp on any early
681                    // Err in the write/sync/rename path so a repeatedly-
682                    // failing rewrite (ENOSPC/EIO) does not fill the inode
683                    // table with UUID-suffixed orphan .tmp files.
684                    let mut guard = pv_core::fsutil::TempPathGuard::new(tmp.clone());
685                    {
686                        // Explicit 0o600 mode so the retention rewrite
687                        // doesn't silently ratchet segment permissions
688                        // back to 0o644 under the ambient umask — the
689                        // previous `File::create` recipe would land at
690                        // 0o644 on default-umask hosts and, because
691                        // `rename` preserves the tmp's mode, every
692                        // retention pass re-opened the confidentiality
693                        // gap the initial 0o600 append open closes.
694                        let mut tmp_options = fs::OpenOptions::new();
695                        tmp_options.write(true).create_new(true);
696                        #[cfg(unix)]
697                        {
698                            use std::os::unix::fs::OpenOptionsExt as _;
699                            tmp_options.mode(0o600);
700                        }
701                        let mut f = tmp_options.open(&tmp)?;
702                        // Streaming pass 3: kept lines flow straight to the
703                        // tmp segment.
704                        {
705                            use std::io::Write as _;
706                            let mut writer = std::io::BufWriter::new(&mut f);
707                            for_each_segment_line(&part.path, cutoff, |line, is_expired| {
708                                if !is_expired {
709                                    writer.write_all(line.as_bytes())?;
710                                    writer.write_all(b"\n")?;
711                                }
712                                Ok(())
713                            })?;
714                            writer.flush()?;
715                        }
716                        f.sync_all()?;
717                    }
718                    fs::rename(&tmp, &part.path)?;
719                    guard.disarm();
720                    // Everything between the rename above and a successful
721                    // writer reopen below runs while `part.writer` still holds
722                    // the fd of the pre-rewrite (now unlinked) inode. If any of
723                    // it fails, propagating the error WITHOUT poisoning the
724                    // partition would leave that stale handle live: every
725                    // subsequent publish would append to the unlinked inode —
726                    // acked as durable, invisible to `fetch`, and irrecoverably
727                    // freed at process exit. Poison the partition on any
728                    // post-rename failure so appends fail closed until a
729                    // restart reopens the renamed segment (same fail-closed
730                    // posture as an unrepaired append truncation).
731                    let post_rename: Result<(), BusError> = (|| {
732                        if let Some(parent) = part.path.parent() {
733                            pv_core::fsutil::sync_directory(parent)?;
734                        }
735                        // Prune the idempotency map + sidecar of any UID whose offset
736                        // was expired: without this, a subsequent `publish_idempotent`
737                        // with a still-cached UID returns an ack pointing at data that
738                        // no longer exists, and the caller's follow-up `fetch(offset)`
739                        // silently returns the wrong event or nothing.
740                        //
741                        // Expired offsets were collected in streaming pass 1
742                        // (only parseable lines expire, so the set covers
743                        // exactly the removed records). The previous
744                        // survivors+[min,max] range heuristic was unsound at
745                        // both edges: a wall-clock regression could expire a
746                        // MIDDLE-offset record whose UID then survived
747                        // pruning (stale ack → wrong-record fetch), and a
748                        // trailing unparseable kept line's offset fell
749                        // OUTSIDE the range so its UID was dropped
750                        // (duplicate re-append on the next
751                        // publish_idempotent).
752                        let before = part.seen_event_uids.len();
753                        part.seen_event_uids
754                            .retain(|_, offset| !expired_offsets.contains(offset));
755                        if part.seen_event_uids.len() != before {
756                            // Sidecar is now stale — rewrite it atomically with only
757                            // the surviving mappings so recovery cannot resurrect a
758                            // just-expired UID.
759                            let mut lines: Vec<(String, u64)> = part
760                                .seen_event_uids
761                                .iter()
762                                .map(|(uid, offset)| (uid.clone(), *offset))
763                                .collect();
764                            lines.sort_by_key(|(_, offset)| *offset);
765                            let mut sidecar = Vec::new();
766                            for (uid, offset) in lines {
767                                let mapping = serde_json::to_string(&EventUidOffset {
768                                    event_uid: uid,
769                                    offset,
770                                })?;
771                                sidecar.extend_from_slice(mapping.as_bytes());
772                                sidecar.push(b'\n');
773                            }
774                            rewrite_atomic(&part.idempotency_path, &sidecar)?;
775                        }
776                        // Segment reopen after retention rewrite. Use
777                        // O_NOFOLLOW so a co-tenant that raced the
778                        // rename window with a symlink plant cannot
779                        // hijack the next authenticated append (a
780                        // spool-plant confused-deputy). No create ⇒
781                        // 0o600 mode is decorative but preserved for
782                        // uniformity with the initial-open site.
783                        let mut reopen = fs::OpenOptions::new();
784                        reopen.append(true);
785                        #[cfg(unix)]
786                        {
787                            use std::os::unix::fs::OpenOptionsExt as _;
788                            reopen
789                                .custom_flags(pv_core::fsutil::unix_o_nofollow())
790                                .mode(0o600);
791                        }
792                        part.writer = reopen.open(&part.path)?;
793                        Ok(())
794                    })();
795                    if let Err(error) = post_rename {
796                        part.poisoned = Some(format!(
797                            "retention post-rename failure ({error}); the append handle may still \
798                         reference the replaced segment inode (restart to recover)"
799                        ));
800                        return Err(error);
801                    }
802                    Ok(expired_count)
803                })();
804                match outcome {
805                    Ok(count) => expired_total += count,
806                    Err(error) => {
807                        tracing::warn!(
808                            topic = %t.name,
809                            %error,
810                            "retention failed for this partition; remaining topics/partitions continue"
811                        );
812                        if first_error.is_none() {
813                            first_error = Some((t.name.clone(), error));
814                        }
815                    }
816                }
817            }
818        }
819        match first_error {
820            Some((topic, error)) => Err(BusError::Backend(format!(
821                "retention failed for topic {topic:?} (remaining topics were still processed, \
822                 {expired_total} records expired): {error}"
823            ))),
824            None => Ok(expired_total),
825        }
826    }
827}

829fn write_cold_event_once(directory: &Path, event: &StoredEvent) -> Result<(), BusError> {
830    let path = directory.join(format!("{:020}.json", event.offset));
831    let bytes = serde_json::to_vec(event)?;
832    if path.exists() {
833        return if fs::read(&path)? == bytes {
834            // Idempotent re-export (e.g. a retention pass that crashed
835            // after the cold copy landed but before the hot rewrite).
836            Ok(())
837        } else {
838            // Path-leak hazard: `write_cold_event_once`
839            // errors bubble up through `enforce_retention` and can
840            // reach a
841            // tracing::warn!(%error) on the maintenance path.
842            // Basename the absolute cold-destination path so a
843            // duplicate-object collision doesn't ship the
844            // deployment topology to OTLP.
845            Err(BusError::Backend(format!(
846                "cold object {} already exists with different content",
847                pv_core::fsutil::basename(&path)
848            )))
849        };
850    }
851    // Crash-atomic write-once. Writing directly at the final name
852    // (create_new → write_all → sync_all) meant a crash/ENOSPC mid-write
853    // left a torn file there, and every subsequent retention pass hit
854    // AlreadyExists + content mismatch → enforce_retention aborted for
855    // all topics, forever. Stage in a UUID-suffixed tmp, fsync, then
856    // rename — the final path only ever holds complete objects (same
857    // shape as `rewrite_atomic`; the caller fsyncs the directory after
858    // the batch). True conflicts are still detected by the pre-existing
859    // content comparison above.
860    let tmp = path.with_extension(format!("json.{}.tmp", pv_core::new_event_uid()));
861    let mut guard = pv_core::fsutil::TempPathGuard::new(tmp.clone());
862    {
863        // Explicit 0o600 mode — cold objects carry full `StoredEvent`
864        // records (same session-data class as the hot segment). The
865        // previous `File::create` recipe landed at 0o644 under the
866        // ambient umask; `rename` preserves the tmp's mode, so cold
867        // storage inherited world-readable posture without
868        // announcement.
869        let mut tmp_options = fs::OpenOptions::new();
870        tmp_options.write(true).create_new(true);
871        #[cfg(unix)]
872        {
873            use std::os::unix::fs::OpenOptionsExt as _;
874            tmp_options.mode(0o600);
875        }
876        let mut file = tmp_options.open(&tmp)?;
877        file.write_all(&bytes)?;
878        file.sync_all()?;
879    }
880    fs::rename(&tmp, &path)?;
881    guard.disarm();
882    Ok(())
883}

885/// Pre-claim phase: resolve and schema-validate every referenced schema
886/// without touching the filesystem, so a bad `schema_ref` fails provision
887/// retry-clean (before the single-winner claim).
888fn resolve_referenced_schemas(
889    manifest: &BridgeManifest,
890) -> Result<Vec<(String, serde_json::Value)>, BusError> {
891    let mut resolved = Vec::new();
892    for topic in &manifest.topics {
893        let Some(reference) = &topic.schema_ref else {
894            continue;
895        };
896        let value = crate::manifest::schema_document(reference)?;
897        jsonschema::options()
898            .should_validate_formats(true)
899            .build(&value)
900            .map_err(|error| BusError::Backend(format!("invalid schema {reference:?}: {error}")))?;
901        resolved.push((reference.clone(), value));
902    }
903    Ok(resolved)
904}

906/// Post-claim phase: persist the pre-resolved schema copies.
907fn write_referenced_schemas(
908    data_dir: &Path,
909    resolved: &[(String, serde_json::Value)],
910) -> Result<(), BusError> {
911    for (reference, value) in resolved {
912        let destination = data_dir.join(reference);
913        if let Some(parent) = destination.parent() {
914            pv_core::fsutil::create_dir_all_synced(parent)?;
915        }
916        // Fsync-safe atomic replace: a crash mid-`fs::write` (the old
917        // shape) left torn JSON that made the `load_validators` disk
918        // fallback fail `open()` permanently for non-builtin refs.
919        rewrite_atomic(&destination, &serde_json::to_vec_pretty(value)?)?;
920    }
921    Ok(())
922}

924fn load_validators(
925    data_dir: &Path,
926    manifest: &BridgeManifest,
927) -> Result<HashMap<String, jsonschema::Validator>, BusError> {
928    let mut validators = HashMap::new();
929    for topic in &manifest.topics {
930        let Some(reference) = &topic.schema_ref else {
931            continue;
932        };
933        let schema = crate::manifest::schema_document(reference).or_else(|_| {
934            // Cap the schema file read. Schemas are control-
935            // plane data (bounded well below 1 MiB in practice); a hostile
936            // plant of a multi-GiB schema file would OOM the broker before
937            // the JSON parser could complain.
938            let bytes =
939                pv_core::fsutil::read_capped(&data_dir.join(reference), pv_core::fsutil::MAX_CONTROL_BYTES)?;
940            serde_json::from_slice(&bytes).map_err(BusError::from)
941        })?;
942        let validator = jsonschema::options()
943            .should_validate_formats(true)
944            .build(&schema)
945            .map_err(|error| BusError::Backend(format!("invalid schema {reference:?}: {error}")))?;
946        validators.insert(topic.name.clone(), validator);
947    }
948    Ok(validators)
949}

951/// Count complete lines; truncate a torn trailing line if present.
952///
953/// Streams the segment (same pattern as
954/// `recover_event_uids`) instead of `fs::read`-ing it whole: segments
955/// grow with accepted event traffic and are only time-bounded by
956/// retention, so recovery of a large partition otherwise loaded an
957/// arbitrarily large file fully into memory — the one unbounded
958/// allocation left in startup recovery. Each line is capped; a line
959/// above the cap fails the open loudly (fail closed, like a corrupt
960/// watermark) rather than risking the allocation.
961fn recover_segment(path: &Path, persisted_offset: u64) -> Result<(u64, u64), BusError> {
962    /// Generous per-record bound: the harness caps request bodies at
963    /// 16 MiB, so no legitimately published event line approaches this.
964    const MAX_SEGMENT_LINE_BYTES: u64 = 16 * 1024 * 1024;
965    use std::io::BufRead as _;
966    let file = match fs::File::open(path) {
967        Ok(file) => file,
968        Err(error) if error.kind() == std::io::ErrorKind::NotFound => return Ok((0, 0)),
969        Err(error) => return Err(BusError::Io(error)),
970    };
971    let mut reader = std::io::BufReader::new(file);
972    let mut buf: Vec<u8> = Vec::new();
973    let mut complete_len: u64 = 0;
974    let mut next_offset = 0u64;
975    let mut parseable_lines = 0u64;
976    let mut unparseable_lines = 0u64;
977    let mut torn = false;
978    loop {
979        buf.clear();
980        let read = {
981            use std::io::Read as _;
982            let mut limited = (&mut reader).take(MAX_SEGMENT_LINE_BYTES);
983            limited.read_until(b'\n', &mut buf)?
984        };
985        if read == 0 {
986            break;
987        }
988        if buf.last() != Some(&b'\n') {
989            if read as u64 == MAX_SEGMENT_LINE_BYTES {
990                return Err(BusError::Backend(format!(
991                    "segment {} contains a record above {MAX_SEGMENT_LINE_BYTES} bytes; refusing to open the partition",
992                    pv_core::fsutil::basename(path)
993                )));
994            }
995            // Torn trailing line (crash mid-append).
996            torn = true;
997            break;
998        }
999        complete_len = complete_len.saturating_add(read as u64);
1000        let line = buf.get(..buf.len().saturating_sub(1)).unwrap_or_default();
1001        if line.is_empty() {
1002            continue;
1003        }
1004        match serde_json::from_slice::<StoredEvent>(line) {
1005            Ok(event) => {
1006                parseable_lines = parseable_lines.saturating_add(1);
1007                next_offset = next_offset.max(event.offset.saturating_add(1));
1008            }
1009            // A single unparseable middle line must not brick the broker.
1010            // The corrupted bytes are left on disk as forensic evidence;
1011            // subsequent publishes append past the surviving max offset.
1012            // The line still occupies one logical offset (segment lines are
1013            // appended in offset order), so advance next_offset past it —
1014            // computing next_offset from parseable lines only let a corrupt
1015            // *trailing* line regress next_offset, and the next publish
1016            // reused that line's live offset (offset collision in the
1017            // audit stream).
1018            Err(error) => {
1019                unparseable_lines = unparseable_lines.saturating_add(1);
1020                next_offset = next_offset.saturating_add(1);
1021                tracing::warn!(
1022                    %error,
1023                    path = %pv_core::fsutil::basename(path),
1024                    "skipping unparseable segment record during recovery",
1025                );
1026            }
1027        }
1028    }
1029    // Unparseable lines advance `next_offset` RELATIVE to the running
1030    // value while parseable lines anchor it to the ABSOLUTE offsets
1031    // stored in the records. Segment lines are appended in contiguous
1032    // offset order, so mixing the two is exact whenever at least one
1033    // line parses — but a segment with ONLY unparseable complete lines
1034    // yields a bare line count with no anchor. When retention has ever
1035    // rewritten this segment (persisted watermark > 0) the true first
1036    // offset is unknowable: `max(count, watermark)` masks any records
1037    // appended after the last watermark persist, and the next publish
1038    // would re-stamp an acknowledged offset (offset collision in the
1039    // audit stream, with the stale sidecar UID re-homing acks onto the
1040    // new record). Fail closed like the oversized-record arm above —
1041    // the operator moves the segment aside for forensics and the
1042    // partition reopens at the watermark. A never-rewritten segment
1043    // (watermark 0) starts at absolute offset 0, so the count IS the
1044    // anchor and recovery stays exact.
1045    if unparseable_lines > 0 && parseable_lines == 0 && persisted_offset > 0 {
1046        return Err(BusError::Backend(format!(
1047            "segment {} contains only unparseable records ({unparseable_lines}) but the \
1048             partition has a retention watermark ({persisted_offset}); absolute offsets are \
1049             unrecoverable — refusing to open the partition (move the segment aside to repair)",
1050            pv_core::fsutil::basename(path)
1051        )));
1052    }
1053    if torn {
1054        // In-place truncate + fsync: unlike the old whole-file
1055        // `rewrite_atomic` (which required buffering the entire segment),
1056        // `set_len` touches only file metadata — a crash mid-truncate
1057        // leaves either the torn or the truncated length, never an
1058        // empty/renamed-away segment, so the total-segment-loss hazard
1059        // the old comment guarded against cannot occur.
1060        //
1061        // O_NOFOLLOW: this reopens a path derived from a broker
1062        // partition name; a pre-planted symlink at that path would
1063        // otherwise let a co-tenant redirect the `set_len` +
1064        // `sync_all` to an arbitrary file the daemon UID can touch
1065        // (canonical confused-deputy).
1066        let mut reopen = fs::OpenOptions::new();
1067        reopen.write(true);
1068        #[cfg(unix)]
1069        {
1070            use std::os::unix::fs::OpenOptionsExt as _;
1071            reopen.custom_flags(pv_core::fsutil::unix_o_nofollow());
1072        }
1073        let file = reopen.open(path)?;
1074        file.set_len(complete_len)?;
1075        file.sync_all()?;
1076    }
1077    Ok((next_offset, u64::from(torn)))
1078}

1080fn recover_event_uids(path: &Path) -> Result<HashMap<String, u64>, BusError> {
1081    if !path.exists() {
1082        return Ok(HashMap::new());
1083    }
1084    // Stream the sidecar with BufReader rather than
1085    // reading it whole and capping at MAX_ATIF_BYTES. The sidecar grows
1086    // one line per idempotent publish, compacted only when retention
1087    // expires UIDs — at 30-day default hot retention, ~1M live events
1088    // per partition (≈0.4 ev/s) crosses 64 MiB. The previous hard cap
1089    // would then fail `open()` on every restart, permanently bricking
1090    // the broker even though the segment on disk is authoritative and
1091    // `recover_segment_event_uids` will fix the map either way. Bound
1092    // per-line via `read_line` (line length is only limited by
1093    // `EventUidOffset`'s JSON encoding at ~200 B); an oversize line is
1094    // treated as sidecar corruption and skipped, matching the parse
1095    // failure discipline just below.
1096    let file = std::fs::File::open(path)?;
1097    let mut reader = std::io::BufReader::new(file);
1098    let mut seen = HashMap::new();
1099    let mut line_buffer = Vec::new();
1100    let mut valid_bytes: u64 = 0;
1101    let mut has_torn_tail = false;
1102    let mut logged_oversize_line = false;
1103    // Sibling once-per-file guards for the parse-failure and
1104    // duplicate-UID branches below. A crashed or hostile writer can
1105    // leave sidecars with millions of mangled-JSON lines (up to
1106    // hot-retention × events/partition, ≈ 1M per partition); without
1107    // per-branch guards the recovery pass would emit one warn per
1108    // corrupt line during the very restart it's diagnosing, bricking
1109    // the log pipeline. Same discipline as `logged_oversize_line`
1110    // above.
1111    let mut logged_parse_failure = false;
1112    let mut logged_duplicate_uid = false;
1113    // A single oversized-line hard cap: 4 KiB is 20x the fattest
1114    // sidecar record we produce. The reader is `take`-bounded to the
1115    // cap (+1 to detect overshoot) so a planted multi-GiB single line
1116    // cannot grow the buffer to file size; the overflow tail is
1117    // drained in cap-sized chunks and the whole line is skipped,
1118    // matching the parse-failure discipline below.
1119    const MAX_SIDECAR_LINE: usize = 4096;
1120    loop {
1121        line_buffer.clear();
1122        let bytes_read = {
1123            use std::io::Read as _;
1124            let mut limited = (&mut reader).take(MAX_SIDECAR_LINE as u64 + 1);
1125            match limited.read_until(b'\n', &mut line_buffer) {
1126                Ok(0) => break,
1127                Ok(bytes) => bytes,
1128                Err(error) => return Err(BusError::Io(error)),
1129            }
1130        };
1131        let terminated = line_buffer.last() == Some(&b'\n');
1132        if !terminated && bytes_read > MAX_SIDECAR_LINE {
1133            // Oversized line cut off by the `take` bound: drain the
1134            // rest of it in bounded chunks so the skip stays O(cap)
1135            // in memory whatever the line length on disk.
1136            let mut drained: u64 = bytes_read as u64;
1137            let mut found_newline = false;
1138            loop {
1139                line_buffer.clear();
1140                use std::io::Read as _;
1141                let mut limited = (&mut reader).take(MAX_SIDECAR_LINE as u64 + 1);
1142                match limited.read_until(b'\n', &mut line_buffer) {
1143                    Ok(0) => break,
1144                    Ok(bytes) => {
1145                        drained = drained.saturating_add(bytes as u64);
1146                        if line_buffer.last() == Some(&b'\n') {
1147                            found_newline = true;
1148                            break;
1149                        }
1150                    }
1151                    Err(error) => return Err(BusError::Io(error)),
1152                }
1153            }
1154            if !found_newline {
1155                // Oversized AND unterminated: the classic torn tail,
1156                // just fat. Truncate to the last complete line.
1157                has_torn_tail = true;
1158                break;
1159            }
1160            if !logged_oversize_line {
1161                tracing::warn!(
1162                    path = %pv_core::fsutil::basename(path),
1163                    line_bytes = drained,
1164                    "oversized event-uid sidecar record skipped during recovery"
1165                );
1166                logged_oversize_line = true;
1167            }
1168            valid_bytes = valid_bytes.saturating_add(drained);
1169            continue;
1170        }
1171        if !terminated {
1172            // Trailing partial line (crash between newline appends);
1173            // the atomic-append discipline in `publish_with_uid`
1174            // guarantees any complete line is fully synced before the
1175            // next one starts, so this is the classic torn tail.
1176            has_torn_tail = true;
1177            break;
1178        }
1179        if bytes_read > MAX_SIDECAR_LINE {
1180            if !logged_oversize_line {
1181                tracing::warn!(
1182                    path = %pv_core::fsutil::basename(path),
1183                    line_bytes = bytes_read,
1184                    "oversized event-uid sidecar record skipped during recovery"
1185                );
1186                logged_oversize_line = true;
1187            }
1188            valid_bytes = valid_bytes.saturating_add(bytes_read as u64);
1189            continue;
1190        }
1191        valid_bytes = valid_bytes.saturating_add(bytes_read as u64);
1192        let payload = line_buffer.strip_suffix(b"\n").unwrap_or(&line_buffer);
1193        if payload.is_empty() {
1194            continue;
1195        }
1196        let mapping: EventUidOffset = match serde_json::from_slice(payload) {
1197            Ok(mapping) => mapping,
1198            // Sidecar corruption skip-and-log: the segment is the
1199            // ground truth (see `recover_segment_event_uids` below), and
1200            // an unreadable idempotency line at most costs a duplicate
1201            // ack for the same UID.
1202            Err(error) => {
1203                if !logged_parse_failure {
1204                    tracing::warn!(
1205                        %error,
1206                        path = %pv_core::fsutil::basename(path),
1207                        "skipping unparseable event-uid sidecar record during recovery; \
1208                         subsequent parse failures on this sidecar suppressed to avoid log storm"
1209                    );
1210                    logged_parse_failure = true;
1211                }
1212                continue;
1213            }
1214        };
1215        if let Some(existing) = seen.insert(mapping.event_uid.clone(), mapping.offset) {
1216            if existing != mapping.offset {
1217                // Sidecar (idempotency journal) can legitimately hold
1218                // stale UID→offset pairs after a partial retention
1219                // (segment rewritten atomically, sidecar rewrite lost
1220                // in a crash). The segment on disk is the source of
1221                // truth; log and let `recover_segment_event_uids` fix
1222                // the mapping. Refusing to open the broker here would
1223                // brick the whole tier over a benign inconsistency.
1224                if !logged_duplicate_uid {
1225                    tracing::warn!(
1226                        event_uid = %mapping.event_uid,
1227                        prior_offset = existing,
1228                        current_offset = mapping.offset,
1229                        path = %pv_core::fsutil::basename(path),
1230                        "sidecar has duplicate UID entry; segment offset will win after \
1231                         full recovery. Subsequent duplicate-UID entries on this sidecar \
1232                         suppressed to avoid log storm"
1233                    );
1234                    logged_duplicate_uid = true;
1235                }
1236            }
1237        }
1238    }
1239    if has_torn_tail {
1240        // Truncate to the last complete line so future appends land
1241        // cleanly (the on-disk record set is unchanged). A transient
1242        // I/O error on `read_valid_prefix` (bit-rot, ENOENT race
1243        // with a concurrent unlink between the earlier BufReader
1244        // open and this reopen) must NOT collapse to an
1245        // `unwrap_or_default()` empty Vec — that atomically renames
1246        // a zero-byte file on top of the sidecar, permanently
1247        // destroying every persisted `EventUidOffset` mapping under
1248        // the guise of "torn-tail repair". Fail closed instead —
1249        // the caller retries recovery on the next daemon boot with
1250        // the sidecar bytes still intact. Same posture as
1251        // `read_high_water` (line 1132), which explicitly refused
1252        // its old "fail-open Ok(0)" for the identical rationale.
1253        let valid_prefix = read_valid_prefix(path, valid_bytes).map_err(BusError::Io)?;
1254        rewrite_atomic(path, &valid_prefix)?;
1255    }
1256    Ok(seen)
1257}

1259fn read_valid_prefix(path: &Path, len: u64) -> std::io::Result<Vec<u8>> {
1260    use std::io::Read as _;
1261    let file = std::fs::File::open(path)?;
1262    let mut reader = file.take(len);
1263    let mut out = Vec::with_capacity(len.try_into().unwrap_or(usize::MAX));
1264    reader.read_to_end(&mut out)?;
1265    Ok(out)
1266}

1268fn recover_segment_event_uids(path: &Path, seen: &mut HashMap<String, u64>) -> Result<(), BusError> {
1269    if !path.exists() {
1270        return Ok(());
1271    }
1272    for line in BufReader::new(fs::File::open(path)?).lines() {
1273        let line = line?;
1274        if line.is_empty() {
1275            continue;
1276        }
1277        let event: StoredEvent = match serde_json::from_str(&line) {
1278            Ok(event) => event,
1279            // Same skip-and-log policy as `recover_segment`: an unreadable
1280            // segment record has already been flagged there; here it just
1281            // means we cannot reconstruct its UID → offset mapping.
1282            Err(error) => {
1283                tracing::warn!(
1284                    %error,
1285                    path = %pv_core::fsutil::basename(path),
1286                    "skipping unparseable segment record while rebuilding UID index",
1287                );
1288                continue;
1289            }
1290        };
1291        let Some(uid) = event_uid_from_value(&event.value) else {
1292            continue;
1293        };
1294        if let Some(existing) = seen.insert(uid.to_owned(), event.offset) {
1295            if existing != event.offset {
1296                // Segment is authoritative — the sidecar was stale
1297                // (see recover_event_uids for the crash mode). Log
1298                // and keep the segment offset (the last write wins
1299                // via insert). Refusing to open would leave the
1300                // broker un-openable on a benign inconsistency.
1301                tracing::warn!(
1302                    event_uid = uid,
1303                    prior_offset = existing,
1304                    current_offset = event.offset,
1305                    "duplicate UID between sidecar and segment; segment offset wins"
1306                );
1307            }
1308        }
1309    }
1310    Ok(())
1311}

1330fn read_high_water(path: &Path) -> Result<u64, BusError> {
1331    // A watermark is at most u64 in decimal (~20 chars). Cap
1332    // the read so a hostile plant of a giant p<N>.next-offset cannot OOM
1333    // the broker at startup. Use MAX_CONTROL_BYTES (1 MiB) for the
1334    // shared trust boundary; a real watermark is orders of magnitude
1335    // smaller.
1336    match pv_core::fsutil::read_capped_string(path, pv_core::fsutil::MAX_CONTROL_BYTES) {
1337        Ok(value) => match value.trim().parse::<u64>() {
1338            Ok(offset) => Ok(offset),
1339            // A corrupt watermark file IS fatal. The old fail-open
1340            // fallback (`Ok(0)`) claimed the segment is authoritative
1341            // and "the next successful publish rewrites the watermark
1342            // and self-heals" — both false: `persist_high_water` runs
1343            // only from `enforce_retention`, and the watermark exists
1344            // precisely for the case where retention has emptied the
1345            // segment. Falling back to 0 there restarts `next_offset`
1346            // at 0, reusing offsets of already-cold-exported records
1347            // (offset collisions in the audit stream, permanent
1348            // "cold object … already exists with different content"
1349            // conflicts). Fail closed like every other corrupt
1350            // control file in this crate; NotFound below stays Ok(0)
1351            // because a missing watermark is a legitimate fresh state.
1352            Err(error) => Err(BusError::Backend(format!(
1353                "high-watermark file {} is corrupt ({error}); refusing to open the partition",
1354                pv_core::fsutil::basename(path)
1355            ))),
1356        },
1357        Err(error) if error.kind() == std::io::ErrorKind::NotFound => Ok(0),
1358        Err(error) => Err(BusError::Io(error)),
1359    }
1360}

1362fn persist_high_water(path: &Path, next_offset: u64) -> Result<(), BusError> {
1363    let parent = path
1364        .parent()
1365        .ok_or_else(|| BusError::Backend("high-watermark has no parent".to_owned()))?;
1366    let temporary = path.with_extension(format!("{}.tmp", pv_core::new_event_uid()));
1367    // RAII cleanup on any early Err — same discipline as
1368    // `rewrite_atomic` above. On a bad-disk day, watermark writes fire
1369    // on every publish; orphan tmp accumulation would be fastest here.
1370    let mut guard = pv_core::fsutil::TempPathGuard::new(temporary.clone());
1371    {
1372        // Explicit 0o600 mode — the watermark leaks throughput
1373        // pattern (a next-offset u64), which is minor per-file but
1374        // adds up over a busy broker. Cheap uniformity with the
1375        // segment/sidecar posture; removes umask-dependence from
1376        // the durability-critical control files.
1377        let mut tmp_options = fs::OpenOptions::new();
1378        tmp_options.write(true).create_new(true);
1379        #[cfg(unix)]
1380        {
1381            use std::os::unix::fs::OpenOptionsExt as _;
1382            tmp_options.mode(0o600);
1383        }
1384        let mut file = tmp_options.open(&temporary)?;
1385        file.write_all(next_offset.to_string().as_bytes())?;
1386        file.sync_all()?;
1387    }
1388    fs::rename(&temporary, path)?;
1389    guard.disarm();
1390    pv_core::fsutil::sync_directory(parent)?;
1391    Ok(())
1392}

1394/// Partition a segment's lines into (kept, expired) by `stored_at < cutoff`.
1395/// Stream every line of a segment through `visit(line, is_expired)`,
1396/// with per-line buffers bounded at 16 MiB (the segment line cap used by
1397/// the read paths). Replaces the old `split_by_time`, which buffered the
1398/// ENTIRE segment into two `Vec<String>`s on every hourly retention pass
1399/// — even when nothing had expired — so a large hot partition (or one
1400/// hostile giant line, unbounded under `BufRead::lines`) could OOM the
1401/// maintenance task. Lines that fail to parse as `StoredEvent` are
1402/// treated as unexpired, exactly like the old shape. A line at the cap
1403/// with no newline is refused as corrupt (the partition lock is held, so
1404/// there is no in-flight append to race).
1405fn for_each_segment_line(
1406    path: &Path,
1407    cutoff_ms: u64,
1408    mut visit: impl FnMut(&str, bool) -> Result<(), BusError>,
1409) -> Result<(), BusError> {
1410    const MAX_SEGMENT_LINE_BYTES: u64 = 16 * 1024 * 1024;
1411    if !path.exists() {
1412        return Ok(());
1413    }
1414    let mut reader = BufReader::new(fs::File::open(path)?);
1415    let mut buf: Vec<u8> = Vec::new();
1416    loop {
1417        buf.clear();
1418        let read = {
1419            use std::io::BufRead as _;
1420            use std::io::Read as _;
1421            let mut limited = (&mut reader).take(MAX_SEGMENT_LINE_BYTES);
1422            limited.read_until(b'\n', &mut buf)?
1423        };
1424        if read == 0 {
1425            return Ok(());
1426        }
1427        if buf.last() != Some(&b'\n') && read as u64 >= MAX_SEGMENT_LINE_BYTES {
1428            return Err(BusError::Backend(format!(
1429                "segment line exceeds the {MAX_SEGMENT_LINE_BYTES}-byte cap in {}",
1430                path.display()
1431            )));
1432        }
1433        let line = std::str::from_utf8(&buf)
1434            .map_err(|error| BusError::Backend(format!("segment line is not UTF-8: {error}")))?
1435            .trim_end_matches('\n');
1436        if line.is_empty() {
1437            continue;
1438        }
1439        let is_expired = serde_json::from_str::<StoredEvent>(line)
1440            .map(|e| e.stored_at < cutoff_ms)
1441            .unwrap_or(false);
1442        visit(line, is_expired)?;
1443    }
1444}
1445
1446/// Streaming pass 1 of retention: collect the OFFSETS of expired
1447/// records (for the idempotency-map prune) without buffering lines —
1448/// 8 bytes per expired record instead of the whole segment.
1449fn expired_offsets(path: &Path, cutoff_ms: u64) -> Result<std::collections::HashSet<u64>, BusError> {
1450    let mut offsets = std::collections::HashSet::new();
1451    for_each_segment_line(path, cutoff_ms, |line, is_expired| {
1452        if is_expired {
1453            if let Ok(event) = serde_json::from_str::<StoredEvent>(line) {
1454                offsets.insert(event.offset);
1455            }
1456        }
1457        Ok(())
1458    })?;
1459    Ok(offsets)
1460}

1462impl EmbeddedBroker {
1463    fn publish_with_uid(
1464        &self,
1465        topic: &str,
1466        key: &str,
1467        value: &serde_json::Value,
1468        event_uid: Option<&str>,
1469    ) -> Result<PublishAck, BusError> {
1470        let parts = self
1471            .partitions
1472            .get(topic)
1473            .ok_or_else(|| BusError::UnknownTopic(topic.to_owned()))?;
1474        if let Some(validator) = self.validators.get(topic) {
1475            let errors: Vec<String> = validator
1476                .iter_errors(value)
1477                .take(3)
1478                .map(|error| error.to_string())
1479                .collect();
1480            if !errors.is_empty() {
1481                return Err(BusError::Backend(format!(
1482                    "event rejected by schema for topic {topic:?}: {}",
1483                    errors.join("; ")
1484                )));
1485            }
1486        }
1487        if let Some(expected) = event_uid {
1488            match event_uid_from_value(value) {
1489                Some(actual) if expected != actual => {
1490                    return Err(BusError::Backend(
1491                        "idempotency UID does not match event metadata UID".to_owned(),
1492                    ));
1493                }
1494                // A missing metadata.uid would make crash recovery lose the UID
1495                // → sidecar because recover_segment_event_uids scans metadata.uid;
1496                // require the UID to be embedded so recovery is deterministic.
1497                None => {
1498                    return Err(BusError::Backend(
1499                        "idempotent publish requires event metadata.uid to match the UID".to_owned(),
1500                    ));
1501                }
1502                _ => {}
1503            }
1504        }
1505        let partition = partition_for(key, u32::try_from(parts.len()).unwrap_or(u32::MAX));
1506        let slot = parts
1507            .get(partition as usize)
1508            .ok_or_else(|| BusError::Backend(format!("partition {partition} out of range")))?;
1509        let mut part = slot.lock();
1510        if let Some(reason) = &part.poisoned {
1511            return Err(BusError::Backend(format!(
1512                "partition {partition} of topic {topic:?} is poisoned by an unrepaired append failure \
1513                 (restart to recover): {reason}"
1514            )));
1515        }
1516        if let Some(uid) = event_uid {
1517            if let Some(offset) = part.seen_event_uids.get(uid).copied() {
1518                return Ok(PublishAck {
1519                    topic: topic.to_owned(),
1520                    partition,
1521                    offset,
1522                });
1523            }
1524        }
1525        let offset = part.next_offset;
1526        let record = StoredEvent {
1527            partition,
1528            offset,
1529            key: key.to_owned(),
1530            value: value.clone(),
1531            stored_at: pv_core::time::now_ms(),
1532        };
1533        let line = serde_json::to_string(&record)?;
1534        // Compute the successor BEFORE any bytes touch the disk: at the
1535        // (tamper-shaped) u64::MAX boundary the old post-append check
1536        // returned an error AFTER the record was durably fsynced without
1537        // ever advancing `next_offset`, so every retry appended another
1538        // record stamped with the same colliding offset.
1539        let next_after_append = part
1540            .next_offset
1541            .checked_add(1)
1542            .ok_or_else(|| BusError::Backend("embedded offset overflow".to_owned()))?;
1543        // Capture the durable length before appending so a failed append can
1544        // be repaired. Without the truncate-on-error below, two failure
1545        // modes leave the partition permanently inconsistent while the same
1546        // writer stays live:
1547        //   1. a torn write (ENOSPC/EIO mid-`write_all`) leaves partial
1548        //      bytes that the NEXT publish appends after, merging both into
1549        //      one unparseable line that ends in '\n' — `fetch` then errors
1550        //      on every read, and startup recovery only repairs torn
1551        //      *trailing* lines (no newline at EOF), not mid-file ones;
1552        //   2. a failed `sync_data` after a complete write leaves the
1553        //      record durable while `next_offset` was never advanced, so
1554        //      the next publish stamps a different record with the same
1555        //      offset (duplicate offsets hidden by max(offset)+1 recovery).
1556        let known_good = part.writer.metadata()?.len();
1557        let append = (|| {
1558            part.writer.write_all(line.as_bytes())?;
1559            part.writer.write_all(b"\n")?;
1560            part.writer.flush()?;
1561            part.writer.sync_data()
1562        })();
1563        if let Err(error) = append {
1564            if let Err(repair) = part
1565                .writer
1566                .set_len(known_good)
1567                .and_then(|()| part.writer.sync_data())
1568            {
1569                // The segment may still carry torn bytes; refuse further
1570                // appends (fail-closed) rather than corrupt the next record.
1571                part.poisoned = Some(format!("append failed ({error}); truncation failed ({repair})"));
1572            }
1573            return Err(BusError::Io(error));
1574        }
1575        part.next_offset = next_after_append;
1576        if let Some(uid) = event_uid {
1577            part.seen_event_uids.insert(uid.to_owned(), offset);
1578            // The sidecar is a rebuildable cache: `open()` reconciles it
1579            // from the segment itself (`recover_segment_event_uids` —
1580            // idempotent publishes are required to embed metadata.uid),
1581            // and the in-memory map updated above already serves
1582            // same-process retries. The record is durable (fsynced append
1583            // above), so a sidecar failure must NOT fail the publish:
1584            // reporting an error for an operation whose primary effect
1585            // succeeded invites the caller to re-mint the event under a
1586            // fresh UID — a genuine duplicate. Log loudly and return the
1587            // ack instead.
1588            let sidecar_write = (|| -> Result<(), BusError> {
1589                let mapping = serde_json::to_string(&EventUidOffset {
1590                    event_uid: uid.to_owned(),
1591                    offset,
1592                })?;
1593                // Directory entry for a first-time-created sidecar is only
1594                // durable after sync_directory(parent). Sync the parent when
1595                // we discover we're creating the file so a subsequent power
1596                // loss cannot lose the whole sidecar (which would silently
1597                // convert future publish_idempotent calls into duplicate
1598                // appends).
1599                let sidecar_created = std::fs::symlink_metadata(&part.idempotency_path).is_err();
1600                // Same 0o600 + O_NOFOLLOW discipline the segment append
1601                // gets — sidecar entries are (event_uid, offset) pairs,
1602                // and event_uids are session-derived identifiers.
1603                let mut sidecar_options = fs::OpenOptions::new();
1604                sidecar_options.create(true).append(true);
1605                #[cfg(unix)]
1606                {
1607                    use std::os::unix::fs::OpenOptionsExt as _;
1608                    sidecar_options
1609                        .custom_flags(pv_core::fsutil::unix_o_nofollow())
1610                        .mode(0o600);
1611                }
1612                let mut idempotency = sidecar_options.open(&part.idempotency_path)?;
1613                idempotency.write_all(mapping.as_bytes())?;
1614                idempotency.write_all(b"\n")?;
1615                idempotency.sync_data()?;
1616                if sidecar_created {
1617                    if let Some(parent) = part.idempotency_path.parent() {
1618                        pv_core::fsutil::sync_directory(parent)?;
1619                    }
1620                }
1621                Ok(())
1622            })();
1623            if let Err(error) = sidecar_write {
1624                tracing::warn!(
1625                    %error,
1626                    topic,
1627                    partition,
1628                    offset,
1629                    "idempotency sidecar append failed; the publish is durable and the \
1630                     UID index will be rebuilt from the segment at the next open"
1631                );
1632            }
1633        }
1634        Ok(PublishAck {
1635            topic: topic.to_owned(),
1636            partition,
1637            offset,
1638        })
1639    }
1640}

1665    /// O(1) event-UID lookup via the partition's in-memory
1666    /// `seen_event_uids` map. The default trait impl at
1667    /// `bus.rs::find_event_by_uid` linearly scans the segment from
1668    /// offset 0 with `fetch`, which for `EmbeddedBroker` re-opens
1669    /// the segment file each call and streams from the head with
1670    /// a `BufReader` — so each fetch page for offset > 0 also walks
1671    /// the whole prefix. `resolve_lifecycle_ack` in the harness
1672    /// calls this on EVERY `emit_bridge_event` (twice per session
1673    /// close — RECEIPT + SESSION_CLOSE), which meant close latency
1674    /// grew O(total events per partition²) amortised. The Kafka
1675    /// and NATS bus implementations both override with O(1) lookups;
1676    /// only the embedded broker was still on the trait default.
1677    fn find_event_by_uid(
1678        &self,
1679        topic: &str,
1680        key: &str,
1681        event_uid: &str,
1682    ) -> Result<Option<PublishAck>, BusError> {
1683        let parts = self
1684            .partitions
1685            .get(topic)
1686            .ok_or_else(|| BusError::UnknownTopic(topic.to_owned()))?;
1687        let partition = partition_for(key, u32::try_from(parts.len()).unwrap_or(u32::MAX));
1688        let slot = parts
1689            .get(partition as usize)
1690            .ok_or_else(|| BusError::Backend(format!("partition {partition} out of range")))?;
1691        let part = slot.lock();
1692        Ok(part
1693            .seen_event_uids
1694            .get(event_uid)
1695            .copied()
1696            .map(|offset| PublishAck {
1697                topic: topic.to_owned(),
1698                partition,
1699                offset,
1700            }))
1701    }

1703    fn fetch(
1704        &self,
1705        topic: &str,
1706        partition: u32,
1707        offset: u64,
1708        max: usize,
1709    ) -> Result<Vec<StoredEvent>, BusError> {
1710        let parts = self
1711            .partitions
1712            .get(topic)
1713            .ok_or_else(|| BusError::UnknownTopic(topic.to_owned()))?;
1714        let slot = parts
1715            .get(partition as usize)
1716            .ok_or_else(|| BusError::Backend(format!("partition {partition} out of range")))?;
1717        // Hold the partition lock across the entire read so a concurrent
1718        // enforce_retention rename cannot swap the segment mid-read.
1719        let part = slot.lock();
1720        if max == 0 || !part.path.exists() {
1721            return Ok(Vec::new());
1722        }
1723        let reader = BufReader::new(fs::File::open(&part.path)?);
1724        let mut out = Vec::with_capacity(max.min(1024));
1725        for line in reader.lines() {
1726            let line = line?;
1727            if line.is_empty() {
1728                continue;
1729            }
1730            let ev: StoredEvent = match serde_json::from_str(&line) {
1731                Ok(ev) => ev,
1732                // Same policy as `recover_segment`: a single unparseable
1733                // complete line (bit-rot, tamper, a crash mode recovery
1734                // does not repair) must not brick the read side. Retention
1735                // keeps unparseable lines forever (`split_by_time` treats
1736                // them as non-expired), so erroring here made every fetch
1737                // spanning the line fail permanently while publishes kept
1738                // acking events that could never be delivered.
1739                Err(error) => {
1740                    tracing::warn!(
1741                        %error,
1742                        path = %pv_core::fsutil::basename(&part.path),
1743                        "skipping unparseable segment record during fetch",
1744                    );
1745                    continue;
1746                }
1747            };
1748            if ev.offset < offset {
1749                continue;
1750            }
1751            out.push(ev);
1752            if out.len() >= max {
1753                break;
1754            }
1755        }
1756        Ok(out)
1757    }

1772    fn maintenance(&self, now_ms: u64) -> Result<u64, BusError> {
1773        let expired = EmbeddedBroker::enforce_retention(self, now_ms)?;
1774        // Drain cold-export intents that a previous `ColdArchive::commit`
1775        // converted from a transient remote failure into a durable
1776        // "queued for retry" file. Only the Kafka/NATS maintenance paths
1777        // used to drain the queue; the embedded broker never did, so the
1778        // retry promise was never fulfilled — a silent permanent
1779        // cold-tier gap after the record left the hot segment.
1780        #[cfg(feature = "cold-store")]
1781        if let Some(archive) = &self.cold_archive {
1782            archive.retry_pending_with(|pending| {
1783                // Embedded intents are staged and committed inside one
1784                // `put()` call during retention, so a surviving
1785                // offset-None intent means a crash landed between stage
1786                // and commit while the record was still hot (the segment
1787                // rewrite only happens after export). Resolve the offset
1788                // from the partition's idempotency map instead of
1789                // re-publishing, which would append a duplicate record.
1790                let parts = self
1791                    .partitions
1792                    .get(&pending.topic)
1793                    .ok_or_else(|| BusError::UnknownTopic(pending.topic.clone()))?;
1794                let slot = parts.get(pending.partition as usize).ok_or_else(|| {
1795                    BusError::Backend(format!("partition {} out of range", pending.partition))
1796                })?;
1797                let offset = slot
1798                    .lock()
1799                    .seen_event_uids
1800                    .get(&pending.event_uid)
1801                    .copied()
1802                    .ok_or_else(|| {
1803                        BusError::Backend(format!(
1804                            "cold intent {:?} has no broker offset and no live segment \
1805                             record to resolve it from; refusing to fabricate one",
1806                            pending.event_uid
1807                        ))
1808                    })?;
1809                Ok(PublishAck {
1810                    topic: pending.topic.clone(),
1811                    partition: pending.partition,
1812                    offset,
1813                })
1814            })?;
1815        }
1816        Ok(expired)
1817    }
