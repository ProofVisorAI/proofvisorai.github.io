13/// Retention policy for one topic.
14#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
15#[serde(deny_unknown_fields)]
16pub struct RetentionSpec {
17    /// Hot retention window in hours (default 720 = 30 days, per the brief).
18    pub hot_hours: u32,
19    /// Optional cold-tier export directory/bucket URI (customer-owned storage).
20    #[serde(default, skip_serializing_if = "Option::is_none")]
21    pub cold_uri: Option<String>,
22}
23
24impl Default for RetentionSpec {
25    fn default() -> Self {
26        Self {
27            hot_hours: 720,
28            cold_uri: None,
29        }
30    }
31}

33/// One topic declaration.
34#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
35#[serde(deny_unknown_fields)]
36pub struct TopicSpec {
37    /// Topic name (`agent.tool_call`, …).
38    pub name: String,
39    /// Partition count (partitioned by `ai_agent.instance_uid`).
40    pub partitions: u32,
41    /// Retention policy. Required (declared as such in the shipped
42    /// JSON Schema at `schemas/bridge-manifest.schema.json`). If we
43    /// silently defaulted here the `pvctl manifest-validate` verdict
44    /// would split from every external JSON-Schema linter's — an
45    /// operator seeing the CLI's OK plus their platform gate's FAIL
46    /// on the same file has to guess which is correct. Retention is
47    /// also semantically load-bearing: an operator who reads the
48    /// schema's `hot_hours: default: 720` and omits the block gets
49    /// 30 days of hot storage they never asked for.
50    pub retention: RetentionSpec,
51    /// JSON Schema reference events on this topic must satisfy.
52    #[serde(default, skip_serializing_if = "Option::is_none")]
53    pub schema_ref: Option<String>,
54}

75/// Manifest validation errors.
76#[derive(Debug, thiserror::Error, PartialEq, Eq)]
77#[non_exhaustive]
78pub enum ManifestError {
79    /// Unsupported version.
80    #[error("unsupported manifest_version {0} (this build supports {MANIFEST_VERSION})")]
81    Version(u32),
82    /// YAML/JSON parse failure.
83    #[error("manifest parse: {0}")]
84    Parse(String),
85    /// Structural problem.
86    #[error("manifest invalid: {0}")]
87    Invalid(String),
88}

90impl BridgeManifest {
91    /// The default manifest covering every OCSF event class topic.
92    pub fn default_for(name: &str) -> Self {
93        Self {
94            manifest_version: MANIFEST_VERSION,
95            name: name.to_owned(),
96            replication_factor: default_replication_factor(),
97            topics: pv_events::EventClass::all()
98                .iter()
99                .map(|c| TopicSpec {
100                    name: c.topic().to_owned(),
101                    partitions: 8,

103                    schema_ref: Some("schemas/ocsf-agent-event.schema.json".to_owned()),
104                })
105                .collect(),
106        }

109    /// Parse from YAML.
110    ///
111    /// Enforces a 256 KiB input cap and rejects any document containing
112    /// YAML anchors (`&`) or aliases (`*`). Together these close the
113    /// billion-laughs attack surface: `serde_yaml` 0.9 expands aliases
114    /// eagerly with no depth or expansion cap, so a maliciously crafted
115    /// manifest of a few KiB would OOM the process at parse time.
116    /// Legitimate bridge manifests never need aliases — reject rather
117    /// than trying to bound expansion, which would require a fork.
118    pub fn from_yaml(yaml: &str) -> Result<Self, ManifestError> {
119        const MAX_YAML_BYTES: usize = 256 * 1024;
120        if yaml.len() > MAX_YAML_BYTES {
121            return Err(ManifestError::Parse(format!(
122                "manifest is {} bytes, exceeds cap of {MAX_YAML_BYTES}",
123                yaml.len()
124            )));
125        }
126        // Cheap syntactic scan for YAML aliases. Legitimate anchors/
127        // aliases would need explicit `&name`/`*name` syntax; a naive
128        // string search catches every case that serde_yaml would
129        // actually expand. False positives on strings that happen to
130        // contain `&` or `*` are reduced (not eliminated — the scan is
131        // not YAML-context-aware, so a quoted string like `"a &name"`
132        // still trips it; failing closed is acceptable for manifests)
133        // by checking that the char precedes a valid anchor-name
134        // character. Note libyaml's anchor-name
135        // class is alphanumeric | `_` | `-` (unsafe-libyaml IS_ALPHA);
136        // the original predicate omitted `-`, so hyphen-led anchors
137        // (`&-a`) expanded while evading this guard — empirically
138        // confirmed against the workspace's serde_yaml and pinned by
139        // the hyphen regression test below.
140        for (marker, kind) in [('&', "anchor"), ('*', "alias")] {
141            let mut chars = yaml.char_indices().peekable();
142            while let Some((_, ch)) = chars.next() {
143                if ch != marker {
144                    continue;
145                }
146                if let Some(&(_, next)) = chars.peek() {
147                    if next.is_ascii_alphanumeric() || next == '_' || next == '-' {
148                        return Err(ManifestError::Parse(format!(
149                            "manifest contains a YAML {kind} ('{marker}<name>'); \
150                             ProofVisor refuses anchor/alias syntax to close the \
151                             billion-laughs attack surface (serde_yaml expands aliases \
152                             with no cap). Rewrite the document without &/* references."
153                        )));
154                    }
155                }
156            }
157        }
158        let m: Self = serde_yaml::from_str(yaml).map_err(|e| ManifestError::Parse(e.to_string()))?;
159        m.validate()?;
160        Ok(m)
161    }

163    /// Serialize to YAML.
164    pub fn to_yaml(&self) -> Result<String, ManifestError> {
165        serde_yaml::to_string(self).map_err(|e| ManifestError::Parse(e.to_string()))
166    }

168    /// Validate structural invariants.
169    pub fn validate(&self) -> Result<(), ManifestError> {
170        if self.manifest_version != MANIFEST_VERSION {
171            return Err(ManifestError::Version(self.manifest_version));
172        }
173        if self.name.is_empty() {
174            return Err(ManifestError::Invalid("name is empty".into()));
175        }
176        // `from_yaml`'s anchor/alias scan (the billion-laughs guard)
177        // rejects any '&'/'*' followed by an anchor-name character
178        // anywhere in the document — including inside quoted strings.
179        // Refuse those characters in every free-text field `to_yaml`
180        // serializes, so `provision()` can never persist a manifest that
181        // `open()` then refuses to re-parse (a name like "R&D-east" used
182        // to provision once and permanently brick the directory as
183        // already-provisioned-but-unopenable).
184        refuse_yaml_marker_chars("name", &self.name)?;
185        if self.topics.is_empty() {
186            return Err(ManifestError::Invalid("no topics declared".into()));
187        }
188        // Schema parity: the shipped bridge-manifest schema pins
189        // `topics.maxItems: 256`; without the same cap here a manifest
190        // that external schema tooling rejects passed `pvctl
191        // manifest-validate` (split offline verdicts), and the count
192        // was otherwise bounded only indirectly by the YAML byte cap.
193        if self.topics.len() > 256 {
194            return Err(ManifestError::Invalid(format!(
195                "{} topics declared, above the schema maximum of 256",
196                self.topics.len()
197            )));
198        }
199        if !(1..=5).contains(&self.replication_factor) {
200            return Err(ManifestError::Invalid(
201                "replication_factor must be between 1 and 5".to_owned(),
202            ));
203        }
204        let mut names: Vec<&str> = self.topics.iter().map(|t| t.name.as_str()).collect();
205        names.sort_unstable();
206        let before = names.len();
207        names.dedup();
208        if names.len() != before {
209            return Err(ManifestError::Invalid("duplicate topic names".into()));
210        }
211        // The embedded broker uses topic
212        // names verbatim as directory names, so on case-insensitive
213        // filesystems (macOS APFS/HFS+ — the documented dev platform)
214        // two topics differing only by case ("Audit" vs "audit")
215        // resolve to ONE directory and silently interleave segments,
216        // offsets, and idempotency state. Reject case-insensitive
217        // collisions everywhere so a manifest behaves identically on
218        // macOS and Linux (Kafka refuses the same collision).
219        let mut folded: Vec<String> = self.topics.iter().map(|t| t.name.to_ascii_lowercase()).collect();
220        folded.sort_unstable();
221        let before = folded.len();
222        folded.dedup();
223        if folded.len() != before {
224            return Err(ManifestError::Invalid(
225                "topic names collide case-insensitively (unsafe on case-insensitive filesystems)".into(),
226            ));
227        }
228        // The NATS backend maps topics to stream
229        // names via `.`→`_` (`pv_{topic.replace('.', "_")}`), which is
230        // not injective: `agent.x` and `agent_x` collide onto stream
231        // `pv_agent_x`, and the second provision's update_stream
232        // silently REPLACES the first topic's subjects — destroying its
233        // capture. Reject the collision at manifest validation for all
234        // backends so a manifest is portable across them.
235        let mut stream_folded: Vec<String> = self
236            .topics
237            .iter()
238            .map(|t| t.name.to_ascii_lowercase().replace('.', "_"))
239            .collect();
240        stream_folded.sort_unstable();
241        let before = stream_folded.len();
242        stream_folded.dedup();
243        if stream_folded.len() != before {
244            return Err(ManifestError::Invalid(
245                "topic names collide after '.'→'_' folding (ambiguous NATS stream names)".into(),
246            ));
247        }
248        for t in &self.topics {
249            if t.name.is_empty()
250                || !t
251                    .name
252                    .bytes()
253                    .all(|byte| byte.is_ascii_alphanumeric() || matches!(byte, b'.' | b'_' | b'-'))
254                || t.name == "."
255                || t.name == ".."
256            {
257                return Err(ManifestError::Invalid(format!("unsafe topic name {:?}", t.name)));
258            }
259            // Length parity with the schema's `maxLength: 249`. 249 is
260            // Kafka's documented topic-name limit — the tightest real
261            // backend bound, and safely below filesystem NAME_MAX (255)
262            // for the embedded broker's per-topic directory and NATS's
263            // `pv_`-prefixed stream name. Without a Rust-side bound, a
264            // longer name passed `pvctl manifest-validate` while
265            // external schema tooling rejected it (split verdicts) and
266            // only failed at provision time.
267            if t.name.len() > 249 {
268                return Err(ManifestError::Invalid(format!(
269                    "topic name {:?}... is {} bytes, above the 249-byte limit",
270                    t.name.get(..24).unwrap_or(&t.name),
271                    t.name.len()
272                )));
273            }
274            if t.partitions == 0 {
275                return Err(ManifestError::Invalid(format!(
276                    "topic {:?} has 0 partitions",
277                    t.name
278                )));
279            }
280            // Cap partitions and hot_hours. `partitions` is
281            // u32, and without a cap `partitions: 4294967295` would
282            // pass validate() and drive the embedded broker to create
283            // 4 billion partition directories on startup — guaranteed
284            // inode/OOM exhaustion. 1024 matches the practical Kafka
285            // per-topic sanity cap. 87_600 hours is 10 years —
286            // realistic ceiling for a long-lived audit trail.
287            const MAX_PARTITIONS: u32 = 1024;
288            const MAX_HOT_HOURS: u32 = 24 * 365 * 10;
289            /// Schema parity: bridge-manifest.schema.json's `maxLength`
290            /// for `schema_ref` and `cold_uri`.
291            const MAX_REFERENCE_BYTES: usize = 4096;
292            if t.partitions > MAX_PARTITIONS {
293                return Err(ManifestError::Invalid(format!(
294                    "topic {:?} partitions {} exceeds cap of {MAX_PARTITIONS}",
295                    t.name, t.partitions
296                )));
297            }
298            if t.retention.hot_hours == 0 {
299                return Err(ManifestError::Invalid(format!(
300                    "topic {:?} has 0h retention",
301                    t.name
302                )));
303            }
304            if t.retention.hot_hours > MAX_HOT_HOURS {
305                return Err(ManifestError::Invalid(format!(
306                    "topic {:?} hot_hours {} exceeds cap of {MAX_HOT_HOURS} (10 years)",
307                    t.name, t.retention.hot_hours
308                )));
309            }
310            if let Some(reference) = &t.schema_ref {
311                // schema_ref and cold_uri are the remaining serialized
312                // fields that admit '&'/'*' (topic names are already
313                // restricted to [A-Za-z0-9._-]); same round-trip rationale
314                // as the deployment-name check above.
315                refuse_yaml_marker_chars("schema_ref", reference)?;
316                // Schema parity: bridge-manifest.schema.json pins
317                // minLength 1 and maxLength 4096 on schema_ref and
318                // cold_uri — a value passing here while external schema
319                // validation refuses it would split the two verification
320                // paths.
321                if reference.is_empty() {
322                    return Err(ManifestError::Invalid(format!(
323                        "topic {:?} schema_ref is empty (schema requires minLength 1); omit the field instead",
324                        t.name
325                    )));
326                }
327                if reference.len() > MAX_REFERENCE_BYTES {
328                    return Err(ManifestError::Invalid(format!(
329                        "schema_ref exceeds the schema maxLength of {MAX_REFERENCE_BYTES} bytes ({})",
330                        reference.len()
331                    )));
332                }
333                let path = Path::new(reference);
334                if path.is_absolute()
335                    || path
336                        .components()
337                        .any(|component| !matches!(component, std::path::Component::Normal(_)))
338                {
339                    return Err(ManifestError::Invalid(format!("unsafe schema_ref {reference:?}")));
340                }
341            }
342            if let Some(uri) = &t.retention.cold_uri {
343                refuse_yaml_marker_chars("cold_uri", uri)?;
344                // Schema parity (minLength 1) AND retention safety: an
345                // empty cold_uri reaches retention enforcement as
346                // `Path::new("")`, exporting the cold archive relative
347                // to the process CWD before the hot records are deleted
348                // — evidence lands wherever the daemon happened to be
349                // started. Refuse it up front.
350                if uri.is_empty() {
351                    return Err(ManifestError::Invalid(format!(
352                        "topic {:?} cold_uri is empty (schema requires minLength 1); omit the field to disable cold export",
353                        t.name
354                    )));
355                }
356                if uri.len() > MAX_REFERENCE_BYTES {
357                    return Err(ManifestError::Invalid(format!(
358                        "cold_uri exceeds the schema maxLength of {MAX_REFERENCE_BYTES} bytes ({})",
359                        uri.len()
360                    )));
361                }
362                // A local (non-scheme) cold_uri
363                // is used directly as a filesystem root at retention
364                // enforcement time (`Path::new(cold).join(...)` in
365                // embedded.rs). A `..` component would let the cold-
366                // archive tree land outside the intended prefix — a
367                // manifest-supplied path escape (CWE-22 class). The
368                // scheme forms (`s3://`, `gs://`, `file://`) are
369                // handled by the cold-store feature layer and are not
370                // subject to this check. Absolute paths are allowed:
371                // legitimate deployments point `cold_uri` at a
372                // dedicated partition or mount (e.g. `/mnt/cold`).
373                if !uri.contains("://") {
374                    let path = Path::new(uri);
375                    if path
376                        .components()
377                        .any(|component| matches!(component, std::path::Component::ParentDir))
378                    {
379                        return Err(ManifestError::Invalid(format!(
380                            "unsafe cold_uri {uri:?} (must not contain `..` components; \
381                             use a scheme URI or a concrete absolute/relative path with \
382                             no parent-directory traversal)"
383                        )));
384                    }
385                }
386            }
387        }
388        Ok(())
389    }
390}

392/// Reject '&' and '*' in a free-text manifest field. These are the exact
393/// markers `from_yaml`'s anchor/alias scan keys on; allowing them in a
394/// serialized field would let `to_yaml` produce a document the scan
395/// refuses to re-parse. Rejected wholesale (not just marker+name-char)
396/// so the rule stays obvious and robust against scan refinements.
397fn refuse_yaml_marker_chars(field: &str, value: &str) -> Result<(), ManifestError> {
398    if value.contains('&') || value.contains('*') {
399        return Err(ManifestError::Invalid(format!(
400            "{field} {value:?} contains '&' or '*', which the YAML anchor/alias \
401             guard refuses at parse time; remove those characters so the \
402             provisioned manifest can be re-opened"
403        )));
404    }
405    Ok(())
406}

408#[cfg(any(feature = "nats", feature = "kafka"))]
409pub(crate) fn compile_topic_validators(
410    manifest: &BridgeManifest,
411) -> Result<HashMap<String, jsonschema::Validator>, crate::BusError> {
412    let mut validators = HashMap::new();
413    for topic in &manifest.topics {
414        let Some(reference) = &topic.schema_ref else {
415            continue;
416        };
417        let schema = schema_document(reference)?;
418        let validator = jsonschema::options()
419            .should_validate_formats(true)
420            .build(&schema)
421            .map_err(|error| crate::BusError::Backend(format!("invalid schema {reference:?}: {error}")))?;
422        validators.insert(topic.name.clone(), validator);
423    }
424    Ok(validators)
425}

427#[cfg(any(feature = "nats", feature = "kafka"))]
428pub(crate) fn validate_topic_event(
429    validators: &HashMap<String, jsonschema::Validator>,
430    topic: &str,
431    value: &serde_json::Value,
432) -> Result<(), crate::BusError> {
433    let Some(validator) = validators.get(topic) else {
434        return Ok(());
435    };
436    let errors: Vec<String> = validator
437        .iter_errors(value)
438        .take(3)
439        .map(|error| error.to_string())
440        .collect();
441    if errors.is_empty() {
442        Ok(())
443    } else {
444        Err(crate::BusError::Backend(format!(
445            "event rejected by schema for topic {topic:?}: {}",
446            errors.join("; ")
447        )))
448    }
449}

451pub(crate) fn schema_document(reference: &str) -> Result<serde_json::Value, crate::BusError> {
452    if reference == "schemas/ocsf-agent-event.schema.json" {
453        // The embedded copy lives INSIDE
454        // the crate (`crates/pv-bridge/schemas/`) — an `include_str!`
455        // that reaches outside the package root breaks `cargo package`
456        // (and therefore every crates.io publish). A sync test in this
457        // module asserts byte-equality with the workspace-level copy.
458        return serde_json::from_str(include_str!("../schemas/ocsf-agent-event.schema.json"))
459            .map_err(crate::BusError::from);
460    }
461    let direct = PathBuf::from(reference);
462    if direct.exists() {
463        // Cap the read (parity with the schema-read cap in embedded.rs): a hostile
464        // plant of a multi-GiB schema file would OOM the process before
465        // the JSON parser could complain.
466        let bytes = pv_core::fsutil::read_capped(&direct, pv_core::fsutil::MAX_CONTROL_BYTES)?;
467        return serde_json::from_slice(&bytes).map_err(crate::BusError::from);
468    }
469    let workspace = Path::new(env!("CARGO_MANIFEST_DIR"))
470        .join("../..")
471        .join(reference);
472    if workspace.exists() {
473        let bytes = pv_core::fsutil::read_capped(&workspace, pv_core::fsutil::MAX_CONTROL_BYTES)?;
474        serde_json::from_slice(&bytes).map_err(crate::BusError::from)
475    } else {
476        Err(crate::BusError::Backend(format!(
477            "schema reference {reference:?} could not be resolved"
478        )))
479    }
480}
