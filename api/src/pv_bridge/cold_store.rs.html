9#[derive(Clone)]
10struct ColdTarget {

12    prefix: Path,
13}

15pub(crate) struct ColdArchive {
16    targets: HashMap<String, ColdTarget>,
17    executor: crate::bus::ConnectorExecutor,
18    pending_dir: std::path::PathBuf,
19    control_key: parking_lot::RwLock<[u8; 32]>,
20    /// Serializes pending-intent file read/modify/remove between
21    /// publishers (`stage`/`commit`) and the maintenance retry scan, so
22    /// neither side observes (or clobbers) an intent mid-update. Never
23    /// held across a remote put or a broker resolve: those run up to the
24    /// 10 s `ConnectorExecutor` timeout, and the embedded resolver takes
25    /// a partition lock that retention holds while staging (ABBA).
26    intent_lock: parking_lot::Mutex<()>,
27}
28
29/// Grace age below which an offset-None intent is presumed to be a
30/// publisher's in-flight stage → produce → commit window and is skipped
31/// by the maintenance retry scan. Publishers set `stored_at` to "now"
32/// immediately before staging and the `ConnectorExecutor` caps the
33/// produce at 10 s, so 60 s comfortably exceeds the longest possible
34/// stage→commit window. Resolving a younger intent would re-publish the
35/// event mid-produce (duplicate) and make the racing `commit()` fail
36/// with NotFound or a conflicting offset (typically triggering a caller
37/// retry — a third copy). A genuine crash leftover ages past the grace
38/// and is resolved on a later scan.
39const RETRY_GRACE_MS: u64 = 60_000;

66    /// Same as [`Self::from_manifest`] but uses `pending_default` when the
67    /// operator-set `PV_COLD_OUTBOX_DIR` env var is absent (falling back to
68    /// `data/cold-outbox` when the caller passes no default either).
69    /// Callers with a natural data directory (e.g. the embedded
70    /// broker) pass `data_dir.join("cold-outbox")` so two brokers in the
71    /// same process do not cross-consume each other's intents via the
72    /// CWD-relative fallback.
73    pub(crate) fn from_manifest_with_pending_default(
74        manifest: &BridgeManifest,
75        pending_default: Option<std::path::PathBuf>,
76    ) -> Result<Option<Self>, BusError> {
77        let mut targets = HashMap::new();
78        for topic in &manifest.topics {
79            let Some(uri) = topic.retention.cold_uri.as_deref() else {
80                continue;
81            };
82            let url = cold_url(uri)?;
83            let options = aws_env_options(std::env::vars());
84            let (store, prefix) = object_store::parse_url_opts(&url, options).map_err(|error| {
85                // Object_store's
86                // "feature for AmazonS3 not enabled" names an internal
87                // feature the operator cannot act on; map it to the
88                // cargo feature that actually fixes the build.
89                let detail = error.to_string();
90                if detail.contains("not enabled") && url.scheme() == "s3" {
91                    BusError::Backend(format!(
92                        "cold store {uri:?}: this build has no S3 support — rebuild with \
93                         `--features cold-store-aws` (or `full`)"
94                    ))
95                } else {
96                    BusError::Backend(format!("cold store {uri:?}: {detail}"))
97                }
98            })?;
99            targets.insert(
100                topic.name.clone(),
101                ColdTarget {
102                    store: Arc::from(store),
103                    prefix,
104                },
105            );
106        }
107        if targets.is_empty() {
108            return Ok(None);
109        }
110        let pending_dir = std::env::var_os("PV_COLD_OUTBOX_DIR")
111            .map(std::path::PathBuf::from)
112            .or(pending_default)
113            .unwrap_or_else(|| std::path::PathBuf::from("data/cold-outbox"));
114        Ok(Some(Self {
115            targets,
116            executor: crate::bus::ConnectorExecutor::new("proofvisor-cold-store")?,
117            pending_dir,
118            control_key: parking_lot::RwLock::new([0; 32]),
119            intent_lock: parking_lot::Mutex::new(()),
120        }))
121    }

123    pub(crate) fn set_control_key(&self, key: [u8; 32]) -> Result<(), BusError> {
124        // Refuse known-weak keys. The default init at
125        // construction is `[0; 32]` (both weak patterns are already
126        // banned for the primary signing seed;
127        // parity here closes the last legal surface for the
128        // pattern). Silently accepting a weak key would let a
129        // startup-order wiring bug (bus impl forgets to plumb
130        // through set_control_key, or a caller passes an
131        // uninitialized array) produce a cold-outbox whose
132        // authentication tag is forgeable by anyone who guessed
133        // the pattern.
134        if key == [0u8; 32] || key == [0xFFu8; 32] {
135            return Err(BusError::Backend(
136                "cold-outbox control key is all-zero or all-0xFF; refusing (known-weak pattern)".to_owned(),
137            ));
138        }
139        *self.control_key.write() = key;
140        Ok(())
141    }

143    pub(crate) fn put(&self, topic: &str, event: &StoredEvent) -> Result<(), BusError> {
144        if !self.targets.contains_key(topic) {
145            return Ok(());
146        }
147        let event_uid = event
148            .value
149            .get("metadata")
150            .and_then(|metadata| metadata.get("uid"))
151            .and_then(serde_json::Value::as_str)
152            .map(str::to_owned)
153            .unwrap_or_else(|| {
154                pv_core::digest::sha256_hex(
155                    format!("{topic}:{}:{}", event.partition, event.offset).as_bytes(),
156                )
157            });
158        self.stage(topic, event, &event_uid)?;
159        self.commit(topic, &event_uid, event.offset)
160    }

162    /// Records the durable publish intent for `event_uid`. Returns
163    /// `Ok(true)` when a matching intent for this UID already existed —
164    /// i.e. this publish is a retry of an earlier attempt that staged
165    /// (and possibly produced) before completing — and `Ok(false)` for a
166    /// freshly-created intent or a topic with no cold target.
167    pub(crate) fn stage(&self, topic: &str, event: &StoredEvent, event_uid: &str) -> Result<bool, BusError> {
168        if !self.targets.contains_key(topic) {
169            return Ok(false);
170        }
171        let pending = PendingColdEvent {
172            topic: topic.to_owned(),
173            event_uid: event_uid.to_owned(),
174            partition: event.partition,
175            key: event.key.clone(),
176            value: event.value.clone(),
177            stored_at: event.stored_at,
178            offset: None,
179        };
180        let pending_path = self.pending_path(topic, event_uid);
181        // Existence-check + persist is a read-modify-write on the intent
182        // file; hold the intent lock so the retry scan cannot interleave.
183        let _guard = self.intent_lock.lock();
184        if pending_path.exists() {
185            let existing = read_pending(&pending_path, &self.control_key.read())?;
186            validate_same_intent(&existing, &pending)?;
187            return Ok(true);
188        }
189        persist_pending(&pending_path, &pending, &self.control_key.read())?;
190        Ok(false)
191    }

193    pub(crate) fn commit(&self, topic: &str, event_uid: &str, offset: u64) -> Result<(), BusError> {
194        if !self.targets.contains_key(topic) {
195            return Ok(());
196        }
197        let pending_path = self.pending_path(topic, event_uid);
198        // Read-check-persist under the intent lock so the maintenance
199        // retry scan can never observe (or rewrite) the intent mid-update;
200        // the remote put runs outside the lock (it is idempotent — a
201        // concurrent retry of the same resolved intent writes identical
202        // bytes and both removals tolerate NotFound).
203        let pending = {
204            let _guard = self.intent_lock.lock();
205            let mut pending = match read_pending(&pending_path, &self.control_key.read()) {
206                Ok(pending) => pending,
207                // A racing maintenance pass may have resolved, exported, and
208                // removed this intent while the publisher sat between
209                // `stage` and `commit` (stalled thread or wall-clock jump
210                // past RETRY_GRACE_MS). The export already happened, so the
211                // publish as a whole succeeded — mirror the NotFound
212                // tolerance of `retry_pending_with` instead of failing a
213                // fully-successful publish.
214                Err(BusError::Io(error)) if error.kind() == std::io::ErrorKind::NotFound => {
215                    return Ok(());
216                }
217                Err(error) => return Err(error),
218            };
219            if pending.topic != topic || pending.event_uid != event_uid {
220                return Err(BusError::Backend(
221                    "cold intent does not match broker acknowledgment".to_owned(),
222                ));
223            }
224            if pending.offset.is_some_and(|existing| existing != offset) {
225                return Err(BusError::Backend(
226                    "cold intent received conflicting broker offsets".to_owned(),
227                ));
228            }
229            pending.offset = Some(offset);
230            persist_pending(&pending_path, &pending, &self.control_key.read())?;
231            pending
232        };
233        match self.put_remote(&pending) {
234            Ok(()) => remove_pending(&pending_path),
235            Err(error) => {
236                tracing::warn!(
237                    %error,
238                    topic,
239                    event_uid,
240                    offset,
241                    "cold export intent is durable and queued for retry"
242                );
243                Ok(())
244            }
245        }
246    }

248    pub(crate) fn retry_pending_with<F>(&self, mut resolve: F) -> Result<u64, BusError>
249    where
250        F: FnMut(&PendingColdEvent) -> Result<crate::PublishAck, BusError>,
251    {
252        let entries = match std::fs::read_dir(&self.pending_dir) {
253            Ok(entries) => entries,
254            Err(error) if error.kind() == std::io::ErrorKind::NotFound => return Ok(0),
255            Err(error) => return Err(BusError::Io(error)),
256        };
257        let mut completed = 0u64;
258        let mut first_error: Option<BusError> = None;
259        for entry in entries {
260            // A single unreadable dirent (broken symlink appearing
261            // mid-scan, racing FS cleanup) must not abort the whole pass —
262            // that would head-of-line-block cold export for every healthy
263            // intent, the exact failure mode the per-intent isolation
264            // below exists to prevent. Skip it, surface the first error.
265            let path = match entry {
266                Ok(entry) => entry.path(),
267                Err(error) => {
268                    tracing::warn!(%error, "cold outbox directory entry unreadable; skipped this pass");
269                    if first_error.is_none() {
270                        first_error = Some(BusError::Io(error));
271                    }
272                    continue;
273                }
274            };
275            if path.extension().and_then(std::ffi::OsStr::to_str) != Some("json") {
276                continue;
277            }
278            // Isolate per-intent failures so
279            // ONE poisoned intent does not head-of-line-block cold export
280            // for every other durable intent — but WITHOUT the quarantine
281            // rename the first revision used. Renaming to `.corrupt-*`
282            // destroyed the durability contract for transient classes (a
283            // plain network failure during an outage removed the intent
284            // from the retry queue forever — the only remaining copy of
285            // the event once retention rewrote the hot segment) and
286            // silenced the loud-refusal contract for integrity classes
287            // (MAC tamper, content conflict) that the outbox tests pin.
288            // Instead: keep scanning past the failure so healthy intents
289            // still export this pass, leave the failed intent in place
290            // for the next pass, and surface the first error to the
291            // maintenance caller so `pv_bridge_maintenance_errors_total`
292            // fires.
293            match self.retry_pending_one(&path, &mut resolve) {
294                Ok(true) => completed = completed.saturating_add(1),
295                Ok(false) => {}
296                Err(error) => {
297                    tracing::warn!(
298                        %error,
299                        path = %pv_core::fsutil::basename(&path),
300                        "cold outbox intent failed; left in place for the next retry pass"
301                    );
302                    if first_error.is_none() {
303                        first_error = Some(error);
304                    }
305                }
306            }
307        }
308        match first_error {
309            Some(error) => Err(error),
310            None => Ok(completed),
311        }
312    }

314    /// Returns `Ok(true)` when the intent was exported and removed,
315    /// `Ok(false)` for the benign skip cases (racing commit already
316    /// exported it; young in-flight offset-None intent inside the
317    /// publisher's grace window) so the caller does not count a skip as
318    /// a completed export in the maintenance action metric.
319    fn retry_pending_one<F>(&self, path: &std::path::Path, resolve: &mut F) -> Result<bool, BusError>
320    where
321        F: FnMut(&PendingColdEvent) -> Result<crate::PublishAck, BusError>,
322    {
323        // Read under the intent lock so a concurrent stage/commit
324        // rewrite is never observed mid-swap; NotFound means a racing
325        // commit already exported and removed the intent.
326        let mut pending = {
327            let _guard = self.intent_lock.lock();
328            match read_pending(path, &self.control_key.read()) {
329                Ok(pending) => pending,
330                Err(BusError::Io(error)) if error.kind() == std::io::ErrorKind::NotFound => {
331                    return Ok(false);
332                }
333                Err(error) => return Err(error),
334            }
335        };
336        if path != self.pending_path(&pending.topic, &pending.event_uid) {
337            return Err(BusError::Backend(
338                "cold outbox path does not match its payload".to_owned(),
339            ));
340        }
341        if pending.offset.is_none() {
342            // In-flight publisher window: skip young offset-None
343            // intents entirely (see RETRY_GRACE_MS). `stored_at` is
344            // MAC-authenticated, so a filesystem tamperer cannot park
345            // an intent in the grace window forever.
346            if pv_core::time::now_ms().saturating_sub(pending.stored_at) < RETRY_GRACE_MS {
347                return Ok(false);
348            }
349            let ack = resolve(&pending)?;
350            if ack.topic != pending.topic || ack.partition != pending.partition {
351                return Err(BusError::Backend(
352                    "broker resolver returned an acknowledgment for another cold intent".to_owned(),
353                ));
354            }
355            // Re-read after the (unlocked) resolve: a racing commit may
356            // have exported+removed the intent (skip it) or persisted
357            // the broker-acked offset (which wins over ours).
358            let _guard = self.intent_lock.lock();
359            match read_pending(path, &self.control_key.read()) {
360                Ok(current) if current.offset.is_some() => pending = current,
361                Ok(_) => {
362                    pending.offset = Some(ack.offset);
363                    persist_pending(path, &pending, &self.control_key.read())?;
364                }
365                Err(BusError::Io(error)) if error.kind() == std::io::ErrorKind::NotFound => {
366                    return Ok(false);
367                }
368                Err(error) => return Err(error),
369            }
370        }
371        self.put_remote(&pending)?;
372        remove_pending(path)?;
373        Ok(true)
374    }

376    fn put_remote(&self, pending: &PendingColdEvent) -> Result<(), BusError> {
377        let Some(target) = self.targets.get(&pending.topic).cloned() else {
378            return Err(BusError::Backend(format!(
379                "cold target for topic {:?} is unavailable",
380                pending.topic
381            )));
382        };
383        let location = target
384            .prefix
385            .clone()
386            .join(pending.topic.as_str())
387            .join(format!("p{}", pending.partition))
388            .join(format!(
389                "{:020}.json",
390                pending
391                    .offset
392                    .ok_or_else(|| BusError::Backend("cold intent has no broker offset".to_owned()))?
393            ));
394        let location_display = location.to_string();
395        let payload = serde_json::to_vec(&StoredEvent {
396            partition: pending.partition,
397            offset: pending.offset.unwrap_or(0),
398            key: pending.key.clone(),
399            value: pending.value.clone(),
400            stored_at: pending.stored_at,
401        })?;
402        self.executor
403            .run(move || async move {
404                match target
405                    .store
406                    .put_opts(
407                        &location,
408                        payload.clone().into(),
409                        object_store::PutMode::Create.into(),
410                    )
411                    .await
412                {
413                    Ok(_) => Ok(()),
414                    Err(object_store::Error::AlreadyExists { .. }) => {
415                        let existing = target
416                            .store
417                            .get(&location)
418                            .await
419                            .map_err(|error| error.to_string())?
420                            .bytes()
421                            .await
422                            .map_err(|error| error.to_string())?;
423                        if existing.as_ref() == payload.as_slice() {
424                            Ok(())
425                        } else {
426                            Err("existing cold object has different content".to_owned())
427                        }
428                    }
429                    Err(error) => Err(error.to_string()),
430                }
431            })?
432            .map_err(|error| BusError::Backend(format!("cold export {location_display}: {error}")))
433    }

443fn persist_pending(
444    path: &std::path::Path,
445    pending: &PendingColdEvent,
446    control_key: &[u8; 32],
447) -> Result<(), BusError> {
448    use std::io::Write as _;
449    let parent = path
450        .parent()
451        .ok_or_else(|| BusError::Backend("cold outbox has no parent".to_owned()))?;
452    // Sync newly created ancestors too: the intent this file records is
453    // the only durable trace of a staged cold export, and an unsynced
454    // ancestor dirent can drop the whole outbox subtree on power loss.
455    pv_core::fsutil::create_dir_all_synced(parent)?;
456    let temporary = path.with_extension(format!("{}.tmp", pv_core::new_event_uid()));
457    // Arm the RAII guard immediately after choosing the
458    // temp path so that any Err on mac/serialize/write/sync/rename
459    // cleans up the tmp instead of leaving an orphan. Repeated
460    // ENOSPC/EIO on a bad-disk day would otherwise unboundedly
461    // consume inodes (UUIDv7-suffixed .tmp names never collide).
462    let mut guard = pv_core::fsutil::TempPathGuard::new(temporary.clone());
463    // Explicit 0o600 mode + create_new — cold-outbox pending intent
464    // files hold the full sealed event value (session-derived data,
465    // same class as `write_cold_event_once` and the hot segment
466    // JSONL that were hardened elsewhere in this crate). The
467    // previous `File::create` recipe landed at 0o644 under the
468    // default umask, and `rename` propagated that mode into the
469    // durable pending file at `{pending_dir}/{sha256(topic:uid)}.json`.
470    // `create_new(true)` closes the (formally-possible) symlink-plant
471    // window at the UUIDv7-suffixed tmp path.
472    let mut tmp_options = std::fs::OpenOptions::new();
473    tmp_options.write(true).create_new(true);
474    #[cfg(unix)]
475    {
476        use std::os::unix::fs::OpenOptionsExt as _;
477        tmp_options.mode(0o600);
478    }
479    let mut file = tmp_options.open(&temporary)?;
480    let mac = pending_mac(control_key, pending)?;
481    file.write_all(&serde_json::to_vec(&PendingEnvelope {
482        payload: pending.clone(),
483        mac,
484    })?)?;
485    file.sync_all()?;
486    std::fs::rename(&temporary, path)?;
487    guard.disarm();
488    pv_core::fsutil::sync_directory(parent)?;
489    Ok(())
490}

492/// Pending intents embed one full event value; the harness caps request
493/// bodies at 16 MiB, so no legitimate intent approaches this bound.
494const MAX_PENDING_INTENT_BYTES: u64 = 16 * 1024 * 1024;
495
496fn read_pending(path: &std::path::Path, control_key: &[u8; 32]) -> Result<PendingColdEvent, BusError> {
497    // Cap the read BEFORE the MAC is verified: every other control-plane
498    // read in this crate is size-capped because filesystem tampering is
499    // in-model here, but the MAC can only reject a planted file after it
500    // has been fully buffered — a multi-GiB plant would OOM the broker
501    // on the very next stage/commit/maintenance pass.
502    let bytes = pv_core::fsutil::read_capped(path, MAX_PENDING_INTENT_BYTES)?;
503    let envelope: PendingEnvelope = serde_json::from_slice(&bytes)?;
504    verify_pending_mac(control_key, &envelope.payload, &envelope.mac)?;
505    Ok(envelope.payload)
506}
507
508fn pending_mac(control_key: &[u8; 32], pending: &PendingColdEvent) -> Result<String, BusError> {
509    // Refuse to sign under a known-weak key. `set_control_key`
510    // blocks *installing* [0; 32] / [0xFF; 32],
511    // but the archive is *constructed* with [0; 32] and a bus impl that
512    // publishes before an operator installs a real key (or that forgets
513    // set_control_key entirely) would sign under the weak default. Fail
514    // closed at the sign site so the default-init window can never
515    // produce a forgeable envelope.
516    if control_key == &[0u8; 32] || control_key == &[0xFFu8; 32] {
517        return Err(BusError::Backend(
518            "cold-outbox control key is uninitialized/known-weak; refusing to sign".to_owned(),
519        ));
520    }
521    use hmac::{Hmac, KeyInit as _, Mac as _};
522    use sha2::Sha256;
523    let value = serde_json::to_value(pending)?;
524    let canonical =
525        pv_receipts::canonicalize(&value).map_err(|error| BusError::Backend(error.to_string()))?;
526    let mut mac =
527        Hmac::<Sha256>::new_from_slice(control_key).map_err(|error| BusError::Backend(error.to_string()))?;
528    // Wire format: outbox entries on disk carry this label; never rename.
529    mac.update(b"agentvisor-cold-outbox-v1\0");
530    mac.update(canonical.as_bytes());
531    Ok(hex::encode(mac.finalize().into_bytes()))
532}

534/// Verify the MAC in constant time (`hmac::Mac::verify_slice`), which
535/// eliminates the timing side-channel that a variable-time `String != String`
536/// comparison would expose (CWE-208 class).
537fn verify_pending_mac(
538    control_key: &[u8; 32],
539    pending: &PendingColdEvent,
540    presented_hex: &str,
541) -> Result<(), BusError> {
542    // Refuse to verify under a known-weak key. Mirrors the
543    // sign-time check in `pending_mac`. If the archive is still holding
544    // its default [0; 32] key when a stale envelope is read back on
545    // startup, we do NOT want to accept it as authentic — a
546    // filesystem-tamper attacker knows the weak key too.
547    if control_key == &[0u8; 32] || control_key == &[0xFFu8; 32] {
548        return Err(BusError::Backend("cold outbox authentication failed".to_owned()));
549    }
550    use hmac::{Hmac, KeyInit as _, Mac as _};
551    use sha2::Sha256;
552    let value = serde_json::to_value(pending)?;
553    let canonical =
554        pv_receipts::canonicalize(&value).map_err(|error| BusError::Backend(error.to_string()))?;
555    let mut mac =
556        Hmac::<Sha256>::new_from_slice(control_key).map_err(|error| BusError::Backend(error.to_string()))?;
557    mac.update(b"agentvisor-cold-outbox-v1\0");
558    mac.update(canonical.as_bytes());
559    // Same 128-char cap as journal.rs — HMAC-SHA256 is
560    // 64 hex chars; refuse a giant `presented_hex` so a fs-tamper
561    // attacker cannot force a huge hex::decode allocation before
562    // verify_slice fails.
563    if presented_hex.len() > 128 {
564        return Err(BusError::Backend("cold outbox authentication failed".to_owned()));
565    }
566    let presented_bytes = hex::decode(presented_hex)

572fn validate_same_intent(existing: &PendingColdEvent, pending: &PendingColdEvent) -> Result<(), BusError> {
573    if existing.topic != pending.topic
574        || existing.event_uid != pending.event_uid
575        || existing.partition != pending.partition
576        || existing.key != pending.key
577        || existing.value != pending.value
578    {
579        return Err(BusError::Backend(
580            "cold outbox UID is bound to a different event".to_owned(),
581        ));
582    }
583    Ok(())
584}

586fn remove_pending(path: &std::path::Path) -> Result<(), BusError> {
587    let parent = path
588        .parent()
589        .ok_or_else(|| BusError::Backend("cold outbox has no parent".to_owned()))?;
590    match std::fs::remove_file(path) {
591        Ok(()) => pv_core::fsutil::sync_directory(parent).map_err(BusError::Io),
592        Err(error) if error.kind() == std::io::ErrorKind::NotFound => Ok(()),
593        Err(error) => Err(BusError::Io(error)),
594    }
595}

597/// Honor only `AWS_*`-prefixed env vars (credentials plus AWS-scoped
598/// configuration like `AWS_ENDPOINT`), lowercased for
599/// `parse_url_opts` (its config-key parser accepts lowercase names only).
600/// Passing the whole environment would let generic variable names —
601/// `ENDPOINT`, `REGION`, `TIMEOUT`, `TOKEN`, `PROXY_URL` — silently
602/// reconfigure the cold-store client; standard AWS tooling scopes
603/// credential discovery to `AWS_*`.
604fn aws_env_options(vars: impl Iterator<Item = (String, String)>) -> Vec<(String, String)> {
605    vars.filter_map(|(key, value)| key.starts_with("AWS_").then(|| (key.to_ascii_lowercase(), value)))
606        .collect()
607}
608
609fn cold_url(value: &str) -> Result<url::Url, BusError> {
610    if value.contains("://") {
611        return url::Url::parse(value)
612            .map_err(|error| BusError::Backend(format!("invalid cold_uri {value:?}: {error}")));
613    }
614    pv_core::fsutil::create_dir_all_synced(std::path::Path::new(value))?;
615    // `Url::from_directory_path` rejects relative paths outright, so a
616    // portable manifest entry like `cold_uri: "data/cold"` used to create
617    // the directory and then fail provisioning. Canonicalize against the
618    // CWD first (same resolution rule as the cold-outbox default).
619    let absolute = std::fs::canonicalize(value)?;
620    url::Url::from_directory_path(&absolute)
621        .map_err(|()| BusError::Backend(format!("invalid cold directory {value:?}")))
622}

1149    /// Byte-level tamper uniformity: mutating any single nibble of the MAC
1150    /// field must yield the same "authentication failed" error text. A
1151    /// short-circuit comparator would either fail earlier (leaking via
1152    /// timing) or emit a different error for a "wrong-format" MAC vs a
1153    /// "wrong-value" MAC — either would form a CWE-208 oracle.
1154    #[test]
1155    fn cold_outbox_mac_tamper_at_any_nibble_returns_the_same_error() {
1156        let directory = tempfile::tempdir().unwrap();
1157        let outbox = tempfile::tempdir().unwrap();
1158        let uri = url::Url::from_directory_path(directory.path())
1159            .unwrap()
1160            .to_string();
1161        let mut manifest = BridgeManifest::default_for("cold-uniform");
1162        let topic = &mut manifest.topics[0];
1163        topic.retention.cold_uri = Some(uri);
1164        let topic_name = topic.name.clone();
1165        let mut archive = ColdArchive::from_manifest(&manifest).unwrap().unwrap();
1166        archive.pending_dir = outbox.path().to_path_buf();
1167        archive.set_control_key([99; 32]).unwrap();
1168        archive
1169            .stage(
1170                &topic_name,
1171                &StoredEvent {
1172                    partition: 0,
1173                    offset: 0,
1174                    key: "instance".to_owned(),
1175                    value: serde_json::json!({"metadata": {"uid": "cold-uniform-1"}}),
1176                    stored_at: 1,
1177                },
1178                "cold-uniform-1",
1179            )
1180            .unwrap();
1181        let path = std::fs::read_dir(outbox.path())
1182            .unwrap()
1183            .next()
1184            .unwrap()
1185            .unwrap()
1186            .path();
1187        let envelope: serde_json::Value = serde_json::from_slice(&std::fs::read(&path).unwrap()).unwrap();
1188        let mac_hex = envelope["mac"].as_str().unwrap().to_owned();
1189
1190        let mut errors = std::collections::HashSet::new();
1191        for i in 0..mac_hex.len() {
1192            let mut bytes: Vec<u8> = mac_hex.as_bytes().to_vec();
1193            bytes[i] = if bytes[i] == b'0' { b'f' } else { b'0' };
1194            let tampered_hex = String::from_utf8(bytes).unwrap();
1195            let mut tampered_envelope = envelope.clone();
1196            tampered_envelope["mac"] = serde_json::json!(tampered_hex);
1197            std::fs::write(&path, serde_json::to_vec(&tampered_envelope).unwrap()).unwrap();
1198            let err = archive
1199                .retry_pending_with(|_| Err(BusError::Backend("unexpected".to_owned())))
1200                .unwrap_err()
1201                .to_string();
1202            errors.insert(err);
1203        }
1204        assert_eq!(
1205            errors.len(),
1206            1,
1207            "cold-outbox MAC verification must return a single error text regardless of which \
1208             nibble failed; got {} distinct texts (i.e. a timing/oracle side channel): {errors:?}",
1209            errors.len()
1210        );
1211    }

1244    /// Sign-time refusal of known-weak keys. The install-time guard
1245    /// only covers `set_control_key`. If a bus impl publishes before
1246    /// (or without) installing a real key, the archive is still holding
1247    /// its default `[0; 32]` — and would previously have signed
1248    /// envelopes under an all-zero MAC key any attacker could forge.
1249    /// `stage()` (which calls `persist_pending` → `pending_mac`) must
1250    /// fail closed in that state.
1251    #[test]
1252    fn stage_refuses_to_sign_under_default_weak_key() {
1253        let directory = tempfile::tempdir().unwrap();
1254        let outbox = tempfile::tempdir().unwrap();
1255        let uri = url::Url::from_directory_path(directory.path())
1256            .unwrap()
1257            .to_string();
1258        let mut manifest = BridgeManifest::default_for("cold-default-weak");
1259        let topic = &mut manifest.topics[0];
1260        topic.retention.cold_uri = Some(uri);
1261        let topic_name = topic.name.clone();
1262        let mut archive = ColdArchive::from_manifest(&manifest).unwrap().unwrap();
1263        archive.pending_dir = outbox.path().to_path_buf();
1264        // NOTE: no set_control_key call — the archive is still holding
1265        // its constructor default of [0; 32].
1266        let event = StoredEvent {
1267            partition: 0,
1268            offset: 0,
1269            key: "instance".to_owned(),
