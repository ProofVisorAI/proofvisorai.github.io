1//! Token revocation list kept in the harness state store (pillar 1).
2//!
3//! With `state_backend = "redis"`, every replica reads the same list, so a
4//! token revoked through one replica is refused by all of them on the next
5//! request. The single-process backend keeps
6//! [`pv_identity::InMemoryRevocationStore`] instead, whose sweep bounds its
7//! memory; see `proofvisord`'s startup wiring.
8
9use pv_identity::{RevocationStore, RevokeOutcome, RevokedBy};
10use pv_state::StateStore;
11use std::sync::Arc;
12
13/// Revocation list backed by a [`StateStore`]. Each revoked JTI is one
14/// counter; the Redis backend expires it 24 h after the write, far past the
15/// 15-minute NHI lifetime, so the list cleans itself up.
16pub struct StateRevocationStore {
17    store: Arc<dyn StateStore>,
18    prefix: String,
19    instance_prefix: String,
20}
21
22impl StateRevocationStore {
23    /// Wrap the harness state store.
24    pub fn new(store: Arc<dyn StateStore>) -> Self {
25        Self {
26            store,
27            prefix: "pv:revoked:".into(),
28            instance_prefix: "pv:revoked-instance:".into(),
29        }
30    }
31
32    /// Isolate a harness-issued token namespace from upstream token IDs.
33    /// Namespace names are supplied by the application, never by token claims.
34    pub fn with_namespace(store: Arc<dyn StateStore>, namespace: &str) -> Self {
35        Self {
36            store,
37            prefix: format!("pv:revoked-{namespace}:"),
38            instance_prefix: format!("pv:revoked-{namespace}-instance:"),
39        }
40    }
41
42    /// The key holds a digest of the JTI, not the JTI itself: the issuer
43    /// controls the claim's length and characters, the key must not.
44    fn key(&self, jti: &str) -> String {
45        format!("{}{}", self.prefix, pv_core::digest::sha256_hex(jti.as_bytes()))
46    }
47
48    fn issued_key(&self, issuer: &str, jti: &str) -> String {
49        // Separate from the global key's 64-digit suffix. Hash each component
50        // independently so no delimiter in either claim can alias another pair.
51        format!(
52            "{}issuer:{}:{}",
53            self.prefix,
54            pv_core::digest::sha256_hex(issuer.as_bytes()),
55            pv_core::digest::sha256_hex(jti.as_bytes())
56        )
57    }
58
59    fn instance_key(&self, instance_uid: &str) -> String {
60        format!(
61            "{}{}",
62            self.instance_prefix,
63            pv_core::digest::sha256_hex(instance_uid.as_bytes())
64        )
65    }
66}

68/// Identity validation runs on tokio workers in several handlers, and a
69/// Redis lookup is a synchronous network round-trip that must never stall
70/// a worker (the rule `prepare_chat_nonblocking` enforces for budgets). On
71/// a multi-thread worker, `block_in_place` first hands the worker's other
72/// tasks to another thread. Anywhere else (the blocking pool, no runtime,
73/// or a current-thread runtime, where `block_in_place` would panic) the
74/// call just runs.
75fn off_the_reactor<T>(call: impl FnOnce() -> T) -> T {
76    match tokio::runtime::Handle::try_current() {
77        Ok(handle) if handle.runtime_flavor() == tokio::runtime::RuntimeFlavor::MultiThread => {
78            tokio::task::block_in_place(call)
79        }
80        _ => call(),
81    }
82}

84impl RevocationStore for StateRevocationStore {
85    fn revoke(&self, jti: &str, expires_at: u64) -> Result<(), String> {
86        self.try_revoke(jti, expires_at).map(|_| ())
87    }
88
89    fn try_revoke(&self, jti: &str, _expires_at: u64) -> Result<RevokeOutcome, String> {
90        off_the_reactor(|| self.store.add(&self.key(jti), 1))
91            .map(|marks| {
92                if marks == 1 {
93                    RevokeOutcome::Revoked
94                } else {
95                    RevokeOutcome::AlreadyRevoked
96                }
97            })
98            .map_err(|error| error.to_string())
99    }
100
101    fn try_revoke_token(&self, issuer: &str, jti: &str, _expires_at: u64) -> Result<RevokeOutcome, String> {
102        off_the_reactor(|| self.store.add(&self.issued_key(issuer, jti), 1))
103            .map(|marks| {
104                if marks == 1 {
105                    RevokeOutcome::Revoked
106                } else {
107                    RevokeOutcome::AlreadyRevoked
108                }
109            })
110            .map_err(|error| error.to_string())
111    }
112
113    fn check_token(&self, issuer: &str, jti: &str, uid: &str, iat: u64) -> Result<Option<RevokedBy>, String> {
114        let marks = off_the_reactor(|| self.store.get(&self.issued_key(issuer, jti)))
115            .map_err(|error| error.to_string())?;
116        if marks > 0 {
117            return Ok(Some(RevokedBy::Jti));
118        }
119        self.check(jti, uid, iat)
120    }
121
122    fn is_revoked(&self, jti: &str) -> Result<bool, String> {
123        off_the_reactor(|| self.store.get(&self.key(jti)))
124            .map(|marks| marks > 0)
125            .map_err(|error| error.to_string())
126    }
127
128    fn revoke_instance(&self, instance_uid: &str, cutoff: u64, _expires_at: u64) -> Result<(), String> {
129        // Encode cutoff+1 so a revocation through epoch zero remains distinct
130        // from StateStore::get's missing-key sentinel of zero.
131        let encoded = cutoff
132            .checked_add(1)
133            .ok_or_else(|| "instance cutoff overflow".to_owned())?;
134        off_the_reactor(|| self.store.fetch_max(&self.instance_key(instance_uid), encoded))
135            .map(|_| ())
136            .map_err(|error| error.to_string())
137    }
138
139    fn check(&self, jti: &str, instance_uid: &str, iat: u64) -> Result<Option<RevokedBy>, String> {
140        if self.is_revoked(jti)? {
141            return Ok(Some(RevokedBy::Jti));
142        }
143        let encoded = off_the_reactor(|| self.store.get(&self.instance_key(instance_uid)))
144            .map_err(|error| error.to_string())?;
145        Ok(encoded
146            .checked_sub(1)
147            .filter(|cutoff| iat <= *cutoff)
148            .map(|cutoff| RevokedBy::Instance { cutoff }))
149    }
150}

152/// Three consecutive dependency failures open the revocation circuit.
153const BREAKER_FAILURE_THRESHOLD: u32 = 3;
154/// An open circuit permits one recovery probe each second.
155const BREAKER_OPEN_TIME: std::time::Duration = std::time::Duration::from_secs(1);
156/// Bound memory independently of the number of token identifiers an issuer creates.
157const LOCAL_REVOKED_MAX: usize = 100_000;
158/// Bound simultaneous synchronous storage calls even before the first failures return.
159const MAX_IN_FLIGHT_READS: usize = 32;
160
161trait RevocationClock: Send + Sync {
162    fn monotonic(&self) -> std::time::Instant;
163    fn epoch_seconds(&self) -> u64;
164}
165
166struct SystemRevocationClock;
167impl RevocationClock for SystemRevocationClock {
168    fn monotonic(&self) -> std::time::Instant {
169        std::time::Instant::now()
170    }
171    fn epoch_seconds(&self) -> u64 {
172        pv_core::time::now_ms() / 1000
173    }
174}

203/// Fail-closed protection for distributed revocation storage. Only confirmed
204/// revocations and locally requested revocations are cached. A successful
205/// "not revoked" answer is never cached, so replicas see later revocations on
206/// the next request. Failed writes still return errors to their callers.
207///
208/// Reads are limited to 32 concurrent dependency calls. After three consecutive
209/// errors, reads fail immediately for one second; then one caller probes
210/// recovery. Writes always reach the backing store, even while the circuit is
211/// open. Positive JTI and instance entries share a strict 100,000-entry bound.
212pub struct GuardedRevocationStore {
213    inner: Arc<dyn RevocationStore>,
214    state: parking_lot::Mutex<GuardState>,
215    metrics: GuardMetrics,
216    namespace: &'static str,
217    clock: Arc<dyn RevocationClock>,
218}
219
220impl GuardedRevocationStore {
221    /// Wrap a Redis-backed store. Namespace is `nhi` or `exchanged`; any other
222    /// caller-supplied name shares the fixed `other` metric label.
223    pub fn new(
224        inner: Arc<dyn RevocationStore>,
225        metrics: &pv_core::metrics::Registry,
226        namespace: &'static str,
227    ) -> Self {
228        Self::with_clock(inner, metrics, namespace, Arc::new(SystemRevocationClock))
229    }
230
231    fn with_clock(
232        inner: Arc<dyn RevocationStore>,
233        registry: &pv_core::metrics::Registry,
234        namespace: &'static str,
235        clock: Arc<dyn RevocationClock>,
236    ) -> Self {
237        let namespace = match namespace {
238            "nhi" => "nhi",
239            "exchanged" => "exchanged",
240            _ => "other",
241        };
242        let lookup = |outcome: &str| {
243            registry.counter(
244                &format!("pv_revocation_lookups_total{{namespace=\"{namespace}\",outcome=\"{outcome}\"}}"),
245                "Revocation dependency lookup outcomes",
246            )
247        };
248        let metrics = GuardMetrics {
249            ok: lookup("ok"),
250            failed: lookup("failed"),
251            short_circuited: lookup("short_circuited"),
252            local_revoked: lookup("local_revoked"),
253            opened: registry.counter(
254                &format!("pv_revocation_breaker_opened_total{{namespace=\"{namespace}\"}}"),
255                "Revocation dependency circuit openings",
256            ),
257            available: registry.gauge(
258                &format!("pv_revocation_list_available{{namespace=\"{namespace}\"}}"),
259                "Whether the revocation dependency circuit is closed",
260            ),
261            entries: registry.gauge(
262                &format!("pv_revocation_local_entries{{namespace=\"{namespace}\"}}"),
263                "Locally cached known revocations",
264            ),
265        };
266        metrics.available.set(1);
267        metrics.entries.set(0);
268        Self {
269            inner,
270            state: parking_lot::Mutex::new(GuardState::default()),
271            metrics,
272            namespace,
273            clock,
274        }
275    }
276
277    fn cache_key(kind: &str, id: &str) -> String {
278        format!("{kind}:{}", pv_core::digest::sha256_hex(id.as_bytes()))
279    }
280
281    fn issued_cache_key(issuer: &str, jti: &str) -> String {
282        format!(
283            "issuer:{}:{}",
284            pv_core::digest::sha256_hex(issuer.as_bytes()),
285            pv_core::digest::sha256_hex(jti.as_bytes())
286        )
287    }
288
289    fn cached(&self, key: &str) -> Option<CachedRevocation> {
290        let now = self.clock.epoch_seconds();
291        let mut state = self.state.lock();
292        match state.revoked.get(key).copied() {
293            Some(entry) if entry.expires_at >= now => Some(entry),
294            Some(_) => {
295                state.revoked.remove(key);
296                self.metrics.entries.set(state.revoked.len() as u64);
297                None
298            }
299            None => None,
300        }
301    }
302
303    fn remember(&self, key: String, cutoff: Option<u64>, expires_at: u64) {
304        if expires_at < self.clock.epoch_seconds() {
305            return;
306        }
307        let mut state = self.state.lock();
308        if let Some(entry) = state.revoked.get_mut(&key) {
309            entry.expires_at = entry.expires_at.max(expires_at);
310            entry.cutoff = entry.cutoff.max(cutoff);
311        } else if state.revoked.len() < LOCAL_REVOKED_MAX {
312            state.revoked.insert(key, CachedRevocation { cutoff, expires_at });
313        }
314        self.metrics.entries.set(state.revoked.len() as u64);
315    }
316
317    fn retention_from_read(&self) -> u64 {
318        self.clock
319            .epoch_seconds()
320            .saturating_add(pv_identity::MAX_TTL_SECS)
321            .saturating_add(2 * pv_identity::validator::DEFAULT_LEEWAY_SECS)
322    }
323
324    fn begin_lookup(&self) -> Result<LookupPermit<'_>, String> {
325        let now = self.clock.monotonic();
326        let mut state = self.state.lock();
327        if state.in_flight >= MAX_IN_FLIGHT_READS
328            || state
329                .open_until
330                .is_some_and(|until| now < until || state.probe.is_some())
331        {
332            self.metrics.short_circuited.inc();
333            return Err("revocation dependency temporarily unavailable".into());
334        }
335        let probe = if state.open_until.is_some() {
336            state.next_probe = state.next_probe.wrapping_add(1);
337            let probe = state.next_probe;
338            state.probe = Some(probe);
339            Some(probe)
340        } else {
341            None
342        };
343        state.in_flight += 1;
344        Ok(LookupPermit {
345            guard: self,
346            probe,
347            finished: false,
348        })
349    }
350
351    fn record_result(&self, success: bool, probe: Option<u64>) {
352        let now = self.clock.monotonic();
353        let mut state = self.state.lock();
354        let owns_probe = probe.is_some() && state.probe == probe;
355        if owns_probe {
356            state.probe = None;
357        }
358        if success {
359            let recovered = state.open_until.take().is_some();
360            state.failures = 0;
361            state.last_success = Some(now);
362            self.metrics.available.set(1);
363            if recovered {
364                tracing::info!(
365                    namespace = self.namespace,
366                    "revocation dependency reachable again"
367                );
368            }
369        } else {
370            state.failures = state.failures.saturating_add(1);
371            if state.failures >= BREAKER_FAILURE_THRESHOLD || owns_probe {
372                if state.open_until.is_none() {
373                    self.metrics.opened.inc();
374                    tracing::error!(
375                        namespace = self.namespace,
376                        "revocation dependency circuit opened after repeated errors"
377                    );
378                }
379                state.open_until = Some(now + BREAKER_OPEN_TIME);
380                self.metrics.available.set(0);
381            }
382        }
383    }
384
385    fn lookup<T>(&self, read: impl FnOnce(&dyn RevocationStore) -> Result<T, String>) -> Result<T, String> {
386        let mut permit = self.begin_lookup()?;
387        let result = read(self.inner.as_ref());
388        if result.is_ok() {
389            self.metrics.ok.inc();
390        } else {
391            self.metrics.failed.inc();
392        }
393        self.record_result(result.is_ok(), permit.probe);
394        permit.finished = true;
395        result
396    }
397}

399struct LookupPermit<'a> {
400    guard: &'a GuardedRevocationStore,
401    probe: Option<u64>,
402    finished: bool,
403}
404
405impl Drop for LookupPermit<'_> {
406    fn drop(&mut self) {
407        if !self.finished {
408            self.guard.metrics.failed.inc();
409            self.guard.record_result(false, self.probe);
410        }
411        let mut state = self.guard.state.lock();
412        state.in_flight = state.in_flight.saturating_sub(1);
413    }
414}

416impl RevocationStore for GuardedRevocationStore {
417    fn health(&self) -> Option<pv_identity::RevocationHealth> {
418        let state = self.state.lock();
419        Some(pv_identity::RevocationHealth {
420            available: state.open_until.is_none(),
421            local_entries: state.revoked.len(),
422            seconds_since_last_success: state
423                .last_success
424                .map(|last| self.clock.monotonic().saturating_duration_since(last).as_secs()),
425        })
426    }
427
428    fn sweep_expired(&self, now_s: u64) {
429        {
430            let mut state = self.state.lock();
431            state.revoked.retain(|_, entry| entry.expires_at >= now_s);
432            self.metrics.entries.set(state.revoked.len() as u64);
433        }
434        self.inner.sweep_expired(now_s);
435    }
436
437    fn revoke(&self, jti: &str, expires_at: u64) -> Result<(), String> {
438        self.try_revoke(jti, expires_at).map(|_| ())
439    }
440
441    fn try_revoke(&self, jti: &str, expires_at: u64) -> Result<RevokeOutcome, String> {
442        self.remember(Self::cache_key("jti", jti), None, expires_at);
443        let result = self.inner.try_revoke(jti, expires_at);
444        self.record_result(result.is_ok(), None);
445        result
446    }
447
448    fn try_revoke_token(&self, issuer: &str, jti: &str, expiry: u64) -> Result<RevokeOutcome, String> {
449        self.remember(Self::issued_cache_key(issuer, jti), None, expiry);
450        let result = self.inner.try_revoke_token(issuer, jti, expiry);
451        self.record_result(result.is_ok(), None);
452        result
453    }
454
455    fn check_token(&self, issuer: &str, jti: &str, uid: &str, iat: u64) -> Result<Option<RevokedBy>, String> {
456        let scoped = Self::issued_cache_key(issuer, jti);
457        let global = Self::cache_key("jti", jti);
458        if self.cached(&scoped).is_some() || self.cached(&global).is_some() {
459            self.metrics.local_revoked.inc();
460            return Ok(Some(RevokedBy::Jti));
461        }
462        let instance = Self::cache_key("instance", uid);
463        if let Some(cutoff) = self
464            .cached(&instance)
465            .and_then(|entry| entry.cutoff)
466            .filter(|cutoff| iat <= *cutoff)
467        {
468            self.metrics.local_revoked.inc();
469            return Ok(Some(RevokedBy::Instance { cutoff }));
470        }
471        let revoked = self.lookup(|inner| inner.check_token(issuer, jti, uid, iat))?;
472        match revoked {
473            // The backing result may be scoped or global. Cache it only for
474            // this issuer, because widening a scoped denial would cross trust
475            // boundaries. Global writes still populate the global cache above.
476            Some(RevokedBy::Jti) => self.remember(scoped, None, self.retention_from_read()),
477            Some(RevokedBy::Instance { cutoff }) => {
478                self.remember(instance, Some(cutoff), self.retention_from_read())
479            }
480            None => {}
481        }
482        Ok(revoked)
483    }
484
485    fn is_revoked(&self, jti: &str) -> Result<bool, String> {
486        let key = Self::cache_key("jti", jti);
487        if self.cached(&key).is_some() {
488            self.metrics.local_revoked.inc();
489            return Ok(true);
490        }
491        let revoked = self.lookup(|inner| inner.is_revoked(jti))?;
492        if revoked {
493            self.remember(key, None, self.retention_from_read());
494        }
495        Ok(revoked)
496    }
497
498    fn revoke_instance(&self, uid: &str, cutoff: u64, expires_at: u64) -> Result<(), String> {
499        self.remember(Self::cache_key("instance", uid), Some(cutoff), expires_at);
500        let result = self.inner.revoke_instance(uid, cutoff, expires_at);
501        self.record_result(result.is_ok(), None);
502        result
503    }
504
505    fn check(&self, jti: &str, uid: &str, iat: u64) -> Result<Option<RevokedBy>, String> {
506        let jti_key = Self::cache_key("jti", jti);
507        if self.cached(&jti_key).is_some() {
508            self.metrics.local_revoked.inc();
509            return Ok(Some(RevokedBy::Jti));
510        }
511        let instance_key = Self::cache_key("instance", uid);
512        if let Some(cutoff) = self
513            .cached(&instance_key)
514            .and_then(|entry| entry.cutoff)
515            .filter(|cutoff| iat <= *cutoff)
516        {
517            self.metrics.local_revoked.inc();
518            return Ok(Some(RevokedBy::Instance { cutoff }));
519        }
520        let revoked = self.lookup(|inner| inner.check(jti, uid, iat))?;
521        match revoked {
522            Some(RevokedBy::Jti) => self.remember(jti_key, None, self.retention_from_read()),
523            Some(RevokedBy::Instance { cutoff }) => {
524                self.remember(instance_key, Some(cutoff), self.retention_from_read())
525            }
526            None => {}
527        }
528        Ok(revoked)
529    }

701    #[derive(Default)]
702    struct TestDependency {
703        store: pv_identity::InMemoryRevocationStore,
704        offline: std::sync::atomic::AtomicBool,
705        panic_next: std::sync::atomic::AtomicBool,
706        reads: std::sync::atomic::AtomicUsize,
707        writes: std::sync::atomic::AtomicUsize,
708        pause_next: parking_lot::Mutex<Option<PausedLookup>>,
709    }
710    impl TestDependency {
711        fn read(&self) -> Result<(), String> {
712            use std::sync::atomic::Ordering::SeqCst;
713            self.reads.fetch_add(1, SeqCst);
714            if let Some((entered, resume)) = self.pause_next.lock().take() {
715                entered.send(()).unwrap();
716                resume.wait();
717            }
718            assert!(!self.panic_next.swap(false, SeqCst), "injected dependency panic");
719            if self.offline.load(SeqCst) {
720                Err("dependency offline".into())
721            } else {
722                Ok(())
723            }
724        }
725        fn write(&self) -> Result<(), String> {
726            use std::sync::atomic::Ordering::SeqCst;
727            self.writes.fetch_add(1, SeqCst);
728            if self.offline.load(SeqCst) {
729                Err("dependency offline".into())
730            } else {
731                Ok(())
732            }
733        }
734    }
735    impl RevocationStore for TestDependency {
736        fn is_revoked(&self, jti: &str) -> Result<bool, String> {
737            self.read()?;
738            self.store.is_revoked(jti)
739        }
740        fn revoke(&self, jti: &str, expiry: u64) -> Result<(), String> {
741            self.try_revoke(jti, expiry).map(|_| ())
742        }
743        fn try_revoke(&self, jti: &str, expiry: u64) -> Result<RevokeOutcome, String> {
744            self.write()?;
745            self.store.try_revoke(jti, expiry)
746        }
747        fn try_revoke_token(&self, issuer: &str, jti: &str, expiry: u64) -> Result<RevokeOutcome, String> {
748            self.write()?;
749            self.store.try_revoke_token(issuer, jti, expiry)
750        }
751        fn check_token(
752            &self,
753            issuer: &str,
754            jti: &str,
755            uid: &str,
756            iat: u64,
757        ) -> Result<Option<RevokedBy>, String> {
758            self.read()?;
759            self.store.check_token(issuer, jti, uid, iat)
760        }
761        fn check(&self, jti: &str, uid: &str, iat: u64) -> Result<Option<RevokedBy>, String> {
762            self.read()?;
763            self.store.check(jti, uid, iat)
764        }
765        fn revoke_instance(&self, uid: &str, cutoff: u64, expiry: u64) -> Result<(), String> {
766            self.write()?;
767            self.store.revoke_instance(uid, cutoff, expiry)
768        }
769    }
