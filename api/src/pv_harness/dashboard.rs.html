1//! Read-only operator dashboard.
2//!
3//! Serves a single-page HTML/CSS/JS application at `/dashboard` and a small
4//! JSON API under `/api/v1/dashboard/*` that lets the page render live
5//! sessions, per-session details, and aggregate stats. The assets are
6//! bundled into the binary via [`include_str!`] so there is no filesystem
7//! dependency at runtime.
8//!
9//! Trust posture: the dashboard exposes the same session metadata that
10//! already lands on disk and in `/metrics`. It is not authenticated by
11//! itself — front it with the same ingress-level control operators use
12//! for `/metrics`. There is no mutating endpoint here; every route is
13//! `GET`.
14

37/// Record latency + outcome for a dashboard request. Kept next to the
38/// handlers so every endpoint funnels through one registration point —
39/// a new endpoint that calls `record()` is instrumented automatically,
40/// and one that doesn't shows up as a missing series in the
41/// dashboard-metrics tests (render() itself never panics on absent keys).
42///
43/// R59 hot-path: metric handles are pre-resolved into
44/// `HotMetrics.dashboard_request_counters` /
45/// `dashboard_request_duration_histograms` — same rationale as R58's
46/// request-metrics middleware pre-resolution. Prior code did 2
47/// per-endpoint-hit `format!` allocs + 4 shared-Registry mutex ops.
48fn record(
49    state: &AppState,
50    endpoint: crate::pipeline::DashboardEndpoint,
51    status: crate::pipeline::DashboardStatus,
52    started: Instant,
53) {
54    // Safe: `DashboardEndpoint::index()` returns `0..3` and
55    // `DashboardStatus::index()` returns `0..2`; `HotMetrics`
56    // allocates arrays of exactly those sizes at boot.
57    #[allow(clippy::indexing_slicing)]
58    {
59        state.hot_metrics.dashboard_request_duration_histograms[endpoint.index()]
60            .observe_us(elapsed_us(started));
61        state.hot_metrics.dashboard_request_counters[endpoint.index()][status.index()].inc();
62    }
63}

98impl SessionSummary {
99    pub(crate) fn from_session(session: &crate::session::Session) -> Self {
100        let latest = session.current_identity();
101        let stop_id = u8::try_from(session.recorded_stop_reason_id()).unwrap_or(0);
102        let stop = pv_events::StopReason::from_id(stop_id);
103        Self {
104            id: session.id.clone(),
105            workflow: session.workflow.as_str(),
106            last_activity_ms: session.last_activity_ms.load(Ordering::Acquire),
107            open: !session.is_closed(),
108            closed: session.closed.load(Ordering::Acquire) != 0,
109            artifact_committed: session.artifact_committed_flag(),
110            close_complete: session.close_complete_flag(),
111            capture_failed: session.capture_failed(),
112            active_streams: session.active_streams_count(),
113            pending_jobs: session.pending_jobs_count(),
114            stop_reason_id: stop_id,
115            stop_reason: stop.caption(),
116            tool_calls: session.totals.tool_calls.load(Ordering::Acquire),
117            tool_allowed: session.totals.tool_allowed.load(Ordering::Acquire),
118            tool_blocked: session.totals.tool_blocked.load(Ordering::Acquire),
119            prompt_tokens: session.totals.prompt_tokens.load(Ordering::Acquire),
120            completion_tokens: session.totals.completion_tokens.load(Ordering::Acquire),
121            cached_tokens: session.totals.cached_tokens.load(Ordering::Acquire),
122            cost_usd_micros: session.totals.cost_usd_micros.load(Ordering::Acquire),
123            identity: Identity {
124                instance_uid: latest.instance_uid,
125                charter: latest.charter.name.clone(),
126                version: latest.version,
127                ttl_remaining_s: latest.ttl_remaining_s,
128            },
129            has_receipt: session.receipt.lock().is_some(),
130        }
131    }
132}

167fn static_asset(body: &'static str, content_type: &'static str) -> Response {
168    let mut response = Response::new(axum::body::Body::from(body));
169    response
170        .headers_mut()
171        .insert(header::CONTENT_TYPE, HeaderValue::from_static(content_type));
172    response.headers_mut().insert(
173        header::CACHE_CONTROL,
174        HeaderValue::from_static("no-store, max-age=0"),
175    );
176    response.headers_mut().insert(
177        HeaderName::from_static("x-content-type-options"),
178        HeaderValue::from_static("nosniff"),
179    );
180    // defense-in-depth against a future XSS regression
181    // in `highlightJson` or any innerHTML-writing dashboard code.
182    // The dashboard is a same-origin single-page app that only
183    // needs its own JS/CSS/img and its own /api/v1/dashboard/*
184    // fetches; a strict CSP shrinks the blast radius to zero even
185    // if attacker-controlled text ever reaches innerHTML.
186    // `frame-ancestors 'none'` + `X-Frame-Options: DENY` block
187    // clickjacking / iframe embed for the day the dashboard grows
188    // an identity gate. `Referrer-Policy: no-referrer` prevents
189    // session ids from leaking through an outbound link.
190    response.headers_mut().insert(
191        header::CONTENT_SECURITY_POLICY,
192        HeaderValue::from_static(
193            "default-src 'none'; script-src 'self'; style-src 'self'; \
194             img-src 'self' data:; font-src 'self'; \
195             connect-src 'self'; frame-ancestors 'none'; base-uri 'none'; \
196             form-action 'none'",
197        ),
198    );
199    response.headers_mut().insert(
200        HeaderName::from_static("x-frame-options"),
201        HeaderValue::from_static("DENY"),
202    );
203    response
204        .headers_mut()
205        .insert(header::REFERRER_POLICY, HeaderValue::from_static("no-referrer"));
206    response
207}

209/// Wrap dashboard JSON responses with headers that
210/// forbid any intermediary shared cache from re-serving one caller's
211/// session detail to another. The static HTML/CSS/JS assets already
212/// set these; the JSON handlers didn't. Also sets `Vary:
213/// Authorization` for the day the dashboard gets an identity gate.
214fn no_store_json_response(value: impl serde::Serialize) -> Response {
215    let mut response = Json(value).into_response();
216    response.headers_mut().insert(
217        header::CACHE_CONTROL,
218        HeaderValue::from_static("no-store, max-age=0"),
219    );
220    response
221        .headers_mut()
222        .insert(header::PRAGMA, HeaderValue::from_static("no-cache"));
223    response
224        .headers_mut()
225        .insert(header::VARY, HeaderValue::from_static("Authorization"));
226    response.headers_mut().insert(
227        HeaderName::from_static("x-content-type-options"),
228        HeaderValue::from_static("nosniff"),
229    );
230    // Mirror the same anti-XSS / anti-clickjacking /
231    // anti-referrer-leak headers on JSON responses. A future
232    // dashboard client that navigates directly to
233    // /api/v1/dashboard/sessions/{id} in a top-level window must
234    // not become a referrer source for session-id leakage, and
235    // must not be framable.
236    response.headers_mut().insert(
237        header::CONTENT_SECURITY_POLICY,
238        HeaderValue::from_static("default-src 'none'; frame-ancestors 'none'; base-uri 'none'"),
239    );
240    response.headers_mut().insert(
241        HeaderName::from_static("x-frame-options"),
242        HeaderValue::from_static("DENY"),
243    );
244    response
245        .headers_mut()
246        .insert(header::REFERRER_POLICY, HeaderValue::from_static("no-referrer"));
247    response
248}

250/// GET /api/v1/dashboard/sessions?limit=&status=
251pub async fn list_sessions(
252    State(state): State<AppState>,
253    axum::extract::Query(query): axum::extract::Query<ListQuery>,
254) -> Response {
255    let started = Instant::now();
256    let limit = query.limit.unwrap_or(DEFAULT_LIST_LIMIT).min(MAX_LIST_LIMIT);
257    let snapshot = state.sessions.open_sessions_including_closed();
258    let total_before_truncate = snapshot.len();
259    let mut summaries: Vec<SessionSummary> = snapshot
260        .iter()
261        .filter(|session| match query.status.as_deref() {
262            Some("open") => !session.is_closed(),
263            Some("closed") => session.is_closed(),
264            Some("failed") => session.capture_failed(),
265            _ => true,
266        })
267        .filter(|session| match query.workflow.as_deref() {
268            Some("signed") => session.workflow == Workflow::Signed,
269            Some("unsigned") => session.workflow == Workflow::Unsigned,
270            _ => true,
271        })
272        .map(|session| SessionSummary::from_session(session))
273        .collect();
274    let matched = summaries.len();
275    summaries.sort_by_key(|s| std::cmp::Reverse(s.last_activity_ms));
276    summaries.truncate(limit);
277    let response = no_store_json_response(json!({
278        "sessions": summaries,
279        "generated_at_ms": pv_core::time::now_ms(),
280        // Total number of sessions currently in the registry (before any
281        // filter is applied). This is what /stats also sees.
282        "total_before_truncate": total_before_truncate,
283        // Number of sessions that matched the requested filter (before the
284        // `limit` cap). Useful for pagination hints and diagnostics.
285        "matched": matched,
286    }));
287    record(
288        &state,
289        crate::pipeline::DashboardEndpoint::List,
290        crate::pipeline::DashboardStatus::Ok,
291        started,
292    );
293    response
294}

296/// GET /api/v1/dashboard/sessions/{id}
297pub async fn session_detail(State(state): State<AppState>, Path(id): Path<String>) -> Response {
298    let started = Instant::now();
299    let Some(session) = state.sessions.get(&id) else {
300        record(
301            &state,
302            crate::pipeline::DashboardEndpoint::Detail,
303            crate::pipeline::DashboardStatus::NotFound,
304            started,
305        );
306        let mut response = no_store_json_response(json!({"error": "session not found in registry"}));
307        *response.status_mut() = StatusCode::NOT_FOUND;
308        return response;
309    };
310    let summary = SessionSummary::from_session(&session);
311    // Clone out under the lock and drop it before serialization so the
312    // reconciler / worker never blocks on a slow serde_json step.
313    let receipt_clone: Option<pv_receipts::Receipt> = session.receipt.lock().clone();
314    // Previously returned `path.display().to_string()`,
315    // which disclosed the absolute spool directory (e.g.
316    // `/var/lib/proofvisor/spool/<uid>.json`) to any unauthenticated
317    // caller of the dashboard. The dashboard's design intent (see
318    // module doc-comment) is "the same session metadata that lands on
319    // disk and in /metrics" — filesystem paths appear in neither.
320    // Disclosing the absolute layout gives a probe attacker a
321    // starting point for later privilege-escalation or path-
322    // traversal attempts. Return only the file name; the UI never
323    // uses the path for anything besides display.
324    let atif_filename: Option<String> = session
325        .atif_path
326        .lock()
327        .as_ref()
328        .and_then(|path| path.file_name().map(|name| name.to_string_lossy().into_owned()));
329    // Chain head/count are exposed as a small provenance stub so the UI can
330    // show "47 events, head=b2b7…". The full chain lives in the journal on
331    // disk; we don't stream it through the dashboard.
332    let (chain_head_hex, chain_count) = {
333        let chain = session.chain.lock();
334        (chain.head_hex(), chain.count())
335    };
336    let receipt: Option<Value> = receipt_clone.and_then(|r| serde_json::to_value(&r).ok());
337    let response = no_store_json_response(json!({
338        "summary": summary,
339        "chain": {
340            "head_hex": chain_head_hex,
341            "count": chain_count,
342        },
343        "receipt": receipt,
344        "atif_filename": atif_filename,
345    }));
346    record(
347        &state,
348        crate::pipeline::DashboardEndpoint::Detail,
349        crate::pipeline::DashboardStatus::Ok,
350        started,
351    );
352    response
353}

355/// GET /api/v1/dashboard/stats
356pub async fn stats(State(state): State<AppState>) -> Response {
357    let started = Instant::now();
358    let sessions = state.sessions.open_sessions_including_closed();
359    let mut totals = Stats {
360        generated_at_ms: pv_core::time::now_ms(),
361        session_count: sessions.len(),
362        open_count: 0,
363        closed_count: 0,
364        capture_failed_count: 0,
365        total_prompt_tokens: 0,
366        total_completion_tokens: 0,
367        total_cached_tokens: 0,
368        total_cost_usd_micros: 0,
369        total_tool_calls: 0,
370        total_tool_allowed: 0,
371        total_tool_blocked: 0,
372        workflow_signed: 0,
373        workflow_unsigned: 0,
374    };
375    for session in &sessions {
376        if session.is_closed() {
377            totals.closed_count += 1;
378        } else {
379            totals.open_count += 1;
380        }
381        if session.capture_failed() {
382            totals.capture_failed_count += 1;
383        }
384        match session.workflow {
385            Workflow::Signed => totals.workflow_signed += 1,
386            Workflow::Unsigned => totals.workflow_unsigned += 1,
387        }
388        totals.total_prompt_tokens = totals
389            .total_prompt_tokens
390            .saturating_add(session.totals.prompt_tokens.load(Ordering::Acquire));
391        totals.total_completion_tokens = totals
392            .total_completion_tokens
393            .saturating_add(session.totals.completion_tokens.load(Ordering::Acquire));
394        totals.total_cached_tokens = totals
395            .total_cached_tokens
396            .saturating_add(session.totals.cached_tokens.load(Ordering::Acquire));
397        totals.total_cost_usd_micros = totals
398            .total_cost_usd_micros
399            .saturating_add(session.totals.cost_usd_micros.load(Ordering::Acquire));
400        totals.total_tool_calls = totals
401            .total_tool_calls
402            .saturating_add(session.totals.tool_calls.load(Ordering::Acquire));
403        totals.total_tool_allowed = totals
404            .total_tool_allowed
405            .saturating_add(session.totals.tool_allowed.load(Ordering::Acquire));
406        totals.total_tool_blocked = totals
407            .total_tool_blocked
408            .saturating_add(session.totals.tool_blocked.load(Ordering::Acquire));
409    }
410    let response = no_store_json_response(totals);
411    record(
412        &state,
413        crate::pipeline::DashboardEndpoint::Stats,
414        crate::pipeline::DashboardStatus::Ok,
415        started,
416    );
417    response
418}

529    async fn get_json(router: &axum::Router, path: &str) -> (StatusCode, Value) {
530        let response = router
531            .clone()
532            .oneshot(
533                Request::builder()
534                    .method(Method::GET)
535                    .uri(path)
536                    .body(Body::empty())
537                    .unwrap(),
538            )

540            .unwrap();
541        let status = response.status();
542        let bytes = to_bytes(response.into_body(), 256 * 1024).await.unwrap();

544        (status, value)
545    }

777    async fn detail_endpoint_returns_receipt_when_present() {
778        // Round-trip a signed receipt into a Session and confirm the
779        // dashboard surfaces it as JSON with the signature intact —
780        // catching any regression that would silently drop cryptographic
781        // material from the operator view.
782        use base64::Engine as _;
783        use pv_receipts::{CostSummary, Receipt, ReceiptBody, ReceiptSubject, Signer as _, ToolCallSummary};
784        let config = HarnessConfig::for_tests("http://127.0.0.1:9", "/tmp", "/tmp");
785        let state = build_state(config);
786        open_a_session(&state, "sess-with-receipt");
787        let session = state.sessions.get("sess-with-receipt").unwrap();
788        let signer = Ed25519Signer::from_seed(&[7; 32]);
789        let key_id = signer.key_id().to_owned();
790        let public_key_b64 = base64::engine::general_purpose::STANDARD.encode(signer.public_key_bytes());
791        let body = ReceiptBody {
792            receipt_version: 1,
793            receipt_id: "test-receipt".to_owned(),
794            session_id: session.id.clone(),
795            issued_at: 1_700_000_000_000,
796            issued_at_iso: "2023-11-14T22:13:20.000Z".to_owned(),
797            ai_agent: session.identity.clone(),
798            subject: ReceiptSubject::EventChain {
799                chain_head: "ab".repeat(32),
800                event_count: 1,
801            },
802            tool_calls: ToolCallSummary::default(),
803            cost: CostSummary::default(),
804            stop_reason_id: 1,
805            stop_reason: "Stop".to_owned(),
806            key_id,
807            public_key_b64,
808        };
809        let receipt = Receipt::issue(body, &signer).expect("sign");
810        session.restore_receipt(receipt.clone());
811        let router = build_router(state.clone());
812        let (status, body) = get_json(&router, "/api/v1/dashboard/sessions/sess-with-receipt").await;

887        // 128 parallel dashboard reads while the hot path is admitting
888        // more sessions — verifies the observability code does not
889        // deadlock, crash on shard lock contention, or panic on a
890        // registry that grows mid-scan.
891        let config = HarnessConfig::for_tests("http://127.0.0.1:9", "/tmp", "/tmp");
892        let state = build_state(config);
893        // Seed enough sessions that iteration touches every DashMap
894        // shard and shows non-trivial payloads to the readers.
895        for i in 0..64 {
896            open_a_session(&state, &format!("seed-{i}"));
897        }
898        let router = build_router(state.clone());
899
900        let mut readers = Vec::new();
901        for _ in 0..128 {
902            let router = router.clone();
903            readers.push(tokio::spawn(async move {
904                let (status, body) = get_json(&router, "/api/v1/dashboard/stats").await;
905                assert_eq!(status, StatusCode::OK);
906                assert!(body["session_count"].as_u64().unwrap() >= 64);
907            }));
908        }
909        // Simultaneously open more sessions on the hot path.
910        let mut writers = Vec::new();
911        for i in 64..96 {
912            let state = state.clone();
913            writers.push(tokio::spawn(async move {
914                let mut headers = axum::http::HeaderMap::new();
915                headers.insert(
916                    SESSION_HEADER,
917                    HeaderValue::from_str(&format!("hot-{i}")).unwrap(),
918                );
919                let payload = serde_json::json!({
920                    "model": "test",
921                    "messages": [{"role":"user","content":"go"}],
922                });
923                state.prepare_chat(&headers, payload, None).unwrap();
924            }));
925        }
926        for reader in readers {
927            reader.await.unwrap();
928        }
929        for writer in writers {
930            writer.await.unwrap();
931        }
