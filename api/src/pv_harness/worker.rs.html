111/// Accounting totals folded from journal records during recovery.
112///
113/// The same
114/// accounting fold used to be implemented THREE times — the live path
115/// in `process_job`, signed recovery in `recover_signed_journals`, and
116/// unsigned consolidation in `consolidate_step_journals`. Any change to
117/// a counted dimension had to be made in three places or a
118/// crash-recovered receipt attested different numbers than a
119/// clean-close receipt for identical traffic — and no test caught it,
120/// because the paths were asserted separately. `ActiveJournalRecord`'s
121/// fields are now the single source: the live path applies them via
122/// [`ActiveJournalRecord::apply_to_totals`], recovery folds them via
123/// [`ActiveJournalRecord::fold_into`], and the proof-harness test
124/// (`unified_fold_matches_live_path`) pins the two applications to
125/// identical results over representative record shapes.
126#[derive(Debug, Default, PartialEq, Eq)]
127pub(crate) struct RecoveredTotals {
128    pub(crate) tool_calls: u64,
129    pub(crate) tool_allowed: u64,
130    pub(crate) tool_blocked: u64,
131    pub(crate) prompt_tokens: u64,
132    pub(crate) completion_tokens: u64,
133    pub(crate) cached_tokens: u64,
134    pub(crate) cost_usd_micros: u64,
135}
136
137impl RecoveredTotals {
138    /// The tool-accounting invariant every fold must check: classified
139    /// outcomes can never exceed observed calls.
140    pub(crate) fn validate_tool_accounting(&self) -> Result<(), String> {
141        if self
142            .tool_allowed
143            .checked_add(self.tool_blocked)
144            .is_none_or(|classified| classified > self.tool_calls)
145        {
146            return Err("journal has inconsistent tool accounting".to_owned());
147        }
148        Ok(())
149    }
150
151    /// Publish the folded totals onto a session (Release ordering, the
152    /// same discipline both recovery paths used inline).
153    pub(crate) fn store_on(&self, totals: &crate::session::Totals) {
154        use std::sync::atomic::Ordering;
155        totals.tool_calls.store(self.tool_calls, Ordering::Release);
156        totals.tool_allowed.store(self.tool_allowed, Ordering::Release);
157        totals.tool_blocked.store(self.tool_blocked, Ordering::Release);
158        totals.prompt_tokens.store(self.prompt_tokens, Ordering::Release);
159        totals
160            .completion_tokens
161            .store(self.completion_tokens, Ordering::Release);
162        totals.cached_tokens.store(self.cached_tokens, Ordering::Release);
163        totals
164            .cost_usd_micros
165            .store(self.cost_usd_micros, Ordering::Release);
166    }
167}

169impl ActiveJournalRecord {
170    /// Fold this record's counted dimensions into recovery totals with
171    /// overflow + JCS-bound checks. THE accounting rule — do not
172    /// re-implement per call site.
173    pub(crate) fn fold_into(&self, totals: &mut RecoveredTotals) -> Result<(), String> {
174        fn checked(current: u64, value: u64, field: &str) -> Result<u64, String> {
175            current
176                .checked_add(value)
177                .filter(|total| *total <= pv_core::error::JCS_SAFE_MAX)
178                .ok_or_else(|| format!("recovered {field} overflow"))
179        }
180        // R64 F3 (landed R69): the cost dimension has a TIGHTER
181        // ceiling than the JCS-safe integer bound. `pv_atif::validate::
182        // MAX_COST_USD` refuses any `total_cost_usd > 9e9` (== 9e15
183        // micros), while `JCS_SAFE_MAX = 2^53 = 9.007e15` micros —
184        // a $7.2M window in which the fold used to say "safe" while
185        // the next strict-write would refuse the trajectory, wedging
186        // recovery. Cap the cost fold at
187        // `pv_core::units::COST_MICROS_MAX` so any cost that folds
188        // cleanly also passes ATIF strict validation.
189        fn checked_cost(current: u64, value: u64) -> Result<u64, String> {
190            current
191                .checked_add(value)
192                .filter(|total| *total <= pv_core::units::COST_MICROS_MAX)
193                .ok_or_else(|| "recovered cost exceeds ATIF strict validator ceiling".to_owned())
194        }
195        totals.tool_calls = checked(totals.tool_calls, self.tool_calls, "tool calls")?;
196        totals.tool_allowed = checked(totals.tool_allowed, self.tool_allowed, "allowed tools")?;
197        totals.tool_blocked = checked(totals.tool_blocked, self.tool_blocked, "blocked tools")?;
198        totals.prompt_tokens = checked(totals.prompt_tokens, self.prompt_tokens, "prompt tokens")?;
199        if self.prompt_token_correction != 0 {
200            totals.prompt_tokens = totals
201                .prompt_tokens
202                .checked_add_signed(self.prompt_token_correction)
203                .filter(|total| *total <= pv_core::error::JCS_SAFE_MAX)
204                .ok_or_else(|| "recovered prompt token correction underflow/overflow".to_owned())?;
205        }
206        totals.completion_tokens = checked(
207            totals.completion_tokens,
208            self.completion_tokens,
209            "completion tokens",
210        )?;
211        totals.cached_tokens = checked(totals.cached_tokens, self.cached_tokens, "cached tokens")?;
212        totals.cost_usd_micros = checked_cost(totals.cost_usd_micros, self.cost_usd_micros)?;
213        Ok(())
214    }
215
216    /// Apply this record's counted dimensions to a live session's
217    /// atomic totals — the live-path twin of [`Self::fold_into`], with
218    /// the same JCS-bound discipline. The record's fields were
219    /// conditionalized ONCE at construction (compression carries
220    /// prompt tokens; terminal responses carry completion/cached/cost;
221    /// tool records carry call outcomes), so applying them
222    /// unconditionally here is exactly the old per-class branching,
223    /// without the second copy of the rules.
224    ///
225    /// All-or-nothing across fields (R64 F1). The previous shape was
226    /// a loop of `checked_atomic_add` + a trailing `fetch_update` —
227    /// each iteration COMMITTED its store before the next validated,
228    /// so a mid-record failure (JCS bound crossed on
229    /// `completion_tokens`; `prompt_token_correction` underflow) left
230    /// `session.totals` half-updated (e.g., `tool_calls` bumped but
231    /// `completion_tokens` not). The subsequent invariant
232    /// `allowed+blocked <= calls` could then fail spuriously, and a
233    /// clean-close receipt over the partial totals attested different
234    /// numbers than a `fold_into`-based recovery replay of the exact
235    /// same on-disk journal — reintroducing the pre-R59 "recovered
236    /// receipt differs from live receipt for identical traffic"
237    /// hazard the unified fold exists to prevent. Snapshot → fold →
238    /// store restores the discipline: fold_into runs on a local
239    /// `RecoveredTotals`, either all fields validate or the Err
240    /// bubbles up with no store; only on success do we publish.
241    ///
242    /// Safe under sequential-per-session semantics: the only
243    /// production caller is `finish_job` (worker.rs
244    /// below), which runs one step at a time per session inside the
245    /// worker loop. Metrics scrape may still observe a mid-publish
246    /// window across the seven per-field stores in `store_on`, but
247    /// that is inherent to per-field atomics and unchanged from the
248    /// prior shape.
249    pub(crate) fn apply_to_totals(&self, totals: &crate::session::Totals) -> Result<(), String> {
250        use std::sync::atomic::Ordering;
251        let mut snapshot = RecoveredTotals {
252            tool_calls: totals.tool_calls.load(Ordering::Acquire),
253            tool_allowed: totals.tool_allowed.load(Ordering::Acquire),
254            tool_blocked: totals.tool_blocked.load(Ordering::Acquire),
255            prompt_tokens: totals.prompt_tokens.load(Ordering::Acquire),
256            completion_tokens: totals.completion_tokens.load(Ordering::Acquire),
257            cached_tokens: totals.cached_tokens.load(Ordering::Acquire),
258            cost_usd_micros: totals.cost_usd_micros.load(Ordering::Acquire),
259        };
260        self.fold_into(&mut snapshot)?;
261        snapshot.store_on(totals);
262        Ok(())
263    }
264}

292/// Which admission stage a drop should be counted against. Each variant
293/// maps to a distinct `pv_events_dropped_total{stage="..."}` counter,
294/// so operators can PromQL-alert on the actual bottleneck class.
295#[derive(Debug, Clone, Copy, PartialEq, Eq)]
296enum DropStage {
297    /// Initial worker-side admission. `try_reserve` / `try_reserve_pair`'s
298    /// worker half. Bumps `pv_events_dropped_total{stage="worker_queue"}`.
299    WorkerQueue,
300    /// Downstream response-capture submission (`ResponsePermit::submit`).
301    /// The permit itself only holds the response-capacity semaphore; the
302    /// mpsc slot is contested at submit time, so exhaustion at THIS
303    /// stage — which is a real observable class of failure — bumps
304    /// `pv_events_dropped_total{stage="response_slot"}`.
305    ResponseSlot,
306}
307
308impl DropStage {
309    fn full_counter_key(self) -> &'static str {
310        match self {
311            Self::WorkerQueue => "pv_events_dropped_total{stage=\"worker_queue\"}",
312            Self::ResponseSlot => "pv_events_dropped_total{stage=\"response_slot\"}",
313        }
314    }
315
316    fn label(self) -> &'static str {
317        match self {
318            Self::WorkerQueue => "worker_queue",
319            Self::ResponseSlot => "response_slot",
320        }
321    }
322}

349/// Reservation for the downstream response-capture worker job. Draws
350/// from a distinct capacity semaphore so operators can tell via
351/// `pv_events_dropped_total{stage="response_slot"}` vs
352/// `stage="worker_queue"` which class of admission is the bottleneck.
353///
354/// Unlike [`WorkerPermit`], the response permit only holds the
355/// capacity semaphore — the mpsc queue slot is re-acquired at submit
356/// time via [`Self::submit`]. This keeps the initial admission cheap
357/// (one semaphore acquire instead of two mpsc reservations that would
358/// otherwise compete for the same shard's slot with the worker
359/// permit) and lets the response job land whenever worker capacity
360/// and the shard have room, which is the common case since response
361/// capture happens
362/// tens-of-seconds after the initial worker job has drained.
363///
364/// Held by the streaming-response wrapper in `routes.rs`
365/// (`AbortFinalizingStream`) for the lifetime of the forwarded response;
366/// drops on stream completion or client abort.
367pub struct ResponsePermit {
368    _capacity_permit: tokio::sync::OwnedSemaphorePermit,
369}
370
371impl ResponsePermit {
372    /// Commit a response-capture job. The capacity slot reserved at
373    /// admission travels INTO the envelope: the submission never
374    /// contends on the main worker semaphore, so a backlog of new
375    /// admissions cannot starve the capture job of an
376    /// already-admitted stream — the guarantee `try_reserve_pair`'s
377    /// acquire-both-or-fail exists to provide. (A previous revision
378    /// dropped this permit and drew a fresh slot from the WORKER
379    /// semaphore, which both forfeited the reservation under load and
380    /// charged the resulting drop to `stage="response_slot"` when the
381    /// exhausted budget was actually worker capacity.) The only
382    /// remaining failure mode is the shard's bounded mpsc queue being
383    /// momentarily full, which returns `SubmitError::Full` and bumps
384    /// `pv_events_dropped_total{stage="response_slot"}`.
385    pub fn submit(self, worker: &WorkerHandle, job: WorkerJob) -> Result<(), SubmitError> {
386        worker.submit_with_permit(job, self._capacity_permit, DropStage::ResponseSlot)
387    }
388}

390/// Fused reservation covering a worker job AND its downstream response
391/// slot. Acquired atomically at request admission: on any failure the
392/// worker slot (if already held) is released via RAII before the error
393/// surfaces, so callers cannot end up with an orphaned half-permit.
394pub struct WorkerAndResponsePermit {
395    /// Permit for the initial worker job (dispatch / quota / receipt-sign).
396    pub worker: WorkerPermit,
397    /// Permit for the downstream response-capture job.
398    pub response: ResponsePermit,
399}
400
401impl WorkerPermit {
402    /// Commit a job into the previously reserved queue slot.
403    pub fn submit(self, job: WorkerJob) {
404        job.session.worker_job_started();
405        self.pending.fetch_add(1, Ordering::AcqRel);
406        let span = worker_span(&job);
407        self.permit.send(Envelope {
408            job,
409            completion: None,
410            span,
411            _capacity_permit: self.capacity_permit,
412        });
413    }
414}

416impl WorkerHandle {
417    /// Reserve queue capacity before committing quota or other state.
418    pub fn try_reserve(&self, session_id: &str) -> Result<WorkerPermit, SubmitError> {
419        let capacity_permit = self.try_capacity()?;
420        let Some(sender) = self.sender_for(session_id).cloned() else {
421            return Err(SubmitError::Closed);
422        };
423        match sender.try_reserve_owned() {
424            Ok(permit) => Ok(WorkerPermit {
425                permit,
426                pending: Arc::clone(&self.pending),
427                capacity_permit,
428            }),
429            Err(mpsc::error::TrySendError::Full(_sender)) => {
430                self.metrics
431                    .counter(
432                        "pv_events_dropped_total{stage=\"worker_queue\"}",
433                        "Worker jobs or response-slot reservations dropped, labeled by admission stage",
434                    )
435                    .inc();
436                Err(SubmitError::Full)
437            }
438            Err(mpsc::error::TrySendError::Closed(_sender)) => {
439                self.metrics
440                    .counter(
441                        "pv_events_dropped_total{stage=\"worker_closed\"}",
442                        "Worker jobs or response-slot reservations dropped, labeled by admission stage",
443                    )
444                    .inc();
445                Err(SubmitError::Closed)
446            }
447        }
448    }
449
450    /// Reserve the downstream response-capture slot. Distinct from
451    /// [`Self::try_reserve`] because response capture is a separate
452    /// stage with its own admission budget; keeping the counters
453    /// separate lets operators see which one ran out. The mpsc queue
454    /// slot is re-acquired at submit time via [`ResponsePermit::submit`].
455    fn try_reserve_response(&self) -> Result<ResponsePermit, SubmitError> {
456        Arc::clone(&self.response_capacity)
457            .try_acquire_owned()
458            .map(|p| ResponsePermit { _capacity_permit: p })
459            .map_err(|_| {
460                self.metrics
461                    .counter(
462                        "pv_events_dropped_total{stage=\"response_slot\"}",
463                        "Worker jobs or response-slot reservations dropped, labeled by admission stage",
464                    )
465                    .inc();
466                SubmitError::Full
467            })
468    }
469
470    /// Atomic acquire-both-or-fail for admission: obtains a worker
471    /// permit AND a response permit for the same request, distinct
472    /// counters on failure. On any error path (worker slot or response
473    /// slot exhaustion, closed channel), a partially-taken worker
474    /// permit drops via RAII before the error surfaces so callers
475    /// never observe an orphaned half-reservation.
476    pub fn try_reserve_pair(&self, session_id: &str) -> Result<WorkerAndResponsePermit, SubmitError> {
477        // Stage 1: worker slot. Bumps stage="worker_queue" on failure.
478        let worker = self.try_reserve(session_id)?;
479        // Stage 2: response slot. Bumps stage="response_slot" on
480        // failure. If this fails, `worker` drops via NLL — its
481        // OwnedSemaphorePermit + mpsc::OwnedPermit release cleanly
482        // and the counters agree with the caller's view (only the
483        // response-slot counter incremented).
484        let response = self.try_reserve_response()?;
485        Ok(WorkerAndResponsePermit { worker, response })
486    }

488    /// Submit without waiting. Used by rejection/failure paths; the hot path
489    /// reserves capacity up front via [`WorkerHandle::try_reserve`] and
490    /// submits through [`WorkerPermit::submit`].
491    pub fn try_submit(&self, job: WorkerJob) -> Result<(), SubmitError> {
492        self.try_submit_labeled(job, DropStage::WorkerQueue)
493    }
494
495    /// Same as [`Self::try_submit`], but the failure counters carry a
496    /// caller-supplied stage label so response-slot exhaustion mid-stream
497    /// can be distinguished from admission-side worker-queue exhaustion.
498    /// See [`ResponsePermit::submit`].
499    fn try_submit_labeled(&self, job: WorkerJob, stage: DropStage) -> Result<(), SubmitError> {
500        let capacity_permit = self.try_capacity_labeled(stage)?;
501        self.submit_with_permit(job, capacity_permit, stage)
502    }
503
504    /// Queue-admission tail shared by [`Self::try_submit_labeled`]
505    /// (which draws a fresh worker-capacity slot) and
506    /// [`ResponsePermit::submit`] (which carries the response-capacity
507    /// slot reserved at admission). The permit — whichever semaphore
508    /// it came from — rides inside the envelope and releases to its
509    /// origin when the worker finishes the job.
510    fn submit_with_permit(
511        &self,
512        job: WorkerJob,
513        capacity_permit: tokio::sync::OwnedSemaphorePermit,
514        stage: DropStage,
515    ) -> Result<(), SubmitError> {
516        let Some(sender) = self.sender_for(&job.session.id).cloned() else {
517            return Err(SubmitError::Closed);
518        };
519        job.session.worker_job_started();
520        self.pending.fetch_add(1, Ordering::AcqRel);
521        let span = worker_span(&job);
522        match sender.try_send(Envelope {
523            job,
524            completion: None,
525            span,
526            _capacity_permit: capacity_permit,
527        }) {
528            Ok(()) => Ok(()),
529            Err(mpsc::error::TrySendError::Full(envelope)) => {
530                let session_id = envelope.job.session.id.clone();
531                envelope.job.session.worker_job_finished();
532                self.worker_job_finished();
533                self.metrics
534                    .counter(
535                        stage.full_counter_key(),
536                        "Worker jobs or response-slot reservations dropped, labeled by admission stage",
537                    )
538                    .inc();
539                tracing::warn!(
540                    session = %session_id,
541                    stage = %stage.label(),
542                    "worker queue full; request must fail closed"
543                );
544                Err(SubmitError::Full)
545            }
546            Err(mpsc::error::TrySendError::Closed(envelope)) => {
547                let session_id = envelope.job.session.id.clone();
548                envelope.job.session.worker_job_finished();
549                self.worker_job_finished();
550                self.metrics
551                    .counter(
552                        "pv_events_dropped_total{stage=\"worker_closed\"}",
553                        "Worker jobs or response-slot reservations dropped, labeled by admission stage",
554                    )
555                    .inc();
556                tracing::warn!(
557                    session = %session_id,
558                    stage = %stage.label(),
559                    "worker queue closed; request must fail closed"
560                );
561                Err(SubmitError::Closed)
562            }
563        }
564    }

566    /// Submit, waiting up to `wait` for queue capacity, without waiting for
567    /// the job to be processed. For background work that must not be lost
568    /// to a moment of backpressure (the semantic judge's events).
569    pub async fn submit_within(&self, job: WorkerJob, wait: std::time::Duration) -> Result<(), String> {
570        let sender_for_session = self
571            .sender_for(&job.session.id)
572            .cloned()
573            .ok_or_else(|| "worker queue is closed".to_owned())?;
574        let capacity_permit = tokio::time::timeout(wait, Arc::clone(&self.capacity).acquire_owned())
575            .await
576            .map_err(|_| "worker queue stayed full".to_owned())?
577            .map_err(|_| "worker capacity semaphore is closed".to_owned())?;
578        let session = Arc::clone(&job.session);
579        session.worker_job_started();
580        self.pending.fetch_add(1, Ordering::AcqRel);
581        let span = worker_span(&job);
582        let envelope = Envelope {
583            job,
584            completion: None,
585            span,
586            _capacity_permit: capacity_permit,
587        };
588        match tokio::time::timeout(wait, sender_for_session.send(envelope)).await {
589            Ok(Ok(())) => Ok(()),
590            Ok(Err(error)) => {
591                error.0.job.session.worker_job_finished();
592                self.worker_job_finished();
593                Err("worker queue is closed".to_owned())
594            }
595            Err(_) => {
596                // The envelope was dropped with the send: its session and
597                // pending counts are released here.
598                session.worker_job_finished();
599                self.worker_job_finished();
600                Err("worker queue stayed full".to_owned())
601            }
602        }
603    }

605    /// Submit and wait for completion. Intended for lifecycle operations and
606    /// deterministic integration tests, never streaming hot paths.
607    pub async fn submit_and_wait(&self, job: WorkerJob) -> Result<(), String> {
608        let sender_for_session = self
609            .sender_for(&job.session.id)
610            .cloned()
611            .ok_or_else(|| "worker queue is closed".to_owned())?;
612        let (sender, receiver) = oneshot::channel();
613        let capacity_permit = Arc::clone(&self.capacity)
614            .acquire_owned()
615            .await
616            .map_err(|_| "worker capacity semaphore is closed".to_owned())?;
617        job.session.worker_job_started();
618        self.pending.fetch_add(1, Ordering::AcqRel);
619        let span = worker_span(&job);
620        sender_for_session
621            .send(Envelope {
622                job,
623                completion: Some(sender),
624                span,
625                _capacity_permit: capacity_permit,
626            })
627            .await
628            .map_err(|error| {
629                error.0.job.session.worker_job_finished();
630                self.worker_job_finished();
631                "worker queue is closed".to_owned()
632            })?;
633        receiver
634            .await
635            .map_err(|_| "worker completion channel closed".to_owned())?
636    }

638    /// Wait until every accepted job has completed processing.
639    pub async fn wait_idle(&self) {
640        loop {
641            // Same pinned Notified must span enable() and .await; a fresh
642            // notified() after enable is dropped would miss a notify_waiters()
643            // firing in the interval.
644            let notified = self.drained.notified();
645            let mut notified = std::pin::pin!(notified);
646            notified.as_mut().enable();
647            if self.pending.load(Ordering::Acquire) == 0 {
648                return;
649            }
650            notified.await;
651        }
652    }

658    }

661        if self.pending.fetch_sub(1, Ordering::AcqRel) == 1 {

663        }

764/// Start the worker pool with authenticated journals, redaction, and an
765/// optional content sink that receives every journaled step.
766#[allow(clippy::too_many_arguments)]
767pub fn spawn_worker_with_outputs(
768    capacity: usize,
769    bridge: Arc<dyn EventBus>,
770    embedder: Arc<dyn Embedder>,
771    vector_sink: Arc<dyn VectorSink>,
772    spool_dir: Option<std::path::PathBuf>,
773    journal_key: [u8; 32],
774    metrics: Arc<Registry>,
775    outputs: crate::content::AuditOutputs,
776) -> WorkerHandle {
777    // Sharding is decoupled from `capacity`: routing is by
778    // `partition_for(session_id, senders.len())`, so every spawned
779    // shard is routable by construction. Under `capacity = 1` all
780    // shards still spawn; the global semaphore continues to enforce
781    // the caller's admission cap so total in-flight work is still
782    // bounded by `capacity`.
783    //
784    // The count was hardcoded at 16 regardless of core
785    // count, so the throughput ceiling (~24k events/s measured, and
786    // 1/16th of the id space frozen behind any one stalled session)
787    // did not improve with more cores. Scale with available
788    // parallelism, floored at the historical 16 so small machines
789    // keep the old fan-out. Per-session ordering is unaffected: a
790    // session's shard is stable for the process lifetime because
791    // routing derives from the spawned count.
792    const MIN_SHARDS: usize = 16;
793    let capacity = capacity.max(1);
794    let shard_count = std::thread::available_parallelism()
795        .map(usize::from)
796        .unwrap_or(MIN_SHARDS)
797        .max(MIN_SHARDS);
798    // Per-shard channel size: enough that a single shard can absorb
799    // the full admission burst even if every envelope happens to hash
800    // to it. TWO same-sized semaphores feed the same shard channels —
801    // worker capacity (admission jobs) and response capacity
802    // (capture jobs riding their admission-time reservation) — so the
803    // worst-case in-flight envelope count against one shard is
804    // 2 × `capacity`. Sizing at `capacity` alone let response
805    // envelopes fill a shard and (a) fail-close response captures
806    // whose slot was reserved at admission, and (b) break
807    // `submit_and_wait`'s cancellation-safety invariant that
808    // `send().await` never blocks while a permit is held (a
809    // cancellation while blocked leaks `pending` and deadlocks
810    // `wait_idle`).
811    let per_shard_capacity = capacity.saturating_mul(2);
812    let global_capacity = Arc::new(tokio::sync::Semaphore::new(capacity));
813    // Response-slot capacity mirrors worker capacity by default. Keeping
814    // them equal preserves the historical behaviour (both counters were
815    // silently drawn from the same pool) while distinct semaphores let
816    // operators observe which class exhausts first via
817    // `pv_events_dropped_total{stage="response_slot"}` vs
818    // `stage="worker_queue"`. A follow-up can expose an independent
819    // `response_capacity` config field once operators have telemetry.
820    let response_capacity = Arc::new(tokio::sync::Semaphore::new(capacity));
821    let pending = Arc::new(std::sync::atomic::AtomicU64::new(0));
822    let drained = Arc::new(tokio::sync::Notify::new());
823    let mut senders = Vec::with_capacity(shard_count);
824    for _ in 0..shard_count {
825        let (sender, receiver) = mpsc::channel::<Envelope>(per_shard_capacity);
826        senders.push(sender);
827        spawn_worker_shard(
828            receiver,
829            Arc::clone(&bridge),
830            Arc::clone(&embedder),
831            Arc::clone(&vector_sink),
832            spool_dir.clone(),
833            journal_key,
834            Arc::clone(&metrics),
835            Arc::clone(&pending),
836            Arc::clone(&drained),
837            outputs.clone(),
838        );
839    }
840    WorkerHandle {
841        senders: Arc::new(senders),
842        metrics,
843        pending,
844        drained,
845        capacity: global_capacity,
846        response_capacity,
847    }
848}

850/// Sessions a shard may process concurrently. One
851/// stalled session (slow journal fsync, hung broker publish) used to
852/// freeze its entire shard — 31× measured head-of-line blocking for
853/// every other session hashed there. Eight slots bound the
854/// wasted-capacity blast radius of pathological neighbors to 1/8 of
855/// a shard while keeping per-shard task fan-out small.
856const MAX_ACTIVE_SESSIONS_PER_SHARD: usize = 8;
857
858/// Group commit: how many backlogged envelopes of ONE
859/// session a single dispatch drains into a batch (one journal
860/// fdatasync per batch instead of one per event). Sized to cap both
861/// the added completion latency of the batch's first job and the
862/// replay window a crash-before-sync leaves (nothing acknowledged is
863/// ever lost — Phase B runs only after the batch sync).
864const MAX_BATCH_PER_SESSION: usize = 16;
865
866#[allow(clippy::too_many_arguments)]
867fn spawn_worker_shard(
868    mut receiver: mpsc::Receiver<Envelope>,
869    bridge: Arc<dyn EventBus>,
870    embedder: Arc<dyn Embedder>,
871    vector_sink: Arc<dyn VectorSink>,
872    spool_dir: Option<std::path::PathBuf>,
873    journal_key: [u8; 32],
874    worker_metrics: Arc<Registry>,
875    worker_pending: Arc<std::sync::atomic::AtomicU64>,
876    worker_drained: Arc<tokio::sync::Notify>,
877    outputs: crate::content::AuditOutputs,
878) {
879    // To avoid intra-shard head-of-line blocking, the shard is
880    // a DISPATCHER over per-session FIFO queues, not a serial loop.
881    // Invariants:
882    //   * Per-session ordering: at most ONE envelope per session is in
883    //     flight; the next spawns only after its predecessor's task
884    //     joins. Arrival order per session is preserved (mpsc order →
885    //     VecDeque order).
886    //   * Cross-session progress: up to MAX_ACTIVE_SESSIONS_PER_SHARD
887    //     sessions run concurrently, so one stalled session occupies
888    //     one slot instead of the whole shard.
889    //   * Memory stays bounded by the GLOBAL capacity semaphore: every
890    //     queued envelope still holds its `_capacity_permit`, so
891    //     draining the channel into local queues does not raise the
892    //     in-flight ceiling.
893    //   * Shutdown: the loop exits only when the channel is closed AND
894    //     every local queue and in-flight task has drained, preserving
895    //     the drain contract (`worker_pending` accounting is inside
896    //     `process_envelope`, unchanged).
897    tokio::spawn(async move {
898        let mut queues: std::collections::HashMap<String, std::collections::VecDeque<Envelope>> =
899            std::collections::HashMap::new();
900        // Sessions with an envelope IN FLIGHT. Tracked separately from
901        // `queues`: a session can be active AND have a backlog queued
902        // behind its in-flight envelope, and the slot-release scan
903        // must never dispatch from an active session's backlog — that
904        // would run two envelopes of one session concurrently and
905        // corrupt the per-session journal index sequence.
906        let mut active: std::collections::HashSet<String> = std::collections::HashSet::new();
907        let mut tasks: tokio::task::JoinSet<String> = tokio::task::JoinSet::new();
908        let mut receiver_open = true;
909
910        let spawn_batch = |tasks: &mut tokio::task::JoinSet<String>, batch: Vec<Envelope>| {
911            let session_id = batch
912                .first()
913                .map(|envelope| envelope.job.session.id.clone())
914                .unwrap_or_default();
915            let bridge = Arc::clone(&bridge);
916            let embedder = Arc::clone(&embedder);
917            let vector_sink = Arc::clone(&vector_sink);
918            let spool_dir = spool_dir.clone();
919            let worker_metrics = Arc::clone(&worker_metrics);
920            let worker_pending = Arc::clone(&worker_pending);
921            let worker_drained = Arc::clone(&worker_drained);
922            let outputs = outputs.clone();
923            tasks.spawn(async move {
924                use futures::future::FutureExt as _;
925                // Supervise the whole envelope (routing + process_job)
926                // exactly like the pre-dispatcher shard loop did: the
927                // INNER tokio::spawn(process_job) already catches job
928                // panics; this catch_unwind covers the outer routing so
929                // a panic cannot lose the session's dispatch slot. The
930                // task cannot panic between here and returning the id.
931                let outcome = std::panic::AssertUnwindSafe(async {
932                    let metrics = Arc::clone(&worker_metrics);
933                    let outcome = std::panic::AssertUnwindSafe(process_envelope_batch(
934                        batch,
935                        bridge,
936                        embedder,
937                        vector_sink,
938                        spool_dir,
939                        journal_key,
940                        worker_metrics,
941                        worker_pending,
942                        worker_drained,
943                        outputs,
944                    ))
945                    .catch_unwind()
946                    .await;
947                    if let Err(panic) = outcome {
948                        let msg = panic
949                            .downcast_ref::<&'static str>()
950                            .copied()
951                            .or_else(|| panic.downcast_ref::<String>().map(String::as_str))
952                            .unwrap_or("panic payload was not a string");
953                        metrics
954                            .counter(
955                                "pv_worker_shard_panics_total",
956                                "Worker shard driver panicked outside a job; supervised via catch_unwind",
957                            )
958                            .inc();
959                        tracing::error!(
960                            panic = %msg,
961                            "worker envelope routing panicked; continuing"
962                        );
963                    }
964                })
965                .catch_unwind()
966                .await;
967                drop(outcome);
968                session_id
969            });
970        };
971
972        loop {
973            if !receiver_open && tasks.is_empty() && queues.is_empty() {
974                break;
975            }
976            tokio::select! {
977                maybe_envelope = receiver.recv(), if receiver_open => {
978                    match maybe_envelope {
979                        Some(envelope) => {
980                            let session_id = envelope.job.session.id.clone();
981                            if !active.contains(&session_id)
982                                // Defense in depth: a WAITING backlog
983                                // (queued while at cap) must drain
984                                // first or this envelope would jump
985                                // its predecessors. Unreachable today
986                                // (every join refills slots from the
987                                // waiting set before the next recv),
988                                // but cheap to make impossible.
989                                && !queues.contains_key(&session_id)
990                                && active.len() < MAX_ACTIVE_SESSIONS_PER_SHARD
991                            {
992                                // Idle session, free slot: dispatch now.
993                                active.insert(session_id);
994                                spawn_batch(&mut tasks, vec![envelope]);
995                            } else {
996                                // Active (order! behind the in-flight
997                                // envelope) or no free slot: queue.
998                                queues.entry(session_id).or_default().push_back(envelope);
999                            }
1000                        }
1001                        None => receiver_open = false,
1002                    }
1003                }
1004                Some(joined) = tasks.join_next(), if !tasks.is_empty() => {
1005                    match joined {
1006                        Ok(session_id) => {
1007                            let backlog = queues.get_mut(&session_id).map_or(Vec::new(), |queue| {
1008                                let take = queue.len().min(MAX_BATCH_PER_SESSION);
1009                                queue.drain(..take).collect::<Vec<_>>()
1010                            });
1011                            if !backlog.is_empty() {
1012                                // Same session next-in-line keeps its
1013                                // slot: per-session order is preserved
1014                                // because dispatch for an ACTIVE session
1015                                // happens only here, after its
1016                                // predecessor joined. The backlog runs
1017                                // as ONE group-committed batch (§7.3).
1018                                if queues.get(&session_id).is_some_and(std::collections::VecDeque::is_empty) {
1019                                    queues.remove(&session_id);
1020                                }
1021                                spawn_batch(&mut tasks, backlog);
1022                            } else {
1023                                queues.remove(&session_id);
1024                                active.remove(&session_id);
1025                                // Slot freed: activate WAITING sessions
1026                                // (queued work, not active — enqueued
1027                                // while the shard was at cap).
1028                                let waiting = queues
1029                                    .keys()
1030                                    .filter(|id| !active.contains(*id))
1031                                    .take(
1032                                        MAX_ACTIVE_SESSIONS_PER_SHARD.saturating_sub(active.len()),
1033                                    )
1034                                    .cloned()
1035                                    .collect::<Vec<_>>();
1036                                for id in waiting {
1037                                    let batch = queues.get_mut(&id).map_or(Vec::new(), |queue| {
1038                                        let take = queue.len().min(MAX_BATCH_PER_SESSION);
1039                                        queue.drain(..take).collect::<Vec<_>>()
1040                                    });
1041                                    if batch.is_empty() {
1042                                        continue;
1043                                    }
1044                                    if queues.get(&id).is_some_and(std::collections::VecDeque::is_empty) {
1045                                        queues.remove(&id);
1046                                    }
1047                                    active.insert(id);
1048                                    spawn_batch(&mut tasks, batch);
1049                                }
1050                            }
1051                        }
1052                        Err(error) => {
1053                            // Only runtime teardown/abort can land here
1054                            // (the task body is panic-supervised and
1055                            // infallible). The session's queue entry
1056                            // stays; it re-activates on the next slot
1057                            // release via the waiting scan above.
1058                            tracing::error!(%error, "worker envelope task failed to join");
1059                        }
1060                    }
1061                }
1062                else => {
1063                    // Receiver closed and no tasks in flight. Queues can
1064                    // only be non-empty here after a JoinError leak
1065                    // (runtime teardown); surface it rather than hang.
1066                    if !queues.is_empty() {
1067                        tracing::error!(
1068                            sessions = queues.len(),
1069                            "worker shard exiting with undispatched envelopes after task join failure"
1070                        );
1071                    }
1072                    break;
1073                }
1074            }
1075        }
1076    });
1077}

1079/// Group commit: fdatasync a session's events journal
1080/// once, covering every deferred append a batch wrote before it.
1081async fn sync_session_journal(directory: &std::path::Path, session: &Session) -> Result<(), String> {
1082    let digest = pv_core::digest::sha256_hex(session.id.as_bytes());
1083    let stem = digest.get(..32).unwrap_or(&digest);
1084    let path = directory.join(format!("{stem}.events.ndjson"));
1085    tokio::task::spawn_blocking(move || {
1086        // Single-invariant discipline: EVERY open on a spool-derived
1087        // path uses `O_NOFOLLOW`. The events-journal append path
1088        // (worker.rs `append_active_event_journal`), torn-tail
1089        // `set_len` reopen (reconciler.rs), broker segment reopen
1090        // (pv-bridge/embedded.rs), and sidecar reopen all pass
1091        // `O_NOFOLLOW`; this group-commit sync was the last outlier.
1092        // In the current 0o700 daemon-owned threat model a symlink
1093        // plant is not reachable; but a future refactor that widened
1094        // spool group permissions would silently regress durability
1095        // of group-committed batches (the sync would flush a decoy
1096        // inode instead of the real journal-fd). Close the surface
1097        // preemptively so the invariant "spool open ⇒ O_NOFOLLOW"
1098        // holds uniformly.
1099        let mut options = std::fs::OpenOptions::new();
1100        options.read(true);
1101        #[cfg(unix)]
1102        {
1103            use std::os::unix::fs::OpenOptionsExt as _;
1104            options.custom_flags(pv_core::fsutil::unix_o_nofollow());
1105        }
1106        // Windows' FlushFileBuffers needs write access ("Access is
1107        // denied" on a read-only handle). Opening for write neither
1108        // creates nor truncates, so the journal is untouched.
1109        #[cfg(not(unix))]
1110        options.write(true);
1111        match options.open(&path) {
1112            Ok(journal) => journal.sync_data().map_err(|error| error.to_string()),
1113            // No journal file: every job in the batch was skipped by
1114            // the capture-failed guard before appending anything.
1115            Err(error) if error.kind() == std::io::ErrorKind::NotFound => Ok(()),
1116            Err(error) => Err(error.to_string()),
1117        }
1118    })
1119    .await
1120    .map_err(|error| error.to_string())?
1121}

1123/// Group commit: process a SAME-SESSION batch of
1124/// envelopes with one journal fdatasync.
1125///
1126/// Phase A appends every job's record with the per-append sync
1127/// deferred; one `sync_session_journal` then makes the whole batch
1128/// durable; Phase B (publish, ack, totals, marker clear — everything
1129/// externally visible) runs per job strictly afterwards, so no job's
1130/// effects can precede its record's durability. A crash before the
1131/// batch sync is indistinguishable from crashing before processing:
1132/// nothing was published or acked, the response markers are still on
1133/// disk, and recovery quarantines exactly as it would have.
1134///
1135/// Fail-stop semantics match the serial path: a Phase A error marks
1136/// the session capture-failed and the remaining jobs resolve as
1137/// skipped/failed (their guards would have skipped them serially); a
1138/// Phase B error fails the batch's remaining jobs (their durable
1139/// records replay through crash recovery's re-publish path).
1140#[allow(clippy::too_many_arguments)]
1141async fn process_envelope_batch(
1142    envelopes: Vec<Envelope>,
1143    bridge: Arc<dyn EventBus>,
1144    embedder: Arc<dyn Embedder>,
1145    vector_sink: Arc<dyn VectorSink>,
1146    spool_dir: Option<std::path::PathBuf>,
1147    journal_key: [u8; 32],
1148    worker_metrics: Arc<Registry>,
1149    worker_pending: Arc<std::sync::atomic::AtomicU64>,
1150    worker_drained: Arc<tokio::sync::Notify>,
1151    outputs: crate::content::AuditOutputs,
1152) {
1153    let mut iterator = envelopes.into_iter();
1154    let Some(first) = iterator.next() else { return };
1155    let rest: Vec<Envelope> = iterator.collect();
1156    if rest.is_empty() {
1157        return process_envelope(
1158            first,
1159            bridge,
1160            embedder,
1161            vector_sink,
1162            spool_dir,
1163            journal_key,
1164            worker_metrics,
1165            worker_pending,
1166            worker_drained,
1167            outputs,
1168        )
1169        .await;
1170    }
1171    let session = Arc::clone(&first.job.session);
1172    // Per-envelope bookkeeping: guards + permits live for the whole
1173    // batch; completions resolve individually below.
1174    let mut guards = Vec::new();
1175    let mut completions = Vec::new();
1176    let mut jobs = Vec::new();
1177    for envelope in std::iter::once(first).chain(rest) {
1178        let Envelope {
1179            job,
1180            completion,
1181            span,
1182            _capacity_permit: capacity_permit,
1183        } = envelope;
1184        guards.push((
1185            PendingGuard::new(Arc::clone(&worker_pending), Arc::clone(&worker_drained)),
1186            SessionPendingGuard {
1187                session: Arc::clone(&job.session),
1188            },
1189            capacity_permit,
1190        ));
1191        completions.push(completion);
1192        jobs.push((job, span));
1193    }
1194    let batch_metrics = Arc::clone(&worker_metrics);
1195    let batch_session = Arc::clone(&session);
1196    let spool = spool_dir.clone();
1197    let result = tokio::spawn(async move {
1198        let mut outcomes: Vec<Result<(), String>> = Vec::new();
1199        let mut persisted: Vec<Option<PersistedJob>> = Vec::new();
1200        let mut failed = false;
1201        // Phase A: append every record, sync deferred.
1202        for (job, span) in jobs {
1203            if failed {
1204                // Serial semantics: after a sibling failure the
1205                // capture-failed guard would have skipped this job.
1206                outcomes.push(Ok(()));
1207                persisted.push(None);
1208                continue;
1209            }
1210            let outcome = persist_job(
1211                job,
1212                Arc::clone(&embedder),
1213                Arc::clone(&vector_sink),
1214                spool.as_deref(),
1215                journal_key,
1216                &batch_metrics,
1217                /* sync_journal */ false,
1218                &outputs,
1219            )
1220            .instrument(span)
1221            .await;
1222            match outcome {
1223                Ok(slot) => {
1224                    persisted.push(slot);
1225                    outcomes.push(Ok(()));
1226                }
1227                Err(error) => {
1228                    // Latch NOW so recovery semantics match serial
1229                    // processing (later jobs must not consume seqs).
1230                    batch_session.mark_capture_failed();
1231                    persisted.push(None);
1232                    outcomes.push(Err(error));
1233                    failed = true;
1234                }
1235            }
1236        }
1237        // The batch fsync: everything Phase A appended becomes
1238        // durable in one fdatasync.
1239        if let Some(directory) = spool.as_deref() {
1240            if persisted.iter().any(Option::is_some) {
1241                if let Err(error) = sync_session_journal(directory, &batch_session).await {
1242                    // None of the batch is provably durable: fail every
1243                    // job that had persisted (fail-closed, like a
1244                    // failed per-append sync on the serial path).
1245                    for (index, slot) in persisted.iter_mut().enumerate() {
1246                        if slot.take().is_some() {
1247                            if let Some(outcome) = outcomes.get_mut(index) {
1248                                *outcome = Err(format!("batch journal sync failed: {error}"));
1249                            }
1250                        }
1251                    }
1252                }
1253            }
1254        }
1255        // Phase B, in order, fail-stop.
1256        let mut tail_failed = false;
1257        for (index, slot) in persisted.into_iter().enumerate() {
1258            let Some(slot) = slot else { continue };
1259            if tail_failed {
1260                if let Some(outcome) = outcomes.get_mut(index) {
1261                    *outcome = Err("skipped after a batch sibling's post-persist failure".to_owned());
1262                }
1263                continue;
1264            }
1265            if let Err(error) = finish_job(slot, Arc::clone(&bridge), spool.as_deref(), journal_key).await {
1266                if let Some(outcome) = outcomes.get_mut(index) {
1267                    *outcome = Err(error);
1268                }
1269                tail_failed = true;
1270            }
1271        }
1272        outcomes
1273    })
1274    .await;
1275    let outcomes = match result {
1276        Ok(outcomes) => outcomes,
1277        Err(error) => {
1278            worker_metrics
1279                .counter(
1280                    "pv_worker_panics_total",
1281                    "Worker job panics isolated by supervisor",
1282                )
1283                .inc();
1284            let failure = format!("worker batch panicked: {error}");
1285            vec![Err(failure); completions.len()]
1286        }
1287    };
1288    let mut any_failed = false;
1289    for (completion, outcome) in completions.into_iter().zip(outcomes) {
1290        if let Err(error) = &outcome {
1291            any_failed = true;
1292            tracing::warn!(session = %session.id, %error, "capture job failed; session is fail-closed");
1293            worker_metrics
1294                .counter("pv_worker_errors_total", "Worker jobs that failed")
1295                .inc();
1296        }
1297        if let Some(completion) = completion {
1298            let _ = completion.send(outcome);
1299        }
1300    }
1301    if any_failed {
1302        session.mark_capture_failed();
1303    }
1304    drop(guards);
1305}

1307/// One envelope's worth of shard-driver work, factored out so
1308/// [`spawn_worker_shard`] can wrap the whole body in `catch_unwind`.
1309#[allow(clippy::too_many_arguments)]
1310async fn process_envelope(
1311    envelope: Envelope,
1312    bridge: Arc<dyn EventBus>,
1313    embedder: Arc<dyn Embedder>,
1314    vector_sink: Arc<dyn VectorSink>,
1315    spool_dir: Option<std::path::PathBuf>,
1316    journal_key: [u8; 32],
1317    worker_metrics: Arc<Registry>,
1318    worker_pending: Arc<std::sync::atomic::AtomicU64>,
1319    worker_drained: Arc<tokio::sync::Notify>,
1320    outputs: crate::content::AuditOutputs,
1321) {
1322    // `worker_pending` was previously decremented at the
1323    // bottom of this function. Any panic between here and that line
1324    // (tokio::spawn(...).await JoinError construction under runtime
1325    // shutdown races, or a panic inside tracing::warn's Display
1326    // formatter when the error string exceeds the allocator's
1327    // fragmentation threshold) would leak the pending count — and
1328    // `WorkerHandle::wait_idle()` at shutdown would spin on
1329    // `notify.notified().await` forever, tripping the 30 s drain
1330    // budget and terminating without capturing evidence. Encode the
1331    // decrement in an RAII guard so unwinding is a valid release
1332    // point.
1333    let _pending_guard = PendingGuard::new(Arc::clone(&worker_pending), Arc::clone(&worker_drained));
1334    let Envelope {
1335        job,
1336        completion,
1337        span,
1338        _capacity_permit: capacity_permit,
1339    } = envelope;
1340    let session = Arc::clone(&job.session);
1341    // Guard the session-level pending-jobs decrement in
1342    // the same RAII shape as `PendingGuard`. The bare
1343    // `session.worker_job_finished()` after `tokio::spawn(...).await`
1344    // used to leak the session-level counter on any panic or drop
1345    // between here and line ~712. A stuck `session.pending_jobs` means
1346    // `close_session_locked -> wait_for_worker_jobs().await` blocks
1347    // forever on `jobs_drained.notified()`, holding the session's
1348    // lifecycle lock and starving every subsequent close / promote /
1349    // recovery-adopt on that id. Same leak class as the worker-level
1350    // counter, closed one call
1351    // frame up.
1352    let _session_pending_guard = SessionPendingGuard {
1353        session: Arc::clone(&session),
1354    };
1355    let result = tokio::spawn(
1356        {
1357            let job_metrics = Arc::clone(&worker_metrics);
1358            async move {
1359                process_job(
1360                    job,
1361                    bridge,
1362                    embedder,
1363                    vector_sink,
1364                    spool_dir,
1365                    journal_key,
1366                    job_metrics,
1367                    outputs,
1368                )
1369                .await
1370            }
1371        }
1372        .instrument(span),
1373    )
1374    .await;
1375    let outcome = match result {
1376        Ok(result) => result,
1377        Err(error) => {
1378            worker_metrics
1379                .counter(
1380                    "pv_worker_panics_total",
1381                    "Worker job panics isolated by supervisor",
1382                )
1383                .inc();
1384            Err(format!("worker job panicked: {error}"))
1385        }
1386    };
1387    if let Err(error) = &outcome {
1388        // The session flips fail-closed silently otherwise: name the
1389        // session and the root cause (e.g. EACCES on the journal
1390        // file) so operators can trace "capture is incomplete"
1391        // refusals back to this event.
1392        tracing::warn!(session = %session.id, %error, "capture job failed; session is fail-closed");
1393        session.mark_capture_failed();
1394        worker_metrics
1395            .counter("pv_worker_errors_total", "Worker jobs that failed")
1396            .inc();
1397    }
1398    // `_session_pending_guard`'s Drop calls
1399    // `worker_job_finished()` — replaces the bare call previously
1400    // here so a panic between `spawn.await` and this point cannot
1401    // leak the session pending counter.
1402    drop(capacity_permit);
1403    // The pending guards (named bindings above) drop at end of scope —
1404    // AFTER the completion send below, not here. That order is fine:
1405    // the receiver needs only the outcome, and `wait_idle` /
1406    // `wait_for_worker_jobs` re-check the pending counters in a loop,
1407    // so a momentary completed-but-still-pending window is benign.
1408    if let Some(completion) = completion {
1409        let _ = completion.send(outcome);
1410    }
1411}

1413/// RAII decrement for `worker_pending`. Runs on every path — normal
1414/// return, error return, or panic — so `wait_idle` never sees a
1415/// permanently non-zero pending count after a panic in
1416/// [`process_envelope`].
1417struct PendingGuard {
1418    pending: Arc<std::sync::atomic::AtomicU64>,
1419    drained: Arc<tokio::sync::Notify>,
1420}
1421
1422impl PendingGuard {
1423    fn new(pending: Arc<std::sync::atomic::AtomicU64>, drained: Arc<tokio::sync::Notify>) -> Self {
1424        Self { pending, drained }
1425    }
1426}
1427
1428impl Drop for PendingGuard {
1429    fn drop(&mut self) {
1430        if self.pending.fetch_sub(1, Ordering::AcqRel) == 1 {
1431            self.drained.notify_waiters();
1432        }
1433    }
1434}

1436/// RAII pair to [`PendingGuard`] for the session-level
1437/// pending-jobs counter. `Session::worker_job_finished()` was
1438/// previously called as a bare method after `tokio::spawn(...).await`,
1439/// but a panic in the routing-side epilogue (Display-side allocator
1440/// failure, catch_unwind of the wrapper future, runtime shutdown
1441/// mid-envelope) between `.await` and that call left
1442/// `session.pending_jobs` stuck. `close_session_locked` then blocked
1443/// forever on `wait_for_worker_jobs().await`, holding the session's
1444/// lifecycle lock — the exact leak class already closed on the
1445/// worker-level counter.
1446struct SessionPendingGuard {
1447    session: Arc<crate::session::Session>,
1448}
1449
1450impl Drop for SessionPendingGuard {
1451    fn drop(&mut self) {
1452        self.session.worker_job_finished();
1453    }
1454}

1516/// Redact both copies of the audit data before they reach durable storage.
1517fn redact_audit_capture(
1518    mut value: Value,
1519    mut atif_step: Option<pv_atif::Step>,
1520    class: EventClass,
1521    engine: &pv_redact::RedactionEngine,
1522) -> (Value, Option<pv_atif::Step>) {
1523    // Tool authorization stores the event payload as a JSON string in the
1524    // observation. Preserve that schema, but reuse the redacted payload for
1525    // exact copies. Arbitrary tool output is not a payload-relative document.
1526    let original_payload = (class == EventClass::ToolCall)
1527        .then(|| value.get("payload").map(Value::to_string))
1528        .flatten();
1529    if let Some(payload) = value.get_mut("payload") {
1530        engine.redact_value(payload);
1531    }
1532    if let Some(step) = &mut atif_step {
1533        engine.redact_value(&mut step.message);
1534        if let Some(reasoning) = &mut step.reasoning_content {
1535            let mut text = Value::String(std::mem::take(reasoning));
1536            engine.redact_value(&mut text);
1537            if let Value::String(redacted) = text {
1538                *reasoning = redacted;
1539            }
1540        }
1541        if let Some(calls) = &mut step.tool_calls {
1542            for call in calls {
1543                engine.redact_user_value(&mut call.arguments);
1544                // The semantic judge's record, redacted like its copy in
1545                // the event payload.
1546                if let Some(extra) = &mut call.extra {
1547                    engine.redact_value(extra);
1548                }
1549            }
1550        }
1551        if let Some(observation) = &mut step.observation {
1552            for result in &mut observation.results {
1553                if let Some(content) = &mut result.content {
1554                    if original_payload
1555                        .as_deref()
1556                        .is_some_and(|original| content.as_str() == Some(original))
1557                    {
1558                        if let Some(payload) = value.get("payload") {
1559                            *content = Value::String(payload.to_string());
1560                        }
1561                    }
1562                    engine.redact_value(content);
1563                }
1564            }
1565        }
1566    }
1567    (value, atif_step)
1568}

1570/// Phase A: analyze, build the event + journal record, and append it
1571/// to the session journal (fsynced only when `sync_journal` — a batch
1572/// syncs once after its last append). Returns `None` when the session
1573/// is poisoned (capture-failed guard) and the job must be skipped
1574/// without consuming a sequence number.
1575#[allow(clippy::too_many_arguments)]
1576async fn persist_job(
1577    mut job: WorkerJob,
1578    embedder: Arc<dyn Embedder>,
1579    vector_sink: Arc<dyn VectorSink>,
1580    spool_dir: Option<&std::path::Path>,
1581    journal_key: [u8; 32],
1582    metrics: &Arc<Registry>,
1583    sync_journal: bool,
1584    outputs: &crate::content::AuditOutputs,
1585) -> Result<Option<PersistedJob>, String> {
1586    // A queued envelope for a session that has already been poisoned must not
1587    // consume a sequence number or write to the journal — otherwise the seq
1588    // its OcsfEvent carries (from `next_seq`, advanced by the failed prior
1589    // envelope) would not match the entry's position on disk, breaking
1590    // recovery's `event.metadata.sequence != index` check.
1591    if job.session.capture_failed() {
1592        return Ok(None);
1593    }
1594    // The judge, when attached. Read once, outside any await.
1595    let judge = outputs.judge.read().clone();
1596    let captured_output = judge.as_ref().and_then(|_| captured_output(&job));
1597    let response_marker = job.response_marker.clone();
1598    let is_llm_agent_response = job
1599        .atif
1600        .as_ref()
1601        .is_some_and(|capture| capture.source == pv_atif::Source::Agent && capture.llm_call_count != Some(0));
1602    let step_tokens = job
1603        .metrics
1604        .prompt_tokens
1605        .unwrap_or(0)
1606        .saturating_add(job.metrics.completion_tokens.unwrap_or(0));
1607    let breaker = if job.analyze_loop {
1608        let text = job.text.clone();
1609        // vector-sink and embedder errors must not
1610        // brick the session — the sink's trait doc-comment describes it
1611        // as "off-path observability, not participating in the hot
1612        // path". A single 2 s Qdrant timeout or one transient
1613        // ONNX/HashEmbedder error used to `?`-propagate all the way to
1614        // `mark_capture_failed` (sticky) → 503 forever. Downgrade all
1615        // three failure modes (embed, query, record) to warn+counter +
1616        // continue-without-breaker: the audit chain still gets the
1617        // step; only the semantic-loop breaker is skipped for that
1618        // step. Journal and receipt integrity still fail-close as
1619        // before via other paths.
1620        let embedding_result = tokio::task::spawn_blocking(move || embedder.try_embed(&text))
1621            .await
1622            .map_err(|error| error.to_string())?;
1623        match embedding_result {
1624            Ok(embedding) => {
1625                let similarity = match vector_sink
1626                    .nearest_similarity(&job.session.session_scope, &embedding)
1627                    .await
1628                {
1629                    Ok(similarity) => similarity,
1630                    Err(error) => {
1631                        tracing::warn!(
1632                            session = %job.session.id,
1633                            %error,
1634                            "vector-sink nearest_similarity failed; skipping breaker for this step"
1635                        );
1636                        metrics
1637                            .counter(
1638                                "pv_vector_sink_errors_total",
1639                                "Vector-sink errors demoted to warn",
1640                            )
1641                            .inc();
1642                        None
1643                    }
1644                };
1645                // Model output only (never request text): an ambiguous step
1646                // gets the semantic judge's second opinion when configured.
1647                let model_output = job.class != EventClass::Compression;
1648                let opinion = match &judge {
1649                    Some(hooks) if model_output && hooks.service().breaker().is_some() => {
1650                        let delta = job.session.loop_state.peek_delta(&embedding, similarity);
1651                        let previous = job.session.last_model_output();
1652                        hooks
1653                            .second_opinion(&job.session, delta, previous.as_deref(), &job.text)
1654                            .await
1655                    }
1656                    _ => None,
1657                };
1658                let verdict = job.session.loop_state.observe_embedding_with_opinion(
1659                    embedding.clone(),
1660                    step_tokens,
1661                    similarity,
1662                    opinion,
1663                );
1664                if model_output && judge.is_some() {
1665                    job.session.set_last_model_output(&job.text);
1666                }
1667                if let Err(error) = vector_sink.record(&job.session.session_scope, &embedding).await {
1668                    tracing::warn!(
1669                        session = %job.session.id,
1670                        %error,
1671                        "vector-sink record failed; the local breaker still observed this step"
1672                    );
1673                    metrics
1674                        .counter(
1675                            "pv_vector_sink_errors_total",
1676                            "Vector-sink errors demoted to warn",
1677                        )
1678                        .inc();
1679                }
1680                Some(verdict)
1681            }
1682            Err(error) => {
1683                tracing::warn!(
1684                    session = %job.session.id,
1685                    %error,
1686                    "embedder try_embed failed; skipping breaker for this step"
1687                );
1688                metrics
1689                    .counter("pv_embedder_errors_total", "Embedder errors demoted to warn")
1690                    .inc();
1691                None
1692            }
1693        }
1694    } else {
1695        None
1696    };
1697    // Accounting must key on the class the job was *submitted* with: a
1698    // breaker trip replaces `class` with StopReason below, and deciding the
1699    // prompt/completion buckets from the replaced class silently dropped a
1700    // tripped chat admission's prompt tokens from the journal record and the
1701    // receipt totals — undercounting exactly the runaway sessions the
1702    // breaker exists to attest.
1703    let submitted_class = job.class;
1704    let (class, status, stop_reason, payload) = match breaker {
1705        Some(BreakerVerdict::Tripped {
1706            delta,
1707            streak,
1708            tokens_consumed,
1709            action,
1710        }) => {
1711            // The trip is recorded in the trajectory, but operators watching
1712            // logs/metrics must also see why this session's next request
1713            // will be rejected/aborted/injected.
1714            tracing::warn!(
1715                session = %job.session.id,
1716                delta,
1717                streak,
1718                tokens_consumed,
1719                action = ?action,
1720                "semantic loop circuit breaker tripped"
1721            );
1722            metrics
1723                .counter("pv_breaker_trips_total", "Semantic loop breaker trips")
1724                .inc();
1725            // Latch at the trip itself — not only at the NEXT
1726            // admission's breaker-open refusal: a session whose breaker
1727            // trips and is then closed without another admission
1728            // attempt (explicit /close, idle sweep) would otherwise
1729            // recycle its id with a fresh breaker. Terminal actions
1730            // only, matching the admission arms: an Inject-configured
1731            // breaker resolves the trip by rewriting the next request,
1732            // so its trip must not make the id terminal.
1733            if matches!(
1734                action,
1735                pv_loopdetect::BreakerAction::Abort | pv_loopdetect::BreakerAction::Reject
1736            ) {
1737                job.session.latch_enforcement(StopReason::LoopDetected);
1738            }
1739            (
1740                EventClass::StopReason,
1741                StatusId::Failure,
1742                Some(StopReason::LoopDetected),
1743                serde_json::json!({
1744                    "delta": delta,
1745                    "streak": streak,
1746                    "tokens_consumed": tokens_consumed,
1747                    "action": action,
1748                }),
1749            )
1750        }
1751        _ => (job.class, job.status, job.stop_reason, job.payload),
1752    };
1753
1754    // Kept for the content sink: the event builder consumes it.
1755    let native_finish_reason = job.native_stop_reason.clone();
1756    let severity = judge_event_severity(class, &payload);
1757    let mut builder = OcsfEventBuilder::new(
1758        class,
1759        job.session.id.clone(),
1760        job.identity.clone(),
1761        job.session.next_seq(),
1762    )
1763    .severity(severity)
1764    .status(status)
1765    .payload(payload)
1766    .metrics(job.metrics);

1770            Some(native) => builder.stop_reason_native(reason, native),

1772        };

1887        )
1888    });

1890    // pending guard it takes holds the session's close until the verdict
1891    // is recorded too.
1892    if let (Some(hooks), Some(output)) = (&judge, captured_output) {
1893        hooks.after_capture(Arc::clone(&job.session), output);

1896        session: job.session,

1949        }

1959/// The text and tool calls of a recorded model answer, as one string.
1960fn answer_text(capture: &AtifCapture) -> String {
1961    let mut parts: Vec<String> = Vec::new();
1962    match &capture.message {
1963        Value::String(text) => parts.push(text.clone()),
1964        // Content parts: keep the text ones.
1965        Value::Array(items) => parts.extend(
1966            items
1967                .iter()
1968                .filter_map(|item| item.get("text").and_then(Value::as_str).map(str::to_owned)),
1969        ),
1970        _ => {}
1971    }
1972    for call in capture.tool_calls.iter().flatten() {
1973        parts.push(format!("{}({})", call.function_name, call.arguments));
1974    }
1975    parts.retain(|part| !part.trim().is_empty());
1976    parts.join("\n")
1977}

2058/// Phase B: everything that makes the job externally visible. MUST
2059/// run only after the job's journal record is durable (its own
2060/// synced append, or the batch fsync covering it) — publishing an
2061/// unsynced event would let a crash leave the bus ahead of the
2062/// journal.
2063async fn finish_job(
2064    persisted: PersistedJob,
2065    bridge: Arc<dyn EventBus>,
2066    spool_dir: Option<&std::path::Path>,
2067    journal_key: [u8; 32],
2068) -> Result<(), String> {
2069    let PersistedJob {
2070        session,
2071        identity,
2072        class,
2073        value,
2074        event_uid,
2075        record,
2076        response_marker,
2077        content,
2078    } = persisted;
2079    match session.workflow {
2080        Workflow::Signed => session
2081            .chain
2082            .lock()
2083            .append(&value)
2084            .map_err(|error| error.to_string())?,
2085        Workflow::Unsigned => {
2086            // RAM-cliff guard: the step was already
2087            // journaled durably in `record.atif_step` — do NOT
2088            // retain a second copy in the in-RAM builder (50 turns ×
2089            // 2 KB × 10k sessions was 1.35 GB held for the idle
2090            // window). Close rebuilds the trajectory from the
2091            // journal; RAM keeps only the count.
2092            if record.atif_step.is_none() {
2093                return Err("unsigned ATIF step disappeared".to_owned());
2094            }
2095            session.note_atif_step();
2096        }
2097    }
2098
2099    let key = identity.instance_uid.clone();
2100    let publish_topic = class.topic().to_owned();
2101    let publish_uid = event_uid.clone();
2102    let ack = tokio::task::spawn_blocking(move || {
2103        bridge.publish_idempotent(&publish_topic, &key, &value, &publish_uid)
2104    })
2105    .await
2106    .map_err(|error| error.to_string())?
2107    .map_err(|error| error.to_string())?;
2108    if let Some(directory) = spool_dir {
2109        persist_broker_ack(directory, &session.id, &event_uid, &ack, &journal_key).await?;
2110    }
2111    // The record IS the accounting rule. Its fields
2112    // were conditionalized once at construction above; apply them
2113    // through the same impl the recovery folds replay, so the live
2114    // path and a crash-recovered session can never attest different
2115    // numbers for identical traffic.
2116    record
2117        .apply_to_totals(&session.totals)
2118        .map_err(|error| error.to_string())?;
2119    if let (Some(directory), Some(attempt_id)) = (spool_dir, response_marker.as_deref()) {
2120        clear_response_marker(directory, &journal_key, &session.id, attempt_id).await?;
2121    }
2122    // Content leaves the process only after the step is durable and
2123    // published; the dispatcher never blocks and never fails this job.
2124    if let Some((dispatcher, content)) = content {
2125        dispatcher.submit(content);
2126    }
2127    Ok(())
2128}

2200pub(crate) async fn inflight_response_sessions(
2201    spool_dir: &std::path::Path,
2202    journal_key: &[u8; 32],
2203) -> Result<std::collections::HashSet<String>, String> {
2204    let spool_dir = spool_dir.to_path_buf();
2205    let directory = spool_dir.join(crate::spool::INFLIGHT_RESPONSES);
2206    let journal_key = *journal_key;
2207    tokio::task::spawn_blocking(move || {
2208        let entries = match std::fs::read_dir(&directory) {
2209            Ok(entries) => entries,
2210            Err(error) if error.kind() == std::io::ErrorKind::NotFound => {
2211                return Ok(std::collections::HashSet::new());
2212            }
2213            Err(error) => return Err(error.to_string()),
2214        };
2215        let mut sessions = std::collections::HashSet::new();
2216        for entry in entries {
2217            let path = entry.map_err(|error| error.to_string())?.path();
2218            if path.extension().and_then(std::ffi::OsStr::to_str) != Some("json") {
2219                continue;
2220            }
2221            let marker_bytes = match pv_core::fsutil::read_capped(&path, pv_core::fsutil::MAX_CONTROL_BYTES) {
2222                Ok(bytes) => bytes,
2223                // A live request legitimately clears its marker between
2224                // the directory listing and this read — the response
2225                // completed; it is not in flight.
2226                Err(error) if error.kind() == std::io::ErrorKind::NotFound => continue,
2227                Err(error) => return Err(error.to_string()),
2228            };
2229            // A single unverifiable marker (seed rotation, future field
2230            // added to `InFlightResponse`, torn payload) must not brick
2231            // recovery — which is fatal at boot in main.rs. Mirror the
2232            // tool-intent quarantine discipline: warn, rename to
2233            // `.corrupt-<uid>`, skip. The signed chain never referenced
2234            // this file; nothing else claims it.
2235            let marker: InFlightResponse =
2236                match crate::journal::open(&journal_key, "in-flight-response", 0, &marker_bytes) {
2237                    Ok(marker) => marker,
2238                    Err(error) => {
2239                        quarantine_inflight_marker(&path, &error);
2240                        continue;
2241                    }
2242                };
2243            if path != response_marker_path(&spool_dir, &marker.session_id, &marker.attempt_id) {
2244                // Same defense: a marker whose payload's ids don't
2245                // reconstruct its filename must not stall recovery.
2246                quarantine_inflight_marker(
2247                    &path,
2248                    "in-flight response marker path does not match its payload",
2249                );
2250                continue;
2251            }
2252            // CRASH-WINDOW FALSE-POSITIVE GUARD (R63 Finding 1).
2253            // `finish_job` fsyncs the terminal response
2254            // record BEFORE `clear_response_marker` runs, and the
2255            // interval between the two is broker publish
2256            // (spawn_blocking) + persist_broker_ack (spool write) +
2257            // apply_to_totals — hundreds of milliseconds to seconds
2258            // under load. A crash inside that window leaves this
2259            // marker on disk even though the journal already
2260            // witnesses the response as terminal. Without this
2261            // consultation the downstream quarantine paths at
2262            // reconciler.rs:1982/2458/2709 flip the session to
2263            // `capture_failed`: signed sessions never mint receipts,
2264            // unsigned sessions never write ATIF trajectories — a
2265            // silent-data-loss on a properly-fsynced completion.
2266            // Consult the journal (MAC-verified per line, so a
2267            // hostile local process cannot inject a fake terminal
2268            // to mask a real capture failure); if it records a
2269            // terminal `response_attempt` for this attempt_id, the
2270            // marker is a crash-window artifact — opportunistically
2271            // delete it and skip.
2272            match journal_witnesses_terminal_attempt(
2273                &spool_dir,
2274                &journal_key,
2275                &marker.session_id,
2276                &marker.attempt_id,
2277            ) {
2278                Ok(true) => {
2279                    match std::fs::remove_file(&path) {
2280                        Ok(()) => tracing::info!(
2281                            session = %marker.session_id,
2282                            attempt = %marker.attempt_id,
2283                            "cleared stale in-flight response marker (journal witnesses terminal record)"
2284                        ),
2285                        Err(error) if error.kind() == std::io::ErrorKind::NotFound => {}
2286                        Err(error) => tracing::warn!(
2287                            marker = %pv_core::fsutil::basename(&path),
2288                            %error,
2289                            "failed to clear stale in-flight response marker after journal witness"
2290                        ),
2291                    }
2292                    continue;
2293                }
2294                Ok(false) => {}
2295                Err(error) => {
2296                    // Journal may itself be torn or MAC-mismatched
2297                    // (a separate quarantine path will handle that).
2298                    // Do NOT silently drop the marker on a witness
2299                    // failure — fall through and let downstream
2300                    // quarantine treat this session as capture_failed
2301                    // (the safe default: false-positive quarantine
2302                    // preserves evidence; missed quarantine loses it).
2303                    tracing::warn!(
2304                        session = %marker.session_id,
2305                        attempt = %marker.attempt_id,
2306                        %error,
2307                        "in-flight response journal-witness check failed; retaining marker"
2308                    );
2309                }
2310            }
2311            sessions.insert(marker.session_id);
2312        }
2313        Ok(sessions)
2314    })
2315    .await
2316    .map_err(|error| error.to_string())?
2317}

2319/// Consult `<stem>.events.ndjson` for a MAC-verified `response_attempt`
2320/// record with `terminal=true` and matching `attempt_id`. Used to
2321/// distinguish a crash-window stale in-flight marker from a genuinely
2322/// interrupted response. Truncates at the last complete newline so a
2323/// torn tail (a common crash pattern) doesn't cause a MAC-open error
2324/// on garbage; complete records earlier in the file are still
2325/// verified.
2326///
2327/// **Concurrent-scan invariant** (R64 M1): `inflight_response_sessions`
2328/// is called from EVERY reconciler tick (`recovery.rs:157`), not just
2329/// at boot. Live workers can be actively appending to the same
2330/// `events.ndjson` this helper reads. Safety today relies on THREE
2331/// independent defenses that must all remain intact under future
2332/// refactors:
2333///   1. Newline-boundary truncation in this fn: the writer emits
2334///      `line` then `\n` as two separate `write_all` calls
2335///      (`append_journal` below), so a partial mid-race read whose
2336///      tail lacks `\n` is truncated here — the still-fsyncing record
2337///      is dropped from consideration and the next tick will see it.
2338///   2. `clear_response_marker` is idempotent to `NotFound`
2339///      (below): if a live worker races with our `remove_file` of
2340///      the same marker, the loser observes `NotFound` and treats
2341///      it as success.
2342///   3. `QuarantineIncompleteEffectsPass` retains ONLY sessions
2343///      absent from the live registry (`recovery.rs:165`), so
2344///      even if our scan sees a live session's in-flight marker,
2345///      that session cannot end up in `quarantined_sessions` while
2346///      it is still owned by a worker.
2347///
2348/// Weaken ANY one of those and the helper silently regresses.
2349fn journal_witnesses_terminal_attempt(
2350    spool_dir: &std::path::Path,
2351    journal_key: &[u8; 32],
2352    session_id: &str,
2353    attempt_id: &str,
2354) -> Result<bool, String> {
2355    let digest = pv_core::digest::sha256_hex(session_id.as_bytes());
2356    let stem = digest.get(..32).unwrap_or(&digest);
2357    let journal_path = spool_dir.join(format!("{stem}.events.ndjson"));
2358    let bytes = match pv_core::fsutil::read_capped(&journal_path, pv_core::fsutil::MAX_ATIF_BYTES) {
2359        Ok(bytes) => bytes,
2360        Err(error) if error.kind() == std::io::ErrorKind::NotFound => return Ok(false),
2361        Err(error) => return Err(error.to_string()),
2362    };
2363    if bytes.is_empty() {
2364        return Ok(false);
2365    }
2366    let complete_len = if bytes.last() == Some(&b'\n') {
2367        bytes.len()
2368    } else {
2369        bytes
2370            .iter()
2371            .rposition(|byte| *byte == b'\n')
2372            .map_or(0, |index| index + 1)
2373    };
2374    let domain = format!("{session_id}:active");
2375    let mut index: u64 = 0;
2376    let mut cursor = 0usize;
2377    while cursor < complete_len {
2378        let Some(remaining) = bytes.get(cursor..complete_len) else {
2379            break;
2380        };
2381        let Some(newline_offset) = remaining.iter().position(|byte| *byte == b'\n') else {
2382            break;
2383        };
2384        let Some(line) = remaining.get(..newline_offset) else {
2385            break;
2386        };
2387        cursor += newline_offset + 1;
2388        if line.is_empty() {
2389            // R64 L1: `append_journal` never emits an empty line
2390            // (it writes a non-empty sealed record then `\n`), so
2391            // an empty line mid-file is corruption. Refuse to
2392            // silently skip it — Err defensively retains the
2393            // marker rather than desyncing our per-record `index`
2394            // from the writer's `session.journal_index()`.
2395            return Err("active journal contains an empty line".to_owned());
2396        }
2397        let record: ActiveJournalRecord = crate::journal::open(journal_key, &domain, index, line)
2398            .map_err(|error| format!("active journal record open failed: {error}"))?;
2399        if let Some(attempt) = record.response_attempt {
2400            if attempt.terminal && attempt.id == attempt_id {
2401                return Ok(true);
2402            }
2403        }
2404        index = index
2405            .checked_add(1)
2406            .ok_or_else(|| "active journal index overflow".to_owned())?;
2407    }
2408    Ok(false)
2409}

2512async fn append_journal(
2513    directory: &std::path::Path,
2514    session: &Session,
2515    line: Vec<u8>,
2516    journal_key: [u8; 32],
2517    sync: bool,
2518) -> Result<(), String> {
2519    let directory = directory.to_path_buf();
2520    let session_id = session.id.clone();
2521    let identity = session.identity.clone();
2522    let principal_binding = session.principal_binding();
2523    let workflow = session.workflow.as_str();
2524    tokio::task::spawn_blocking(move || -> Result<(), String> {
2525        use std::io::Write as _;
2526        // Skip the mkdir when the spool dir already
2527        // exists (every request after the first). If the boot-time
2528        // `acquire_spool_lock` root creation was undone (operator
2529        // wipe, mount flip) at runtime, use `create_dir_all_synced`
2530        // here so the recreated tree still lands at 0o700 — same
2531        // discipline the boot path applies.
2532        if !directory.is_dir() {
2533            pv_core::fsutil::create_dir_all_synced(&directory).map_err(|error| error.to_string())?;
2534        }
2535        let digest = pv_core::digest::sha256_hex(session_id.as_bytes());
2536        let stem = digest.get(..32).unwrap_or(&digest);
2537        let metadata_path = directory.join(format!("{stem}.session.json"));
2538        let metadata_payload = serde_json::json!({
2539            "journal_version": 2,
2540            "session_id": session_id,
2541            "identity": identity,
2542            "principal_binding": principal_binding,
2543            "workflow": workflow,
2544        });
2545        // `symlink_metadata` (NOT `exists()`) so a dangling symlink
2546        // at `metadata_path` is detected as "an entry lives here"
2547        // rather than "no entry" — write_atomic's exclusive-create
2548        // temp is elsewhere and its rename REPLACES the symlink at
2549        // the destination (safe) rather than following it, but
2550        // preserving symmetry with the events journal open below
2551        // and future-proofing against a change that switches
2552        // metadata to a non-atomic-rename recipe.
2553        if std::fs::symlink_metadata(&metadata_path).is_err() {
2554            let metadata = crate::journal::seal(&journal_key, "metadata", 0, &metadata_payload)?;
2555            // Use the centralized atomic writer: it uses an RAII guard
2556            // so any intermediate failure (write_all / sync_all /
2557            // rename) cannot leak a zero-byte `.tmp` orphan. A
2558            // repeatedly-failing journal writer used to be able to
2559            // exhaust the ext4 inode table long before disk-full.
2560            //
2561            // Basename the paths in error strings. These
2562            // errors bubble to `tracing::warn!(session = %session.id,
2563            // %error, "capture job failed; session is fail-closed")`
2564            // at process_envelope; that warn exports through
2565            // tracing_opentelemetry -> OTLP -> SIEM, so a single
2566            // disk-full incident used to emit one span per pending
2567            // event with the full absolute journal tree. Basename
2568            // preserves enough context for triage (the stem encodes
2569            // the session id) without leaking the deployment topology.
2570            pv_core::fsutil::write_atomic(&metadata_path, &metadata).map_err(|error| {
2571                format!(
2572                    "write journal metadata {}: {error}",
2573                    pv_core::fsutil::basename(&metadata_path)
2574                )
2575            })?;
2576        } else {
2577            let mut stored: serde_json::Value = crate::journal::open(
2578                &journal_key,
2579                "metadata",
2580                0,
2581                // Cap sealed journal-metadata read at
2582                // MAX_CONTROL_BYTES.
2583                &pv_core::fsutil::read_capped(&metadata_path, pv_core::fsutil::MAX_CONTROL_BYTES)
2584                    .map_err(|error| error.to_string())?,
2585            )?;
2586            // Older anonymous journals have no binding field. They may only
2587            // continue anonymously; a verified principal must never inherit
2588            // a journal whose owner was not durably recorded.
2589            if principal_binding.is_none() && stored.get("principal_binding").is_none() {
2590                if let Some(metadata) = stored.as_object_mut() {
2591                    metadata.insert("principal_binding".into(), Value::Null);
2592                }
2593            }
2594            if stored != metadata_payload {
2595                return Err(
2596                    "journal metadata does not match session workflow, identity, and principal".to_owned(),
2597                );
2598            }
2599        }
2600        let journal_path = directory.join(format!("{stem}.events.ndjson"));
2601        // Track whether the journal is being created by *this* append so
2602        // we can fsync the containing directory once the file exists —
2603        // the metadata fsync above only durably named the metadata
2604        // file, not this new journal file. Without the dirent fsync,
2605        // a power loss on POSIX-conformant filesystems (xfs, btrfs)
2606        // can lose the entry entirely — the file appears not to exist
2607        // on restart, `recover_signed_journals` treats the journal as
2608        // empty, deletes the metadata, and every already-acked event
2609        // becomes orphaned on the broker.
2610        // `symlink_metadata` (NOT `exists()`) so a dangling symlink
2611        // planted at `journal_path` is detected as "an entry lives
2612        // here" rather than "no entry" — pairs with `O_NOFOLLOW`
2613        // below to fail-close instead of following the symlink and
2614        // corrupting an arbitrary path the harness UID can write.
2615        let journal_created = std::fs::symlink_metadata(&journal_path).is_err();
2616        let mut open_options = std::fs::OpenOptions::new();
2617        open_options.create(true).append(true);
2618        #[cfg(unix)]
2619        {
2620            use std::os::unix::fs::OpenOptionsExt as _;
2621            // O_NOFOLLOW: refuse to follow any symlink at
2622            // `journal_path`. Signed events journals authenticate
2623            // via HMAC-per-envelope on read, so a symlink redirect
2624            // wouldn't produce a forged replay, but it would let a
2625            // co-tenant with write access to the spool dir redirect
2626            // the harness's authenticated append into an arbitrary
2627            // path the harness UID can create (a canonical
2628            // symlink-plant confused-deputy). 0o600 mode pairs with
2629            // the same discipline that `write_atomic` uses; here
2630            // the file is APPEND-created rather than atomic-renamed
2631            // so the mode is set inline on the initial open. The
2632            // O_NOFOLLOW constant lives in `pv_core::fsutil` so the
2633            // per-platform value is settled in one auditable place.
2634            open_options
2635                .custom_flags(pv_core::fsutil::unix_o_nofollow())
2636                .mode(0o600);
2637        }
2638        let mut journal = open_options.open(&journal_path).map_err(|error| {
2639            format!(
2640                "open event journal {}: {error}",
2641                pv_core::fsutil::basename(&journal_path)
2642            )
2643        })?;
2644        journal.write_all(&line).map_err(|error| error.to_string())?;
2645        journal.write_all(b"\n").map_err(|error| error.to_string())?;
2646        // Group commit: a same-session batch defers
2647        // this to ONE fdatasync issued by its LAST append —
2648        // `sync_data` flushes all of the file's dirty pages
2649        // regardless of which fd wrote them, so the final synced
2650        // append makes every prior deferred line in the batch
2651        // durable. No job's Phase B runs before the batch sync.
2652        if sync {
2653            journal.sync_data().map_err(|error| error.to_string())?;
2654        }
2655        if journal_created {
2656            pv_core::fsutil::sync_directory(&directory).map_err(|error| {
2657                // Drop the full path entirely; the
2658                // sibling `session = %session.id` field on the
2659                // downstream warn already scopes this to a
2660                // specific session, and the containing directory
2661                // is `atif_spool_dir` — same across the whole
2662                // deployment.
2663                format!("fsync journal directory: {error}")
2664            })?;
2665        }
2666        Ok(())
2667    })
2668    .await
2669    .map_err(|error| error.to_string())?
2670}

2672/// Broker acks append to ONE per-session NDJSON
2673/// journal (`{stem}.acks.ndjson` in the spool root) instead of one
2674/// `write_atomic` file per event under `broker-acks/<hash>/`. Per
2675/// ack this replaces two durable syncs (tmp `sync_all` + parent-dir
2676/// sync) and the tmp-create/rename metadata ops with a single
2677/// appended line + `sync_data`, and stops leaving one file per
2678/// request on disk. Each line is the SAME sealed record bytes the
2679/// per-event file carried (domain "broker-ack", index 0): lines are
2680/// self-describing and order-independent, so reordering/duplication
2681/// grants nothing, and a truncated tail line simply reads as "no
2682/// ack" — the event is re-published, which the bridge tolerates
2683/// (same recovery semantics as a lost ack file).
2684pub(crate) async fn persist_broker_ack(
2685    directory: &std::path::Path,
2686    session_id: &str,
2687    event_uid: &str,
2688    ack: &PublishAck,
2689    journal_key: &[u8; 32],
2690) -> Result<(), String> {
2691    let path = ack_journal_path(directory, session_id);
2692    let record = BrokerAckRecord {
2693        session_id: session_id.to_owned(),
2694        event_uid: event_uid.to_owned(),
2695        ack: ack.clone(),
2696    };
2697    let sealed = crate::journal::seal(journal_key, "broker-ack", 0, &record)?;
2698    tokio::task::spawn_blocking(move || -> Result<(), String> {
2699        use std::io::Write as _;
2700        // `symlink_metadata` (NOT `exists()`) so a dangling symlink
2701        // planted at `path` is detected as "an entry lives here"
2702        // rather than "no entry" — pairs with `O_NOFOLLOW` below to
2703        // fail-close instead of following the symlink and corrupting
2704        // an arbitrary path the harness UID can write. Same posture
2705        // as the events-journal append; `BrokerAckRecord` carries
2706        // plaintext session id + event uid + broker offset, which
2707        // is the "low-entropy identifier" class that the deterministic
2708        // `sha256(session-id)[..32]` stem was already supposed to hide.
2709        let created = std::fs::symlink_metadata(&path).is_err();
2710        let mut open_options = std::fs::OpenOptions::new();
2711        open_options.create(true).append(true);
2712        #[cfg(unix)]
2713        {
2714            use std::os::unix::fs::OpenOptionsExt as _;
2715            open_options
2716                .custom_flags(pv_core::fsutil::unix_o_nofollow())
2717                .mode(0o600);
2718        }
2719        let mut journal = open_options
2720            .open(&path)
2721            .map_err(|error| format!("open ack journal {}: {error}", pv_core::fsutil::basename(&path)))?;
2722        journal.write_all(&sealed).map_err(|error| error.to_string())?;
2723        journal.write_all(b"\n").map_err(|error| error.to_string())?;
2724        journal.sync_data().map_err(|error| error.to_string())?;
2725        if created {
2726            // First ack for this session: make the dirent durable so a
2727            // crash cannot lose the whole journal (same discipline as
2728            // the events journal).
2729            let parent = path
2730                .parent()
2731                .ok_or_else(|| "ack journal has no parent".to_owned())?;
2732            pv_core::fsutil::sync_directory(parent).map_err(|error| error.to_string())?;
2733        }
2734        Ok(())
2735    })
2736    .await
2737    .map_err(|error| error.to_string())?
2738}

2740/// Per-RECORD bound for ack-journal lines. Individual sealed ack
2741/// records are ~300 bytes, so 1 MiB (the MAX_CONTROL_BYTES class) is
2742/// far above any legitimate record while keeping a hostile oversized
2743/// line from ballooning memory. The journal FILE is deliberately
2744/// unbounded: it grows one line per published event for the session's
2745/// whole life, and every previous WHOLE-FILE cap here (1 MiB in the
2746/// first shape, then 16 MiB) recreated the same outage one order of
2747/// magnitude later — the first restart after a busy session crossed
2748/// the cap made EVERY `read_broker_ack` for it fail and recovery
2749/// skipped the session forever, stranding acknowledged evidence.
2750/// Stream line-by-line instead so journal length can never invalidate
2751/// the evidence inside it.
2752const MAX_ACK_RECORD_BYTES: u64 = 1024 * 1024;
2753
2754pub(crate) async fn read_broker_ack(
2755    directory: &std::path::Path,
2756    session_id: &str,
2757    event_uid: &str,
2758    journal_key: &[u8; 32],
2759) -> Result<Option<PublishAck>, String> {
2760    // New layout first: scan the per-session ack journal. Recovery-only
2761    // path, so the linear scan per lookup is acceptable; memory is
2762    // bounded per RECORD (see MAX_ACK_RECORD_BYTES), not per file.
2763    {
2764        let path = ack_journal_path(directory, session_id);
2765        let session = session_id.to_owned();
2766        let uid = event_uid.to_owned();
2767        let key = *journal_key;
2768        let found = tokio::task::spawn_blocking(move || -> Result<Option<PublishAck>, String> {
2769            use std::io::{BufRead as _, Read as _};
2770            let file = match std::fs::File::open(&path) {
2771                Ok(file) => file,
2772                Err(error) if error.kind() == std::io::ErrorKind::NotFound => return Ok(None),
2773                Err(error) => return Err(error.to_string()),
2774            };
2775            let mut reader = std::io::BufReader::new(file);
2776            let mut buf: Vec<u8> = Vec::new();
2777            loop {
2778                buf.clear();
2779                let read = {
2780                    let mut limited = (&mut reader).take(MAX_ACK_RECORD_BYTES);
2781                    limited
2782                        .read_until(b'\n', &mut buf)
2783                        .map_err(|error| error.to_string())?
2784                };
2785                if read == 0 {
2786                    break;
2787                }
2788                if buf.last() != Some(&b'\n') && read as u64 == MAX_ACK_RECORD_BYTES {
2789                    // Oversized line (hostile plant or corruption): drain
2790                    // to its newline in bounded chunks and keep scanning —
2791                    // one bad line must not poison the intact acks around
2792                    // it (same skip posture as the MAC-failure arm below).
2793                    loop {
2794                        buf.clear();
2795                        let drained = {
2796                            let mut limited = (&mut reader).take(MAX_ACK_RECORD_BYTES);
2797                            limited
2798                                .read_until(b'\n', &mut buf)
2799                                .map_err(|error| error.to_string())?
2800                        };
2801                        if drained == 0 || buf.last() == Some(&b'\n') {
2802                            break;
2803                        }
2804                    }
2805                    continue;
2806                }
2807                let line = buf.strip_suffix(b"\n").unwrap_or(&buf);
2808                if line.is_empty() {
2809                    continue;
2810                }
2811                // A torn tail line (crash mid-append) or a tampered line
2812                // must not poison the earlier, intact acks: skip lines
2813                // that fail to open and keep scanning. "No ack found"
2814                // degrades to a re-publish, exactly like a lost file.
2815                let Ok(record) = crate::journal::open::<BrokerAckRecord>(&key, "broker-ack", 0, line) else {
2816                    continue;
2817                };
2818                if record.session_id == session && record.event_uid == uid {
2819                    return Ok(Some(record.ack));
2820                }
2821            }
2822            Ok(None)
2823        })
2824        .await
2825        .map_err(|error| error.to_string())??;
2826        if found.is_some() {
2827            return Ok(found);
2828        }
2829    }
2830    // Legacy layout fallback (older per-event files), so a
2831    // deployment upgraded mid-session still sees its earlier acks.
2832    let path = broker_ack_path(directory, session_id, event_uid);
2833    // Bounded read — same policy as every other sealed marker in the
2834    // spool (close-complete marker, promotion marker, response marker,
2835    // ATIF sidecar). A local fs-tamper attacker plant at
2836    // `spool/broker-acks/<hash>/<hash>.json` would otherwise blow up
2837    // memory during finalize before MAC rejection: `journal::open` calls
2838    // `serde_json::from_slice` on the raw bytes, and a multi-GB file
2839    // reaches serde_json's own limit only after the initial `read` has
2840    // already allocated the whole buffer. `MAX_CONTROL_BYTES` (1 MiB)
2841    // is far above any legitimate ack record.
2842    let bytes = match tokio::task::spawn_blocking({
2843        let path = path.clone();
2844        move || pv_core::fsutil::read_capped(&path, pv_core::fsutil::MAX_CONTROL_BYTES)
2845    })
2846    .await
2847    .map_err(|error| error.to_string())?
2848    {
2849        Ok(bytes) => bytes,
