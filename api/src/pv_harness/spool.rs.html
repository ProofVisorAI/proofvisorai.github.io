15/// Listing a tool-execution folder and creating, removing or renaming
16/// files in it take turns. On macOS, a listing that runs while another
17/// thread removes files from the folder can miss files that nobody
18/// touched, and concurrent closes then left other sessions' files behind.
19/// Each folder has its own turns: work in one never waits for another.
20fn turns(directory: &std::path::Path) -> std::sync::Arc<parking_lot::Mutex<()>> {
21    type Turns = std::collections::HashMap<std::path::PathBuf, std::sync::Arc<parking_lot::Mutex<()>>>;
22    static TURNS: std::sync::LazyLock<parking_lot::Mutex<Turns>> = std::sync::LazyLock::new(Default::default);
23    std::sync::Arc::clone(TURNS.lock().entry(directory.to_path_buf()).or_default())
24}

26/// The paths in the tool-execution folder `directory`: at most
27/// `max + 1`, so a caller can tell that it holds more than `max`.
28pub(crate) fn list_tool_executions(
29    directory: &std::path::Path,
30    max: usize,
31) -> std::io::Result<Vec<std::path::PathBuf>> {
32    let turn = turns(directory);
33    let _turn = turn.lock();
34    std::fs::read_dir(directory)?
35        .take(max.saturating_add(1))
36        .map(|entry| entry.map(|entry| entry.path()))
37        .collect()
38}

40/// Read the tool intent at `path` (at most `MAX_CONTROL_BYTES`) in a
41/// turn of its folder: a claim creates and writes an intent in one turn,
42/// so this never reads one that is still empty.
43pub(crate) fn read_tool_intent(path: &std::path::Path) -> std::io::Result<Vec<u8>> {
44    let turn = turns(path.parent().unwrap_or(std::path::Path::new("")));
45    let _turn = turn.lock();
46    pv_core::fsutil::read_capped(path, pv_core::fsutil::MAX_CONTROL_BYTES)
47}
48
49/// Run `change`, which creates, removes or renames files in the
50/// tool-execution folder `directory`, while no listing of it or intent
51/// read in it runs.
52pub(crate) fn change_tool_executions<T>(directory: &std::path::Path, change: impl FnOnce() -> T) -> T {
53    let turn = turns(directory);
54    let _turn = turn.lock();
55    change()
56}

62    /// Register item 30 sub-clause: "write a one-page 'spool layout
63    /// and recovery' doc". docs/reference/SPOOL-AND-RECOVERY.md exists
64    /// and its file inventory must actually enumerate every real
65    /// on-disk artifact the harness writes. When a new suffix or
66    /// subdirectory constant is added here, this test forces it into
67    /// the doc — otherwise the layout grows silently and an operator
68    /// inspecting a live spool sees files they can't reason about.
69    ///
70    /// Behaviorally verified: adding a new constant here and NOT
71    /// touching the doc fails this test with a message naming the
72    /// exact missing string.
73    #[test]
74    fn spool_constants_are_documented_in_spool_and_recovery() {
75        let doc = include_str!("../../../docs/reference/SPOOL-AND-RECOVERY.md");
76        for (name, value) in [
77            ("OUTBOX", OUTBOX),
78            ("TOOL_EXECUTIONS", TOOL_EXECUTIONS),
79            ("INFLIGHT_RESPONSES", INFLIGHT_RESPONSES),
80            ("TOOL_INTENT_SUFFIX", TOOL_INTENT_SUFFIX),
81            ("TOOL_OUTCOME_SUFFIX", TOOL_OUTCOME_SUFFIX),
82            ("TOOL_AUDITED_SUFFIX", TOOL_AUDITED_SUFFIX),
83        ] {
84            assert!(
85                doc.contains(value),
86                "spool constant {name} = {value:?} is not documented in \
87                 docs/reference/SPOOL-AND-RECOVERY.md. Every on-disk artifact \
88                 the harness writes must appear in the file-inventory table \
89                 so operators inspecting a live spool can identify it."
90            );
91        }
92    }
