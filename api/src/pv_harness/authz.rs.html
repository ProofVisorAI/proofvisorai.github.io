12/// Ed25519 JWT signer for exchange tokens and intent tokens.
13pub struct TokenSigner {
14    key: jsonwebtoken::EncodingKey,
15    kid: String,
16    public_key: [u8; 32],
17}
18
19impl TokenSigner {
20    /// Build from a 32-byte Ed25519 seed. Derives the kid from the
21    /// public key using the same scheme as the receipt signer.
22    pub fn from_seed(seed: &[u8; 32]) -> Self {
23        use pv_receipts::Signer as _;
24        let receipt_signer = pv_receipts::Ed25519Signer::from_seed(seed);
25        let kid = receipt_signer.key_id().to_owned();
26        let public_key = receipt_signer.public_key_bytes();
27        let pkcs8 = ed25519_seed_to_pkcs8(seed);
28        let key = jsonwebtoken::EncodingKey::from_ed_der(&pkcs8);
29        Self { key, kid, public_key }
30    }
31
32    /// Sign claims as an EdDSA JWT with the configured kid.
33    pub fn sign<T: Serialize>(&self, claims: &T) -> Result<String, String> {
34        self.sign_with_typ(claims, "JWT")
35    }
36
37    /// Sign with an explicit JWT `typ` header (RFC 8725 §3.11).
38    pub fn sign_with_typ<T: Serialize>(&self, claims: &T, typ: &str) -> Result<String, String> {
39        let mut header = jsonwebtoken::Header::new(jsonwebtoken::Algorithm::EdDSA);
40        header.kid = Some(self.kid.clone());
41        header.typ = Some(typ.to_owned());
42        jsonwebtoken::encode(&header, claims, &self.key).map_err(|e| e.to_string())
43    }
44
45    /// The key id (first 32 hex chars of SHA-256 of the public key).
46    pub fn kid(&self) -> &str {
47        &self.kid
48    }
49
50    /// The raw Ed25519 public key, base64url without padding (a JWK `x`).
51    pub fn public_key_b64url(&self) -> String {
52        use base64::Engine as _;
53        base64::engine::general_purpose::URL_SAFE_NO_PAD.encode(self.public_key)
54    }
55
56    /// Public verification key for JWTs minted by this signer.
57    pub fn decoding_key(&self) -> jsonwebtoken::DecodingKey {
58        jsonwebtoken::DecodingKey::from_ed_der(&self.public_key)
59    }
60
61    /// Return a JWKS document containing this signer's public key,
62    /// suitable for serving at `/.well-known/jwks.json` so backends
63    /// can verify signed tokens.
64    pub fn jwks_json(&self) -> serde_json::Value {
65        use base64::Engine as _;
66        let x = base64::engine::general_purpose::URL_SAFE_NO_PAD.encode(self.public_key);
67        serde_json::json!({
68            "keys": [{
69                "kty": "OKP",
70                "crv": "Ed25519",
71                "x": x,
72                "kid": self.kid,
73                "alg": "EdDSA",
74                "use": "sig",
75            }]
76        })
77    }
78}

80/// Encode a raw 32-byte Ed25519 seed as a PKCS8v1 DER private key
81/// (the format `jsonwebtoken::EncodingKey::from_ed_der` expects).
82fn ed25519_seed_to_pkcs8(seed: &[u8; 32]) -> Vec<u8> {
83    let mut der = Vec::with_capacity(48);
84    der.extend_from_slice(&[0x30, 0x2e]);
85    der.extend_from_slice(&[0x02, 0x01, 0x00]);
86    der.extend_from_slice(&[0x30, 0x05, 0x06, 0x03, 0x2b, 0x65, 0x70]);
87    der.extend_from_slice(&[0x04, 0x22, 0x04, 0x20]);
88    der.extend_from_slice(seed);
89    der
90}

92/// Result of an authorization decision.
93#[derive(Debug, Clone, PartialEq, Eq)]
94pub enum AuthzDecision {
95    /// The call is permitted under the resolved business intent.
96    Permit {
97        /// The resolved business intent for this tool call.
98        intent: String,
99    },
100    /// The call is denied with a structured code and reason.
101    Deny {
102        /// Machine-readable denial code.
103        code: DenialCode,
104        /// Human-readable explanation.
105        reason: String,
106    },

170/// JWT `typ` of intent tokens. A distinct explicit type (RFC 8725 §3.11)
171/// keeps a backend from accepting an intent token as an access token: both
172/// are signed by the same exchange key and carry the backend as `aud`.
173///
174/// Wire format: backends verify this `typ` (and `av-tool+jwt` for
175/// exchanged tokens); the `av` names predate ProofVisor. Never rename.
176pub const INTENT_TOKEN_TYP: &str = "av-intent+jwt";

178/// The acting agent inside an intent token (RFC 8693 `act` shape).
179#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
180pub struct IntentActor {
181    /// The calling agent's `instance_uid`.
182    pub sub: String,
183}

185/// Claims inside a signed intent token.
186#[derive(Debug, Serialize, Deserialize)]
187#[allow(missing_docs)]
188pub struct IntentClaims {
189    pub iss: String,
190    pub aud: String,
191    /// The human principal the call is attributable to (the caller token's
192    /// `sub`), or the caller's instance id when no identity was validated.
193    pub sub: String,

195    #[serde(default, skip_serializing_if = "Option::is_none")]
196    pub act: Option<IntentActor>,
197    pub tool: String,
198    pub intent: String,
199    /// Ids of the missions that constrained this call.
200    #[serde(default, skip_serializing_if = "Vec::is_empty")]
201    pub missions: Vec<String>,
202    pub iat: u64,
203    pub exp: u64,
204    pub jti: String,
205}

207/// What the policy decision point knows about the caller.
208#[derive(Debug, Clone, Copy, PartialEq, Eq)]
209pub struct CallerFacts<'a> {

218impl CallerFacts<'static> {

227/// The local policy decision point. Evaluates tool calls against the
228/// intent map, the global mission, and every per-agent mission that
229/// selects the caller.
230pub struct PolicyDecisionPoint {
231    intent_map: BTreeMap<String, String>,
232    require_mapping: bool,
233    mission: Option<MissionConfig>,
234    missions: Vec<MissionConfig>,
235    intent_token_ttl_s: u64,
236    issuer: String,
237    signer: Option<Arc<TokenSigner>>,
238}
239
240impl PolicyDecisionPoint {
241    /// Build from harness config with an optional token signer.
242    pub fn from_config(config: &HarnessConfig, signer: Option<Arc<TokenSigner>>) -> Self {
243        Self {
244            intent_map: config.intent_map.clone(),
245            require_mapping: config.require_intent_mapping,
246            mission: config.mission.clone(),
247            missions: config.missions.clone(),
248            intent_token_ttl_s: config.intent_token_ttl_s,
249            issuer: config.audience.clone(),
250            signer,
251        }
252    }
253
254    /// Every mission that constrains `caller`: the global mission, then
255    /// each `[[missions]]` entry whose selectors match.
256    pub fn applicable_missions<'s>(&'s self, caller: &CallerFacts<'_>) -> Vec<&'s MissionConfig> {
257        self.mission
258            .iter()
259            .chain(
260                self.missions
261                    .iter()
262                    .filter(|mission| mission.selects(caller.instance_uid, caller.charter, caller.subject)),
263            )
264            .collect()
265    }
266
267    /// Decide whether `tool` may run at `now_s` for a caller without a
268    /// validated identity. Signs nothing.
269    pub fn decide(&self, tool: &str, now_s: u64) -> AuthzDecision {
270        self.decide_for(tool, now_s, &CallerFacts::ANONYMOUS)
271    }
272
273    /// Decide whether `tool` may run at `now_s` for `caller`. Signs
274    /// nothing, so the pipeline can run it before the budget is charged.
275    /// Missions only narrow: every applicable mission must allow the call.
276    pub fn decide_for(&self, tool: &str, now_s: u64, caller: &CallerFacts<'_>) -> AuthzDecision {
277        // Step 1: resolve intent from tool name.
278        let intent = match self.intent_map.get(tool) {
279            Some(intent) => intent.clone(),
280            None => {
281                if self.require_mapping {
282                    return AuthzDecision::Deny {
283                        code: DenialCode::UnmappedTool,
284                        reason: format!("tool {tool:?} has no configured intent mapping"),
285                    };
286                }
287                tool.to_owned()
288            }
289        };
290
291        // Step 2: check every applicable mission.
292        for mission in self.applicable_missions(caller) {
293            if now_s >= mission.expires_at {
294                return AuthzDecision::Deny {
295                    code: DenialCode::MissionExpired,
296                    reason: format!(
297                        "mission {:?} expired at {} (now {})",
298                        mission.id, mission.expires_at, now_s
299                    ),
300                };
301            }
302            if !mission.allowed_intents.iter().any(|ai| ai == &intent) {
303                return AuthzDecision::Deny {
304                    code: DenialCode::MissionDenied,
305                    reason: format!("intent {intent:?} not permitted by mission {:?}", mission.id),
306                };
307            }
308        }
309
310        AuthzDecision::Permit { intent }
311    }
312
313    /// The business intent `tool` maps to: its `intent_map` entry, or the
314    /// tool name itself. A plain lookup that never denies.
315    pub fn intent_for(&self, tool: &str) -> String {
316        self.intent_map
317            .get(tool)
318            .cloned()
319            .unwrap_or_else(|| tool.to_owned())
320    }
321
322    /// Sign a short-lived intent token for a permitted call, bound to the
323    /// backend that receives it. `sub` is the human principal and `act.sub`
324    /// the calling agent; the token never outlives an applicable mission.
325    /// `None` when no signer is configured.
326    pub fn mint_intent_token(
327        &self,
328        tool: &str,
329        intent: &str,
330        caller: &CallerFacts<'_>,
331        audience: &str,
332        now_s: u64,
333    ) -> Option<String> {
334        let signer = self.signer.as_ref()?;
335        let missions = self.applicable_missions(caller);
336        let exp = missions
337            .iter()
338            .map(|mission| mission.expires_at)
339            .fold(now_s.saturating_add(self.intent_token_ttl_s), u64::min);
340        let claims = IntentClaims {
341            iss: self.issuer.clone(),
342            aud: audience.to_owned(),
343            sub: caller.subject.unwrap_or(caller.instance_uid).to_owned(),
344            act: caller.subject.map(|_| IntentActor {
345                sub: caller.instance_uid.to_owned(),
346            }),
347            tool: tool.to_owned(),
348            intent: intent.to_owned(),
349            missions: missions.iter().map(|mission| mission.id.clone()).collect(),
350            iat: now_s,
351            exp,
352            jti: pv_core::ids::new_event_uid(),
353        };
354        match signer.sign_with_typ(&claims, INTENT_TOKEN_TYP) {
355            Ok(jwt) => Some(jwt),
356            Err(e) => {
357                tracing::error!(error = %e, "intent token signing failed");
358                None
359            }
360        }
361    }
362
363    /// The configured intent token TTL.
364    pub fn intent_token_ttl_s(&self) -> u64 {
365        self.intent_token_ttl_s
366    }
367}

511    fn verify_intent_token(signer: &TokenSigner, jwt: &str) -> jsonwebtoken::TokenData<IntentClaims> {
512        let dk = signer.decoding_key();
513        let mut validation = jsonwebtoken::Validation::new(jsonwebtoken::Algorithm::EdDSA);
514        validation.validate_exp = false;
515        validation.validate_aud = false;
516        validation.required_spec_claims.clear();
517        jsonwebtoken::decode::<IntentClaims>(jwt, &dk, &validation).unwrap()
518    }

642    fn intent_token_is_typed_and_bound_to_issuer_audience_and_subject() {
643        let signer = Arc::new(TokenSigner::from_seed(&[99u8; 32]));
644        let mut p = pdp(&[("search", "data.read")], false, None);
645        p.signer = Some(Arc::clone(&signer));
646        let jwt = p
647            .mint_intent_token("search", "data.read", &facts(), "backend-a", 1000)
648            .expect("signer present → must produce JWT");
649        let header = jsonwebtoken::decode_header(&jwt).unwrap();
650        assert_eq!(header.alg, jsonwebtoken::Algorithm::EdDSA);
651        assert_eq!(header.typ.as_deref(), Some(INTENT_TOKEN_TYP));
652        assert_eq!(header.kid.as_deref(), Some(signer.kid()));
653        let decoded = verify_intent_token(&signer, &jwt);
654        assert_eq!(decoded.claims.iss, "proofvisor");
655        assert_eq!(decoded.claims.aud, "backend-a");
656        assert_eq!(decoded.claims.sub, "user:alice@corp.com", "sub is the human");
657        assert_eq!(
658            decoded.claims.act,
659            Some(IntentActor { sub: "inst-1".into() }),
660            "act.sub is the calling agent"
661        );
662        assert_eq!(decoded.claims.tool, "search");
663        assert_eq!(decoded.claims.intent, "data.read");
664        assert_eq!(decoded.claims.iat, 1000);
665        assert_eq!(decoded.claims.exp - decoded.claims.iat, 60);
666        assert!(!decoded.claims.jti.is_empty());
