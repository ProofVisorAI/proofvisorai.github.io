12/// Workflow kind (brief Modules G/H).
13#[derive(Debug, Clone, Copy, PartialEq, Eq)]
14#[non_exhaustive]
15pub enum Workflow {
16    /// Consequential actions: event chain + receipt at close.

18    /// Exploratory: ATIF capture, receipt only on promotion.
19    Unsigned,
20}

25        match self {
26            Self::Signed => "signed",
27            Self::Unsigned => "unsigned",

41/// A live session.
42pub struct Session {
43    /// Session id.
44    pub id: String,
45    /// A per-open UUID that disambiguates
46    /// recycled session ids in the off-path vector store. Client
47    /// callers may reuse the same `id` after a finalize; the
48    /// `SessionRegistry::get_or_open` path historically returned a
49    /// fresh `Session` under the same key. A Qdrant vector search
50    /// filtered ONLY on `session_id` would then return the prior
51    /// incarnation's vectors, causing false semantic-loop signals
52    /// from an unrelated past session. Every vector record and
53    /// query is now scoped by
54    /// `session_scope = "{id}#{generation_uid}"` — the UUID is
55    /// fresh for every `Session::new`, so a recycled id lands in
56    /// a distinct scope. The bare `id` remains the primary key for
57    /// on-disk artifacts (spool paths, journal filenames), which
58    /// use crash-safe generation logic of their own.
59    pub session_scope: String,
60    /// Workflow kind.
61    pub workflow: Workflow,
62    /// Agent identity bound at open (from NHI validation).
63    pub identity: AgentIdentity,
64    /// Most recently validated identity, including current TTL.
65    latest_identity: Mutex<AgentIdentity>,
66    principal_binding: Mutex<Option<Option<String>>>,
67    /// Monotonic per-session event sequence (authoritative order).
68    seq: AtomicU64,
69    /// Monotonic authenticated-journal record index.
70    journal_index: AtomicU64,
71    /// Loop-breaker state.
72    pub loop_state: pv_loopdetect::SessionLoopState,
73    /// The model's previous output (bounded), for the semantic judge's
74    /// second opinion on the loop breaker. Kept only while a judge is
75    /// attached.
76    last_model_output: parking_lot::Mutex<Option<String>>,
77    /// Signed workflow: incremental event chain.
78    pub chain: Mutex<EventChain>,
79    /// Unsigned workflow: ATIF steps.
80    pub atif: Mutex<pv_atif::TrajectoryBuilder>,
81    /// Aggregates for the receipt.
82    pub totals: Totals,
83    /// Last-activity timestamp (dashboard/display only), epoch ms.
84    /// Wall-clock time subject to VM/NTP jumps; not safe to use for

99    /// `last_stop_reason_id`, which later successful responses
100    /// overwrite and which quota-BACKEND failures and blocked tool
101    /// calls also write, making it an unreliable proxy for "enforcement

106    pub closed: AtomicU64,

114    pub receipt: Mutex<Option<pv_receipts::Receipt>>,
115    /// Persisted unsigned ATIF artifact.
116    pub atif_path: Mutex<Option<PathBuf>>,
117    /// Set once an unsigned artifact is promoted.
118    promoted: AtomicU64,
119    /// Worker jobs accepted but not yet fully captured.
120    pending_jobs: AtomicU64,
121    /// Notification used by finalization to await a drained worker queue.
122    jobs_drained: tokio::sync::Notify,
123    /// Background semantic-judge verdicts not yet recorded. Kept apart
124    /// from `pending_jobs` so that requests waiting for their own audit
125    /// record never wait for the judge; only finalization waits for both.
126    background_jobs: AtomicU64,
127    /// Notification used by finalization to await background verdicts.
128    background_drained: tokio::sync::Notify,
129    /// Set when an upstream action could not be captured completely.
130    capture_failed: AtomicU64,
131    /// Automatic promotions still owed for a semantic-judge trace verdict
132    /// in this unsigned session: attempts left (0 = none owed).
133    judged_promotion: AtomicU64,
134    /// RAM-cliff guard: count of unsigned ATIF steps
135    /// journaled by the worker. The steps themselves live ONLY in the
136    /// events journal; close rebuilds the trajectory from disk.
137    atif_steps: AtomicU64,

139    /// holder of Open/Draining/Sealed/Complete, advanced only via
140    /// CAS `transition` calls that name their legal source states.

147/// The session lifecycle chain, held in
148/// one `AtomicU8` and advanced only through CAS transitions that
149/// name their expected source state — illegal transitions become
150/// impossible rather than untested.
151///
152/// Two flags deliberately stay OUTSIDE the chain (mirroring the S2
153/// plan's `admission_open` exception): `capture_failed` and
154/// `promoted` are orthogonal *properties*, not chain positions — a
155/// capture-failed session still seals and completes (the
156/// capture-failed seal path), and promotion happens to already-
157/// complete sessions. Folding them in would multiply states for
158/// combinations that all remain reachable.
159#[repr(u8)]
160#[derive(Debug, Clone, Copy, PartialEq, Eq)]
161#[non_exhaustive]
162pub enum SessionState {
163    /// Admitting requests.
164    Open = 0,
165    /// `try_close` claimed the close; streams may still be draining.
166    /// A failed finalize returns to `Open` via `reset_close`.
167    Draining = 1,
168    /// The receipt/ATIF artifact is durably committed; the fallible
169    /// close tail (bridge emit + journal removal) is still pending.
170    Sealed = 2,
171    /// The close ran to full completion; the registry may evict.
172    Complete = 3,
173}
174
175impl SessionState {
176    fn decode(raw: u8) -> Self {
177        match raw {
178            1 => Self::Draining,
179            2 => Self::Sealed,
180            3 => Self::Complete,
181            _ => Self::Open,
182        }
183    }
184}

355    /// Attempt to claim the close transition. Only one caller can hold the
356    /// claim at a time; a failed finalize resets it (`reset_close`) so the
357    /// close can be retried.
358    pub fn try_close(&self) -> bool {
359        let claimed = self
360            .closed
361            .compare_exchange(0, 1, Ordering::AcqRel, Ordering::Acquire)
362            .is_ok();
363        if claimed {
364            self.transition(&[SessionState::Open], SessionState::Draining);
365        }
366        claimed
367    }

369    /// The lifecycle chain position.
370    pub fn lifecycle_state(&self) -> SessionState {
371        SessionState::decode(self.lifecycle.load(Ordering::Acquire))
372    }

374    /// One CAS transition of the lifecycle state machine (S2 step 3:
375    /// the enum IS the state; the mirror flags are deleted), tolerant
376    /// of two documented behaviors of the close machinery:
377    ///
378    /// * **Idempotent/late re-marks** — a current state at or past
379    ///   `to` in the chain is a no-op. This covers idempotent flag
380    ///   stores AND the `closed` claim flag's post-seal semantics:
381    ///   `reset_close`/`try_close` toggle the claim after an
382    ///   artifact is already committed (failed close-tail retries),
383    ///   where `is_closed()` stays true via `artifact_committed`
384    ///   and the chain must stay Sealed.
385    /// * **Multiple legal sources** — signed recovery seals adopted
386    ///   sessions without a prior claim (Open→Sealed); a live close
387    ///   seals under its claim (Draining→Sealed).
388    ///
389    /// Any other source state is a model violation: the transition
390    /// is REFUSED (the CAS never fires — illegal transitions are
391    /// impossible, not merely untested), and debug builds (the
392    /// entire test suite) additionally panic naming the transition.
393    fn transition(&self, from: &[SessionState], to: SessionState) {
394        let mut current = self.lifecycle.load(Ordering::Acquire);
395        loop {
396            let state = SessionState::decode(current);
397            if from.contains(&state) {
398                match self.lifecycle.compare_exchange_weak(
399                    current,
400                    to as u8,
401                    Ordering::AcqRel,
402                    Ordering::Acquire,
403                ) {
404                    Ok(_) => break,
405                    Err(actual) => current = actual,
406                }
407            } else if state == to || (state as u8) > (to as u8) {
408                break;
409            } else {
410                debug_assert!(
411                    false,
412                    "illegal lifecycle transition {state:?}->{to:?} (expected from {from:?})"
413                );
414                break;
415            }
416        }
417    }

485    pub(crate) async fn wait_for_streams(&self) {
486        loop {
487            // Subscribe first (via enable), then check the guard; the same Notified
488            // must remain pinned through the await, or notify_waiters() firing in
489            // the interval between drop and re-subscribe is lost forever.
490            let notified = self.streams_drained.notified();
491            let mut notified = std::pin::pin!(notified);
492            notified.as_mut().enable();
493            if self.active_streams.load(Ordering::Acquire) == 0 {
494                return;
495            }
496            notified.await;
497        }
498    }

534    /// Recreate a closed unsigned session from a persisted ATIF trajectory.
535    ///
536    /// `next_seq` is the number of per-session events already published to
537    /// the bridge for this session (steps ↔ journal records are 1:1, so the
538    /// step count is exact). Without restoring it, an adopted session
539    /// reports `peek_seq() == 0` and a post-crash SESSION_CLOSE or
540    /// retroactive-receipt event is minted with `metadata.sequence = 0`,
541    /// colliding with the session's first step event already on the bridge
542    /// (both sibling recovery paths call `restore_next_seq`; this one used
543    /// to be the exception).
544    pub fn recover_unsigned(
545        id: String,
546        identity: AgentIdentity,
547        breaker: pv_loopdetect::BreakerConfig,
548        path: PathBuf,
549        metrics: Option<&pv_atif::FinalMetrics>,
550        next_seq: u64,
551    ) -> Result<Self, String> {
552        Self::recover_unsigned_with_principal(id, identity, breaker, path, metrics, next_seq, None)
553    }

555    /// Recover an authenticated ATIF artifact with its persisted owner. Legacy
556    /// artifacts without an owner bind anonymously, so a later authenticated
557    /// caller cannot claim them merely by sharing an agent instance id.
558    pub(crate) fn recover_unsigned_with_principal(
559        id: String,
560        identity: AgentIdentity,
561        breaker: pv_loopdetect::BreakerConfig,
562        path: PathBuf,
563        metrics: Option<&pv_atif::FinalMetrics>,
564        next_seq: u64,
565        principal: Option<&str>,
566    ) -> Result<Self, String> {
567        let session = Self::new(id, Workflow::Unsigned, identity, breaker);
568        session.bind_principal(principal);
569        session.restore_next_seq(next_seq);
570        // A recovered artifact is by definition past close: claim the
571        // close (fresh session — the claim cannot fail) and seal, so
572        // the lifecycle chain walks the same Open→Draining→Sealed
573        // path a live close does.
574        let claimed = session.try_close();
575        debug_assert!(claimed, "recover_unsigned starts from a fresh session");
576        session.mark_artifact_committed();
577        *session.atif_path.lock() = Some(path);
578        if let Some(metrics) = metrics {
579            let prompt_tokens = recovered_counter(metrics.total_prompt_tokens, "prompt tokens")?;
580            let completion_tokens = recovered_counter(metrics.total_completion_tokens, "completion tokens")?;
581            let cached_tokens = recovered_counter(metrics.total_cached_tokens, "cached tokens")?;
582            session
583                .totals
584                .prompt_tokens
585                .store(prompt_tokens, Ordering::Release);
586            session
587                .totals
588                .completion_tokens
589                .store(completion_tokens, Ordering::Release);
590            session
591                .totals
592                .cached_tokens
593                .store(cached_tokens, Ordering::Release);
594            if let Some(extra) = metrics.extra.as_ref() {
595                // Cost has two possible sources on the persisted extra:
596                //   1. `cost_usd_micros` — the current writer's exact
597                //      u64 field.
598                //   2. missing / null — an older writer's format,
599                //      pre-dating the cost_usd_micros key.
600                // Falling straight to `recovered_counter(...)?` on the
601                // missing key returned 0, silently zeroing the session's
602                // recovered cost even when `metrics.total_cost_usd` (the
603                // ATIF-level fallback consumed by the non-extra branch
604                // below) was present. Prefer the exact u64 when it's
605                // there, otherwise use the ATIF-level float — same logic
606                // as the pre-extra branch.
607                let cost_usd_micros = match extra.get("cost_usd_micros").and_then(serde_json::Value::as_u64) {
608                    Some(v) => recovered_counter(Some(v), "cost")?,
609                    None => cost_from_atif_total(metrics.total_cost_usd)?,
610                };
611                session
612                    .totals
613                    .cost_usd_micros
614                    .store(cost_usd_micros, Ordering::Release);
615                let tool_calls = recovered_counter(
616                    extra.get("tool_calls").and_then(serde_json::Value::as_u64),
617                    "tool calls",
618                )?;
619                let tool_allowed = recovered_counter(
620                    extra.get("tool_allowed").and_then(serde_json::Value::as_u64),
621                    "allowed tools",
622                )?;
623                let tool_blocked = recovered_counter(
624                    extra.get("tool_blocked").and_then(serde_json::Value::as_u64),
625                    "blocked tools",
626                )?;
627                if tool_allowed
628                    .checked_add(tool_blocked)
629                    .is_none_or(|classified| classified > tool_calls)
630                {
631                    return Err("recovered tool accounting is inconsistent".to_owned());
632                }
633                session.totals.tool_calls.store(tool_calls, Ordering::Release);
634                session.totals.tool_allowed.store(tool_allowed, Ordering::Release);
635                session.totals.tool_blocked.store(tool_blocked, Ordering::Release);
636                if let Some(id) = extra.get("stop_reason_id").and_then(serde_json::Value::as_u64) {
637                    if id > u64::from(u8::MAX) {
638                        return Err("recovered stop reason exceeds u8".to_owned());
639                    }
640                    session.last_stop_reason_id.store(id, Ordering::Release);
641                }
642                // Latch restoration across restart, from the EXPLICITLY
643                // persisted latch value — never inferred from the stop
644                // reason, which Inject-action breaker trips also record
645                // without latching live (a restart used to upgrade such
646                // deliberately non-terminal trips into permanent id
647                // lockouts). Absent key (pre-latch artifacts,
648                // consolidation-rebuilt artifacts) means no latch.
649                if let Some(latched) = extra
650                    .get("enforcement_latched")
651                    .and_then(serde_json::Value::as_u64)
652                {
653                    if latched > u64::from(u8::MAX) {
654                        return Err("recovered enforcement latch exceeds u8".to_owned());
655                    }
656                    if latched != 0 {
657                        session.enforcement_tripped.store(latched, Ordering::Release);
658                    }
659                }
660            } else if let Some(cost) = metrics.total_cost_usd {
661                session
662                    .totals
663                    .cost_usd_micros
664                    .store(cost_from_atif_total(Some(cost))?, Ordering::Release);
665            }
666        }
667        Ok(session)
668    }

931    /// Build the receipt body for this session (signed close or promotion).
932    pub fn receipt_body(
933        &self,
934        subject: pv_receipts::ReceiptSubject,
935        stop: StopReason,
936    ) -> pv_receipts::ReceiptBody {
937        let recorded_id = self.last_stop_reason_id.load(Ordering::Acquire);
938        let recorded = StopReason::from_id(u8::try_from(recorded_id).unwrap_or(0));
939        // Distinguish "never recorded" (raw id 0) from "recorded an id
940        // this build cannot map" (nonzero → `from_id` says Unknown —
941        // a NEWER node's stop-reason id recovered from its artifact).
942        // Only the former may fall back to the caller's stop: rewriting
943        // a foreign id into e.g. SessionClosed would forge a specific
944        // reason the session never had into the SIGNED receipt. The
945        // honest representation for a foreign id is Unknown.
946        let stop = if recorded_id == 0 { stop } else { recorded };
947        pv_receipts::receipt::new_body(
948            self.id.clone(),
949            self.current_identity(),
950            subject,
951            pv_receipts::ToolCallSummary {
952                total: self.totals.tool_calls.load(Ordering::Acquire),
953                allowed: self.totals.tool_allowed.load(Ordering::Acquire),
954                blocked: self.totals.tool_blocked.load(Ordering::Acquire),
955            },
956            pv_receipts::CostSummary {
957                prompt_tokens: self.totals.prompt_tokens.load(Ordering::Acquire),
958                completion_tokens: self.totals.completion_tokens.load(Ordering::Acquire),
959                cached_tokens: self.totals.cached_tokens.load(Ordering::Acquire),
960                cost_usd_micros: self.totals.cost_usd_micros.load(Ordering::Acquire),
961            },
962            stop,
963        )
964    }

967fn recovered_counter(value: Option<u64>, field: &str) -> Result<u64, String> {
968    let value = value.unwrap_or(0);
969    if value > pv_core::error::JCS_SAFE_MAX {
970        return Err(format!("recovered {field} exceeds JCS-safe bounds"));
971    }

975/// Convert an ATIF-level `total_cost_usd` float (dollars) into u64
976/// micros with the same finite/nonnegative/JCS-safe guards the
977/// non-extra branch enforces inline. Returns 0 for a missing value,
978/// mirroring `recovered_counter`. Shared between the two recovery
979/// branches so a cross-version artifact — extra present but without
980/// `cost_usd_micros`, with `total_cost_usd` at the ATIF level — no
981/// longer silently zeroes the recovered cost.
982fn cost_from_atif_total(cost: Option<f64>) -> Result<u64, String> {
983    let Some(cost) = cost else {
984        return Ok(0);
985    };
986    if !cost.is_finite() || cost < 0.0 {
987        return Err("recovered cost is not finite and nonnegative".to_owned());
988    }
989    let micros = (cost * pv_core::units::USD_MICROS_PER_DOLLAR as f64).round();
990    if micros > pv_core::error::JCS_SAFE_MAX as f64 {
991        return Err("recovered cost exceeds JCS-safe bounds".to_owned());
992    }
993    #[allow(clippy::cast_possible_truncation, clippy::cast_sign_loss)]
994    Ok(micros as u64)
995}

1033    /// Get or open a session.
1034    ///
1035    /// If an entry exists but has completed close (receipt/ATIF durably
1036    /// committed, journal removed) and has not yet been reaped by the
1037    /// idle sweeper, treat the id as free and open a fresh session —
1038    /// otherwise a well-behaved client that reuses a session id after a
1039    /// server-side close (retry after a 5xx, network partition, TTL
1040    /// refresh) would see `400 session is already closed` for the entire
1041    /// eviction window. A session that started close but has not yet
1042    /// finished it (`close_complete = 0`) is *not* replaced: reopening
1043    /// would race the in-flight close and split the audit trail.
1044    ///
1045    /// This is the right shape for **chat requests**: the client is
1046    /// starting a new turn and it's fine to give them a fresh state
1047    /// under the same id. For **tool interception**, use
1048    /// [`Self::get_or_open_no_reopen`] — a tool call references an
1049    /// in-progress conversation, and silently resurrecting a closed
1050    /// session would let the client extend the audit trail past its
1051    /// signed receipt.

1053        &self,

1056        identity: &AgentIdentity,
1057        breaker: &pv_loopdetect::BreakerConfig,

1060            id, workflow, identity, breaker, /* reopen_after_close */ true,
1061        )
1062    }

1066    /// sees `is_closed() == true` and can refuse with `BadRequest`.

1081    fn get_or_open_inner(
1082        &self,
1083        id: &str,
1084        workflow: Workflow,
1085        identity: &AgentIdentity,
1086        breaker: &pv_loopdetect::BreakerConfig,
1087        reopen_after_close: bool,
1088    ) -> Arc<Session> {
1089        use dashmap::mapref::entry::Entry;
1090        match self.sessions.entry(id.to_owned()) {
1091            Entry::Occupied(mut occupied) => {
1092                if reopen_after_close && occupied.get().close_complete_flag() {
1093                    // Enforcement-triggered closes must NOT silently
1094                    // recycle: a session closed because its token budget
1095                    // was exhausted (or its loop breaker aborted) would
1096                    // otherwise reopen under the SAME id with a fresh
1097                    // budget and fresh breaker — verified live as a
1098                    // strict 200/403/200/403 alternation that rides the
1099                    // per-session cap at ~50% duty forever. Per-session
1100                    // budgets cannot stop a client spreading across NEW
1101                    // ids (inherent to per-session semantics), but the
1102                    // same id must stay terminally refused so each
1103                    // incarnation is an explicit, distinct audit entity.
1104                    // Keyed on the sticky enforcement latch, not the
1105                    // last-write-wins stop reason — later responses
1106                    // overwrite that record, and non-enforcement
1107                    // failures (quota-backend errors, blocked tool
1108                    // calls) also write it, making it an unreliable
1109                    // proxy in both directions.
1110                    if occupied.get().enforcement_tripped() {
1111                        return occupied.get().clone();
1112                    }
1113                    let fresh = Arc::new(Session::new(
1114                        id.to_owned(),
1115                        workflow,
1116                        identity.clone(),
1117                        breaker.clone(),
1118                    ));
1119                    occupied.insert(Arc::clone(&fresh));
1120                    fresh
1121                } else {
1122                    occupied.get().clone()
1123                }
1124            }
1125            Entry::Vacant(vacant) => {
1126                let fresh = Arc::new(Session::new(
1127                    id.to_owned(),
1128                    workflow,
1129                    identity.clone(),
1130                    breaker.clone(),
1131                ));
1132                vacant.insert(Arc::clone(&fresh));
1133                fresh
1134            }
1135        }
1136    }

1188    /// Evict signed sessions whose close ran to full completion and that
1189    /// have been idle longer than `idle_s`, returning the evicted sessions.
1190    ///
1191    /// Only signed sessions whose close *fully completed* are eligible: a
1192    /// completed close removed the on-disk journal, so nothing re-inserts
1193    /// them, and a later request or lifecycle call for the id behaves
1194    /// exactly as it would after a process restart. `close_complete` (not
1195    /// `artifact_committed`, which is set before the fallible bridge emits
1196    /// and journal removal) is the gate — a failed or in-flight close must
1197    /// stay resident, or a client reusing the id could open a fresh session
1198    /// whose journal appends collide with the still-on-disk records.
1199    /// Unsigned sessions must stay resident — the recovery scan re-inserts
1200    /// them from their spool artifact on the next tick anyway, and evicting
1201    /// one lets a client reuse its id against the still-present artifact and
1202    /// provenance files, poisoning the new incarnation's close. Capture-failed
1203    /// (quarantined) sessions also stay: they are bounded by real crash
1204    /// events and their in-registry seal is what keeps the fail-closed
1205    /// refusal cheap. Without eviction the registry grows by one entry per
1206    /// client-chosen session id for the process lifetime. The three
1207    /// client-mintable retained classes (enforcement-latched,
1208    /// capture-failed, empty-unsigned quarantine) are each bounded by an
1209    /// overflow cap below.
1210    pub fn evict_finalized(&self, idle_s: u64) -> Vec<Arc<Session>> {
1211        // Idle comparison uses the monotonic clock so a
1212        // forward wall-clock jump (VM resume, NTP correction) cannot
1213        // make finalized sessions look premature-idle. Wall-clock
1214        // `last_activity_ms` stays available for dashboard display.
1215        let idle_duration = Duration::from_secs(idle_s);
1216        let mut evicted = Vec::new();
1217        self.sessions.retain(|_, session| {
1218            let evict = session.workflow == Workflow::Signed
1219                && session.close_complete_flag()
1220                && !session.capture_failed()
1221                // Enforcement-latched sessions are retained like
1222                // capture-failed ones: evicting them vacated the id, and
1223                // the next get_or_open built a fresh session with a
1224                // fresh budget/breaker — the same-id refusal silently
1225                // expired after idle_s for signed workflows. Bounded by
1226                // the overflow pass below (unlike capture_failed, the
1227                // latch is client-mintable at line rate — one over-cap
1228                // request per fresh id — so retention needs a cap).
1229                && !session.enforcement_tripped()
1230                && session.active_streams.load(Ordering::Acquire) == 0
1231                && session.pending_jobs.load(Ordering::Acquire) == 0
1232                && session.idle_duration() >= idle_duration;
1233            if evict {
1234                evicted.push(Arc::clone(session));
1235            }
1236            !evict
1237        });
1238        // Overflow pass: cap latched-retained signed sessions. Beyond the
1239        // cap, evict oldest-first — the same-id refusal expires for the
1240        // evicted ids (degraded to the pre-latch idle-bounded behavior),
1241        // which is the accepted cost of bounding client-mintable registry
1242        // growth. Unsigned latched sessions are governed by the existing
1243        // unsigned retention posture and are not evicted here.
1244        const MAX_LATCHED_RETAINED: usize = 4096;
1245        let mut latched: Vec<Arc<Session>> = self
1246            .sessions
1247            .iter()
1248            .filter(|entry| {
1249                entry.workflow == Workflow::Signed
1250                    && entry.close_complete_flag()
1251                    && entry.enforcement_tripped()
1252                    && !entry.capture_failed()
1253                    && entry.active_streams.load(Ordering::Acquire) == 0
1254                    && entry.pending_jobs.load(Ordering::Acquire) == 0
1255            })
1256            .map(|entry| Arc::clone(&entry))
1257            .collect();
1258        if latched.len() > MAX_LATCHED_RETAINED {
1259            let mut with_idle: Vec<_> = latched
1260                .into_iter()
1261                .map(|s| {
1262                    let d = s.idle_duration();
1263                    (s, d)
1264                })
1265                .collect();
1266            with_idle.sort_by_key(|(_, d)| std::cmp::Reverse(*d));
1267            latched = with_idle.into_iter().map(|(s, _)| s).collect();
1268            let excess = latched.len() - MAX_LATCHED_RETAINED;
1269            tracing::warn!(
1270                excess,
1271                cap = MAX_LATCHED_RETAINED,
1272                "latched-retained signed sessions exceed the cap; evicting oldest \
1273                 (their same-id enforcement refusal expires)"
1274            );
1275            for session in latched.into_iter().take(excess) {
1276                self.sessions.remove(&session.id);
1277                evicted.push(session);
1278            }
1279        }
1280        // Overflow pass: cap capture-failed (quarantined) session
1281        // retention. Same shape as the latched pass above but for a
1282        // distinct class: capture-failed sessions are supposed to be
1283        // bounded by real crash events, but the paired
1284        // `Finalizer::quarantined_sessions` set is now FIFO-capped at
1285        // MAX_QUARANTINED_SESSIONS. Their in-registry retention needed
1286        // a matching cap to prevent the Session Arcs from growing
1287        // indefinitely across a long-lived process's lifetime. Beyond
1288        // the cap, evict oldest-first — the quarantine artifact on
1289        // disk is what fundamentally gates resurrection, so evicting
1290        // the in-memory Arc is safe (a subsequent recovery pass
1291        // re-inserts if the artifact still lives on disk).
1292        const MAX_CAPTURE_FAILED_RETAINED: usize = 4096;
1293        let mut capture_failed: Vec<Arc<Session>> = self
1294            .sessions
1295            .iter()
1296            .filter(|entry| {
1297                // artifact_committed is the terminal flag the
1298                // capture-failed close path sets (close_session_locked
1299                // seals the session and returns CaptureIncomplete
1300                // WITHOUT ever reaching mark_close_complete — that
1301                // marker belongs to successful closes). Requiring
1302                // close_complete here meant NO capture-failed session
1303                // ever qualified: the cap existed but matched nothing,
1304                // and sustained capture failures grew the registry
1305                // without bound.
1306                entry.artifact_committed_flag()
1307                    && entry.capture_failed()
1308                    && entry.active_streams.load(Ordering::Acquire) == 0
1309                    && entry.pending_jobs.load(Ordering::Acquire) == 0
1310            })
1311            .map(|entry| Arc::clone(&entry))
1312            .collect();
1313        if capture_failed.len() > MAX_CAPTURE_FAILED_RETAINED {
1314            let mut with_idle: Vec<_> = capture_failed
1315                .into_iter()
1316                .map(|s| {
1317                    let d = s.idle_duration();
1318                    (s, d)
1319                })
1320                .collect();
1321            with_idle.sort_by_key(|(_, d)| std::cmp::Reverse(*d));
1322            capture_failed = with_idle.into_iter().map(|(s, _)| s).collect();
1323            let excess = capture_failed.len() - MAX_CAPTURE_FAILED_RETAINED;
1324            tracing::warn!(
1325                excess,
1326                cap = MAX_CAPTURE_FAILED_RETAINED,
1327                "capture-failed sessions exceed the retention cap; evicting oldest \
1328                 (their same-id fail-closed refusal expires until on-disk quarantine \
1329                 artifact re-inserts on the next recovery tick)"
1330            );
1331            for session in capture_failed.into_iter().take(excess) {
1332                self.sessions.remove(&session.id);
1333                evicted.push(session);
1334            }
1335        }
1336        // Overflow pass: cap empty-unsigned quarantine retention. These
1337        // are the reconciler's "no captured steps" refusals — retained
1338        // so the same-id refusal stays cheap and the incident evidence
1339        // survives, but they are client-mintable at line rate (open an
1340        // unsigned session with a fresh id, submit no work, let the
1341        // idle sweep close it) and, having captured nothing, they leave
1342        // NO spool artifact behind: nothing on disk re-inserts or
1343        // collides after eviction. Without a cap the registry grows by
1344        // one permanent `Arc<Session>` per aborted admission for the
1345        // process lifetime. Same accepted tradeoff as the two sibling
1346        // caps: beyond the cap, evict oldest-first and let the evicted
1347        // ids' refusal expire.
1348        const MAX_EMPTY_UNSIGNED_RETAINED: usize = 4096;
1349        let mut empty_quarantined: Vec<Arc<Session>> = self
1350            .sessions
1351            .iter()
1352            .filter(|entry| {
1353                entry.is_empty_unsigned_quarantine()
1354                    && entry.active_streams.load(Ordering::Acquire) == 0
1355                    && entry.pending_jobs.load(Ordering::Acquire) == 0
1356            })
1357            .map(|entry| Arc::clone(&entry))
1358            .collect();
1359        if empty_quarantined.len() > MAX_EMPTY_UNSIGNED_RETAINED {
1360            let mut with_idle: Vec<_> = empty_quarantined
1361                .into_iter()
1362                .map(|s| {
1363                    let d = s.idle_duration();
1364                    (s, d)
1365                })
1366                .collect();
1367            with_idle.sort_by_key(|(_, d)| std::cmp::Reverse(*d));
1368            empty_quarantined = with_idle.into_iter().map(|(s, _)| s).collect();
1369            let excess = empty_quarantined.len() - MAX_EMPTY_UNSIGNED_RETAINED;
1370            tracing::warn!(
1371                excess,
1372                cap = MAX_EMPTY_UNSIGNED_RETAINED,
1373                "empty-unsigned quarantined sessions exceed the retention cap; evicting \
1374                 oldest (their same-id refusal expires; no on-disk artifact is affected)"
1375            );
1376            for session in empty_quarantined.into_iter().take(excess) {
1377                self.sessions.remove(&session.id);
1378                evicted.push(session);
1379            }
1380        }
1381        evicted
1382    }

1384    /// Sessions where `close_session_locked` marked
1385    /// `artifact_committed = 1` but crashed / failed before running
1386    /// the finalization tail (`emit_bridge_event(SESSION_CLOSE)` +
1387    /// `remove_step_journal` + `remove_lifecycle_outbox` +

1413            .collect()

1416    /// Sessions idle longer than `idle_s` (for the sweeper).
1417    pub fn idle_sessions(&self, idle_s: u64) -> Vec<Arc<Session>> {
1418        // Monotonic idle math — see `evict_finalized`.
1419        let idle_duration = Duration::from_secs(idle_s);
1420        self.sessions
1421            .iter()
1422            .filter(|e| {
1423                e.idle_duration() >= idle_duration
1424                    && !e.is_closed()
1425                    // A session with a live forwarded response is not idle even
1426                    // when its admission clock is stale: the last-activity
1427                    // clock is refreshed only at request admission, so a
1428                    // stream that outlives the idle window would otherwise
1429                    // let the sweeper claim the close (sealing the session
1430                    // mid-conversation) and then park inside `wait_for_streams`
1431                    // while holding the shared lifecycle lock until the
1432                    // client's stream ends.
1433                    && e.active_streams.load(Ordering::Acquire) == 0
1434            })
1435            .map(|e| e.clone())
1436            .collect()
1437    }
