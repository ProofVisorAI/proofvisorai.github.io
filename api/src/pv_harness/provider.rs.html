1//! Provider adapter seam.
2//!
3//! The client always speaks OpenAI chat completions to the proxy. The
4//! `provider` config value selects how that request reaches the
5//! upstream and how the upstream answer comes back:
6//!
7//! * `"openai"` (the default) forwards the request body unchanged to
8//!   `upstream_url` + `upstream_chat_path` and relays the response
9//!   bytes as they are. This also fits every OpenAI-compatible server
10//!   (vLLM, LiteLLM, Groq, Together, DeepSeek, OpenRouter, Ollama,
11//!   LM Studio, llama.cpp, xAI, Mistral, Azure OpenAI).
12//! * `"anthropic"` translates the request into the Anthropic Messages
13//!   API (`POST /v1/messages`, `x-api-key`, `anthropic-version`) and
14//!   translates the answer — buffered JSON or named SSE events — back
15//!   into OpenAI chat-completion objects and chunks. See
16//!   [`anthropic`].
17//! * `"gemini"` translates the request into Gemini
18//!   `generateContent` / `streamGenerateContent?alt=sse` and the
19//!   answer back into OpenAI shape. See [`gemini`].
20//!
21//! Because every translated response is OpenAI-shaped by the time it
22//! reaches the relay in `routes.rs`, the audit capture (token
23//! accounting, tool-call reassembly, stop reasons, receipts) runs one
24//! parser — `routes::parse_provider_chunk` — for all three providers.
25//! Native usage that the OpenAI shape cannot express directly is
26//! carried in the OpenAI usage sub-objects the parser already reads
27//! (`prompt_tokens_details.cached_tokens`,
28//! `completion_tokens_details.reasoning_tokens`).
29

41/// One upstream provider's wire dialect.
42#[derive(Debug, Clone, Copy, PartialEq, Eq)]
43pub(crate) enum ProviderKind {
44    /// OpenAI chat completions, forwarded and relayed unchanged.
45    OpenAi,
46    /// Anthropic Messages API, translated both ways.
47    Anthropic,
48    /// Google Gemini `generateContent`, translated both ways.
49    Gemini,
50}
51
52impl ProviderKind {
53    /// Stable name, matching the `provider` config value.
54    pub(crate) fn name(self) -> &'static str {
55        match self {
56            Self::OpenAi => "openai",
57            Self::Anthropic => "anthropic",
58            Self::Gemini => "gemini",
59        }
60    }
61
62    /// The header that carries the API key on this provider's native
63    /// API, used when the operator left `upstream_auth_header` and
64    /// `upstream_auth_scheme` at their OpenAI defaults. The key is
65    /// sent raw (no scheme prefix). `None` keeps the configured
66    /// `authorization: Bearer <key>` behaviour.
67    pub(crate) fn native_auth_header(self) -> Option<&'static str> {
68        match self {
69            Self::OpenAi => None,
70            Self::Anthropic => Some("x-api-key"),
71            Self::Gemini => Some("x-goog-api-key"),
72        }
73    }
74}

101/// Read `n` from an OpenAI request: absent or `null` means one choice;
102/// anything else must be an integer in `1..=MAX_CHOICES`. Fails closed
103/// on `2.0`, `"2"`, `0`, negatives and oversized values — OpenAI-
104/// compatible servers lax-coerce some of those into real multi-choice
105/// requests.
106pub(crate) fn choice_count(payload: &Value) -> Result<u64, String> {
107    match payload.get("n") {
108        None | Some(Value::Null) => Ok(1),
109        Some(n) => match n.as_u64() {
110            Some(count) if n.is_u64() && (1..=MAX_CHOICES).contains(&count) => Ok(count),
111            _ => Err(format!(
112                "n must be an integer between 1 and {MAX_CHOICES}, got {n}"
113            )),
114        },
115    }
116}

118/// Refuse a request for token log-probabilities, which neither native
119/// API can return in OpenAI's shape. Answering without them would hand
120/// the client a completion that silently lacks what it asked for.
121pub(crate) fn refuse_logprobs(payload: &Value) -> Result<(), String> {
122    let asked = payload.get("logprobs").and_then(Value::as_bool) == Some(true)
123        || payload.get("top_logprobs").is_some_and(|value| !value.is_null());
124    if asked {
125        return Err("logprobs and top_logprobs are not available from this provider".to_owned());
126    }
127    Ok(())
128}

159/// Text of an OpenAI message `content` field: a string, or the
160/// concatenated `text` parts of a content array. `None` for null or
161/// missing content.
162pub(crate) fn content_text(content: Option<&Value>) -> Option<String> {
163    match content {
164        Some(Value::String(text)) => Some(text.clone()),
165        Some(Value::Array(parts)) => {
166            let mut text = String::new();
167            for part in parts {
168                if let Some(piece) = part.get("text").and_then(Value::as_str) {
169                    text.push_str(piece);
170                }
171            }
172            Some(text)
173        }
174        _ => None,
175    }
176}

178/// Split an OpenAI-style `data:<mime>;base64,<payload>` URL into its
179/// media type and base64 payload. `None` for anything else (remote
180/// URLs, non-base64 data URLs).
181pub(crate) fn base64_data_url(url: &str) -> Option<(String, String)> {
182    let rest = url.strip_prefix("data:")?;
183    let (meta, data) = rest.split_once(',')?;
184    let meta = meta.strip_suffix(";base64")?;
185    let media_type = if meta.is_empty() {
186        "application/octet-stream"
187    } else {
188        meta
189    };
190    Some((media_type.to_owned(), data.to_owned()))
191}

193/// Parse an OpenAI `function.arguments` string into the JSON object
194/// the native tool APIs require. An empty string means "no
195/// arguments". Non-object JSON and unparsable text are refused: both
196/// native APIs reject them and a silent rewrite would attest a call
197/// the model never made.
198pub(crate) fn arguments_object(arguments: Option<&Value>) -> Result<Value, String> {
199    let raw = match arguments {
200        None | Some(Value::Null) => return Ok(json!({})),
201        Some(Value::String(raw)) => raw.trim(),
202        Some(Value::Object(object)) => return Ok(Value::Object(object.clone())),
203        Some(other) => {
204            return Err(format!(
205                "tool call arguments must be a JSON-encoded string, got {other}"
206            ))
207        }
208    };
209    if raw.is_empty() {
210        return Ok(json!({}));
211    }
212    match serde_json::from_str::<Value>(raw) {
213        Ok(value @ Value::Object(_)) => Ok(value),
214        Ok(_) => Err("tool call arguments must encode a JSON object".to_owned()),
215        Err(error) => Err(format!("tool call arguments are not valid JSON: {error}")),
216    }
217}

237/// Token usage gathered from a native response, in OpenAI terms.
238#[derive(Debug, Clone, Copy, Default, PartialEq, Eq)]
239pub(crate) struct UsageTotals {
240    pub(crate) prompt_tokens: Option<u64>,
241    pub(crate) completion_tokens: Option<u64>,
242    pub(crate) cached_tokens: Option<u64>,
243    pub(crate) reasoning_tokens: Option<u64>,
244}
245
246impl UsageTotals {
247    /// True when the upstream reported anything at all.
248    pub(crate) fn is_known(&self) -> bool {
249        self.prompt_tokens.is_some() || self.completion_tokens.is_some()
250    }
251
252    /// Fold another response's usage into this one (every choice of a
253    /// multi-choice request is billed, so the totals add up).
254    pub(crate) fn add(&mut self, other: &UsageTotals) {
255        fn add_opt(a: Option<u64>, b: Option<u64>) -> Option<u64> {
256            match (a, b) {
257                (None, None) => None,
258                (a, b) => Some(a.unwrap_or(0).saturating_add(b.unwrap_or(0))),
259            }
260        }
261        self.prompt_tokens = add_opt(self.prompt_tokens, other.prompt_tokens);
262        self.completion_tokens = add_opt(self.completion_tokens, other.completion_tokens);
263        self.cached_tokens = add_opt(self.cached_tokens, other.cached_tokens);
264        self.reasoning_tokens = add_opt(self.reasoning_tokens, other.reasoning_tokens);
265    }
266
267    /// The OpenAI `usage` object.
268    pub(crate) fn to_openai(self) -> Value {
269        let prompt = self.prompt_tokens.unwrap_or(0);
270        let completion = self.completion_tokens.unwrap_or(0);
271        let mut usage = json!({
272            "prompt_tokens": prompt,
273            "completion_tokens": completion,
274            "total_tokens": prompt.saturating_add(completion),
275        });
276        if let Some(cached) = self.cached_tokens {
277            set(
278                &mut usage,
279                "prompt_tokens_details",
280                json!({ "cached_tokens": cached }),
281            );
282        }
283        if let Some(reasoning) = self.reasoning_tokens {
284            set(
285                &mut usage,
286                "completion_tokens_details",
287                json!({ "reasoning_tokens": reasoning }),
288            );
289        }
290        usage
291    }
292}

309/// One SSE frame (or a bare JSON body) split into its parts.
310pub(crate) struct SseFrame {
311    /// True when any SSE field line was seen (`data:`, `event:`,
312    /// `id:`, `retry:` or a comment). False means `data` is the raw
313    /// body text.
314    pub(crate) is_sse: bool,
315    /// The `event:` name, empty when unset.
316    pub(crate) event: String,
317    /// Joined `data:` lines (or the raw text). `None` when the frame
318    /// carries nothing attributable (keepalive, comment-only).
319    pub(crate) data: Option<String>,
320}
321
322/// Split one SSE frame per the EventSource spec: strip one leading
323/// BOM, collect `data:` lines (one optional leading space each),
324/// track the `event:` name. Non-SSE text is returned whole.
325pub(crate) fn split_sse_frame(raw: &str) -> SseFrame {
326    let raw = raw.strip_prefix('\u{feff}').unwrap_or(raw);
327    let mut is_sse = false;
328    let mut event = String::new();
329    let mut data: Vec<&str> = Vec::new();
330    for line in raw.split(['\r', '\n']) {
331        if let Some(value) = line.strip_prefix("data:") {
332            is_sse = true;
333            data.push(value.strip_prefix(' ').unwrap_or(value));
334        } else if let Some(value) = line.strip_prefix("event:") {
335            is_sse = true;
336            event = value.strip_prefix(' ').unwrap_or(value).to_owned();
337        } else if line == "data"
338            || line.starts_with("id:")
339            || line.starts_with("retry:")
340            || line.starts_with(':')
341        {
342            is_sse = true;
343        }
344    }
345    let data = if is_sse {
346        if data.iter().all(|entry| entry.trim().is_empty()) {
347            None
348        } else {
349            Some(data.join("\n"))
350        }
351    } else {
352        let trimmed = raw.trim();
353        (!trimmed.is_empty()).then(|| trimmed.to_owned())
354    };
355    SseFrame { is_sse, event, data }
356}

406/// Translate a native error body into the OpenAI error envelope
407/// `{"error": {"message", "type", "param", "code"}}`. `type` follows
408/// OpenAI's taxonomy by HTTP status; `code` carries the upstream's
409/// own error type or status string so the native diagnostic survives.
410pub(crate) fn openai_error_envelope(provider: ProviderKind, status: u16, body: &[u8]) -> Value {
411    let parsed: Option<Value> = serde_json::from_slice(body).ok();
412    let error = parsed.as_ref().and_then(|value| value.get("error"));
413    let message = error
414        .and_then(|error| error.get("message"))
415        .and_then(Value::as_str)
416        .map(str::to_owned)
417        .unwrap_or_else(|| {
418            format!(
419                "upstream {} responded with HTTP {status} and no JSON error body",
420                provider.name()
421            )
422        });
423    let native_code = error
424        .and_then(|error| error.get("type").or_else(|| error.get("status")))
425        .and_then(Value::as_str)
426        .map(str::to_owned);
427    let kind = match status {
428        400 | 404 | 405 | 409 | 413 | 422 => "invalid_request_error",
429        401 => "authentication_error",
430        403 => "permission_error",
431        429 => "rate_limit_error",
432        _ => "api_error",
433    };
434    json!({
435        "error": {
436            "message": message,
437            "type": kind,
438            "param": Value::Null,
439            "code": native_code.map_or(Value::Number(status.into()), Value::String),
440        }
441    })
442}

444/// Output of a streaming translator for one native frame.
445pub(crate) enum TranslatedItem {
446    /// One OpenAI chunk to relay.
447    Chunk(Value),
448    /// The stream ended cleanly; carries the usage it reported.
449    End(UsageTotals),
450}

452/// A stateful translator from one native SSE stream to OpenAI chunks.
453pub(crate) trait StreamTranslator: Send {
454    /// Translate one complete SSE frame (or the unterminated tail at
455    /// EOF). An `Err` fails the stream with the given reason.
456    fn frame(&mut self, raw: &str) -> Result<Vec<Value>, String>;
457    /// Called once at end of stream; returns trailing chunks and the
458    /// usage the stream reported.
459    fn finish(&mut self) -> Result<(Vec<Value>, UsageTotals), String>;
460}
461
462/// The longest unterminated SSE frame the translator will buffer.
463const MAX_TRANSLATE_FRAME_BYTES: usize = 8 * 1024 * 1024;
464
465/// Wraps one upstream byte stream and runs a [`StreamTranslator`] over
466/// its SSE frames.
467pub(crate) struct TranslatingStream {
468    inner: BoxStream<'static, Result<Bytes, std::io::Error>>,
469    translator: Box<dyn StreamTranslator>,
470    buffer: Vec<u8>,
471    queued: VecDeque<TranslatedItem>,
472    finished: bool,
473}
474
475impl TranslatingStream {
476    pub(crate) fn new(
477        inner: BoxStream<'static, Result<Bytes, std::io::Error>>,
478        translator: Box<dyn StreamTranslator>,
479    ) -> Self {
480        Self {
481            inner,
482            translator,
483            buffer: Vec::new(),
484            queued: VecDeque::new(),
485            finished: false,
486        }
487    }
488
489    fn absorb(&mut self, bytes: &[u8]) -> Result<(), String> {
490        self.buffer.extend_from_slice(bytes);
491        while let Some(end) = crate::routes::sse_frame_end(&self.buffer) {
492            let frame: Vec<u8> = self.buffer.drain(..end).collect();
493            let frame = std::str::from_utf8(&frame)
494                .map_err(|error| format!("upstream SSE frame is not UTF-8: {error}"))?;
495            for chunk in self.translator.frame(frame)? {
496                self.queued.push_back(TranslatedItem::Chunk(chunk));
497            }
498        }
499        if self.buffer.len() > MAX_TRANSLATE_FRAME_BYTES {
500            return Err("unterminated upstream SSE frame exceeds limit".to_owned());
501        }
502        Ok(())
503    }
504
505    fn end(&mut self) -> Result<(), String> {
506        let tail = std::mem::take(&mut self.buffer);
507        if !tail.is_empty() {
508            let tail =
509                std::str::from_utf8(&tail).map_err(|error| format!("upstream body is not UTF-8: {error}"))?;
510            for chunk in self.translator.frame(tail)? {
511                self.queued.push_back(TranslatedItem::Chunk(chunk));
512            }
513        }
514        let (chunks, usage) = self.translator.finish()?;
515        for chunk in chunks {
516            self.queued.push_back(TranslatedItem::Chunk(chunk));
517        }
518        self.queued.push_back(TranslatedItem::End(usage));
519        Ok(())
520    }
521}
522
523impl Stream for TranslatingStream {
524    type Item = Result<TranslatedItem, std::io::Error>;
525
526    fn poll_next(mut self: Pin<&mut Self>, context: &mut Context<'_>) -> Poll<Option<Self::Item>> {
527        loop {
528            if let Some(item) = self.queued.pop_front() {
529                return Poll::Ready(Some(Ok(item)));
530            }
531            if self.finished {
532                return Poll::Ready(None);
533            }
534            match self.inner.as_mut().poll_next(context) {
535                Poll::Pending => return Poll::Pending,
536                Poll::Ready(Some(Ok(bytes))) => {
537                    if let Err(reason) = self.absorb(&bytes) {
538                        self.finished = true;
539                        return Poll::Ready(Some(Err(std::io::Error::other(reason))));
540                    }
541                }
542                Poll::Ready(Some(Err(error))) => {
543                    self.finished = true;
544                    return Poll::Ready(Some(Err(error)));
545                }
546                Poll::Ready(None) => {
547                    self.finished = true;
548                    if let Err(reason) = self.end() {
549                        return Poll::Ready(Some(Err(std::io::Error::other(reason))));
550                    }
551                }
552            }
553        }
554    }
555}

557/// Merges the per-choice translated streams of one request into a
558/// single OpenAI SSE stream: chunks interleave as they arrive (each
559/// already carries its `choices[].index`), usage is summed over every
560/// sub-stream, and the stream ends with the usage and `data: [DONE]`.
561///
562/// Usage placement follows what the client asked for. With
563/// `stream_options.include_usage` the final usage chunk is the
564/// standalone `choices: []` chunk OpenAI emits. Without it, chunks
565/// that carry a `finish_reason` are held until every sub-stream ends
566/// and the summed usage rides on the last of them, so a client that
567/// never opted into usage chunks still sees only chunks with a
568/// choice — and the audit capture still records provider-true token
569/// counts instead of estimates.
570pub(crate) struct MergedOpenAiStream {
571    inner: SelectAll<TranslatingStream>,
572    include_usage: bool,
573    id: String,
574    model: String,
575    created: u64,
576    usage: UsageTotals,
577    held_finishes: Vec<Value>,
578    output: VecDeque<Bytes>,
579    done: bool,
580}
581
582impl MergedOpenAiStream {
583    pub(crate) fn new(
584        streams: Vec<TranslatingStream>,
585        include_usage: bool,
586        id: String,
587        model: String,
588        created: u64,
589    ) -> Self {
590        Self {
591            inner: futures::stream::select_all(streams),
592            include_usage,
593            id,
594            model,
595            created,
596            usage: UsageTotals::default(),
597            held_finishes: Vec::new(),
598            output: VecDeque::new(),
599            done: false,
600        }
601    }
602
603    fn finalize(&mut self) {
604        let held = std::mem::take(&mut self.held_finishes);
605        let count = held.len();
606        if self.include_usage || count == 0 {
607            for finish in held {
608                self.output.push_back(sse_bytes(&finish));
609            }
610            if self.usage.is_known() {
611                self.output.push_back(sse_bytes(&usage_chunk(
612                    &self.id,
613                    &self.model,
614                    self.created,
615                    &self.usage,
616                )));
617            }
618        } else {
619            for (position, mut finish) in held.into_iter().enumerate() {
620                if position + 1 == count && self.usage.is_known() {
621                    set(&mut finish, "usage", self.usage.to_openai());
622                }
623                self.output.push_back(sse_bytes(&finish));
624            }
625        }
626        self.output.push_back(done_bytes());
627        self.done = true;
628    }
629}
630
631impl Stream for MergedOpenAiStream {
632    type Item = Result<Bytes, std::io::Error>;
633
634    fn poll_next(mut self: Pin<&mut Self>, context: &mut Context<'_>) -> Poll<Option<Self::Item>> {
635        loop {
636            if let Some(bytes) = self.output.pop_front() {
637                return Poll::Ready(Some(Ok(bytes)));
638            }
639            if self.done {
640                return Poll::Ready(None);
641            }
642            match self.inner.poll_next_unpin(context) {
643                Poll::Pending => return Poll::Pending,
644                Poll::Ready(Some(Ok(TranslatedItem::Chunk(chunk)))) => {
645                    let has_finish = chunk
646                        .pointer("/choices/0/finish_reason")
647                        .is_some_and(|reason| !reason.is_null());
648                    if has_finish && !self.include_usage {
649                        self.held_finishes.push(chunk);
650                    } else {
651                        self.output.push_back(sse_bytes(&chunk));
652                    }
653                }
654                Poll::Ready(Some(Ok(TranslatedItem::End(usage)))) => {
655                    self.usage.add(&usage);
656                }
657                Poll::Ready(Some(Err(error))) => {
658                    self.done = true;
659                    self.output.clear();
660                    return Poll::Ready(Some(Err(error)));
661                }
662                Poll::Ready(None) => self.finalize(),
663            }
664        }
665    }
666}

668/// Merge the non-streaming completions of a multi-choice fan-out into
669/// one OpenAI `chat.completion`: choices renumbered in order, usage
670/// summed, id/model/created from the first.
671pub(crate) fn merge_completions(mut completions: Vec<Value>) -> Result<Value, String> {
672    if completions.is_empty() {
673        return Err("no upstream completion to merge".to_owned());
674    }
675    let mut merged = completions.remove(0);
676    let mut usage = usage_from_openai(merged.get("usage"))?;
677    let mut choices = merged
678        .get("choices")
679        .and_then(Value::as_array)
680        .cloned()
681        .unwrap_or_default();
682    for completion in completions {
683        usage.add(&usage_from_openai(completion.get("usage"))?);
684        if let Some(extra) = completion.get("choices").and_then(Value::as_array) {
685            choices.extend(extra.iter().cloned());
686        }
687    }
688    for (position, choice) in choices.iter_mut().enumerate() {
689        set(choice, "index", json!(position));
690    }
691    set(&mut merged, "choices", Value::Array(choices));
692    set(&mut merged, "usage", usage.to_openai());
693    Ok(merged)
694}
695
696/// Read an OpenAI `usage` object back into totals (for merging).
697fn usage_from_openai(usage: Option<&Value>) -> Result<UsageTotals, String> {
698    let Some(usage) = usage.filter(|usage| !usage.is_null()) else {
699        return Ok(UsageTotals::default());
700    };
701    Ok(UsageTotals {
702        prompt_tokens: usage_u64(usage.get("prompt_tokens"), "prompt_tokens")?,
703        completion_tokens: usage_u64(usage.get("completion_tokens"), "completion_tokens")?,
704        cached_tokens: usage_u64(
705            usage.pointer("/prompt_tokens_details/cached_tokens"),
706            "cached_tokens",
707        )?,
708        reasoning_tokens: usage_u64(
709            usage.pointer("/completion_tokens_details/reasoning_tokens"),
710            "reasoning_tokens",
711        )?,
712    })
713}

877    /// A translator that echoes its frames as content deltas and
878    /// finishes on "END", for exercising the stream plumbing.
879    struct Echo {
880        index: u64,
881        completion: u64,
882    }
883
884    impl StreamTranslator for Echo {
885        fn frame(&mut self, raw: &str) -> Result<Vec<Value>, String> {
886            let frame = split_sse_frame(raw);
887            let Some(data) = frame.data else {
888                return Ok(Vec::new());
889            };
890            if data == "FAIL" {
891                return Err("upstream said fail".to_owned());
892            }
893            if data == "END" {
894                return Ok(vec![chunk("id", "m", 1, self.index, json!({}), Some("stop"))]);
895            }
896            self.completion += 1;
897            Ok(vec![chunk(
898                "id",
899                "m",
900                1,
901                self.index,
902                json!({"content": data}),
903                None,
904            )])
905        }
906        fn finish(&mut self) -> Result<(Vec<Value>, UsageTotals), String> {
907            Ok((
908                Vec::new(),
909                UsageTotals {
910                    prompt_tokens: Some(10),
911                    completion_tokens: Some(self.completion),
912                    cached_tokens: None,
913                    reasoning_tokens: None,
914                },
915            ))
916        }
917    }

919    fn byte_stream(frames: &[&str]) -> BoxStream<'static, Result<Bytes, std::io::Error>> {
920        let owned: Vec<Result<Bytes, std::io::Error>> = frames
921            .iter()
922            .map(|frame| Ok(Bytes::from((*frame).to_owned())))
923            .collect();
924        futures::stream::iter(owned).boxed()
925    }

927    async fn collect(stream: MergedOpenAiStream) -> Result<Vec<Value>, std::io::Error> {
928        let mut frames = Vec::new();
929        let mut stream = stream;
930        while let Some(item) = stream.next().await {
931            let bytes = item?;
932            let text = std::str::from_utf8(&bytes).unwrap();
933            for frame in text.split("\n\n").filter(|frame| !frame.is_empty()) {
934                let data = frame.strip_prefix("data: ").unwrap();
935                if data == "[DONE]" {
936                    frames.push(json!("[DONE]"));
937                } else {
938                    frames.push(serde_json::from_str(data).unwrap());
939                }
940            }
941        }
942        Ok(frames)
943    }
