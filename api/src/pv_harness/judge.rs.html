1//! Semantic judge integration (`[judge]`).
2//!
3//! The judge runs on `/v1/mcp` tool calls whose tool is listed under
4//! `[[judge.tools]]`, between the schema gate and the policy gate. It
5//! sees the schema-validated arguments and operator-controlled context
6//! (intent, mission task, charter, delegation depth, and refusals the
7//! harness recorded for this session and for the caller's delegation
8//! parents), never the raw prompt. Untagged tools never reach it, so
9//! their admission path is unchanged.
10//!
11//! The judge never decides. Its verdicts go to the sandbox's policy
12//! gate, where `[[judge.thresholds]]` (and any WASM ABI v2 policy) turn
13//! them into allow, ask (`APPROVAL_REQUIRED`), or deny, and where every
14//! failure denies or asks, never allows. Each verdict, with its full
15//! distribution and the pinned model, is written into the tool-call
16//! OCSF event (hashed into the signed event chain) and the ATIF tool
17//! call (hashed into a retroactive receipt).
18//!
19//! State kept in this process: the drift monitor, the review queue and
20//! the concurrency limits. The prior-denial log is kept here and, when
21//! the state store is shared between instances (Redis), in the store
22//! too; the pilot cap counters live in the state store.
23

279impl JudgeAdapterConfig {
280    /// The configured or default deadline.
281    pub fn deadline(&self) -> Duration {
282        Duration::from_millis(self.deadline_ms.unwrap_or(match self.kind {
283            JudgeAdapterKind::Local => DEFAULT_LOCAL_DEADLINE_MS,
284            JudgeAdapterKind::Cloud | JudgeAdapterKind::Typesafe => DEFAULT_CLOUD_DEADLINE_MS,
285        }))
286    }
287
288    /// The pinned model. For `typesafe`, `weights_sha256` is derived from
289    /// the version name when left out, and must equal that when given.
290    pub fn pin(&self) -> Result<pv_judge::ModelPin, String> {
291        let weights = if self.kind == JudgeAdapterKind::Typesafe {
292            let derived = pv_judge::typesafe::derived_weights_sha256(&self.model_version);
293            if !self.weights_sha256.is_empty() && self.weights_sha256 != derived {
294                return Err(format!(
295                    "judge adapter {:?}: TypeSafe publishes no weight hash, so weights_sha256 is \
296                     derived from model_version; leave it out or set it to {derived}",
297                    self.name
298                ));
299            }
300            derived
301        } else {
302            self.weights_sha256.clone()
303        };
304        pv_judge::ModelPin::new(&self.model_name, &self.model_version, &weights)
305            .map_err(|error| format!("judge adapter {:?}: {error}", self.name))
306    }

572impl JudgeConfig {
573    /// Built-in predicates plus `predicates`.
574    pub fn registry(&self) -> Result<PredicateRegistry, String> {
575        PredicateRegistry::with_custom(&self.predicates).map_err(|error| format!("judge: {error}"))
576    }

578    /// The threshold policy the sandbox applies.
579    pub fn threshold_policy(&self) -> Result<ThresholdPolicy, String> {
580        let rules = self
581            .thresholds
582            .iter()
583            .map(|threshold| {
584                let ppm = |value: Option<f64>, name: &str| {
585                    value
586                        .map(pv_judge::threshold_to_ppm)
587                        .transpose()
588                        .map_err(|error| {
589                            format!(
590                                "judge threshold {} / {} {name}: {error}",
591                                threshold.predicate, threshold.label
592                            )
593                        })
594                };
595                Ok(ThresholdRule {
596                    predicate_id: threshold.predicate.clone(),
597                    label: threshold.label.clone(),
598                    deny_above_ppm: ppm(threshold.deny_above, "deny_above")?,
599                    ask_above_ppm: ppm(threshold.ask_above, "ask_above")?,
600                    min_confidence_ppm: ppm(
601                        Some(threshold.min_confidence.unwrap_or(DEFAULT_MIN_CONFIDENCE)),
602                        "min_confidence",
603                    )?,
604                })
605            })
606            .collect::<Result<Vec<_>, String>>()?;
607        let failure = FailurePolicy {
608            timeout: self.on_failure.timeout.into(),
609            unavailable: self.on_failure.unavailable.into(),
610            saturated: self.on_failure.saturated.into(),
611            malformed: self.on_failure.malformed.into(),
612        };
613        ThresholdPolicy::new(rules, failure).map_err(|error| format!("judge: {error}"))
614    }

632    /// Every structural problem (no file access).
633    pub fn collect_errors(&self, errors: &mut Vec<String>) {
634        if !(1..=MAX_CONCURRENCY_CEILING).contains(&self.max_concurrency) {
635            errors.push(format!(
636                "judge.max_concurrency must be between 1 and {MAX_CONCURRENCY_CEILING}"
637            ));
638        }
639        let registry = match self.registry() {
640            Ok(registry) => Some(registry),
641            Err(error) => {
642                errors.push(error);
643                None
644            }
645        };
646        let mut adapters: BTreeMap<&str, &JudgeAdapterConfig> = BTreeMap::new();
647        for adapter in &self.adapters {
648            if !is_name(&adapter.name) {
649                errors.push(format!(
650                    "judge adapter name {:?} must be 1..=64 characters of [A-Za-z0-9_.-]",
651                    adapter.name
652                ));
653            }
654            if adapters.insert(adapter.name.as_str(), adapter).is_some() {
655                errors.push(format!(
656                    "judge adapter {:?} is defined more than once",
657                    adapter.name
658                ));
659            }
660            collect_adapter_errors(adapter, errors);
661        }
662        if self.tools.is_empty()
663            && self.trace.is_none()
664            && self.breaker.is_none()
665            && self.standing_rules.is_empty()
666        {
667            errors.push(
668                "[judge] is set but enables nothing: tag tools under [[judge.tools]], or set \
669                 [judge.trace], [judge.breaker] or [[judge.standing_rules]]"
670                    .to_owned(),
671            );
672        }
673        if self
674            .max_concurrency_per_caller
675            .is_some_and(|share| share == 0 || share > self.max_concurrency)
676        {
677            errors.push(
678                "judge.max_concurrency_per_caller must be between 1 and judge.max_concurrency".to_owned(),
679            );
680        }
681        if !(1..=MAX_CONCURRENCY_CEILING).contains(&self.background_max_concurrency) {
682            errors.push(format!(
683                "judge.background_max_concurrency must be between 1 and {MAX_CONCURRENCY_CEILING}"
684            ));
685        }
686        self.collect_background_errors(&adapters, registry.as_ref(), errors);
687        let mut tools = BTreeSet::new();
688        let mut bound_predicates = BTreeSet::new();
689        for tool in &self.tools {
690            if tool.tool.trim().is_empty() || tool.tool.len() > 256 || tool.tool.chars().any(char::is_control)
691            {
692                errors.push(format!(
693                    "judge tool name {:?} is not a valid tool name",
694                    tool.tool
695                ));
696            }
697            if !tools.insert(tool.tool.as_str()) {
698                errors.push(format!("judge tool {:?} is listed more than once", tool.tool));
699            }
700            if tool.predicates.is_empty() {
701                errors.push(format!("judge tool {:?} binds no predicates", tool.tool));
702            }
703            let mut seen = BTreeSet::new();
704            for binding in &tool.predicates {
705                if !seen.insert(binding.id.as_str()) {
706                    errors.push(format!(
707                        "judge tool {:?} binds predicate {} more than once",
708                        tool.tool, binding.id
709                    ));
710                }
711                bound_predicates.insert(binding.id.as_str());
712                let predicate = registry.as_ref().and_then(|registry| registry.get(&binding.id));
713                if registry.is_some() && predicate.is_none() {
714                    errors.push(format!(
715                        "judge tool {:?} binds unknown predicate {:?} (define it under [[judge.predicates]])",
716                        tool.tool, binding.id
717                    ));
718                }
719                let Some(adapter) = adapters.get(binding.adapter.as_str()) else {
720                    errors.push(format!(
721                        "judge tool {:?} binds predicate {} to unknown adapter {:?}",
722                        tool.tool, binding.id, binding.adapter
723                    ));
724                    continue;
725                };
726                if adapter.kind.is_remote() && !tool.allow_cloud {
727                    errors.push(format!(
728                        "judge tool {:?} uses cloud adapter {:?}: its arguments would leave the \
729                         harness; set allow_cloud = true on this tool to opt in",
730                        tool.tool, adapter.name
731                    ));
732                }
733                if let (JudgeAdapterKind::Local, Some(predicate)) = (adapter.kind, predicate) {
734                    let labels: BTreeSet<&str> = adapter.output_labels.iter().map(String::as_str).collect();
735                    let expected: BTreeSet<&str> = predicate.labels.iter().map(String::as_str).collect();
736                    if labels != expected || adapter.output_labels.len() != predicate.labels.len() {
737                        errors.push(format!(
738                            "judge adapter {:?} output_labels {:?} must be exactly the labels of {} ({:?})",
739                            adapter.name, adapter.output_labels, predicate.id, predicate.labels
740                        ));
741                    }
742                }
743            }
744        }
745        let widest = self
746            .tools
747            .iter()
748            .map(|tool| tool.predicates.len())
749            .max()
750            .unwrap_or(0);
751        if widest > self.max_concurrency {
752            errors.push(format!(
753                "judge.max_concurrency ({}) is below the {widest} predicates bound to one tool; \
754                 every call to that tool would be refused as saturated",
755                self.max_concurrency
756            ));
757        }
758        if let Some(share) = self.max_concurrency_per_caller.filter(|share| *share < widest) {
759            errors.push(format!(
760                "judge.max_concurrency_per_caller ({share}) is below the {widest} predicates bound \
761                 to one tool; every call to that tool would be refused as saturated"
762            ));
763        }
764        for threshold in &self.thresholds {
765            match registry
766                .as_ref()
767                .map(|registry| registry.get(&threshold.predicate))
768            {
769                Some(None) => errors.push(format!(
770                    "judge threshold names unknown predicate {:?}",
771                    threshold.predicate
772                )),
773                Some(Some(predicate)) if !predicate.has_label(&threshold.label) => errors.push(format!(
774                    "judge threshold label {:?} is not a label of {}",
775                    threshold.label, threshold.predicate
776                )),
777                _ => {}
778            }
779            if !bound_predicates.contains(threshold.predicate.as_str()) {
780                errors.push(format!(
781                    "judge threshold for {} applies to no tool (bind the predicate under [[judge.tools]])",
782                    threshold.predicate
783                ));
784            }
785        }
786        if let Err(error) = self.threshold_policy() {
787            errors.push(error);
788        }
789        if let Some(pilot) = self.pilot {
790            if pilot.daily_block_cap == 0 {
791                errors.push("judge.pilot.daily_block_cap must be at least 1".to_owned());
792            }
793            if self.mode == JudgeModeConfig::Shadow {
794                errors
795                    .push("[judge.pilot] only applies in enforce mode; shadow mode never blocks".to_owned());
796            }
797        }
798        if let Err(error) = self.drift_config() {
799            errors.push(error);
800        }
801        if self
802            .drift
803            .as_ref()
804            .is_some_and(|drift| drift.state_file.trim().is_empty())
805        {
806            errors.push("judge.drift.state_file must not be empty".to_owned());
807        }
808    }

810    fn collect_background_errors(
811        &self,
812        adapters: &BTreeMap<&str, &JudgeAdapterConfig>,
813        registry: Option<&PredicateRegistry>,
814        errors: &mut Vec<String>,
815    ) {
816        let mut check = |section: &str, adapter_name: &str, predicate_id: &str, allow_cloud: bool| {
817            let predicate = registry.and_then(|registry| registry.get(predicate_id));
818            if registry.is_some() && predicate.is_none() {
819                errors.push(format!(
820                    "[judge.{section}] names unknown predicate {predicate_id:?}"
821                ));
822            }
823            let Some(adapter) = adapters.get(adapter_name) else {
824                errors.push(format!(
825                    "[judge.{section}] names unknown adapter {adapter_name:?}"
826                ));
827                return;
828            };
829            if adapter.kind.is_remote() && !allow_cloud {
830                errors.push(format!(
831                    "[judge.{section}] uses cloud adapter {adapter_name:?}: recorded outputs would \
832                     leave the harness; set allow_cloud = true in [judge.{section}] to opt in"
833                ));
834            }
835            if adapter.deadline() > Duration::from_millis(MAX_BACKGROUND_DEADLINE_MS) {
836                errors.push(format!(
837                    "[judge.{section}] adapter {adapter_name:?} has a deadline above \
838                     {MAX_BACKGROUND_DEADLINE_MS} ms; background verdicts hold session close, \
839                     which has 3 s at shutdown"
840                ));
841            }
842            if let (JudgeAdapterKind::Local, Some(predicate)) = (adapter.kind, predicate) {
843                let labels: BTreeSet<&str> = adapter.output_labels.iter().map(String::as_str).collect();
844                let expected: BTreeSet<&str> = predicate.labels.iter().map(String::as_str).collect();
845                if labels != expected || adapter.output_labels.len() != predicate.labels.len() {
846                    errors.push(format!(
847                        "judge adapter {adapter_name:?} output_labels {:?} must be exactly the labels of {} ({:?})",
848                        adapter.output_labels, predicate.id, predicate.labels
849                    ));
850                }
851            }
852        };
853        if let Some(trace) = &self.trace {
854            check("trace", &trace.adapter, &trace.predicate, trace.allow_cloud);
855        }
856        if let Some(breaker) = &self.breaker {
857            check(
858                "breaker",
859                &breaker.adapter,
860                &breaker.predicate,
861                breaker.allow_cloud,
862            );
863        }
864        // A local model must read a whole window (trace) or both texts
865        // (breaker); with a smaller token limit every window would fail.
866        for (section, adapter) in [
867            ("trace", self.trace.as_ref().map(|trace| trace.adapter.as_str())),
868            (
869                "breaker",
870                self.breaker.as_ref().map(|breaker| breaker.adapter.as_str()),
871            ),
872        ] {
873            let Some(adapter) = adapter.and_then(|name| adapters.get(name)) else {
874                continue;
875            };
876            let tokens = adapter.max_input_tokens.unwrap_or(512);
877            if adapter.kind == JudgeAdapterKind::Local && tokens < MIN_LOCAL_TRACE_TOKENS {
878                errors.push(format!(
879                    "[judge.{section}] local adapter {:?} reads {tokens} tokens, fewer than the \
880                     {MIN_LOCAL_TRACE_TOKENS} a window of {TRACE_WINDOW_CHARS} characters needs; \
881                     set max_input_tokens to at least {MIN_LOCAL_TRACE_TOKENS}",
882                    adapter.name
883                ));
884            }
885        }
886        if let Some(trace) = &self.trace {
887            if trace.sources.is_empty() {
888                errors.push("[judge.trace] sources must name tool, model, or both".to_owned());
889            }
890            if pv_judge::threshold_to_ppm(trace.finding_above).is_err() {
891                errors.push("[judge.trace] finding_above must be a probability in [0, 1]".to_owned());
892            }
893            if !(1..=MAX_TRACE_OUTPUT_CHARS).contains(&trace.max_output_chars) {
894                errors.push(format!(
895                    "[judge.trace] max_output_chars must be between 1 and {MAX_TRACE_OUTPUT_CHARS}"
896                ));
897            }
898        }
899        if let Some(breaker) = &self.breaker {
900            if let Some(adapter) = adapters.get(breaker.adapter.as_str()) {
901                if adapter.deadline() > Duration::from_millis(MAX_BREAKER_DEADLINE_MS) {
902                    errors.push(format!(
903                        "[judge.breaker] adapter {:?} has a deadline above {MAX_BREAKER_DEADLINE_MS} ms; \
904                         the second opinion runs inside the audit worker and holds the session's \
905                         later records for as long as it waits (define a separate adapter with a \
906                         shorter deadline_ms for the breaker)",
907                        breaker.adapter
908                    ));
909                }
910            }
911            let low = pv_judge::threshold_to_ppm(breaker.ambiguous_low);
912            let high = pv_judge::threshold_to_ppm(breaker.ambiguous_high);
913            match (low, high) {
914                (Ok(low), Ok(high)) if low < high => {}
915                _ => errors.push("[judge.breaker] needs 0 <= ambiguous_low < ambiguous_high <= 1".to_owned()),
916            }
917            if pv_judge::threshold_to_ppm(breaker.looping_above).is_err() {
918                errors.push("[judge.breaker] looping_above must be a probability in [0, 1]".to_owned());
919            }
920        }
921        let mut rule_ids = BTreeSet::new();
922        for rule in &self.standing_rules {
923            if !is_name(&rule.id) {
924                errors.push(format!(
925                    "standing rule id {:?} must be 1..=64 characters of [A-Za-z0-9_.-]",
926                    rule.id
927                ));
928            }
929            if !rule_ids.insert(rule.id.as_str()) {
930                errors.push(format!("standing rule {:?} is defined more than once", rule.id));
931            }
932            if rule.text.trim().is_empty() || rule.text.chars().count() > MAX_RULE_CHARS {
933                errors.push(format!(
934                    "standing rule {:?} text must be 1..={MAX_RULE_CHARS} characters",
935                    rule.id
936                ));
937            }
938        }
939    }

1002                errors.push(format!(
1003                    "judge adapter {:?} (local) needs output_labels (one per output logit)",

1005                ));
1006            }

1008                errors.push(format!(
1009                    "judge adapter {:?} max_input_tokens must be between 1 and 8192",
1010                    adapter.name
1011                ));
1012            }
1013        }
1014        JudgeAdapterKind::Cloud => {
1015            if local_fields {
1016                errors.push(format!(
1017                    "judge adapter {:?} is cloud; model_path, tokenizer_path, tokenizer_sha256, \
1018                     output_labels and max_input_tokens belong to local adapters",
1019                    adapter.name
1020                ));
1021            }
1022            match adapter.endpoint.as_deref().map(url::Url::parse) {
1023                None => errors.push(format!("judge adapter {:?} (cloud) needs endpoint", adapter.name)),
1024                Some(Err(_)) => {
1025                    errors.push(format!("judge adapter {:?} endpoint is not a URL", adapter.name))
1026                }
1027                Some(Ok(url)) => {
1028                    if let Err(error) = pv_judge::cloud::validate_endpoint(&url) {
1029                        errors.push(format!("judge adapter {:?}: {error}", adapter.name));
1030                    }
1031                }
1032            }
1033        }
1034        JudgeAdapterKind::Typesafe => {
1035            if local_fields {
1036                errors.push(format!(
1037                    "judge adapter {:?} is typesafe; model_path, tokenizer_path, tokenizer_sha256, \
1038                     output_labels and max_input_tokens belong to local adapters",
1039                    adapter.name
1040                ));
1041            }
1042            if let Some(raw) = adapter.endpoint.as_deref() {
1043                match url::Url::parse(raw) {
1044                    Err(_) => errors.push(format!("judge adapter {:?} endpoint is not a URL", adapter.name)),
1045                    Ok(url) => {
1046                        if let Err(error) = pv_judge::cloud::validate_endpoint(&url) {
1047                            errors.push(format!("judge adapter {:?}: {error}", adapter.name));
1048                        }
1049                    }
1050                }
1051            }
1052            if adapter.bearer_file.as_deref().is_none_or(|p| p.trim().is_empty()) {
1053                errors.push(format!(
1054                    "judge adapter {:?} (typesafe) needs bearer_file: an owner-only file holding the API key",
1055                    adapter.name
1056                ));
1057            }
1058            if !pv_judge::typesafe::is_fixed_version(&adapter.model_version) {
1059                errors.push(format!(
1060                    "judge adapter {:?} (typesafe) model_version {:?} must be a fixed release such as \
1061                     jev-1.14.0; jev-latest and jev-preview move without notice and cannot be pinned",
1062                    adapter.name, adapter.model_version
1063                ));
1064            }
1065        }
1066    }

1081/// Recent refusals by session and by agent instance (bounded). Keys are
1082/// evicted least recently used first, so a burst of new session ids
1083/// cannot push out an agent that is still active, and each key keeps one
1084/// entry per kind of refusal (tool, code, predicate), so repeating one
1085/// refusal cannot push out a different one.
1086#[derive(Default)]
1087struct DenialLog {
1088    by_key: HashMap<String, (u64, VecDeque<PriorDenial>)>,
1089    /// Use order: generation → key.
1090    order: BTreeMap<u64, String>,
1091    generation: u64,
1092}
1093
1094impl DenialLog {
1095    fn touch(&mut self, key: &str) {
1096        self.generation = self.generation.wrapping_add(1);
1097        let generation = self.generation;
1098        if let Some((used, _)) = self.by_key.get_mut(key) {
1099            self.order.remove(used);
1100            *used = generation;
1101            self.order.insert(generation, key.to_owned());
1102        }
1103    }
1104
1105    fn push(&mut self, key: String, denial: PriorDenial) {
1106        if !self.by_key.contains_key(&key) {
1107            while self.by_key.len() >= DENIAL_LOG_KEYS {
1108                let Some((_, oldest)) = self.order.pop_first() else {
1109                    break;
1110                };
1111                self.by_key.remove(&oldest);
1112            }
1113            self.by_key.insert(key.clone(), (0, VecDeque::new()));
1114        }
1115        self.touch(&key);
1116        if let Some((_, entries)) = self.by_key.get_mut(&key) {
1117            entries.retain(|entry| !same_kind(entry, &denial));
1118            entries.push_back(denial);
1119            // Full: drop the oldest refusal that did not come from the
1120            // judge first, so cheap policy refusals cannot push out the
1121            // judge's own.
1122            while entries.len() > DENIALS_PER_KEY {
1123                match entries.iter().position(|entry| entry.predicate_id.is_none()) {
1124                    Some(oldest) => {
1125                        entries.remove(oldest);
1126                    }
1127                    None => {
1128                        entries.pop_front();
1129                    }
1130                }
1131            }
1132        }
1133    }
1134
1135    fn recent(&mut self, key: &str) -> Vec<PriorDenial> {
1136        self.touch(key);
1137        self.by_key
1138            .get(key)
1139            .map(|(_, entries)| entries.iter().rev().cloned().collect())
1140            .unwrap_or_default()
1141    }
1142}

1215    /// Build from adapters that are already constructed (the boot path
1216    /// builds them with [`build_adapters`]; tests pass mocks).

1218        config: &JudgeConfig,
1219        adapters: BTreeMap<String, Arc<dyn pv_judge::JudgeAdapter>>,
1220        metrics: Arc<pv_core::metrics::Registry>,
1221    ) -> Result<Self, String> {
1222        let mut errors = Vec::new();
1223        config.collect_errors(&mut errors);
1224        if !errors.is_empty() {
1225            return Err(errors.join("; "));
1226        }
1227        let registry = Arc::new(config.registry()?);
1228        let mut slots = BTreeMap::new();
1229        let mut adapter_kinds = BTreeMap::new();
1230        let mut max_deadline = Duration::ZERO;
1231        for adapter_config in &config.adapters {
1232            let adapter = adapters
1233                .get(&adapter_config.name)
1234                .cloned()
1235                .ok_or_else(|| format!("judge adapter {:?} was not built", adapter_config.name))?;
1236            let pin = adapter_config.pin()?;
1237            if adapter.id() != &pin {
1238                return Err(format!(
1239                    "judge adapter {:?} loaded {} but the config pins {pin}",
1240                    adapter_config.name,
1241                    adapter.id()
1242                ));
1243            }
1244            let deadline = adapter_config.deadline();
1245            max_deadline = max_deadline.max(deadline);
1246            adapter_kinds.insert(adapter_config.name.clone(), adapter.kind());
1247            slots.insert(adapter_config.name.clone(), AdapterSlot { adapter, deadline });
1248        }
1249        let background = JudgeRunner::new(
1250            slots.clone(),
1251            Arc::clone(&registry),
1252            config.background_max_concurrency,
1253        )?
1254        .waiting_for_permits();
1255        let breaker_runner = JudgeRunner::new(
1256            slots.clone(),
1257            Arc::clone(&registry),
1258            config.background_max_concurrency,
1259        )?;

1261        let tools = config
1262            .tools
1263            .iter()

1266                    .predicates
1267                    .iter()
1268                    .map(|binding| JudgeTask {
1269                        predicate_id: binding.id.clone(),
1270                        adapter: binding.adapter.clone(),
1271                    })
1272                    .collect();

1274            })
1275            .collect();

1278            .drift
1279            .as_ref()
1280            .map(|drift| std::path::PathBuf::from(&drift.state_file));

1285                        format!(
1286                            "judge drift state file {} does not parse ({error}); inspect it, then \
1287                             delete it to start the calibration check over",
1288                            path.display()
1289                        )
1290                    })?;
1291                    monitor
1292                        .restore(state)
1293                        .map_err(|error| format!("judge drift state file {}: {error}", path.display()))?;
1294                    let tripped = monitor.tripped();
1295                    if !tripped.is_empty() {
1296                        tracing::warn!(
1297                            predicates = ?tripped,
1298                            "judge predicates restored in ask-by-default mode from the drift state file"
1299                        );
1300                    }
1301                }
1302                Err(error) if error.kind() == std::io::ErrorKind::NotFound => {}
1303                Err(error) => {
1304                    return Err(format!("read judge drift state file {}: {error}", path.display()));
1305                }
1306            }
1307        }
1308        metrics.gauge(
1309            "pv_judge_drift_tripped",
1310            "Judge predicates in ask-by-default mode after a calibration drift check",

1314            background,

1352    /// Keep the denial log and the drift check in `store` as well, when
1353    /// the store is shared between instances. Has no effect for a store
1354    /// that is not, or after the first call.
1355    pub(crate) fn share_denials(&self, store: Arc<dyn pv_state::StateStore>) {
1356        if store.shares_recent_lists() {
1357            if let Some(monitor) = &self.drift {
1358                // A switch this instance's state file kept (the store drops
1359                // keys a day after their last write) is published again.
1360                for id in monitor.tripped() {
1361                    let key = drift_key(&id, "tripped");
1362                    if store.get(&key).is_ok_and(|value| value == 0) {
1363                        let _ = store.add(&key, 1);
1364                    }
1365                }
1366                // Resets made before this instance joined are not applied
1367                // to its state file's switches (failing closed).
1368                for id in self.runner.registry().iter().map(|def| def.id.as_str()) {
1369                    if let Ok(generation) = store.get(&drift_key(id, "reset")) {
1370                        self.drift_reset_seen.lock().insert(id.to_owned(), generation);
1371                    }
1372                }
1373                let _ = self.shared_drift.set(Arc::clone(&store));
1374            }
1375            let _ = self.shared_denials.set(store);
1376        }
1377    }

1384    /// The newest refusals under `key`: the shared log merged with this
1385    /// instance's, newest first. A shared log that cannot be read leaves
1386    /// this instance's entries and counts the failure.
1387    /// The refusals under `key`, every one made by `instance_uid` (the
1388    /// agent the key belongs to).
1389    fn recent_denials(&self, key: &str, instance_uid: &str) -> Vec<PriorDenial> {
1390        let local = self.denials.lock().recent(key);
1391        let Some(store) = self.shared_denials.get() else {
1392            return local;
1393        };
1394        let now_ms = pv_core::time::now_ms();
1395        let read = [true, false]
1396            .into_iter()
1397            .map(|judged| store.recent(&shared_denial_key(key, judged), SHARED_DENIALS_PER_KEY))
1398            .collect::<Result<Vec<_>, _>>();
1399        let shared = match read {
1400            Ok(lists) => lists
1401                .iter()
1402                .flatten()
1403                .filter(|entry| entry.len() <= MAX_SHARED_DENIAL_BYTES)
1404                .filter_map(|entry| serde_json::from_str::<PriorDenial>(entry).ok())
1405                .filter(|denial| denial.instance_uid == instance_uid && self.plausible_denial(denial, now_ms))
1406                .collect(),
1407            Err(error) => {
1408                tracing::warn!(%error, "shared judge denial log unreadable; using this instance's entries");
1409                self.count_shared_denial_error("read");
1410                Vec::new()
1411            }
1412        };
1413        merge_denials(shared, local)
1414    }

1416    /// A refusal read back from the shared store is used only if it is
1417    /// one the harness itself writes: a known denial code, a known
1418    /// predicate, a tool name of printable characters, and a time within
1419    /// the list's lifetime. The store is shared infrastructure, and this
1420    /// text reaches the judge.
1421    fn plausible_denial(&self, denial: &PriorDenial, now_ms: u64) -> bool {
1422        let known_code =
1423            serde_json::from_value::<pv_sandbox::DenialCode>(Value::String(denial.denial_code.clone()))
1424                .is_ok();
1425        let known_predicate = denial
1426            .predicate_id
1427            .as_deref()
1428            .is_none_or(|id| self.runner.registry().get(id).is_some());
1429        // No future times: an entry dated ahead would sort first and push
1430        // real ones out (an instance whose clock runs ahead has its entries
1431        // used once this clock catches up).
1432        let recent =
1433            denial.at_ms <= now_ms && denial.at_ms >= now_ms.saturating_sub(SHARED_DENIAL_MAX_AGE_MS);
1434        // The agent is checked by the caller, which keeps only entries made
1435        // by the agent the list belongs to: any id it may have is fine.
1436        // Any name the request parser accepts (printable ASCII); the entry
1437        // is at most MAX_SHARED_DENIAL_BYTES in all.
1438        let tool = !denial.tool.is_empty() && denial.tool.bytes().all(|b| b.is_ascii_graphic());
1439        tool && known_code && known_predicate && recent
1440    }

1602    /// Record a reviewer's answer for a sampled verdict. With a shared
1603    /// store, the review may have been sampled on another instance: its
1604    /// id names its predicate and band, a shared marker counts it once,
1605    /// and the statistics and switch are the shared ones.
1606    pub(crate) fn record_review(
1607        &self,
1608        review_id: &str,
1609        correct: bool,
1610    ) -> Result<pv_judge::ReviewResult, pv_judge::DriftError> {
1611        let Some(monitor) = &self.drift else {
1612            return Err(pv_judge::DriftError::UnknownReview);
1613        };
1614        let Some(store) = self.shared_drift.get() else {
1615            return monitor.record_review(review_id, correct);
1616        };
1617        let unavailable = |error: pv_state::StateError| pv_judge::DriftError::Unavailable(error.to_string());
1618        let (predicate, band) = parse_review_id(review_id).ok_or(pv_judge::DriftError::UnknownReview)?;
1619        if self.runner.registry().get(&predicate).is_none() || band >= pv_judge::drift::BAND_LOWER_PPM.len() {
1620            return Err(pv_judge::DriftError::UnknownReview);
1621        }
1622        // Issued at 1 when sampled; the first review makes it 2.
1623        let claim = drift_key(&predicate, &format!("review:{review_id}"));
1624        let claimed = store.add(&claim, 1).map_err(unavailable)?;
1625        if claimed != 2 {
1626            if claimed == 1 {
1627                store.remove(&claim);
1628            }
1629            return Err(pv_judge::DriftError::UnknownReview);
1630        }
1631        // Both counts in one atomic step (one Cluster slot), so another
1632        // instance never reads a review counted as reviewed but not yet as
1633        // correct; a failure gives the claim back, so the review can be
1634        // sent again.
1635        let count = |name: &str| pv_state::Spend {
1636            key: drift_key(&predicate, &format!("band:{band}:{name}")),
1637            amount: 1,
1638            limit: u64::from(u32::MAX),
1639        };
1640        let mut spends = vec![count("reviewed")];
1641        if correct {
1642            spends.push(count("correct"));
1643        }
1644        match store.try_spend_many(&spends) {
1645            Ok(pv_state::TrySpendOutcome::Committed { .. }) => {}
1646            Ok(_) => {
1647                store.refund(&claim, 1);
1648                return Err(pv_judge::DriftError::Unavailable(
1649                    "review counts are full".to_owned(),
1650                ));
1651            }
1652            Err(error) => {
1653                store.refund(&claim, 1);
1654                return Err(unavailable(error));
1655            }
1656        }
1657        store.remove(&claim);
1658        monitor.forget_review(review_id);
1659        let bands = self
1660            .shared_bands(store.as_ref(), &predicate)
1661            .map_err(unavailable)?;
1662        let config = monitor.config();
1663        let monotone =
1664            pv_judge::drift::is_monotone(&bands, config.min_reviews_per_band, config.tolerance_ppm);
1665        let tripped_key = drift_key(&predicate, "tripped");
1666        let was = store.get(&tripped_key).map_err(unavailable)? > 0;
1667        if !monotone && !was {
1668            store.add(&tripped_key, 1).map_err(unavailable)?;
1669        }
1670        let tripped = was || !monotone;
1671        if tripped {
1672            monitor.trip(&predicate);
1673        }
1674        self.shared_tripped.lock().0 = None;
1675        Ok(pv_judge::ReviewResult {
1676            predicate_id: predicate,
1677            tripped_now: tripped && !was,
1678            tripped,
1679        })
1680    }

1682    /// Operator reset of a predicate's drift switch and statistics, here
1683    /// and, with a shared store, on every instance. Returns whether it was
1684    /// tripped.
1685    pub(crate) fn reset_drift(&self, predicate_id: &str) -> Result<bool, pv_judge::DriftError> {
1686        let Some(monitor) = &self.drift else {
1687            return Ok(false);
1688        };
1689        let mut was = false;
1690        if let Some(store) = self.shared_drift.get() {
1691            // The reset must reach the store, or the other instances (and
1692            // this one, which writes its switch back) keep the switch.
1693            let unavailable =
1694                |error: pv_state::StateError| pv_judge::DriftError::Unavailable(error.to_string());
1695            let tripped_key = drift_key(predicate_id, "tripped");
1696            was |= store.get(&tripped_key).map_err(unavailable)? > 0;
1697            // Every instance clears its own switch when this count moves;
1698            // a switch that merely vanished from the store is kept.
1699            let generation = store
1700                .add(&drift_key(predicate_id, "reset"), 1)
1701                .map_err(unavailable)?;
1702            self.drift_reset_seen
1703                .lock()
1704                .insert(predicate_id.to_owned(), generation);
1705            store.remove(&tripped_key);
1706            for band in 0..pv_judge::drift::BAND_LOWER_PPM.len() {
1707                store.remove(&drift_key(predicate_id, &format!("band:{band}:reviewed")));
1708                store.remove(&drift_key(predicate_id, &format!("band:{band}:correct")));
1709            }
1710            self.shared_tripped.lock().0 = None;
1711        }
1712        was |= monitor.reset(predicate_id);
1713        Ok(was)
1714    }

1716    /// The drift check's state for operators: with a shared store, the
1717    /// shared statistics and switches.
1718    pub(crate) fn drift_snapshot(&self) -> Option<pv_judge::DriftSnapshot> {
1719        let monitor = self.drift.as_ref()?;
1720        let Some(store) = self.shared_drift.get() else {
1721            return Some(monitor.snapshot());
1722        };
1723        let tripped = self.drift_tripped();
1724        // Reviews this instance sampled but another one recorded are no
1725        // longer pending.
1726        for id in monitor.pending_ids() {
1727            let still_issued = parse_review_id(&id)
1728                .and_then(|(predicate, _)| store.get(&drift_key(&predicate, &format!("review:{id}"))).ok())
1729                .is_none_or(|issued| issued == 1);
1730            if !still_issued {
1731                monitor.forget_review(&id);
1732            }
1733        }
1734        let mut snapshot = monitor.snapshot();
1735        let shared: Result<Vec<_>, pv_state::StateError> = self
1736            .runner
1737            .registry()
1738            .iter()
1739            .map(|def| def.id.as_str())
1740            .map(|id| {
1741                let bands = self.shared_bands(store.as_ref(), id)?;
1742                let tripped = tripped.iter().any(|t| t == id);
1743                Ok((tripped || bands.iter().any(|band| band.reviewed > 0)).then(|| {
1744                    pv_judge::drift::PredicateDrift {
1745                        predicate_id: id.to_owned(),
1746                        bands,
1747                        tripped,
1748                    }
1749                }))
1750            })
1751            .collect();
1752        match shared {
1753            Ok(predicates) => snapshot.predicates = predicates.into_iter().flatten().collect(),
1754            // The store cannot be read: this instance's view, with the
1755            // switches that apply (never hiding one).
1756            Err(_) => {
1757                for id in &tripped {
1758                    match snapshot
1759                        .predicates
1760                        .iter_mut()

1789    /// The switches to ask-by-default, with a shared store: read at most
1790    /// every [`SHARED_DRIFT_REFRESH`] (on the path of judged calls), by one
1791    /// caller at a time, outside the lock the others read the result
1792    /// under. The switch fails closed:
1793    /// * a switch the store no longer holds (its keys expire a day after
1794    ///   the last write, or the store lost them) but this instance does
1795    ///   is written back, not dropped;
1796    /// * only an operator reset, seen as a new reset count, clears this
1797    ///   instance's switch (and its state file);
1798    /// * when the store cannot be read, this instance's own switches
1799    ///   apply until the next attempt, [`SHARED_DRIFT_REFRESH`] later.
1800    fn shared_tripped(&self) -> Vec<String> {
1801        let (Some(store), Some(monitor)) = (self.shared_drift.get(), &self.drift) else {
1802            return Vec::new();
1803        };
1804        {
1805            let cache = self.shared_tripped.lock();
1806            if cache.0.is_some_and(|at| at.elapsed() < SHARED_DRIFT_REFRESH) {
1807                return cache.1.clone();
1808            }
1809        }
1810        let Some(_refreshing) = self.drift_refreshing.try_lock() else {
1811            // Another caller is reading the store: the last result, or
1812            // this instance's own switches before the first one.
1813            let cache = self.shared_tripped.lock();
1814            return if cache.0.is_some() {
1815                cache.1.clone()
1816            } else {
1817                monitor.tripped()
1818            };
1819        };
1820        let mut reset_here = false;
1821        let mut read = || -> Result<Vec<String>, pv_state::StateError> {
1822            let mut tripped = Vec::new();
1823            for id in self.runner.registry().iter().map(|def| def.id.as_str()) {
1824                let key = drift_key(id, "tripped");
1825                let shared = store.get(&key)? > 0;
1826                let generation = store.get(&drift_key(id, "reset"))?;
1827                let seen = self.drift_reset_seen.lock().get(id).copied().unwrap_or(0);
1828                if generation > seen {
1829                    self.drift_reset_seen.lock().insert(id.to_owned(), generation);
1830                    if !shared && monitor.is_tripped(id) {
1831                        monitor.reset(id);
1832                        reset_here = true;
1833                    }
1834                }
1835                if shared {
1836                    store.add(&key, 0)?;
1837                    monitor.trip(id);
1838                    tripped.push(id.to_owned());
1839                } else if monitor.is_tripped(id) {
1840                    store.add(&key, 1)?;
1841                    tripped.push(id.to_owned());
1842                }
1843                for rest in std::iter::once("reset".to_owned()).chain(
1844                    (0..pv_judge::drift::BAND_LOWER_PPM.len())
1845                        .flat_map(|band| [format!("band:{band}:reviewed"), format!("band:{band}:correct")]),
1846                ) {
1847                    let key = drift_key(id, &rest);
1848                    if store.get(&key)? > 0 {
1849                        store.add(&key, 0)?;
1850                    }
1851                }
1852            }
1853            Ok(tripped)
1854        };
1855        let tripped = read().unwrap_or_else(|error| {
1856            tracing::warn!(%error, "shared drift switches unreadable; this instance's own switches apply");
1857            monitor.tripped()
1858        });
1859        if reset_here {
1860            if let Err(error) = self.persist_drift() {
1861                tracing::error!(error = %error, "an operator's drift reset was not saved on this instance");
1862            }
1863        }
1864        *self.shared_tripped.lock() = (Some(std::time::Instant::now()), tripped.clone());
1865        tripped
1866    }

1892    fn build_context(&self, inputs: &ContextInputs<'_>, shared: bool) -> SessionContext {
1893        let denials = |key: &str, instance_uid: &str| {
1894            if shared {
1895                self.recent_denials(key, instance_uid)
1896            } else {
1897                self.denials.lock().recent(key)
1898            }
1899        };
1900        let tasks: Vec<&str> = inputs
1901            .missions
1902            .iter()
1903            .filter_map(|mission| mission.task.as_deref())
1904            .collect();
1905        let mut allowed: Vec<String> = inputs
1906            .missions
1907            .iter()
1908            .flat_map(|mission| mission.allowed_intents.iter().cloned())
1909            .collect();
1910        allowed.sort();
1911        allowed.dedup();
1912        // Refusals of the tool being judged come first, so other refusals
1913        // cannot push them out of the judge's view.
1914        let same_tool_first = |mut denials: Vec<PriorDenial>| {
1915            denials.sort_by_key(|denial| denial.tool != inputs.tool);
1916            denials
1917        };
1918        let mut prior = Vec::new();
1919        prior.extend(
1920            same_tool_first(denials(
1921                &session_key(inputs.instance_uid, inputs.session_id),
1922                inputs.instance_uid,
1923            ))
1924            .into_iter()
1925            .take(8),
1926        );
1927        for parent in &inputs.parents {
1928            prior.extend(
1929                same_tool_first(denials(&instance_key(parent), parent))
1930                    .into_iter()
1931                    .map(|mut denial| {
1932                        denial.source = DenialSource::Parent;
1933                        denial
1934                    })
1935                    .take(8),
1936            );
1937        }
1938        prior.truncate(pv_judge::render::MAX_RENDERED_DENIALS);
1939        SessionContext {
1940            intent: Some(inputs.intent.clone()),
1941            task: (!tasks.is_empty()).then(|| tasks.join("\n")),
1942            mission_ids: inputs.missions.iter().map(|mission| mission.id.clone()).collect(),
1943            allowed_intents: allowed,
1944            charter: inputs.charter.to_owned(),
1945            delegation_depth: u32::try_from(inputs.delegation_depth).unwrap_or(u32::MAX),
1946            prior_denials: prior,
1947        }
1948    }

1950    /// Rebuild this instance's refusal log from recorded `agent.tool_call`
1951    /// events (the session chain), so a restart does not forget the
1952    /// refusals the judge sees as context. Only refusals the live path
1953    /// records count: policy-stage refusals of tools `known_tool` accepts,
1954    /// with a known code, made in the last 24 hours. Nothing is written
1955    /// to a shared store, which keeps its own. Returns how many refusals
1956    /// were restored.
1957    pub(crate) fn restore_denials(
1958        &self,
1959        events: &[Value],
1960        known_tool: &dyn Fn(&str) -> bool,
1961        now_ms: u64,
1962    ) -> usize {
1963        let mut restored: Vec<(u64, PriorDenial, String)> = events
1964            .iter()
1965            .filter(|event| event.get("class_name").and_then(Value::as_str) == Some("agent.tool_call"))
1966            .filter_map(|event| {
1967                let payload = event.get("payload")?;
1968                if payload.get("allowed") != Some(&Value::Bool(false))
1969                    || payload.get("stage").and_then(Value::as_str) != Some("policy")
1970                {
1971                    return None;
1972                }
1973                let tool = payload.get("tool").and_then(Value::as_str)?;
1974                let code = payload.get("denial_code").and_then(Value::as_str)?;
1975                let at_ms = event.get("time").and_then(Value::as_u64)?;
1976                let session = event.get("session_uid").and_then(Value::as_str)?;
1977                let instance = event.pointer("/ai_agent/instance_uid").and_then(Value::as_str)?;
1978                let known_code =
1979                    serde_json::from_value::<pv_sandbox::DenialCode>(Value::String(code.to_owned())).is_ok();
1980                let recent = at_ms <= now_ms && at_ms >= now_ms.saturating_sub(SHARED_DENIAL_MAX_AGE_MS);
1981                if !known_tool(tool) || !known_code || !recent {
1982                    return None;
1983                }
1984                let predicate_id = (payload.get("policy").and_then(Value::as_str)
1985                    == Some(pv_sandbox::JUDGE_POLICY_NAME))
1986                .then(|| {
1987                    payload
1988                        .pointer("/judge/decision_predicate")
1989                        .and_then(Value::as_str)
1990                })
1991                .flatten()
1992                .filter(|id| self.runner.registry().get(id).is_some())
1993                .map(str::to_owned);
1994                let denial = PriorDenial {
1995                    tool: tool.to_owned(),
1996                    denial_code: code.to_owned(),
1997                    predicate_id,
1998                    at_ms,
1999                    source: DenialSource::Session,
2000                    instance_uid: instance.to_owned(),
2001                };
2002                Some((at_ms, denial, session.to_owned()))
2003            })
2004            .collect();
2005        restored.sort_by_key(|(at_ms, _, _)| *at_ms);
2006        let mut log = self.denials.lock();
2007        for (_, denial, session) in &restored {
2008            log.push(session_key(&denial.instance_uid, session), denial.clone());
2009            log.push(instance_key(&denial.instance_uid), denial.clone());
2010        }
2011        restored.len()
2012    }

2014    /// Record a refusal under the session and the refusing agent.
2015    pub(crate) fn record_denial(
2016        &self,
2017        session_id: &str,
2018        instance_uid: &str,
2019        tool: &str,
2020        denial_code: &str,
2021        predicate_id: Option<String>,
2022    ) {
2023        let denial = PriorDenial {
2024            tool: tool.to_owned(),
2025            denial_code: denial_code.to_owned(),
2026            predicate_id,
2027            at_ms: pv_core::time::now_ms(),
2028            source: DenialSource::Session,
2029            instance_uid: instance_uid.to_owned(),
2030        };
2031        let keys = [session_key(instance_uid, session_id), instance_key(instance_uid)];
2032        if let Some(store) = self.shared_denials.get() {
2033            // The judge's refusals have a list of their own, so cheap
2034            // policy refusals cannot push them out of it.
2035            // A kind written in the last hour is not written again; one
2036            // whose write failed, or that keeps repeating, is.
2037            let judged = denial.predicate_id.is_some();
2038            let since_ms = denial.at_ms.saturating_sub(SHARED_DENIAL_REPEAT_MS);
2039            let marks: Vec<(String, String)> = keys
2040                .iter()
2041                .map(|key| (key.clone(), shared_mark(key, &denial)))
2042                .filter(|(_, mark)| {
2043                    self.last_shared
2044                        .lock()
2045                        .get(mark)
2046                        .is_none_or(|at_ms| *at_ms < since_ms)
2047                })
2048                .collect();
2049            match serde_json::to_string(&denial) {
2050                Ok(entry) => {
2051                    for (key, mark) in marks {
2052                        match store.push_recent(
2053                            &shared_denial_key(&key, judged),
2054                            &entry,
2055                            SHARED_DENIALS_PER_KEY,
2056                        ) {
2057                            Ok(()) => {
2058                                let mut last = self.last_shared.lock();
2059                                if last.len() >= DENIAL_LOG_KEYS {
2060                                    last.retain(|_, at_ms| *at_ms >= since_ms);
2061                                }
2062                                if last.len() >= DENIAL_LOG_KEYS {
2063                                    // Still full: forget the oldest write (at
2064                                    // worst, that kind is written again).
2065                                    if let Some(oldest) = last
2066                                        .iter()
2067                                        .min_by_key(|(_, at_ms)| **at_ms)
2068                                        .map(|(mark, _)| mark.clone())
2069                                    {
2070                                        last.remove(&oldest);
2071                                    }
2072                                }
2073                                last.insert(mark, denial.at_ms);
2074                            }
2075                            Err(error) => {
2076                                tracing::warn!(%error, "shared judge denial log unwritable; kept on this instance");
2077                                self.count_shared_denial_error("write");
2078                            }
2079                        }
2080                    }
2081                }
2082                Err(error) => tracing::warn!(%error, "judge denial not serializable"),
2083            }
2084        }
2085        let mut log = self.denials.lock();
2086        for key in keys {
2087            log.push(key, denial.clone());
2088        }
2089    }

2091    /// Score `tool`'s predicates from synchronous code running on the
2092    /// blocking pool. The work runs on the async runtime; this thread
2093    /// waits at most the longest adapter deadline plus one second, so a
2094    /// misuse (calling from an async worker) degrades into timeouts —
2095    /// which fail closed — instead of a hang or a panic.
2096    pub(crate) fn evaluate_blocking(
2097        self: &Arc<Self>,
2098        tasks: &[JudgeTask],
2099        tool: &str,
2100        arguments: &Value,
2101        context: SessionContext,
2102        caller: &str,
2103    ) -> (Vec<JudgeOutcome>, u64) {
2104        let started = std::time::Instant::now();
2105        let fail_all = |kind: pv_judge::FailureKind, detail: &str| -> Vec<JudgeOutcome> {
2106            tasks
2107                .iter()
2108                .map(|task| {
2109                    JudgeOutcome::Failed(JudgeFailure {
2110                        predicate_id: task.predicate_id.clone(),
2111                        model: self
2112                            .runner
2113                            .adapter(&task.adapter)
2114                            .map(|slot| slot.adapter.id().clone())
2115                            .unwrap_or_else(|| unknown_pin(&task.adapter)),
2116                        kind,
2117                        detail: detail.to_owned(),
2118                        latency_us: micros_since(started),
2119                    })
2120                })
2121                .collect()
2122        };
2123        let Some(_slot) = CallerSlot::take(self, caller, tasks.len()) else {
2124            let outcomes = fail_all(
2125                pv_judge::FailureKind::Saturated,
2126                "this agent already has its share of judge calls in flight",
2127            );
2128            return (outcomes, micros_since(started));
2129        };
2130        let Ok(handle) = tokio::runtime::Handle::try_current() else {
2131            let outcomes = fail_all(
2132                pv_judge::FailureKind::Unavailable,
2133                "no async runtime for the judge",
2134            );
2135            return (outcomes, micros_since(started));
2136        };
2137        let (sender, receiver) = std::sync::mpsc::sync_channel(1);
2138        let service = Arc::clone(self);
2139        let owned_tasks = tasks.to_vec();
2140        let owned_tool = tool.to_owned();
2141        let owned_arguments = arguments.clone();
2142        handle.spawn(async move {
2143            let outcomes = service
2144                .runner
2145                .evaluate(&owned_tasks, &owned_tool, &owned_arguments, &context)
2146                .await;
2147            let _ = sender.send(outcomes);
2148        });
2149        let outcomes = match receiver.recv_timeout(self.max_deadline + Duration::from_secs(1)) {
2150            Ok(outcomes) => outcomes,
2151            Err(_) => fail_all(pv_judge::FailureKind::Timeout, "judge did not answer in time"),
2152        };
2153        let judge_us = micros_since(started);
2154        self.observe(&outcomes, judge_us);
2155        (outcomes, judge_us)
2156    }

2302/// The policy-stage refusals recorded at the end of each partition of the
2303/// embedded Bridge's `agent.tool_call` topic under `data_dir`: at most
2304/// `tail_bytes` read per partition segment (a regular file named
2305/// `p<N>.jsonl`), newest last within a partition. Only refused calls are
2306/// kept while reading, so memory follows the refusals, not the traffic.
2307/// Missing files, a missing topic and unreadable lines yield nothing.
2308pub(crate) fn recorded_tool_calls(data_dir: &std::path::Path, tail_bytes: u64) -> Vec<Value> {
2309    use std::io::{BufRead as _, Read as _, Seek as _};
2310    let directory = data_dir.join("topics").join("agent.tool_call");
2311    let Ok(entries) = std::fs::read_dir(&directory) else {
2312        return Vec::new();
2313    };
2314    let mut events = Vec::new();
2315    for entry in entries.flatten() {
2316        let path = entry.path();
2317        let is_segment = path
2318            .file_name()
2319            .and_then(std::ffi::OsStr::to_str)
2320            .and_then(|name| name.strip_prefix('p')?.strip_suffix(".jsonl"))
2321            .is_some_and(|number| !number.is_empty() && number.bytes().all(|b| b.is_ascii_digit()));
2322        let regular = entry.file_type().is_ok_and(|kind| kind.is_file());
2323        if !is_segment || !regular {
2324            continue;
2325        }
2326        let Ok(mut file) = std::fs::File::open(&path) else {
2327            continue;
2328        };
2329        let Ok(length) = file.metadata().map(|metadata| metadata.len()) else {
2330            continue;
2331        };
2332        let start = length.saturating_sub(tail_bytes);
2333        // Starting mid-file, the first line is a fragment unless the byte
2334        // before the start ends a line.
2335        let mut before = [0u8; 1];
2336        let mid_line = start > 0
2337            && (file.seek(std::io::SeekFrom::Start(start - 1)).is_err()
2338                || file.read_exact(&mut before).is_err()
2339                || before[0] != b'\n');
2340        if file.seek(std::io::SeekFrom::Start(start)).is_err() {
2341            continue;
2342        }
2343        let mut reader = std::io::BufReader::new(file.take(tail_bytes));
2344        let mut line = Vec::new();
2345        if mid_line && reader.read_until(b'\n', &mut line).is_err() {
2346            continue;
2347        }
2348        loop {
2349            line.clear();
2350            match reader.read_until(b'\n', &mut line) {
2351                // An unterminated last line is an append still in flight.
2352                Ok(0) | Err(_) => break,
2353                Ok(_) if line.last() != Some(&b'\n') => break,
2354                Ok(_) => {}
2355            }
2356            // Most lines are not refusals: skip them before parsing.
2357            if !line.windows(6).any(|window| window == b"policy") {
2358                continue;
2359            }
2360            let Ok(mut stored) = serde_json::from_slice::<Value>(&line) else {
2361                continue;
2362            };
2363            let refused = stored.pointer("/value/payload/allowed") == Some(&Value::Bool(false))
2364                && stored.pointer("/value/payload/stage").and_then(Value::as_str) == Some("policy");
2365            if refused {
2366                if let Some(event) = stored.get_mut("value") {
2367                    events.push(event.take());
2368                }
2369            }
2370        }
2371    }
2372    events
2373}

2433/// Shared and local entries for one key, newest first, without
2434/// duplicates (an entry this instance wrote is in both).
2435fn merge_denials(shared: Vec<PriorDenial>, local: Vec<PriorDenial>) -> Vec<PriorDenial> {
2436    // This instance's own refusals first (the shared store may hold
2437    // planted ones), then the shared ones; newest first within each; one
2438    // entry per kind of refusal.
2439    let by_time = |mut denials: Vec<PriorDenial>| {
2440        denials.sort_by_key(|denial| std::cmp::Reverse(denial.at_ms));
2441        denials
2442    };
2443    let mut merged: Vec<PriorDenial> = Vec::new();
2444    for denial in by_time(local).into_iter().chain(by_time(shared)) {
2445        if !merged.iter().any(|kept| same_kind(kept, &denial)) {
2446            merged.push(denial);
2447        }
2448    }
2449    // Over the limit: drop refusals that did not come from the judge
2450    // first, last-listed (shared, oldest) first.
2451    while merged.len() > DENIALS_PER_KEY {
2452        match merged.iter().rposition(|denial| denial.predicate_id.is_none()) {
2453            Some(last) => {
2454                merged.remove(last);
2455            }
2456            None => {
2457                merged.pop();
2458            }
2459        }
2460    }
2461    merged
2462}

2664/// Text for a judge to read: characters that break lines (controls and
2665/// the Unicode line and paragraph separators) become spaces. They carry
2666/// no meaning a judge needs, and JSON escapes them to six characters
2667/// each, which could push a window past what the judge reads.
2668/// `text` in windows of [`TRACE_WINDOW_CHARS`], each overlapping the
2669/// previous one by [`TRACE_WINDOW_OVERLAP`].
2670fn trace_windows(text: &[char]) -> Vec<String> {
2671    let mut windows = Vec::new();
2672    let mut start = 0;
2673    while start < text.len() {
2674        let end = (start + TRACE_WINDOW_CHARS).min(text.len());
2675        windows.push(text.get(start..end).unwrap_or_default().iter().collect());
2676        if end == text.len() {
2677            break;
2678        }
2679        start += TRACE_WINDOW_CHARS - TRACE_WINDOW_OVERLAP;
2680    }
2681    windows
2682}
2683
2684/// The first `max_chars` characters of `text` as the judge reads them:
2685/// line-breaking characters as spaces, and redacted by `redactor`. `text`
2686/// may run [`REDACTION_MARGIN_CHARS`] past `max_chars`: the redaction sees
2687/// that margin, so a secret the cut runs through is still recognized,
2688/// and the margin is then dropped. Redaction can lengthen the text (a
2689/// short address becomes "[REDACTED]"): it is kept whole up to `cap`
2690/// characters, and the flag says whether `cap` cut anything.
2691fn prepare_text(
2692    text: &str,
2693    max_chars: usize,
2694    cap: usize,
2695    redactor: Option<&dyn pv_judge::JudgeAdapter>,
2696) -> (Vec<char>, bool) {
2697    let mut prepared = judgeable_text(text);
2698    let margin = text.chars().count().saturating_sub(max_chars);
2699    if let Some(adapter) = redactor {
2700        adapter.redact_text(&mut prepared);
2701    }
2702    let mut chars: Vec<char> = prepared.chars().collect();
2703    chars.truncate(chars.len().saturating_sub(margin));
2704    let capped = chars.len() > cap;
2705    chars.truncate(cap);
2706    (chars, capped)
2707}

2709/// Where a value sits in a tool's JSON-RPC response, which decides what
2710/// of it the model may read.
2711#[derive(Clone, Copy)]
2712enum ResponsePart {
2713    /// The message itself: its `result` and `error` are read, its
2714    /// `jsonrpc` and `id` are not.
2715    Message,
2716    /// The `result`: its `content` parts, and every other key and value.
2717    Result,
2718    /// The `content` array.
2719    Parts,
2720    /// One content part: every value but its `type` and `mimeType`
2721    /// labels and the base64 `data` of an image or audio part.
2722    Part,
2723    /// An embedded resource: its `text` and `uri`, not its base64 `blob`.
2724    Resource,
2725    /// A label or encoded data (`type`, `mimeType`, `data`, `blob`): read
2726    /// only when it holds whitespace, which a label or base64 never does,
2727    /// so prose put there is still judged.
2728    Label,
2729    /// Anything else: every key and string.
2730    Any,
2731}

2733/// Collected text, kept up to `budget` characters and counted in full.
2734struct TextSink {
2735    text: String,
2736    kept: usize,
2737    total: usize,
2738    budget: usize,
2739}
2740
2741impl TextSink {
2742    fn new(budget: usize) -> Self {
2743        Self {
2744            text: String::new(),
2745            kept: 0,
2746            total: 0,
2747            budget,
2748        }
2749    }
2750
2751    fn push(&mut self, piece: &str) {
2752        let separator = usize::from(self.total > 0);
2753        let chars = piece.chars().count();
2754        if separator == 1 && self.kept < self.budget {
2755            self.text.push('\n');
2756            self.kept += 1;
2757        }
2758        let room = self.budget.saturating_sub(self.kept);
2759        self.text.extend(piece.chars().take(room));
2760        self.kept += chars.min(room);
2761        self.total += separator + chars;
2762    }
2763}

2765/// Walks a JSON value as it is parsed, pushing the text the model may
2766/// read into the sink; nothing else of the value is kept in memory.
2767struct TextWalk<'a> {
2768    sink: &'a mut TextSink,
2769    part: ResponsePart,
2770}
2771
2772impl<'de> serde::de::DeserializeSeed<'de> for TextWalk<'_> {
2773    type Value = ();
2774
2775    fn deserialize<D: serde::Deserializer<'de>>(self, deserializer: D) -> Result<(), D::Error> {
2776        deserializer.deserialize_any(self)
2777    }
2778}
2779
2780impl<'de> serde::de::Visitor<'de> for TextWalk<'_> {
2781    type Value = ();
2782
2783    fn expecting(&self, formatter: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
2784        formatter.write_str("JSON")
2785    }
2786
2787    fn visit_str<E: serde::de::Error>(self, value: &str) -> Result<(), E> {
2788        if !matches!(self.part, ResponsePart::Label) || value.chars().any(char::is_whitespace) {
2789            self.sink.push(value);
2790        }
2791        Ok(())
2792    }
2793
2794    fn visit_bool<E: serde::de::Error>(self, _: bool) -> Result<(), E> {
2795        Ok(())
2796    }
2797
2798    fn visit_i64<E: serde::de::Error>(self, _: i64) -> Result<(), E> {
2799        Ok(())
2800    }
2801
2802    fn visit_u64<E: serde::de::Error>(self, _: u64) -> Result<(), E> {
2803        Ok(())
2804    }
2805
2806    fn visit_f64<E: serde::de::Error>(self, _: f64) -> Result<(), E> {
2807        Ok(())
2808    }
2809
2810    fn visit_unit<E: serde::de::Error>(self) -> Result<(), E> {
2811        Ok(())
2812    }
2813
2814    fn visit_seq<A: serde::de::SeqAccess<'de>>(self, mut items: A) -> Result<(), A::Error> {
2815        let part = match self.part {
2816            ResponsePart::Message => ResponsePart::Message,
2817            ResponsePart::Parts => ResponsePart::Part,
2818            _ => ResponsePart::Any,
2819        };
2820        while items
2821            .next_element_seed(TextWalk {
2822                sink: &mut *self.sink,
2823                part,
2824            })?
2825            .is_some()
2826        {}
2827        Ok(())
2828    }
2829
2830    fn visit_map<A: serde::de::MapAccess<'de>>(self, mut fields: A) -> Result<(), A::Error> {
2831        while let Some(key) = fields.next_key::<String>()? {
2832            let part = match (self.part, key.as_str()) {
2833                (ResponsePart::Message, "result") => Some(ResponsePart::Result),
2834                (ResponsePart::Message, "error") => Some(ResponsePart::Any),
2835                (ResponsePart::Message, "jsonrpc" | "id") => None,
2836                (ResponsePart::Result, "content") => Some(ResponsePart::Parts),
2837                (ResponsePart::Part, "type" | "mimeType" | "data")
2838                | (ResponsePart::Resource, "blob" | "mimeType") => Some(ResponsePart::Label),
2839                (ResponsePart::Part, "resource") => Some(ResponsePart::Resource),
2840                (ResponsePart::Part | ResponsePart::Resource, _) => Some(ResponsePart::Any),
2841                _ => {
2842                    self.sink.push(&key);
2843                    Some(ResponsePart::Any)
2844                }
2845            };
2846            match part {
2847                Some(part) => fields.next_value_seed(TextWalk {
2848                    sink: &mut *self.sink,
2849                    part,
2850                })?,
2851                None => {
2852                    fields.next_value::<serde::de::IgnoredAny>()?;
2853                }
2854            }
2855        }
2856        Ok(())
2857    }
2858}

2860/// The text in a tool's response body, as the model may read it, and its
2861/// length in characters: every key and string in its JSON-RPC `result` or
2862/// `error`, except the `type` and `mimeType` labels of the `content`
2863/// parts. The body is one JSON value, or a server-sent event stream
2864/// decoded as the standard says (the `data` lines of an event joined). A
2865/// body that is neither is read as it is. Only the first `budget`
2866/// characters are kept.
2867fn tool_output_text(body: &str, budget: usize) -> (String, usize) {
2868    let body = body.strip_prefix('\u{feff}').unwrap_or(body);
2869    let walk = |json: &str, sink: &mut TextSink| -> bool {
2870        use serde::de::DeserializeSeed as _;
2871        let mut parser = serde_json::Deserializer::from_str(json);
2872        let walked = TextWalk {
2873            sink,
2874            part: ResponsePart::Message,
2875        }
2876        .deserialize(&mut parser);
2877        walked.is_ok() && parser.end().is_ok()
2878    };
2879    let mut sink = TextSink::new(budget);
2880    if walk(body, &mut sink) {
2881        return (sink.text, sink.total);
2882    }
2883    // An event stream only when every line is an event field or a
2884    // comment; a body with other lines is read whole.
2885    let is_stream = body.lines().flat_map(|line| line.split('\r')).all(|line| {
2886        let field = line.split(':').next().unwrap_or_default();
2887        line.is_empty() || line.starts_with(':') || matches!(field, "data" | "event" | "id" | "retry")
2888    });
2889    let mut decoder = crate::mcp_client::SseDecoder::default();
2890    let events = if is_stream {
2891        decoder
2892            .push(body.as_bytes())
2893            .and_then(|mut events| {
2894                events.extend(decoder.push(b"\n\n")?);
2895                Ok(events)
2896            })

2924/// `text` with every string literal decoded in place as JSON decodes it
2925/// (its quotes dropped), for JSON a strict parser refuses: an unpaired
2926/// surrogate escape becomes U+FFFD, an unknown escape stays as written,
2927/// and an unterminated string runs to the end. Text outside strings is
2928/// kept as it is.
2929fn decode_strings_in_place(text: &str) -> String {
2930    let mut decoded = String::with_capacity(text.len());
2931    let mut chars = text.chars().peekable();
2932    while let Some(c) = chars.next() {
2933        if c != '"' {
2934            decoded.push(c);
2935            continue;
2936        }
2937        let mut literal = String::new();
2938        let mut pending_high: Option<u32> = None;
2939        while let Some(c) = chars.next() {
2940            if c == '"' {
2941                break;
2942            }
2943            if c != '\\' {
2944                if pending_high.take().is_some() {
2945                    literal.push('\u{fffd}');
2946                }
2947                literal.push(c);
2948                continue;
2949            }
2950            let escaped = chars.next().unwrap_or('\\');
2951            let unit = if escaped == 'u' {
2952                let hex: String = chars.by_ref().take(4).collect();
2953                u32::from_str_radix(&hex, 16).ok()
2954            } else {
2955                None
2956            };
2957            if let Some(high) = pending_high.take() {
2958                match unit {
2959                    Some(low @ 0xDC00..=0xDFFF) => {
2960                        let code = 0x10000 + ((high - 0xD800) << 10) + (low - 0xDC00);
2961                        literal.push(char::from_u32(code).unwrap_or('\u{fffd}'));
2962                        continue;
2963                    }
2964                    _ => literal.push('\u{fffd}'),
2965                }
2966            }
2967            match (escaped, unit) {
2968                ('u', Some(high @ 0xD800..=0xDBFF)) => pending_high = Some(high),
2969                ('u', Some(code)) => literal.push(char::from_u32(code).unwrap_or('\u{fffd}')),
2970                ('n', _) => literal.push('\n'),
2971                ('t', _) => literal.push('\t'),
2972                ('r', _) => literal.push('\r'),
2973                ('b', _) => literal.push('\u{8}'),
2974                ('f', _) => literal.push('\u{c}'),
2975                ('"' | '\\' | '/', _) => literal.push(escaped),
2976                _ => {
2977                    literal.push('\\');
2978                    literal.push(escaped);
2979                }
2980            }
2981        }
2982        if pending_high.is_some() {
2983            literal.push('\u{fffd}');
2984        }
2985        decoded.push_str(&literal);
2986    }
2987    decoded
2988}

3518            .inc();

4542    /// A shared in-memory store that can be made to fail: every call,
4543    /// or the next `add` to a key containing a given text.
4544    #[derive(Default)]
4545    struct FlakyShared {
4546        inner: pv_state::InMemoryStore,
4547        down: std::sync::atomic::AtomicBool,
4548        fail_next_add: parking_lot::Mutex<Option<String>>,
4549    }
4550
4551    impl FlakyShared {
4552        fn check(&self, key: &str, adding: bool) -> Result<(), StateError> {
4553            if self.down.load(std::sync::atomic::Ordering::SeqCst) {
4554                return Err(StateError::Backend("down".into()));
4555            }
4556            let mut next = self.fail_next_add.lock();
4557            if adding && next.as_deref().is_some_and(|text| key.contains(text)) {
4558                *next = None;
4559                return Err(StateError::Backend("failed once".into()));
4560            }
4561            Ok(())
4562        }
4563    }
4564
4565    impl StateStore for FlakyShared {
4566        fn add(&self, key: &str, delta: u64) -> Result<u64, StateError> {
4567            self.check(key, true)?;
4568            self.inner.add(key, delta)
4569        }
4570        fn get(&self, key: &str) -> Result<u64, StateError> {
4571            self.check(key, false)?;
4572            self.inner.get(key)
4573        }
4574        fn try_spend_many(&self, spends: &[Spend]) -> Result<TrySpendOutcome, StateError> {
4575            for spend in spends {
4576                self.check(&spend.key, true)?;
4577            }
4578            self.inner.try_spend_many(spends)
4579        }
4580        fn remove(&self, key: &str) {
4581            self.inner.remove(key);
4582        }
4583        fn refund(&self, key: &str, amount: u64) {
4584            self.inner.refund(key, amount);
4585        }
4586        fn shares_recent_lists(&self) -> bool {
4587            true
4588        }
4589        fn push_recent(&self, _key: &str, _value: &str, _keep: usize) -> Result<(), StateError> {
4590            Ok(())
4591        }
4592        fn recent(&self, _key: &str, _max: usize) -> Result<Vec<String>, StateError> {
4593            Ok(Vec::new())
4594        }
4595    }
