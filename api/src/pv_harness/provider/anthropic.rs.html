1//! Anthropic Messages API translation, both directions.
2//!
3//! Request (OpenAI chat completion → `POST /v1/messages`):
4//!
5//! * every `system` (and `developer`) message is concatenated into the
6//!   top-level `system` string, separated by blank lines;
7//! * `user` content becomes text blocks, with base64 `data:` image
8//!   URLs as base64 image sources and `http(s)` image URLs as URL
9//!   image sources;
10//! * `assistant` text becomes a text block and each `tool_calls` entry
11//!   a `tool_use` block (`arguments` parsed into the `input` object);
12//! * consecutive `tool` messages become `tool_result` blocks grouped
13//!   into one user turn;
14//! * `tools` → `name`/`description`/`input_schema`; `tool_choice` →
15//!   `auto`/`none`/`any`/`tool`, with `parallel_tool_calls: false` as
16//!   `disable_parallel_tool_use`;
17//! * `max_tokens` is mandatory for Anthropic: the request's
18//!   `max_completion_tokens` or `max_tokens` is used, else the
19//!   configured `provider_default_max_tokens`;
20//! * `temperature`, `top_p`, `stop` → `stop_sequences`, `stream`, and
21//!   `user` → `metadata.user_id` carry over. `n` is handled by the
22//!   pipeline (one upstream request per choice) and never sent.
23//!
24//! Refused with a 400, because answering without them would silently
25//! give the client something other than what it asked for:
26//! `logprobs`/`top_logprobs`, and `response_format` of type
27//! `json_object` or `json_schema` (the Messages API has no JSON mode;
28//! `{"type": "text"}` is accepted).
29//!
30//! Dropped because the Messages API has no equivalent and they only
31//! steer sampling: `seed`, `presence_penalty`, `frequency_penalty`,
32//! `logit_bias`, `store`, `service_tier`, per-message `name` fields.
33//!
34//! Response: content blocks → `choices[0].message.content` (text) and
35//! `tool_calls` (`tool_use` → `id`, `function.name`, JSON-encoded
36//! `arguments`); `thinking` blocks → `reasoning_content`; `stop_reason`
37//! → `finish_reason` (`end_turn`/`stop_sequence` → `stop`,
38//! `max_tokens`/`model_context_window_exceeded` → `length`,
39//! `tool_use`/`pause_turn` → `tool_calls`, `refusal` →
40//! `content_filter`); usage `input_tokens` + cache creation + cache
41//! read → `prompt_tokens` (cache read also as
42//! `prompt_tokens_details.cached_tokens`), `output_tokens` →
43//! `completion_tokens`.
44

55/// Build the Messages API body for one OpenAI chat request. The
56/// result never carries `n`; multi-choice requests fan out upstream.
57pub(crate) fn build_request(payload: &Value, default_max_tokens: u64) -> Result<Value, String> {
58    let model = payload
59        .get("model")
60        .and_then(Value::as_str)
61        .filter(|model| !model.is_empty())
62        .ok_or_else(|| "model is required".to_owned())?;
63    let messages = payload
64        .get("messages")
65        .and_then(Value::as_array)
66        .ok_or_else(|| "messages must be an array".to_owned())?;
67    super::refuse_logprobs(payload)?;
68    match payload.pointer("/response_format/type").and_then(Value::as_str) {
69        None | Some("text") => {}
70        Some(other) => {
71            return Err(format!(
72                "response_format {other:?} is not available from Anthropic; ask for JSON in the prompt or use a tool"
73            ));
74        }
75    }
76
77    let mut system: Vec<String> = Vec::new();
78    let mut turns: Vec<Value> = Vec::new();
79    let mut pending_results: Vec<Value> = Vec::new();
80    let flush_results = |pending: &mut Vec<Value>, turns: &mut Vec<Value>| {
81        if !pending.is_empty() {
82            turns.push(json!({"role": "user", "content": std::mem::take(pending)}));
83        }
84    };
85    for message in messages {
86        let role = message.get("role").and_then(Value::as_str).unwrap_or_default();
87        match role {
88            "system" | "developer" => {
89                flush_results(&mut pending_results, &mut turns);
90                if let Some(text) = content_text(message.get("content")) {
91                    if !text.is_empty() {
92                        system.push(text);
93                    }
94                }
95            }
96            "user" => {
97                flush_results(&mut pending_results, &mut turns);
98                let blocks = user_blocks(message.get("content"))?;
99                if !blocks.is_empty() {
100                    turns.push(json!({"role": "user", "content": blocks}));
101                }
102            }
103            "assistant" => {
104                flush_results(&mut pending_results, &mut turns);
105                let mut blocks: Vec<Value> = Vec::new();
106                if let Some(text) = content_text(message.get("content")) {
107                    if !text.is_empty() {
108                        blocks.push(json!({"type": "text", "text": text}));
109                    }
110                }
111                if let Some(calls) = message.get("tool_calls").and_then(Value::as_array) {
112                    for call in calls {
113                        let id = call
114                            .get("id")
115                            .and_then(Value::as_str)
116                            .ok_or_else(|| "assistant tool_calls entries need an id".to_owned())?;
117                        let name = call
118                            .pointer("/function/name")
119                            .and_then(Value::as_str)
120                            .ok_or_else(|| "assistant tool_calls entries need function.name".to_owned())?;
121                        let input = arguments_object(call.pointer("/function/arguments"))?;
122                        blocks.push(json!({"type": "tool_use", "id": id, "name": name, "input": input}));
123                    }
124                }
125                if !blocks.is_empty() {
126                    turns.push(json!({"role": "assistant", "content": blocks}));
127                }
128            }
129            "tool" => {
130                let id = message
131                    .get("tool_call_id")
132                    .and_then(Value::as_str)
133                    .ok_or_else(|| "tool messages need a tool_call_id".to_owned())?;
134                let content = match message.get("content") {
135                    Some(Value::Array(parts)) => Value::Array(
136                        parts
137                            .iter()
138                            .filter_map(|part| part.get("text").and_then(Value::as_str))
139                            .map(|text| json!({"type": "text", "text": text}))
140                            .collect(),
141                    ),
142                    other => Value::String(content_text(other).unwrap_or_default()),
143                };
144                pending_results.push(json!({"type": "tool_result", "tool_use_id": id, "content": content}));
145            }
146            other => return Err(format!("unsupported message role {other:?}")),
147        }
148    }
149    flush_results(&mut pending_results, &mut turns);
150
151    let mut body = json!({
152        "model": model,
153        "max_tokens": requested_max_tokens(payload).unwrap_or(default_max_tokens),
154        "messages": turns,
155    });
156    if !system.is_empty() {
157        set(&mut body, "system", Value::String(system.join("\n\n")));
158    }
159    for field in ["temperature", "top_p"] {
160        if let Some(value) = payload.get(field).filter(|value| value.is_number()) {
161            set(&mut body, field, value.clone());
162        }
163    }
164    let stops = stop_sequences(payload);
165    if !stops.is_empty() {
166        set(&mut body, "stop_sequences", Value::Array(stops));
167    }
168    if payload.get("stream").and_then(Value::as_bool) == Some(true) {
169        set(&mut body, "stream", Value::Bool(true));
170    }
171    if let Some(user) = payload.get("user").and_then(Value::as_str) {
172        set(&mut body, "metadata", json!({"user_id": user}));
173    }
174    let tools = tool_definitions(payload.get("tools"))?;
175    if !tools.is_empty() {
176        set(&mut body, "tools", Value::Array(tools));
177        if let Some(choice) = tool_choice(payload)? {
178            set(&mut body, "tool_choice", choice);
179        }
180    }
181    Ok(body)
182}

184/// OpenAI user content → Anthropic content blocks.
185fn user_blocks(content: Option<&Value>) -> Result<Vec<Value>, String> {
186    let mut blocks = Vec::new();
187    match content {
188        Some(Value::String(text)) => {
189            if !text.is_empty() {
190                blocks.push(json!({"type": "text", "text": text}));
191            }
192        }
193        Some(Value::Array(parts)) => {
194            for part in parts {
195                match part.get("type").and_then(Value::as_str).unwrap_or_default() {
196                    "text" => {
197                        if let Some(text) = part.get("text").and_then(Value::as_str) {
198                            if !text.is_empty() {
199                                blocks.push(json!({"type": "text", "text": text}));
200                            }
201                        }
202                    }
203                    "image_url" => {
204                        let url = part
205                            .pointer("/image_url/url")
206                            .or_else(|| part.get("image_url"))
207                            .and_then(Value::as_str)
208                            .ok_or_else(|| "image_url parts need image_url.url".to_owned())?;
209                        let source = if let Some((media_type, data)) = base64_data_url(url) {
210                            json!({"type": "base64", "media_type": media_type, "data": data})
211                        } else if url.starts_with("https://") || url.starts_with("http://") {
212                            json!({"type": "url", "url": url})
213                        } else {
214                            return Err("image_url must be a base64 data URL or an http(s) URL".to_owned());
215                        };
216                        blocks.push(json!({"type": "image", "source": source}));
217                    }
218                    other => return Err(format!("unsupported content part type {other:?}")),
219                }
220            }
221        }
222        None | Some(Value::Null) => {}
223        Some(_) => return Err("message content must be a string or an array of parts".to_owned()),
224    }
225    Ok(blocks)
226}

228/// OpenAI `tools` → Anthropic tool definitions.
229fn tool_definitions(tools: Option<&Value>) -> Result<Vec<Value>, String> {
230    let Some(Value::Array(tools)) = tools else {
231        return Ok(Vec::new());
232    };
233    let mut out = Vec::with_capacity(tools.len());
234    for tool in tools {
235        if tool.get("type").and_then(Value::as_str) != Some("function") {
236            return Err("only tools of type \"function\" are supported".to_owned());
237        }
238        let function = tool
239            .get("function")
240            .ok_or_else(|| "function tools need a function object".to_owned())?;
241        let name = function
242            .get("name")
243            .and_then(Value::as_str)
244            .ok_or_else(|| "function tools need a name".to_owned())?;
245        let mut definition = json!({
246            "name": name,
247            "input_schema": function
248                .get("parameters")
249                .cloned()
250                .unwrap_or_else(|| json!({"type": "object", "properties": {}})),
251        });
252        if let Some(description) = function.get("description").and_then(Value::as_str) {
253            set(
254                &mut definition,
255                "description",
256                Value::String(description.to_owned()),
257            );
258        }
259        out.push(definition);
260    }
261    Ok(out)
262}

264/// OpenAI `tool_choice` (+ `parallel_tool_calls`) → Anthropic
265/// `tool_choice`.
266fn tool_choice(payload: &Value) -> Result<Option<Value>, String> {
267    let disable_parallel = payload.get("parallel_tool_calls").and_then(Value::as_bool) == Some(false);
268    let mut choice = match payload.get("tool_choice") {
269        None | Some(Value::Null) => {
270            if disable_parallel {
271                json!({"type": "auto"})
272            } else {
273                return Ok(None);
274            }
275        }
276        Some(Value::String(mode)) => match mode.as_str() {
277            "auto" => json!({"type": "auto"}),
278            "none" => json!({"type": "none"}),
279            "required" => json!({"type": "any"}),
280            other => return Err(format!("unsupported tool_choice {other:?}")),
281        },
282        Some(object) => {
283            let name = object
284                .pointer("/function/name")
285                .and_then(Value::as_str)
286                .ok_or_else(|| "tool_choice objects need function.name".to_owned())?;
287            json!({"type": "tool", "name": name})
288        }
289    };
290    if disable_parallel && choice.get("type").and_then(Value::as_str) != Some("none") {
291        set(&mut choice, "disable_parallel_tool_use", Value::Bool(true));
292    }
293    Ok(Some(choice))
294}

296/// Anthropic `stop_reason` → OpenAI `finish_reason`. A turn that
297/// carries tool calls finishes with `tool_calls` even when Anthropic
298/// said `end_turn`, matching OpenAI semantics.
299pub(crate) fn map_stop_reason(native: &str, has_tool_calls: bool) -> String {
300    let mapped = match native {
301        "end_turn" | "stop_sequence" => "stop",
302        "max_tokens" | "model_context_window_exceeded" => "length",
303        "tool_use" | "pause_turn" => "tool_calls",
304        "refusal" => "content_filter",
305        other => other,
306    };
307    if mapped == "stop" && has_tool_calls {
308        return "tool_calls".to_owned();
309    }
310    mapped.to_owned()
311}

313/// Anthropic `usage` → totals. `input_tokens` excludes the cache-
314/// accounted input, so the true prompt volume is the sum of all three
315/// input counters.
316pub(crate) fn usage_totals(usage: Option<&Value>) -> Result<UsageTotals, String> {
317    let Some(usage) = usage.filter(|usage| !usage.is_null()) else {
318        return Ok(UsageTotals::default());
319    };
320    let input = usage_u64(usage.get("input_tokens"), "input_tokens")?;
321    let creation = usage_u64(
322        usage.get("cache_creation_input_tokens"),
323        "cache_creation_input_tokens",
324    )?;
325    let read = usage_u64(usage.get("cache_read_input_tokens"), "cache_read_input_tokens")?;
326    let prompt_tokens = match (input, creation, read) {
327        (None, None, None) => None,
328        (a, b, c) => {
329            let sum = a
330                .unwrap_or(0)
331                .saturating_add(b.unwrap_or(0))
332                .saturating_add(c.unwrap_or(0));
333            if sum > pv_core::error::JCS_SAFE_MAX {
334                return Err("upstream input token sum exceeds JCS-safe bounds".to_owned());
335            }
336            Some(sum)
337        }
338    };
339    Ok(UsageTotals {
340        prompt_tokens,
341        completion_tokens: usage_u64(usage.get("output_tokens"), "output_tokens")?,
342        cached_tokens: read,
343        reasoning_tokens: None,
344    })
345}

347/// Text, reasoning and tool calls gathered from a content-block array.
348struct Gathered {
349    text: String,
350    reasoning: String,
351    tool_calls: Vec<Value>,
352}
353
354fn gather_blocks(content: Option<&Value>) -> Result<Gathered, String> {
355    let mut gathered = Gathered {
356        text: String::new(),
357        reasoning: String::new(),
358        tool_calls: Vec::new(),
359    };
360    let Some(Value::Array(blocks)) = content else {
361        return Ok(gathered);
362    };
363    for block in blocks {
364        match block.get("type").and_then(Value::as_str).unwrap_or_default() {
365            "text" => {
366                if let Some(text) = block.get("text").and_then(Value::as_str) {
367                    gathered.text.push_str(text);
368                }
369            }
370            "thinking" => {
371                if let Some(text) = block.get("thinking").and_then(Value::as_str) {
372                    gathered.reasoning.push_str(text);
373                }
374            }
375            "tool_use" => {
376                let input = block
377                    .get("input")
378                    .filter(|input| !input.is_null())
379                    .cloned()
380                    .unwrap_or_else(|| json!({}));
381                let arguments = serde_json::to_string(&input)
382                    .map_err(|error| format!("tool input is not serializable: {error}"))?;
383                gathered.tool_calls.push(json!({
384                    "id": block.get("id").and_then(Value::as_str).map_or_else(super::fresh_tool_call_id, str::to_owned),
385                    "type": "function",
386                    "function": {
387                        "name": block.get("name").and_then(Value::as_str).unwrap_or("unknown"),
388                        "arguments": arguments,
389                    },
390                }));
391            }
392            // redacted_thinking, server_tool_use results, citations:
393            // nothing the OpenAI shape can carry.
394            _ => {}
395        }
396    }
397    Ok(gathered)
398}

400/// Translate one buffered Messages API response into an OpenAI
401/// `chat.completion` with a single choice at `index`.
402pub(crate) fn translate_message(body: &Value, id: &str, created: u64, index: u64) -> Result<Value, String> {
403    if body.get("type").and_then(Value::as_str) == Some("error") {
404        return Err(format!(
405            "upstream error: {}",
406            body.pointer("/error/message")
407                .and_then(Value::as_str)
408                .unwrap_or("(no message)")
409        ));
410    }
411    let gathered = gather_blocks(body.get("content"))?;
412    let finish_reason = body
413        .get("stop_reason")
414        .and_then(Value::as_str)
415        .map(|reason| map_stop_reason(reason, !gathered.tool_calls.is_empty()));
416    let mut message = json!({"role": "assistant"});
417    if gathered.text.is_empty() && !gathered.tool_calls.is_empty() {
418        set(&mut message, "content", Value::Null);
419    } else {
420        set(&mut message, "content", Value::String(gathered.text));
421    }
422    if !gathered.reasoning.is_empty() {
423        set(
424            &mut message,
425            "reasoning_content",
426            Value::String(gathered.reasoning),
427        );
428    }
429    if !gathered.tool_calls.is_empty() {
430        set(&mut message, "tool_calls", Value::Array(gathered.tool_calls));
431    }
432    Ok(json!({
433        "id": id,
434        "object": "chat.completion",
435        "created": created,
436        "model": body.get("model").and_then(Value::as_str).unwrap_or(""),
437        "choices": [{
438            "index": index,
439            "message": message,
440            "logprobs": Value::Null,
441            "finish_reason": finish_reason,
442        }],
443        "usage": usage_totals(body.get("usage"))?.to_openai(),
444    }))
445}

447/// Stateful translator for one Anthropic SSE stream, emitting chunks
448/// for one OpenAI choice.
449pub(crate) struct AnthropicStreamTranslator {
450    index: u64,
451    id: String,
452    model: String,
453    created: u64,
454    usage: UsageTotals,
455    /// Content-block index → OpenAI tool-call index within the choice.
456    tool_slots: BTreeMap<u64, u64>,
457    next_tool_index: u64,
458    saw_tool_call: bool,
459    started: bool,
460}
461
462impl AnthropicStreamTranslator {
463    pub(crate) fn new(index: u64, id: String, fallback_model: String, created: u64) -> Self {
464        Self {
465            index,
466            id,
467            model: fallback_model,
468            created,
469            usage: UsageTotals::default(),
470            tool_slots: BTreeMap::new(),
471            next_tool_index: 0,
472            saw_tool_call: false,
473            started: false,
474        }
475    }
476
477    fn emit(&mut self, mut delta: Value, finish_reason: Option<&str>) -> Value {
478        if !self.started {
479            self.started = true;
480            set(&mut delta, "role", Value::String("assistant".to_owned()));
481            if delta.get("content").is_none() {
482                set(&mut delta, "content", Value::String(String::new()));
483            }
484        }
485        chunk(
486            &self.id,
487            &self.model,
488            self.created,
489            self.index,
490            delta,
491            finish_reason,
492        )
493    }
494
495    fn tool_start(&mut self, block_index: u64, block: &Value) -> Value {
496        let slot = self.next_tool_index;
497        self.next_tool_index = self.next_tool_index.saturating_add(1);
498        self.tool_slots.insert(block_index, slot);
499        self.saw_tool_call = true;
500        let id = block
501            .get("id")
502            .and_then(Value::as_str)
503            .map_or_else(super::fresh_tool_call_id, str::to_owned);
504        let name = block.get("name").and_then(Value::as_str).unwrap_or("unknown");
505        self.emit(
506            json!({"tool_calls": [{"index": slot, "id": id, "type": "function", "function": {"name": name, "arguments": ""}}]}),
507            None,
508        )
509    }
510
511    /// A whole `message` document inside a stream (some gateways send
512    /// one): emit it as chunks.
513    fn whole_message(&mut self, body: &Value) -> Result<Vec<Value>, String> {
514        let gathered = gather_blocks(body.get("content"))?;
515        if let Some(model) = body.get("model").and_then(Value::as_str) {
516            self.model = model.to_owned();
517        }
518        let mut chunks = Vec::new();
519        if !gathered.text.is_empty() {
520            chunks.push(self.emit(json!({"content": gathered.text}), None));
521        }
522        if !gathered.reasoning.is_empty() {
523            chunks.push(self.emit(json!({"reasoning_content": gathered.reasoning}), None));
524        }
525        for mut call in gathered.tool_calls {
526            let slot = self.next_tool_index;
527            self.next_tool_index = self.next_tool_index.saturating_add(1);
528            set(&mut call, "index", json!(slot));
529            self.saw_tool_call = true;
530            chunks.push(self.emit(json!({"tool_calls": [call]}), None));
531        }
532        if let Some(reason) = body.get("stop_reason").and_then(Value::as_str) {
533            let mapped = map_stop_reason(reason, self.saw_tool_call);
534            chunks.push(self.emit(json!({}), Some(&mapped)));
535        }
536        self.usage.add(&usage_totals(body.get("usage"))?);
537        Ok(chunks)
538    }
539}

541impl StreamTranslator for AnthropicStreamTranslator {
542    fn frame(&mut self, raw: &str) -> Result<Vec<Value>, String> {
543        let frame = split_sse_frame(raw);
544        let Some(data) = frame.data else {
545            return Ok(Vec::new());
546        };
547        if data.trim() == "[DONE]" {
548            return Ok(Vec::new());
549        }
550        let value: Value =
551            serde_json::from_str(&data).map_err(|error| format!("invalid upstream JSON frame: {error}"))?;
552        let kind = value
553            .get("type")
554            .and_then(Value::as_str)
555            .unwrap_or_default()
556            .to_owned();
557        // The SSE event name and the payload's `type` are redundant in
558        // this dialect; require agreement so a frame cannot smuggle an
559        // error payload under a content event name (or vice versa).
560        // `message` is the SSE default event name and is treated as
561        // unset.
562        if frame.is_sse && !frame.event.is_empty() && frame.event != "message" && frame.event != kind {
563            return Err(format!(
564                "upstream SSE event name {:?} does not match payload type {kind:?}",
565                frame.event
566            ));
567        }
568        match kind.as_str() {
569            "ping" | "message_stop" | "content_block_stop" => Ok(Vec::new()),
570            "error" => Err(format!(
571                "upstream streamed an error event: {}",
572                value
573                    .pointer("/error/message")
574                    .and_then(Value::as_str)
575                    .unwrap_or("(no message)")
576            )),
577            "message_start" => {
578                if let Some(model) = value.pointer("/message/model").and_then(Value::as_str) {
579                    self.model = model.to_owned();
580                }
581                // message_start reports the prompt side; output_tokens
582                // here is a placeholder that message_delta supersedes.
583                let start = usage_totals(value.pointer("/message/usage"))?;
584                self.usage.prompt_tokens = start.prompt_tokens;
585                self.usage.cached_tokens = start.cached_tokens;
586                if self.usage.completion_tokens.is_none() {
587                    self.usage.completion_tokens = start.completion_tokens;
588                }
589                Ok(vec![self.emit(json!({"content": ""}), None)])
590            }
591            "content_block_start" => {
592                let block_index = usage_u64(value.get("index"), "content block index")?
593                    .ok_or_else(|| "upstream content block has no index".to_owned())?;
594                let block = value
595                    .get("content_block")
596                    .ok_or_else(|| "content_block_start has no content_block".to_owned())?;
597                match block.get("type").and_then(Value::as_str).unwrap_or_default() {
598                    "text" => Ok(block
599                        .get("text")
600                        .and_then(Value::as_str)
601                        .filter(|text| !text.is_empty())
602                        .map(|text| self.emit(json!({"content": text}), None))
603                        .into_iter()
604                        .collect()),
605                    "thinking" => Ok(block
606                        .get("thinking")
607                        .and_then(Value::as_str)
608                        .filter(|text| !text.is_empty())
609                        .map(|text| self.emit(json!({"reasoning_content": text}), None))
610                        .into_iter()
611                        .collect()),
612                    "tool_use" => Ok(vec![self.tool_start(block_index, block)]),
613                    _ => Ok(Vec::new()),
614                }
615            }
616            "content_block_delta" => {
617                let block_index = usage_u64(value.get("index"), "content block index")?
618                    .ok_or_else(|| "upstream content block has no index".to_owned())?;
619                let delta = value
620                    .get("delta")
621                    .ok_or_else(|| "content_block_delta has no delta".to_owned())?;
622                match delta.get("type").and_then(Value::as_str).unwrap_or_default() {
623                    "text_delta" => {
624                        let text = delta.get("text").and_then(Value::as_str).unwrap_or_default();
625                        Ok(vec![self.emit(json!({"content": text}), None)])
626                    }
627                    "thinking_delta" => {
628                        let text = delta.get("thinking").and_then(Value::as_str).unwrap_or_default();
629                        Ok(vec![self.emit(json!({"reasoning_content": text}), None)])
630                    }
631                    "input_json_delta" => {
632                        let slot = *self.tool_slots.get(&block_index).ok_or_else(|| {
633                            format!("input_json_delta for unknown content block {block_index}")
634                        })?;
635                        let partial = delta
636                            .get("partial_json")
637                            .and_then(Value::as_str)
638                            .unwrap_or_default();
639                        Ok(vec![self.emit(
640                            json!({"tool_calls": [{"index": slot, "function": {"arguments": partial}}]}),
641                            None,
642                        )])
643                    }
644                    // signature_delta and friends: nothing to relay.
645                    _ => Ok(Vec::new()),
646                }
647            }
648            "message_delta" => {
649                let delta_usage = usage_totals(value.get("usage"))?;
650                if delta_usage.completion_tokens.is_some() {
651                    self.usage.completion_tokens = delta_usage.completion_tokens;
652                }
653                // Some gateways also repeat the input side here.
654                if delta_usage.prompt_tokens.is_some() {
655                    self.usage.prompt_tokens = delta_usage.prompt_tokens;
656                    self.usage.cached_tokens = delta_usage.cached_tokens;
657                }
658                match value.pointer("/delta/stop_reason").and_then(Value::as_str) {
659                    Some(reason) if !reason.is_empty() => {
660                        let mapped = map_stop_reason(reason, self.saw_tool_call);
661                        Ok(vec![self.emit(json!({}), Some(&mapped))])
662                    }
663                    _ => Ok(Vec::new()),
664                }
665            }
666            "message" => self.whole_message(&value),
667            // Unknown event types carry nothing this translation can
668            // express; Anthropic documents that clients must ignore
669            // them.
670            _ => Ok(Vec::new()),
671        }
672    }
673
674    fn finish(&mut self) -> Result<(Vec<Value>, UsageTotals), String> {
675        Ok((Vec::new(), self.usage))
676    }
