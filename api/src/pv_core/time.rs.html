1//! Time helpers.
2//!
3//! Wall-clock time is used for display and record timestamps; ordering
4//! decisions inside a session always use the per-session sequence number, never
5//! the wall clock (clock skew must not corrupt event-chain order — silent-error
6//! class D13.6 in the plan).
7
8use std::time::{SystemTime, UNIX_EPOCH};
9
10/// Milliseconds since the Unix epoch. Saturates at 0 if the system clock is
11/// before the epoch (never panics).
12pub fn now_ms() -> u64 {
13    SystemTime::now()
14        .duration_since(UNIX_EPOCH)
15        .map(|d| u64::try_from(d.as_millis()).unwrap_or(u64::MAX))
16        .unwrap_or(0)
17}
18
19/// RFC 3339 / ISO-8601 UTC timestamp with millisecond precision, e.g.
20/// `2026-08-10T17:03:05.123Z`. Hand-rolled civil-from-days conversion so we
21/// avoid a chrono dependency in the core crate.
22pub fn iso8601_ms(epoch_ms: u64) -> String {
23    let secs = epoch_ms / crate::units::MS_PER_SEC;
24    let ms = epoch_ms % crate::units::MS_PER_SEC;
25    let days = secs / crate::units::SECS_PER_DAY;
26    let rem = secs % crate::units::SECS_PER_DAY;
27    let (h, m, s) = (rem / 3600, (rem % 3600) / 60, rem % 60);
28    let (year, month, day) = civil_from_days(days as i64);
29    format!("{year:04}-{month:02}-{day:02}T{h:02}:{m:02}:{s:02}.{ms:03}Z")
30}
31
32/// Current time as ISO-8601 UTC.
33pub fn now_iso8601() -> String {
34    iso8601_ms(now_ms())
35}
36
37/// Elapsed microseconds since `started`, saturating at `u64::MAX` if a caller
38/// somehow measures longer than ~585 000 years.
39pub fn elapsed_us(started: std::time::Instant) -> u64 {
40    u64::try_from(started.elapsed().as_micros()).unwrap_or(u64::MAX)
41}

43/// Howard Hinnant's `civil_from_days` algorithm (public domain).
44fn civil_from_days(z: i64) -> (i64, u32, u32) {
45    let z = z + 719_468;
46    let era = if z >= 0 { z } else { z - 146_096 } / 146_097;
47    let doe = z - era * 146_097;
48    let yoe = (doe - doe / 1460 + doe / 36_524 - doe / 146_096) / 365;
49    let y = yoe + era * 400;
50    let doy = doe - (365 * yoe + yoe / 4 - yoe / 100);
51    let mp = (5 * doy + 2) / 153;
52    let d = doy - (153 * mp + 2) / 5 + 1;
53    let m = if mp < 10 { mp + 3 } else { mp - 9 };
54    #[allow(clippy::cast_possible_truncation, clippy::cast_sign_loss)]
55    (if m <= 2 { y + 1 } else { y }, m as u32, d as u32)
56}

85    /// Round-23 mutation finding: `iso8601_ms` takes u64, so the
86    /// negative-era branch of Hinnant's civil_from_days (`z - 146_096`)
87    /// is unreachable through the public API and its mutants survived.
88    /// The algorithm is deliberately kept general (i64) — pin the
89    /// pre-epoch branch directly so a future signed-epoch caller
90    /// inherits verified math, and pin the era boundaries the public
91    /// path CAN reach.
92    #[test]
93    fn civil_from_days_handles_negative_days_and_era_boundaries() {
94        // Day -1 = the day before the epoch.
95        assert_eq!(civil_from_days(-1), (1969, 12, 31));
96        // z == 0 boundary (day -719,468) = 0000-03-01, the era origin.
97        assert_eq!(civil_from_days(-719_468), (0, 3, 1));
98        // One day earlier crosses into the NEGATIVE-era branch: year 0
99        // is a proleptic leap year, so it's 0000-02-29.
100        assert_eq!(civil_from_days(-719_469), (0, 2, 29));
101        // Reachable era boundaries: century leap (2000-02-29) and the
102        // 2100 non-leap century (2100-02-28 is the last Feb day).
103        assert_eq!(civil_from_days(11_016), (2000, 2, 29));
104        assert_eq!(civil_from_days(47_540), (2100, 2, 28));
105        assert_eq!(civil_from_days(47_541), (2100, 3, 1));
106    }

212    /// Wall-clock time must not run backward as observed by consecutive
213    /// `now_ms()` calls under normal operation. Two rapid samples might tie,
214    /// but the second is never less than the first. (Pre-epoch clocks are a
215    /// separate concern: `now_ms` saturates at 0 rather than panicking, which
216    /// this smoke test cannot induce.)
217    #[test]
218    fn now_ms_never_panics_and_is_monotonic_under_normal_operation() {
219        let mut previous = now_ms();
220        for _ in 0..1_000 {
221            let current = now_ms();
222            assert!(
223                current >= previous,
224                "wall clock ran backward: {previous} -> {current}",
225            );

249    /// The Feb-28 -> Mar-1 boundary must respect leap-year rules for every
250    /// 1-, 4-, 100-, and 400-year cycle representable within u64.
251    #[test]
252    fn iso8601_ms_leap_year_boundaries_are_correct() {
253        // (feb_28_ms, expected_next_day_iso).
254        // 2000: 400-year rule -> leap, so Feb 29 exists.
255        // 2100: 100-year but not 400 -> non-leap, Feb 28 -> Mar 1.
256        // 2400: 400-year -> leap, Feb 29 exists.
257        // 2024: simple 4-year -> leap, Feb 29 exists.
258        // 2023: not divisible by 4 -> non-leap, Feb 28 -> Mar 1.
259        let cases: &[(u64, &str)] = &[
260            (951_696_000_000, "2000-02-29T00:00:00.000Z"), // 2000-02-28 + 1 day
261            (4_107_456_000_000, "2100-03-01T00:00:00.000Z"), // 2100-02-28 + 1 day
262            (
263                13_569_465_600_000 + 58 * crate::units::MS_PER_DAY,
264                "2400-02-29T00:00:00.000Z",
265            ), // 2400
266            (1_709_078_400_000, "2024-02-29T00:00:00.000Z"), // 2024-02-28 + 1 day
267            (1_677_542_400_000, "2023-03-01T00:00:00.000Z"), // 2023-02-28 + 1 day
268        ];
269        for &(feb_28_ms, expected_next) in cases {
270            let next_day = iso8601_ms(feb_28_ms + crate::units::MS_PER_DAY);
271            assert_eq!(next_day, expected_next, "feb_28_ms={feb_28_ms}");
272        }
273    }
