1//! Cross-platform filesystem helpers.
2//!
3//! `sync_directory` durably persists directory entries on platforms where the
4//! operating system supports it (Unix `fsync` on a directory file descriptor).
5//! On Windows there is no supported way to flush a directory handle
6//! (`FlushFileBuffers` requires a file, and opening a directory for `File::open`
7//! yields `PermissionDenied`), so this is a no-op there. These helpers do not
8//! establish an equivalent power-loss durability guarantee on Windows.
9
10use std::io;
11use std::path::Path;

13/// POSIX `O_NOFOLLOW` open-flag value for the target platform. Used
14/// alongside `std::os::unix::fs::OpenOptionsExt::custom_flags` to
15/// refuse to follow a symlink at the file path being opened —
16/// closes the symlink-plant confused-deputy class where a co-tenant
17/// with write access to a spool directory pre-plants a symlink at
18/// a deterministic file path so the harness's authenticated append
19/// (running as the harness UID) is redirected to a file the
20/// attacker can then read or overwrite.
21///
22/// Rather than pull in a `libc` / `rustix` dependency for one
23/// constant this returns the POSIX-defined value directly; the
24/// values are stable ABI and are drawn from the libc crate's
25/// `constant.O_NOFOLLOW` documentation. On unknown Unix-like
26/// platforms we fall back to the Linux value (a conservative
27/// choice: a mismatch would surface as an `ELOOP`/`ENOTDIR`/`EACCES`
28/// at first symlink attempt, not silent success).
29///
30/// CAUTION (round-99 find): on Linux the fcntl bits are per-ARCH,
31/// not per-OS. PowerPC swaps the asm-generic layout: 0x20000 there
32/// is O_DIRECT — passing it forced direct-I/O alignment on every
33/// spool append (EINVAL, all sessions fail-closed on ppc64le) while
34/// silently NOT refusing symlinks. Alpha/sparc differ too; neither
35/// has a supported Rust target here.
36#[cfg(unix)]
37pub const fn unix_o_nofollow() -> i32 {
38    #[cfg(all(
39        target_os = "linux",
40        not(any(target_arch = "powerpc", target_arch = "powerpc64"))
41    ))]
42    {
43        0x20000
44    }
45    #[cfg(all(target_os = "linux", any(target_arch = "powerpc", target_arch = "powerpc64")))]
46    {
47        0x8000
48    }
49    #[cfg(any(
50        target_os = "macos",
51        target_os = "ios",
52        target_os = "freebsd",
53        target_os = "openbsd",
54        target_os = "netbsd",
55        target_os = "dragonfly"
56    ))]
57    {
58        0x100
59    }
60    #[cfg(not(any(
61        target_os = "linux",
62        target_os = "macos",
63        target_os = "ios",
64        target_os = "freebsd",
65        target_os = "openbsd",
66        target_os = "netbsd",
67        target_os = "dragonfly"
68    )))]
69    {
70        0x20000
71    }
72}

74/// `O_NONBLOCK` for `OpenOptions::custom_flags`. Opening a FIFO
75/// read-only WITHOUT this flag blocks until a writer appears — a
76/// pre-planted FIFO (swapped in after any pathname precheck; pathname
77/// checks cannot close that race) would hang the caller forever
78/// before the post-open handle check could refuse it. With the flag,
79/// the open returns immediately and the handle check sees a
80/// non-regular file. Regular files ignore `O_NONBLOCK` entirely, so
81/// the flag is free on the happy path. Same per-arch caution as
82/// `unix_o_nofollow` above: Linux asm-generic uses 0o4000; the
83/// PowerPC layout swaps it; BSD/macOS use 0x4.
84#[cfg(unix)]
85pub const fn unix_o_nonblock() -> i32 {
86    // Unlike O_NOFOLLOW, O_NONBLOCK is 0o4000 across Linux
87    // arches (asm-generic AND the PowerPC layout).
88    #[cfg(target_os = "linux")]
89    {
90        0x800
91    }
92    #[cfg(any(
93        target_os = "macos",
94        target_os = "ios",
95        target_os = "freebsd",
96        target_os = "openbsd",
97        target_os = "netbsd",
98        target_os = "dragonfly"
99    ))]
100    {
101        0x4
102    }
103    #[cfg(not(any(
104        target_os = "linux",
105        target_os = "macos",
106        target_os = "ios",
107        target_os = "freebsd",
108        target_os = "openbsd",
109        target_os = "netbsd",
110        target_os = "dragonfly"
111    )))]
112    {
113        0x800
114    }
115}

117/// Return just the file name of `path` as a `&str`,
118/// suitable for `path = %basename(&path)` in tracing macros.
119///
120/// The concern is downstream OTLP export. Every `tracing::warn!(path
121/// = %path.display(), ...)` in this workspace flows through
122/// `tracing_opentelemetry::layer` when the `otel` feature is on, so
123/// absolute deployment paths (`/var/lib/proofvisor/spool/...`,
124/// custom outbox layouts, quarantine locations) land in whatever SIEM
125/// ingests OTLP — the same path-leak class already closed on the
126/// dashboard and on `%error` formatting of `reqwest::Error`,
127/// with a different producer / same sink. `basename` keeps enough
128/// context for operator triage (the file name usually encodes the
129/// session id or offset) without leaking the deployment topology.
130/// Non-UTF-8 file names or paths that end in `..` fall back to `?`;
131/// callers who need the full path server-side may still route it
132/// through a separate operator-only log channel.
133pub fn basename(path: &Path) -> &str {
134    path.file_name().and_then(|name| name.to_str()).unwrap_or("?")
135}

137/// Fsync a directory so its rename/create entries are durable after a crash.
138///
139/// On Unix this opens the directory and calls `sync_all` on the descriptor.
140/// On Windows it is a no-op (see module docs).
141pub fn sync_directory(path: &Path) -> io::Result<()> {
142    #[cfg(unix)]
143    {
144        std::fs::File::open(path)?.sync_all()
145    }
146    #[cfg(not(unix))]
147    {
148        let _ = path;
149        Ok(())
150    }
151}

153/// Create `path` and any missing ancestors, then confirm every parent entry
154/// in the resolved directory hierarchy with fsync. Existing directories need
155/// the same confirmation: they can remain visible after an earlier failed
156/// synchronization or a concurrent creator that has not synchronized yet.
157/// An empty path is a no-op. Symbolic links are resolved before synchronizing
158/// so the actual target hierarchy is confirmed, including relative paths.
159pub fn create_dir_all_synced(path: &Path) -> io::Result<()> {
160    create_dir_all_synced_with_directory_sync(path, sync_directory)
161}
162
163fn create_dir_all_synced_with_directory_sync(
164    path: &Path,
165    synchronize_parent: impl FnMut(&Path) -> io::Result<()>,
166) -> io::Result<()> {
167    if path.as_os_str().is_empty() {
168        return Ok(());
169    }
170    // Skip redundant mkdir syscalls for an existing directory, but never its
171    // durability check. A regular file or dangling symlink must still reach
172    // mkdir and report the collision instead of succeeding silently.
173    if !path.is_dir() {
174        // Pin new directories to 0700, independently of the operator's umask.
175        #[cfg(unix)]
176        {
177            use std::os::unix::fs::DirBuilderExt as _;
178            let mut builder = std::fs::DirBuilder::new();
179            builder.recursive(true).mode(0o700);
180            builder.create(path)?;
181        }
182        #[cfg(not(unix))]
183        {
184            std::fs::create_dir_all(path)?;
185        }
186    }
187    sync_directory_ancestors_with_directory_sync(path, synchronize_parent)
188}

190/// Confirm the parent entries of an existing directory without creating it.
191/// This includes every ancestor of the resolved target, highest parent first.
192/// The directory itself is not synchronized; callers persisting its contents
193/// must also call `sync_directory(path)`. Missing or non-directory paths fail.
194pub fn sync_directory_ancestors(path: &Path) -> io::Result<()> {
195    sync_directory_ancestors_with_directory_sync(path, sync_directory)
196}
197
198fn sync_directory_ancestors_with_directory_sync(
199    path: &Path,
200    mut synchronize_parent: impl FnMut(&Path) -> io::Result<()>,
201) -> io::Result<()> {
202    // Resolving the existing target handles relative paths, `..`, and existing
203    // symlinked ancestors such as macOS /tmp -> /private/tmp. Synchronizing
204    // only the lexical parents could miss the actual newly-created entries.
205    let resolved = std::fs::canonicalize(path)?;
206    if !std::fs::metadata(&resolved)?.is_dir() {
207        return Err(io::Error::new(
208            io::ErrorKind::NotADirectory,
209            "directory path no longer names a directory",
210        ));
211    }
212    let parents: Vec<_> = resolved.ancestors().skip(1).collect();
213    // Highest parent first. Do not cache success or infer it from existence:
214    // after a partial failure, every caller must confirm the chain itself.
215    // Retain all complete directories on error so a retry can repair them.
216    for parent in parents.into_iter().rev() {
217        synchronize_parent(parent)?;
218    }
219    Ok(())
220}

237/// Read a file into memory subject to a hard byte cap, refusing
238/// non-regular files. The size check runs on the OPEN handle (not
239/// the path — closes the TOCTOU race where a symlink target is
240/// swapped between `metadata()` and `read()`), and the read itself
241/// uses `Read::take` so a target that grows after the metadata
242/// check still cannot exceed the cap.
243///
244/// Shared between the CLI and the harness reconciler
245/// so both audit tools and the long-running server
246/// enforce identical resource bounds against on-disk tampering.
247pub fn read_capped(path: &Path, max_bytes: u64) -> io::Result<Vec<u8>> {
248    use std::io::Read as _;
249    // Pre-open type check: `File::open` on a FIFO blocks until a writer
250    // appears, so a special-file path hung the caller forever before
251    // the post-open handle check below could refuse it (observed:
252    // `pvctl receipt-verify <fifo>` hanging indefinitely). `stat` does
253    // not open the file, so it cannot block; the post-open handle check
254    // remains the TOCTOU-safe authority.
255    let pre = std::fs::metadata(path)?;
256    if !pre.is_file() {
257        return Err(io::Error::new(
258            io::ErrorKind::InvalidInput,
259            format!(
260                "{} is not a regular file (type: {:?})",
261                basename(path),
262                pre.file_type()
263            ),
264        ));
265    }
266    // O_NONBLOCK closes the residual race the pathname precheck cannot:
267    // a FIFO swapped in between `metadata` and `open` no longer blocks
268    // the open — it returns immediately and the handle check below
269    // refuses it. Regular files are unaffected by the flag.
270    // (No O_NOFOLLOW here: `pvctl` legitimately reads user-supplied
271    // paths through symlinks; the handle check operates on the resolved
272    // target, which is what the cap protects.)
273    let mut open_options = std::fs::OpenOptions::new();
274    open_options.read(true);
275    #[cfg(unix)]
276    {
277        use std::os::unix::fs::OpenOptionsExt as _;
278        open_options.custom_flags(unix_o_nonblock());
279    }
280    let mut file = open_options.open(path)?;
281    let metadata = file.metadata()?;
282    if !metadata.is_file() {
283        // Error messages must not embed the full
284        // absolute path — they surface into logs (defeating the
285        // basename discipline) and into HTTP close/promote error bodies.
286        return Err(io::Error::new(
287            io::ErrorKind::InvalidInput,
288            format!(
289                "{} is not a regular file (type: {:?})",
290                basename(path),
291                metadata.file_type()
292            ),
293        ));
294    }
295    if metadata.len() > max_bytes {
296        return Err(io::Error::new(
297            io::ErrorKind::InvalidData,
298            format!(
299                "{} is {} bytes; refusing to load more than {max_bytes}",
300                basename(path),
301                metadata.len()
302            ),
303        ));
304    }
305    let mut buf: Vec<u8> = Vec::with_capacity(usize::try_from(metadata.len()).unwrap_or(0));
306    (&mut file)
307        .take(max_bytes.saturating_add(1))
308        .read_to_end(&mut buf)?;
309    if buf.len() as u64 > max_bytes {
310        return Err(io::Error::new(
311            io::ErrorKind::InvalidData,
312            format!(
313                "{} grew past {max_bytes} bytes during read; refusing",
314                basename(path)
315            ),
316        ));
317    }
318    Ok(buf)
319}

321/// UTF-8 variant of [`read_capped`]. Used by the CLI for
322/// operator-supplied config / manifest / bearer token files where

329/// Durably write `bytes` to `path` via create-tmp + fsync + rename + parent fsync.
330///
331/// The temporary file uses a UUIDv7-derived suffix to avoid collision with any
332/// concurrent writer targeting the same final path, and is opened with
333/// `create_new` so a stale suffix collision fails safely instead of clobbering.
334/// Callers that already hold the parent directory pass any `Path`; missing
335/// parents are created before the temp file is opened.
336///
337/// If any step between temp creation and rename fails, the temp file is
338/// unlinked via an RAII guard so we never leak zero-byte `.tmp` files into
339/// the spool. A repeatedly-failing writer would otherwise fill the inode
340/// table on ext4/xfs long before the disk is full — an operational silent
341/// death.
342///
343/// `Ok(())` means both the file and its parent directory were successfully
344/// synchronized on platforms supporting directory fsync. An error can occur
345/// *after* rename: the complete new file may already be visible, but its
346/// durability is uncertain. Callers must fail closed and retain that file for
347/// recovery or an idempotent retry; an error does not mean the path is absent.
348/// In particular, visibility alone is insufficient for an audit marker that
349/// must survive a crash before an upstream request may be sent.
350pub fn write_atomic(path: &Path, bytes: &[u8]) -> io::Result<()> {
351    write_atomic_with_directory_sync(path, bytes, sync_directory)
352}
353
354fn write_atomic_with_directory_sync(
355    path: &Path,
356    bytes: &[u8],
357    synchronize_parent: impl FnOnce(&Path) -> io::Result<()>,
358) -> io::Result<()> {
359    use std::io::Write as _;
360    // `Path::new("file.json").parent()` is `Some("")`, not `None` — the
361    // empty path fails `sync_directory` (and is not a valid directory for
362    // `create_dir_all` on all platforms). Normalize to `.` so relative
363    // leaf paths behave like `./file.json`.
364    let parent = path
365        .parent()
366        .filter(|p| !p.as_os_str().is_empty())
367        .unwrap_or_else(|| Path::new("."));
368    // Confirm the complete parent hierarchy even on retry. Existing
369    // directories may be visible remnants of a failed ancestor fsync;
370    // synchronizing only the leaf after rename cannot repair that gap.
371    create_dir_all_synced(parent)?;
372    let temporary = path.with_extension(format!("{}.tmp", crate::new_event_uid()));
373    let mut guard = TempPathGuard::new(temporary.clone());
374    // The temp file is `create_new` (exclusive create — refuses a
375    // pre-planted symlink, per the audit-integrity contract this
376    // module documents). Pair that with an EXPLICIT 0o600 mode on
377    // Unix so the file's confidentiality does not depend on the
378    // caller's umask. ATIF trajectories carry full prompt/response
379    // transcripts; receipts carry identity, cost, and instance_uid;
380    // `.session.json` sidecars carry auth session state. On a host
381    // where umask is the default 0022, a bare `create_new` produces
382    // 0644 — world-readable by any co-tenant with `execute` on the
383    // spool directory. The 0o600 mode closes that surface uniformly
384    // for every spool site that uses this helper.
385    let mut options = std::fs::OpenOptions::new();
386    options.write(true).create_new(true);
387    #[cfg(unix)]
388    {
389        use std::os::unix::fs::OpenOptionsExt as _;
390        options.mode(0o600);
391    }
392    let mut file = options.open(&temporary)?;
393    file.write_all(bytes)?;
394    file.sync_all()?;
395    std::fs::rename(&temporary, path)?;
396    guard.disarm();
397    // Do not remove the destination on failure: rename committed visibility,
398    // and recovery needs the complete file even when durability is uncertain.
399    synchronize_parent(parent)
400}

402/// RAII guard that unlinks a temp path unless [`disarm`](Self::disarm) is
403/// called. Used to prevent orphan `.tmp` files when an intermediate step
404/// between `File::create` and `rename` fails.
405///
406/// Public so callers with their own atomic-rename recipes (harness
407/// `install_seed_exclusive`, per-crate tmp files) can reuse the same
408/// unlink-on-drop discipline as [`write_atomic`].
409pub struct TempPathGuard {
410    path: Option<std::path::PathBuf>,
411}
412
413impl TempPathGuard {
414    /// Arm the guard: `path` will be unlinked when the guard drops
415    /// unless [`disarm`](Self::disarm) is called first.
416    pub fn new(path: std::path::PathBuf) -> Self {
417        Self { path: Some(path) }
418    }
419
420    /// Consume the guard without unlinking (call once the temp has been
421    /// successfully renamed into its final path).
422    pub fn disarm(&mut self) {
423        self.path = None;
424    }
425}
426
427impl Drop for TempPathGuard {
428    fn drop(&mut self) {
429        if let Some(path) = self.path.take() {
430            let _ = std::fs::remove_file(&path);
431        }
432    }
433}

435/// Remove crash-orphaned `*.tmp` files under `root`, recursively.
436///
437/// The RAII unlink in [`write_atomic`]/[`TempPathGuard`] cannot run when
438/// the process is SIGKILLed between `create_new` and `rename`, so crash
439/// loops accumulate orphaned temp files linearly forever (empirically:
440/// 16 stranded temps across a 19-SIGKILL stress run, surviving every
441/// recovery pass) — the exact inode-exhaustion class the module doc
442/// promises to prevent. Callers MUST invoke this only at boot, before
443/// any concurrent writer exists, because a live writer's in-flight temp
444/// is indistinguishable from an orphan. UUIDv7 suffixes guarantee a
445/// name deleted here can never be re-created by a later writer.
446///
447/// Returns the number of files removed. Errors on individual entries
448/// are skipped (a temp that cannot be removed is the pre-existing
449/// condition, not a boot failure); only the root read errors surface.
450pub fn sweep_orphaned_tmp(root: &Path) -> io::Result<u64> {
451    const MAX_SWEEP_DEPTH: usize = 8;
452    fn walk(dir: &Path, depth: usize, removed: &mut u64) {
453        if depth > MAX_SWEEP_DEPTH {
454            return;
455        }
456        let Ok(entries) = std::fs::read_dir(dir) else {
457            return;
458        };
459        for entry in entries.flatten() {
460            let path = entry.path();
461            let Ok(file_type) = entry.file_type() else {
462                continue;
463            };
464            if file_type.is_dir() {
465                walk(&path, depth + 1, removed);
466            } else if file_type.is_file()
467                && path
468                    .file_name()
469                    .and_then(std::ffi::OsStr::to_str)
470                    .is_some_and(|name| name.ends_with(".tmp"))
471                && std::fs::remove_file(&path).is_ok()
472            {
473                *removed += 1;
474            }
475        }
476    }
477    match std::fs::metadata(root) {
478        Ok(_) => {}
479        Err(error) if error.kind() == io::ErrorKind::NotFound => return Ok(0),
480        Err(error) => return Err(error),
481    }
482    let mut removed = 0u64;
483    walk(root, 0, &mut removed);
484    Ok(removed)
485}
