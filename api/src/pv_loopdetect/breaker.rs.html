8/// Breaker tuning (config-file surface). Defaults implement the brief's rule:
9/// Δ≈0 across 3 consecutive steps while consuming N+ tokens.
10///
11/// Unknown keys are rejected so `[breaker]` typos fail loudly at startup.
12#[derive(Debug, Clone, Serialize, Deserialize)]
13#[serde(deny_unknown_fields)]
14pub struct BreakerConfig {
15    /// Δ threshold: a step with `1 - cosine < delta_epsilon` counts as
16    /// near-zero semantic progress.
17    pub delta_epsilon: f32,
18    /// Consecutive near-zero steps required to trip.
19    pub window: usize,
20    /// Minimum session token consumption before the breaker may trip
21    /// (prevents tripping on short legitimate confirmations).
22    pub min_tokens: u64,
23    /// Action on trip: `reject` (HTTP 403 — deliberately not the brief's
24    /// 429, which SDKs auto-retry; see `PipelineError::status`), `inject`
25    /// (corrective system
26    /// payload), or `abort` (connection abort) — the enforceable equivalents
27    /// of the brief's TCP RST / 429 / corrective-payload triple.
28    pub action: BreakerAction,
29}
30
31/// What the harness does when the breaker trips.
32#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
33#[serde(rename_all = "snake_case")]
34#[non_exhaustive]
35pub enum BreakerAction {
36    /// Respond with a permanent policy refusal (HTTP 403 in the harness;
37    /// deliberately not 429, which mainstream SDKs auto-retry — see
38    /// `PipelineError::status`).
39    Reject,
40    /// Inject a corrective system payload into the conversation.
41    Inject,
42    /// Abort the connection.
43    Abort,
44}
45
46impl Default for BreakerConfig {
47    fn default() -> Self {
48        // ε calibrated against measured HashEmbedder delta distributions
49        // (tests/calibrate.rs): paraphrase-loop steps score Δ ≈ 0.13–0.18,
50        // genuinely progressing steps Δ ≈ 0.88–0.98. 0.30 sits between with
51        // wide margins on both sides (semantic ONNX embedders push loop
52        // deltas even lower, so the margin only grows with better models).
53        Self {
54            delta_epsilon: 0.30,
55            window: 3,
56            min_tokens: 1_000,
57            action: BreakerAction::Reject,
58        }
59    }
60}

62/// Verdict for one observed step.
63#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
64#[non_exhaustive]
65pub enum BreakerVerdict {
66    /// Session is making progress.
67    Progressing {
68        /// Semantic delta vs the previous step (1 − cosine).
69        delta: f32,
70    },
71    /// Near-zero progress but the window/token gate hasn't tripped yet.
72    Suspicious {
73        /// Semantic delta vs the previous step.
74        delta: f32,
75        /// Consecutive near-zero steps so far.
76        streak: usize,
77    },
78    /// Loop detected — enforce now.
79    Tripped {
80        /// Semantic delta vs the previous step.
81        delta: f32,
82        /// Consecutive near-zero steps that tripped the breaker.
83        streak: usize,
84        /// Session tokens consumed at trip time.
85        tokens_consumed: u64,
86        /// Configured action.
87        action: BreakerAction,
88    },
89}

91/// Current breaker state (exposed to metrics / events).
92#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
93#[non_exhaustive]
94pub enum BreakerState {
95    /// Not tripped.
96    Closed,
97    /// Tripped: session enforcement active.
98    Open,
99}

101/// Per-session loop-detection state. Embedding/delta mutation happens on
102/// worker threads; the hot path reads [`SessionLoopState::state`] and
103/// [`SessionLoopState::action`], and calls [`SessionLoopState::reset`] when
104/// an Inject verdict fires.
105pub struct SessionLoopState {
106    cfg: BreakerConfig,
107    inner: Mutex<Inner>,
108}
109
110struct Inner {
111    /// Ring of the most recent usable embeddings. A window (not a
112    /// single slot) is required: a breaker that compares only against
113    /// the IMMEDIATELY preceding embedding lets simple alternation
114    /// (A, B, A, B, …) —
115    /// the canonical agent-loop shape — run forever in the default
116    /// build (cross-history tightening needed a Qdrant sink). The
117    /// delta is now the minimum against any embedding in this window,
118    /// so a step that duplicates ANY recent step grows the streak.
119    recent_embeddings: VecDeque<Vec<f32>>,
120    streak: usize,
121    tokens_consumed: u64,
122    state: BreakerState,
123}
124
125/// How many recent embeddings the alternation window retains. Eight
126/// covers alternation periods up to 8 (A,B,…,H,A,…) while keeping
127/// per-session memory bounded (8 × dim × 4 bytes ≈ 12 KiB at
128/// MiniLM's 384 dims).
129const RECENT_EMBEDDING_WINDOW: usize = 8;
130
131impl SessionLoopState {
132    /// Create with `cfg`.
133    pub fn new(cfg: BreakerConfig) -> Self {
134        Self {
135            cfg,
136            inner: Mutex::new(Inner {
137                recent_embeddings: VecDeque::with_capacity(RECENT_EMBEDDING_WINDOW),
138                streak: 0,
139                tokens_consumed: 0,
140                state: BreakerState::Closed,
141            }),
142        }
143    }
144
145    /// Current breaker state (hot-path read).
146    pub fn state(&self) -> BreakerState {
147        self.inner.lock().state
148    }
149    /// Configured enforcement action when the breaker is open.
150    pub fn action(&self) -> BreakerAction {
151        self.cfg.action
152    }
153
154    /// Observe one reasoning step (worker thread): embed, compute Δ against
155    /// the previous step, update the streak, return the verdict.
156    pub fn observe(&self, embedder: &dyn Embedder, text: &str, step_tokens: u64) -> BreakerVerdict {
157        let embedding = embedder.embed(text);
158        self.observe_embedding(embedding, step_tokens)
159    }
160
161    /// Observe a precomputed embedding. Workers use this path when the same
162    /// vector is also persisted to an off-path vector sink.
163    pub fn observe_embedding(&self, embedding: Vec<f32>, step_tokens: u64) -> BreakerVerdict {
164        self.observe_embedding_with_similarity(embedding, step_tokens, None)
165    }
166
167    /// Observe an embedding plus an optional nearest prior-step similarity
168    /// (scoped to the same session) supplied by a distributed vector engine.
169    pub fn observe_embedding_with_similarity(
170        &self,
171        embedding: Vec<f32>,
172        step_tokens: u64,
173        nearest_similarity: Option<f32>,
174    ) -> BreakerVerdict {
175        self.observe_embedding_with_opinion(embedding, step_tokens, nearest_similarity, None)
176    }
177
178    /// The delta [`Self::observe_embedding_with_similarity`] would compute
179    /// for this embedding right now, without recording anything. A caller
180    /// that wants a second opinion on an ambiguous step reads it first,
181    /// then records the step with [`Self::observe_embedding_with_opinion`]
182    /// (the worker processes one session's steps in order, so nothing
183    /// changes in between).
184    pub fn peek_delta(&self, embedding: &[f32], nearest_similarity: Option<f32>) -> f32 {
185        let inner = self.inner.lock();
186        step_delta(&inner.recent_embeddings, embedding, nearest_similarity)
187    }
188
189    /// Same as [`Self::observe_embedding_with_similarity`], except that
190    /// `near_zero`, when given, replaces the "delta below epsilon"
191    /// decision for this step: `Some(true)` counts it as no progress,
192    /// `Some(false)` as progress. The delta itself is still computed and
193    /// reported. An unusable embedding (non-finite or all zero) always
194    /// counts as no progress, whatever `near_zero` says.
195    pub fn observe_embedding_with_opinion(
196        &self,
197        embedding: Vec<f32>,
198        step_tokens: u64,
199        nearest_similarity: Option<f32>,
200        near_zero: Option<bool>,
201    ) -> BreakerVerdict {
202        let mut inner = self.inner.lock();
203        inner.tokens_consumed = inner.tokens_consumed.saturating_add(step_tokens);
204
205        let embedding_is_hostile = is_hostile(&embedding);
206        let delta = step_delta(&inner.recent_embeddings, &embedding, nearest_similarity);
207        // Only record the embedding for future comparisons when it's
208        // usable — a stored NaN vector would poison every subsequent
209        // step's cosine.
210        if !embedding_is_hostile {
211            inner.recent_embeddings.push_back(embedding);
212            if inner.recent_embeddings.len() > RECENT_EMBEDDING_WINDOW {
213                inner.recent_embeddings.pop_front();
214            }
215        }
216
217        // An unusable embedding always counts as no progress (fail
218        // closed); no opinion can turn it into progress.
219        let near_zero = if embedding_is_hostile {
220            Some(true)
221        } else {
222            near_zero
223        };
224        if near_zero.unwrap_or(delta < self.cfg.delta_epsilon) {
225            inner.streak += 1;
226        } else {
227            inner.streak = 0;
228        }
229
230        if inner.streak >= self.cfg.window && inner.tokens_consumed >= self.cfg.min_tokens {
231            inner.state = BreakerState::Open;
232            BreakerVerdict::Tripped {
233                delta,
234                streak: inner.streak,
235                tokens_consumed: inner.tokens_consumed,
236                action: self.cfg.action,
237            }
238        } else if inner.streak > 0 {
239            BreakerVerdict::Suspicious {
240                delta,
241                streak: inner.streak,
242            }
243        } else {
244            BreakerVerdict::Progressing { delta }
245        }
246    }
247
248    /// Release the retained embedding buffers.
249    ///
250    /// Unsigned sessions stay registered for the process lifetime by
251    /// design, and each one otherwise pins its recent response
252    /// embeddings (window × dim × f32) forever — dead weight after
253    /// close, since a closed session never calls `observe` again.
254    /// Unlike [`Self::reset`], the breaker verdict state is preserved.
255    pub fn release_embedding(&self) {
256        self.inner.lock().recent_embeddings.clear();
257    }
258
259    /// Manually reset (e.g. after a corrective injection gave the agent a new
260    /// direction). Clears the streak and the embedding window and closes the
261    /// breaker.
262    ///
263    /// `tokens_consumed` deliberately SURVIVES the reset.
264    /// It implements the documented "minimum session token consumption"
265    /// floor — zeroing it on every Inject turned the session-lifetime
266    /// floor into a per-cycle floor, letting an agent that ignores the
267    /// corrective message loop forever at ~min_tokens per correction.
268    pub fn reset(&self) {
269        let mut inner = self.inner.lock();
270        inner.streak = 0;
271        inner.state = BreakerState::Closed;
272        inner.recent_embeddings.clear();
273    }
274}

276/// Fail-closed on non-finite or all-zero embeddings: a hostile or
277/// misconfigured embedder can produce NaN/Inf vectors (any arithmetic
278/// through `cosine`'s clamp path would return NaN, and NaN <
279/// delta_epsilon is false, so the streak resets on every step and the
280/// breaker never trips). An all-zero vector is the ONNX embedder's error
281/// fallback — treating it as maximum novelty lets a client with
282/// malformed prompts drive the breaker into "novel-forever" mode. In both
283/// cases the step counts as a duplicate of the previous one (delta = 0),
284/// which conservatively grows the streak.
285fn is_hostile(embedding: &[f32]) -> bool {
286    let embedding_is_finite = embedding.iter().all(|x| x.is_finite());
287    let embedding_is_all_zero = embedding.iter().all(|x| *x == 0.0);
288    !embedding_is_finite || embedding_is_all_zero
289}
290
291/// Semantic delta of a step against the recent window (and the optional
292/// nearest prior-step similarity from a distributed vector engine).
293fn step_delta(
294    recent: &std::collections::VecDeque<Vec<f32>>,
295    embedding: &[f32],
296    nearest_similarity: Option<f32>,
297) -> f32 {
298    let adjacent_delta = if is_hostile(embedding) {
299        0.0
300    } else if recent.is_empty() {
301        1.0 // first step: maximum novelty by definition
302    } else {
303        // Minimum delta against ANY recent embedding — catches
304        // A,B,A,B alternation, not only exact-adjacent repeats.
305        recent
306            .iter()
307            .map(|prev| 1.0 - cosine(prev, embedding))
308            .fold(f32::INFINITY, f32::min)
309    };
310    nearest_similarity
311        .filter(|similarity| similarity.is_finite())
312        .map_or(adjacent_delta, |similarity| {
313            adjacent_delta.min(1.0 - similarity.clamp(-1.0, 1.0))
314        })
315}

634    /// (the ONNX embedder's error fallback) or NaN/Inf vectors must not
635    /// let a caller drive the breaker into "novel-forever" mode.
636    /// Zero-vector adjacent delta is `1.0 - cosine(0, 0)`; `cosine`

652            assert!(

661        assert!(
662            matches!(final_verdict, BreakerVerdict::Tripped { .. }),

698    /// (`1.0 - similarity`) had no direct test — a mutant deleting the
699    /// subtraction turns "nearly identical to history" (similarity ~1,
700    /// delta ~0) into "maximum novelty" (delta ~1) and the breaker
701    /// never trips through the vector-store path. Feed orthogonal
702    /// adjacent embeddings (adjacent delta = 1.0: never trips) with
703    /// near-1 nearest_similarity and require the trip; then prove
704    /// near-0 similarity does NOT trip.
705    #[test]
706    fn near_duplicate_history_similarity_trips_the_breaker() {
707        let cfg = BreakerConfig {
708            min_tokens: 100,
709            ..BreakerConfig::default()
710        };
711        let looping = SessionLoopState::new(cfg.clone());
712        let mut tripped = false;
713        for step in 0..8u32 {
714            // Orthogonal one-hot embeddings: adjacent cosine = 0, so
715            // only the similarity path can produce a small delta.
716            let mut e = vec![0.0f32; 16];
717            e[(step as usize) % 16] = 1.0;
718            let verdict = looping.observe_embedding_with_similarity(e, 50, Some(0.999));
719            if matches!(verdict, BreakerVerdict::Tripped { .. }) {
720                tripped = true;
721                break;
722            }
723        }
724        assert!(
725            tripped,
726            "similarity ~1 against history must trip within the window"
727        );
728
729        let progressing = SessionLoopState::new(cfg);
730        for step in 0..8u32 {
731            let mut e = vec![0.0f32; 16];
732            e[(step as usize) % 16] = 1.0;
733            let verdict = progressing.observe_embedding_with_similarity(e, 50, Some(0.01));
734            assert!(
735                !matches!(verdict, BreakerVerdict::Tripped { .. }),
736                "low similarity must not trip"

744    /// Japanese (no word boundaries) and Arabic must build a streak and

754                min_tokens: 1000,
755                ..BreakerConfig::default()
