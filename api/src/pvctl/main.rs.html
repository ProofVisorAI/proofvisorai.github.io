361/// Resolve sidecar settings (flags, then the settings file, then the
362/// environment, then defaults) and run it.
363async fn run_sidecar(
364    config: Option<PathBuf>,
365    gateway: Option<String>,
366    audience: Option<String>,
367    listen: Option<std::net::SocketAddr>,
368    credential: Option<String>,
369) -> Result<()> {
370    let file = config.as_deref().map(sidecar::SidecarFile::load).transpose()?;
371    let env = |name: &str| std::env::var(name).ok().filter(|value| !value.trim().is_empty());
372    let gateway = gateway
373        .or_else(|| file.as_ref().map(|file| file.gateway.clone()))
374        .or_else(|| env("PV_GATEWAY_URL"))
375        .context("the gateway URL is required (--gateway, --config, or PV_GATEWAY_URL)")?;
376    let audience = audience
377        .or_else(|| file.as_ref().map(|file| file.audience.clone()))
378        .or_else(|| env("PV_AUDIENCE"))
379        // Same pre-rename default as proofvisord's `audience`.
380        .unwrap_or_else(|| "agentvisor-ai".to_owned());
381    let listen = match (listen, file.as_ref()) {
382        (Some(listen), _) => listen,
383        (None, Some(file)) => file
384            .listen
385            .parse()
386            .with_context(|| format!("invalid listen address {:?}", file.listen))?,
387        (None, None) => std::net::SocketAddr::from(([127, 0, 0, 1], 8485)),
388    };
389    let credential = credential
390        .or_else(|| file.as_ref().map(|file| file.credential.clone()))
391        .unwrap_or_else(|| "cf-instance".to_owned());
392    let options = sidecar::SidecarOptions {
393        gateway,
394        audience,
395        credential: sidecar::Credential::parse(&credential)?,
396    };
397    sidecar::run(listen, options).await
398}

400fn main() -> std::process::ExitCode {
401    // Before any thread exists, and before clap reads `env = "PV_…"`
402    // fallbacks: adopt pre-rename AV_* variables (see pv_core::legacy).
403    pv_core::legacy::adopt_legacy_env();
404    let runtime = match tokio::runtime::Builder::new_multi_thread().enable_all().build() {
405        Ok(runtime) => runtime,
406        Err(error) => {
407            eprintln!("Error: could not start the async runtime: {error}");
408            return std::process::ExitCode::FAILURE;
409        }
410    };
411    runtime.block_on(async_main())
412}

414async fn async_main() -> std::process::ExitCode {
415    // Returning `Result<()>` relied on the default
416    // Termination impl printing anyhow's DEBUG format — which appends
417    // a full captured backtrace whenever RUST_BACKTRACE is exported,
418    // leaking 12 lines of internal frames on every well-formed user
419    // error (bad TOML, missing file). Print the Display chain only.
420    match run(Cli::parse()).await {
421        Ok(()) => std::process::ExitCode::SUCCESS,
422        Err(error) => {
423            eprintln!("Error: {error:#}");
424            std::process::ExitCode::FAILURE
425        }
426    }
427}

573/// The wizard reads answers from stdin; secrets come from the terminal
574/// with echo off when interactive, or from the same stream when piped

576async fn wizard_then_maybe_start() -> Result<()> {
577    let outcome = {
578        use std::io::IsTerminal as _;
579        let home = std::env::home_dir().context("could not find your home folder")?;
580        let stdin = std::io::stdin();
581        let secrets = if stdin.is_terminal() {
582            setup::SecretInput::Hidden
583        } else {
584            setup::SecretInput::Plain
585        };
586        let mut input = stdin.lock();
587        setup::wizard(&home, &mut input, &secrets)?
588    };
589    if outcome.start_now {
590        setup::start().await
591    } else {
592        println!("\nYour settings are saved at {}.", outcome.config_path.display());
593        println!("When you're ready: pvctl start");
594        Ok(())
595    }
596}

598fn keygen(path: &Path) -> Result<()> {
599    // `signer.seed()` returns `Zeroizing<[u8; 32]>`
600    // directly, and the hex encoding is separately wrapped so both
601    // stack and heap copies zero on drop rather than lingering in
602    // freed memory recoverable from a core dump.
603    use zeroize::Zeroizing;
604    let signer = Ed25519Signer::generate();
605    let seed = signer.seed();
606    // `&*seed` avoids clippy's needless_borrows lint
607    // suggestion (`*seed`) which would move 32 bytes out of the
608    // Zeroizing wrapper into a fresh un-zeroized temp slot.
609    #[allow(clippy::needless_borrows_for_generic_args)]
610    let encoded = Zeroizing::new(hex::encode(&*seed));
611    if !install_seed_exclusive(path, &encoded)? {
612        anyhow::bail!("refusing to overwrite existing key {}", path.display());
613    }
614    println!(
615        "{}",
616        serde_json::json!({
617            "key_id": signer.key_id(),
618            "public_key_hex": hex::encode(signer.public_key_bytes()),
619            "seed_file": path,
620        })
621    );
622    Ok(())
623}

625/// Load an Ed25519 seed file and print the derived public key + key_id.
626/// This exists so operators can pin a trusted public key for
627/// `pvctl receipt-verify` without ever exposing seed material — the
628/// only bytes read from disk are the hex-encoded seed, and only the
629/// derived public key is printed.
630fn pubkey(path: &Path) -> Result<()> {
631    use zeroize::Zeroizing;
632    let encoded = Zeroizing::new(
633        pv_core::fsutil::read_capped_string(path, pv_core::fsutil::MAX_CONTROL_BYTES)
634            .with_context(|| format!("read signing seed {}", path.display()))?,
635    );
636    let bytes = Zeroizing::new(hex::decode(encoded.trim()).context("decode signing seed as hex")?);
637    let seed: Zeroizing<[u8; 32]> = Zeroizing::new(
638        <[u8; 32]>::try_from(bytes.as_slice())
639            .map_err(|_| anyhow::anyhow!("signing seed must contain exactly 32 bytes"))?,
640    );
641    // Mirror the startup-guard refusal — printing the public key of a
642    // known-weak seed would give an operator false confidence that
643    // their pinned key is unique to this deployment.
644    if *seed == [0u8; 32] || *seed == [0xFFu8; 32] {
645        anyhow::bail!(
646            "signing seed at {} is a known-weak seed with a globally predictable public key; refusing to print",
647            path.display()
648        );
649    }
650    let signer = Ed25519Signer::from_seed(&seed);
651    println!(
652        "{}",
653        serde_json::json!({
654            "key_id": signer.key_id(),
655            "public_key_hex": hex::encode(signer.public_key_bytes()),
656            "seed_file": path,
657        })
658    );
659    Ok(())
660}

662fn install_seed_exclusive(path: &Path, encoded: &str) -> Result<bool> {
663    use std::io::Write as _;
664    let parent = path
665        .parent()
666        .filter(|parent| !parent.as_os_str().is_empty())
667        .unwrap_or_else(|| Path::new("."));
668    // Same rationale as harness main.rs `install_seed_exclusive`:
669    // route through `create_dir_all_synced` so newly-created ancestor
670    // dirents are fsynced (a bare `create_dir_all` leaves them
671    // volatile — power loss after install can drop the whole `keys/`
672    // directory even though the seed itself was fsynced) and so the
673    // leaf dir mode is 0o700 atomically at mkdir on Unix.
674    pv_core::fsutil::create_dir_all_synced(parent)
675        .with_context(|| format!("create key directory {}", parent.display()))?;
676    let temporary = parent.join(format!(".pvctl-key-{}.tmp", pv_core::new_event_uid()));
677    // Parity with setup.rs, harness main.rs, and
678    // cold_store.rs — arm an RAII guard so a transient IO failure
679    // (ENOSPC on write, EIO on sync, EROFS after hard_link, etc.)
680    // cannot leave a `.pvctl-key-<uuidv7>.tmp` containing live
681    // Ed25519 seed material on disk. Mode 0600 limits exposure to
682    // the running user, but the tmp survives reboots, gets picked
683    // up by backups, and appears in coredumps that snapshot the
684    // filesystem — none of which the operator will know to purge.
685    let mut guard = pv_core::fsutil::TempPathGuard::new(temporary.clone());
686    let mut options = std::fs::OpenOptions::new();
687    options.write(true).create_new(true);
688    #[cfg(unix)]
689    {
690        use std::os::unix::fs::OpenOptionsExt as _;
691        options.mode(0o600);
692    }
693    let mut file = options
694        .open(&temporary)
695        .with_context(|| format!("create key temporary file {}", temporary.display()))?;
696    file.write_all(encoded.as_bytes())
697        .with_context(|| format!("write key temporary file {}", temporary.display()))?;
698    file.sync_all()
699        .with_context(|| format!("sync key temporary file {}", temporary.display()))?;
700    match std::fs::hard_link(&temporary, path) {
701        Ok(()) => {
702            std::fs::remove_file(&temporary)
703                .with_context(|| format!("remove key temporary file {}", temporary.display()))?;
704            // The tmp is gone; disarm the guard so its Drop is a
705            // no-op (unlinking a non-existent path would race with a
706            // fresh keygen that happened to reuse the UUIDv7 tail).
707            guard.disarm();
708            // Durability is part of installation: without the parent
709            // fsync, a power loss can drop the dirent and the next
710            // keygen mints a DIFFERENT trust anchor. Fail with
711            // explicit operator guidance — the seed FILE at `path` is
712            // intact and must not be deleted; a re-run retries only
713            // the durability sync.
714            pv_core::fsutil::sync_directory(parent).with_context(|| {
715                format!(
716                    "signing seed installed at {} but the parent directory fsync failed; \
717                     the seed file is intact — do NOT delete it; re-run to retry the \
718                     durability sync",
719                    path.display()
720                )
721            })?;
722            Ok(true)
723        }
724        Err(error) if error.kind() == std::io::ErrorKind::AlreadyExists => {
725            // Guard's Drop unlinks the tmp; no manual removal needed.
726            // Re-establish dirent durability on every run that finds
727            // the seed already installed — this is where the hard-fail
728            // arm's "re-run to retry the durability sync" actually
729            // happens (and it heals installs from builds that only
730            // warned).
731            pv_core::fsutil::sync_directory(parent).with_context(|| {
732                format!(
733                    "signing seed exists at {} but its parent directory fsync failed; \
734                     the seed file is intact — do NOT delete it; re-run to retry",
735                    path.display()
736                )
737            })?;
738            Ok(false)
739        }
740        Err(error) => Err(error).with_context(|| format!("install key {}", path.display())),
741    }
742}

756/// Neutralise terminal-escape injection when printing
757/// attacker-influenced strings.
758///
759/// Anything reachable through a trusted third party — a signed receipt
760/// whose `receipt_id` was minted by a rotated/compromised signer, an
761/// ATIF file that a peer produced, an HTTP body from an upstream —
762/// can carry ESC / CSI / DEL / C1 bytes that reprogram the operator's
763/// terminal (clear screen, spoof a green "OK" line, rewrite the
764/// window title, or poison paste buffers on vulnerable emulators —
765/// CVE-2003-0063 class). Replace every control byte (C0 except TAB,
766/// DEL, and every C1) with U+FFFD before it flows into
767/// println!/eprintln!. Also replace the Trojan-Source family
768/// (CVE-2021-42574: bidi overrides/isolates, zero-width glyphs, and
769/// the U+2028/U+2029 line separators — pv-core's shared dangerous
770/// set): a receipt id like "safe\u{202E}suoicilam" would otherwise
771/// render reversed, letting two visually identical ids differ on
772/// the wire.
773fn sanitize_for_terminal(input: &str) -> String {
774    input
775        .chars()
776        .map(|c| {
777            let cp = c as u32;
778            // Preserve TAB (0x09) as the one C0 whitespace we allow;
779            // callers that need line-based output use format! + '\n'
780            // themselves before this function sees the string.
781            if cp == 0x09 {
782                c
783            } else if cp < 0x20
784                || cp == 0x7f
785                || (0x80..=0x9f).contains(&cp)
786                || pv_core::text::is_bidi_or_zero_width(c)
787            {
788                '\u{FFFD}'
789            } else {
790                c
791            }
792        })
793        .collect()
794}

808/// "the filename is `sha256(session_id)[..32]` with no
809/// lookup command" — the offline-verification workflow had no way to
810/// go from a session id to its receipt. Compute the stem and report
811/// every artifact class for it, including `archived-*` prior
812/// incarnations (a recycled session id archives the previous
813/// incarnation's artifact rather than overwriting it).
814fn receipt_locate(session_id: &str, spool: &Path) -> Result<()> {
815    let digest = pv_core::digest::sha256_hex(session_id.as_bytes());
816    let stem = digest.get(..32).unwrap_or(&digest);
817    let mut artifacts = serde_json::Map::new();
818    let candidates: [(&str, PathBuf); 5] = [
819        ("receipt", spool.join("receipts").join(format!("{stem}.json"))),
820        ("atif_trajectory", spool.join(format!("{stem}.json"))),
821        ("atif_provenance", spool.join(format!("{stem}.atif-auth"))),
822        ("event_journal", spool.join(format!("{stem}.events.ndjson"))),
823        ("journal_metadata", spool.join(format!("{stem}.session.json"))),
824    ];
825    for (label, path) in &candidates {
826        artifacts.insert(
827            (*label).to_owned(),
828            serde_json::json!({
829                "path": path,
830                "exists": path.exists(),
831            }),
832        );
833    }
834    // Archived prior incarnations: `{stem}.archived-*` in either the
835    // spool root (ATIF collisions) or receipts/ (receipt collisions).
836    let mut archived: Vec<String> = Vec::new();
837    for dir in [spool.to_path_buf(), spool.join("receipts")] {
838        let Ok(entries) = std::fs::read_dir(&dir) else {
839            continue;
840        };
841        for entry in entries.flatten() {
842            let name = entry.file_name();
843            let Some(name) = name.to_str() else { continue };
844            if name.starts_with(stem) && name.contains("archived-") {
845                archived.push(entry.path().display().to_string());
846            }
847        }
848    }
849    archived.sort();
850    println!(
851        "{}",
852        serde_json::json!({
853            "session_id": session_id,
854            "stem": stem,
855            "artifacts": artifacts,
856            "archived_prior_incarnations": archived,
857        })
858    );
859    Ok(())
860}

881fn receipt_verify(path: &Path, public_keys_hex: &[String]) -> Result<()> {
882    // Use the strict deserializer that refuses duplicate
883    // JSON keys at any nesting level. `pvctl receipt-verify` is the
884    // primary offline audit tool — a receipt that verified here but
885    // showed different content in `jq` (first-wins) would defeat
886    // the whole audit posture. `Receipt::from_json_slice` closes
887    // that split-brain uniformly.
888    let bytes = read_capped(path, MAX_RECEIPT_BYTES, "receipt")?;
889    let receipt =
890        Receipt::from_json_slice(&bytes).with_context(|| format!("parse receipt {}", path.display()))?;
891    // Accept multiple trusted keys so a key-rotation
892    // window is operable — pv_receipts::Keyring was always
893    // multi-key (selected by the receipt's key_id); only this CLI
894    // flag was single-valued.
895    let mut keyring = Keyring::new();
896    for (index, key_hex) in public_keys_hex.iter().enumerate() {
897        // Accept hex (what `pvctl pubkey` and the startup banner print)
898        // AND standard base64 (what the receipt's `public_key_b64`
899        // field carries) — auditors were hand-converting
900        // between the two encodings of the same 32 bytes.
901        let decoded = hex::decode(key_hex.trim()).or_else(|_| {
902            use base64::Engine as _;
903            base64::engine::general_purpose::STANDARD
904                .decode(key_hex.trim())
905                .map_err(|_| hex::FromHexError::InvalidStringLength)
906        });
907        let public_key: [u8; 32] = decoded
908            .with_context(|| {
909                format!(
910                    "trusted public key #{} is neither hexadecimal nor base64",
911                    index + 1
912                )
913            })?
914            .try_into()
915            .map_err(|_| {
916                anyhow::anyhow!("trusted public key #{} must contain exactly 32 bytes", index + 1)
917            })?;
918        keyring
919            .add_key_bytes(&public_key)
920            .with_context(|| format!("trusted public key #{} is invalid", index + 1))?;
921    }
922    receipt.verify(&keyring).context("receipt verification failed")?;
923    // Sanitise the receipt_id before printing. A signer
924    // trusted by the operator could otherwise mint a receipt whose
925    // receipt_id contains ESC/CSI bytes, reprogramming the auditor's
926    // terminal after the "verified" line.
927    println!("verified {}", sanitize_for_terminal(&receipt.body.receipt_id));
928    Ok(())
929}

931fn atif_validate(path: &Path, mode: ValidationMode) -> Result<()> {
932    let bytes = read_capped(path, MAX_ATIF_BYTES, "trajectory")?;
933    let mode = match mode {
934        ValidationMode::Strict => pv_atif::Mode::Strict,
935        ValidationMode::Compat => pv_atif::Mode::Compat,
936    };
937    // `validate_bytes` refuses duplicate keys before
938    // parsing (parallel to `Receipt::from_json_slice`'s
939    // strict scanner) and runs `validate_value` on the untyped
940    // form, which exercises unknown-field checks that the typed
941    // `Trajectory` (no `deny_unknown_fields`) silently drops.
942    let issues = match pv_atif::validate_bytes(&bytes, mode) {
943        Ok(issues) => issues,
944        Err(reason) => anyhow::bail!("parse {}: {reason}", path.display()),
945    };
946    if !issues.is_empty() {
947        // The reconciler already
948        // caps its render at first 16 + total; the CLI mirrors
949        // that, and also detects the pv_atif truncation
950        // marker (message contains "issue cap") so the summary
951        // says "at least N" rather than reporting the capped
952        // count as if it were exact.
953        const ATIF_HEAD: usize = 16;
954        let truncated = issues.last().is_some_and(|i| i.message.contains("issue cap"));
955        // Number of "real" issues: strip the synthetic marker if
956        // present. The total the user sees is a lower bound when
957        // truncated, exact otherwise.
958        let real_total = if truncated { issues.len() - 1 } else { issues.len() };
959        let shown = issues.iter().take(ATIF_HEAD);
960        for issue in shown {
961            // Sanitise every field of the issue since
962            // both `path` (a JSON pointer built from user JSON keys)
963            // and `message` (may embed a value snippet) come from
964            // attacker-supplied ATIF content.
965            eprintln!(
966                "{}: {}",
967                sanitize_for_terminal(&issue.path),
968                sanitize_for_terminal(&issue.message)
969            );
970        }
971        if real_total > ATIF_HEAD {
972            let suppressed = real_total - ATIF_HEAD;
973            let qualifier = if truncated { "at least " } else { "" };
974            eprintln!("... {qualifier}{suppressed} more issue(s) suppressed (showing first {ATIF_HEAD})");
975        }
976        if truncated {
977            anyhow::bail!("ATIF validation failed with at least {real_total} issue(s) (validator truncated)");
978        }
979        anyhow::bail!("ATIF validation failed with {real_total} issue(s)");
980    }
981    println!("valid {}", path.display());
982    Ok(())
983}

1041async fn session_promote(base_url: &str, id: &str, token_file: Option<&Path>) -> Result<()> {
1042    let url = promote_url(base_url, id)?;
1043    let token = bearer_token(token_file)?;
1044    // Every other CLI probe bounds its wait (see `doctor` and `probe_endpoint`
1045    // at 3 s); without an explicit timeout, a hung harness leaves this call
1046    // waiting forever.
1047    let client = reqwest::Client::builder()
1048        .timeout(std::time::Duration::from_secs(60))
1049        .connect_timeout(std::time::Duration::from_secs(5))
1050        // Never follow redirects: the request carries bearer_auth, and a
1051        // 3xx from an in-path proxy would silently re-send the NHI token
1052        // to whatever the Location header names. Matches every other
1053        // credential-bearing client in this crate (doctor, health).
1054        .redirect(reqwest::redirect::Policy::none())
1055        .build()
1056        .context("build promotion client")?;
1057    let mut request = client.post(url);
1058    if let Some(token) = token.as_ref() {
1059        request = request.bearer_auth(token.as_str());
1060    }
1061    let response = request.send().await.context("promote session")?;
1062    let status = response.status();
1063    // Cap the response body read. reqwest's
1064    // `response.text()` buffers the whole body with no built-in
1065    // limit; a misbehaving harness (or a stalled proxy) could keep
1066    // the CLI holding unbounded RAM waiting for EOF. Stream chunks
1067    // and refuse past MAX_CONTROL_BYTES (1 MiB — a promotion
1068    // receipt is a few hundred bytes).
1069    let body = read_capped_response(response, pv_core::fsutil::MAX_CONTROL_BYTES).await?;
1070    if !status.is_success() {
1071        // Harness error body is attacker-influencable
1072        // via header echoing / error messages — sanitise before
1073        // it lands on the operator's terminal.
1074        anyhow::bail!("promotion failed ({status}): {}", sanitize_for_terminal(&body));
1075    }
1076    // The promoted receipt is JSON — printing it
1077    // through serde_json::to_string is not attacker-controllable at
1078    // this layer, but the raw body could carry stray control bytes
1079    // if a proxy adulterates the response. Sanitise to be safe.
1080    println!("{}", sanitize_for_terminal(&body));
1081    Ok(())
1082}

1100/// Capped, streaming replacement for
1101/// `response.text().await`. Refuses to buffer more than `max_bytes`
1102/// before an EOF and surfaces the cap in the error text.
1103async fn read_capped_response(response: reqwest::Response, max_bytes: u64) -> Result<String> {
1104    use futures::StreamExt as _;
1105    let mut stream = response.bytes_stream();
1106    let mut buf: Vec<u8> = Vec::new();
1107    let cap = usize::try_from(max_bytes).unwrap_or(usize::MAX);
1108    while let Some(chunk) = stream.next().await {
1109        let chunk = chunk.context("read response chunk")?;
1110        if buf.len().saturating_add(chunk.len()) > cap {
1111            anyhow::bail!("response exceeded {max_bytes} bytes; refusing to buffer");
1112        }
1113        buf.extend_from_slice(&chunk);
1114    }
1115    String::from_utf8(buf).context("response body was not valid UTF-8")
1116}

1164/// Replay every judged `agent.tool_call` record in an embedded Bridge.
1165fn judge_export(data_dir: &Path, predicate: Option<&str>, would_block_only: bool) -> Result<()> {
1166    use std::io::Write as _;
1167    const TOPIC: &str = "agent.tool_call";
1168    const PAGE: usize = 10_000;
1169    let manifest_text = read_capped_str(
1170        &data_dir.join("manifest.yaml"),
1171        MAX_CONFIG_BYTES,
1172        "bridge manifest",
1173    )?;
1174    let manifest = BridgeManifest::from_yaml(&manifest_text).map_err(anyhow::Error::new)?;
1175    let partitions = manifest
1176        .topics
1177        .iter()
1178        .find(|topic| topic.name == TOPIC)
1179        .map(|topic| topic.partitions)
1180        .with_context(|| format!("the Bridge has no {TOPIC} topic"))?;
1181    let stdout = std::io::stdout();
1182    let mut out = std::io::BufWriter::new(stdout.lock());
1183    let (mut judged, mut without_inputs, mut exported) = (0u64, 0u64, 0u64);
1184    let (mut unreadable_records, mut unreadable_arguments) = (0u64, 0u64);
1185    for partition in 0..partitions {
1186        let mut offset = 0u64;
1187        loop {
1188            let page = EmbeddedBroker::fetch_read_only(data_dir, TOPIC, partition, offset, PAGE)
1189                .context("read the Bridge")?;
1190            let Some(last) = page.last() else { break };
1191            offset = last.offset.saturating_add(1);
1192            for stored in &page {
1193                let event = &stored.value;
1194                let Some(raw) = event.pointer("/payload/judge") else {
1195                    continue;
1196                };
1197                judged += 1;
1198                let Ok(record) = serde_json::from_value::<pv_judge::JudgeRecord>(raw.clone()) else {
1199                    unreadable_records += 1;
1200                    continue;
1201                };
1202                if would_block_only && !record.would_block {
1203                    continue;
1204                }
1205                let Some(inputs) = &record.inputs else {
1206                    without_inputs += 1;
1207                    continue;
1208                };
1209                // Arguments that are not JSON (a record from a build that
1210                // redacted the text itself) would ask a reviewer to label a
1211                // call nobody can see: skipped and counted.
1212                let Ok(arguments) = serde_json::from_str::<serde_json::Value>(&inputs.arguments_json) else {
1213                    unreadable_arguments += 1;
1214                    continue;
1215                };
1216                for verdict in &record.verdicts {
1217                    if predicate.is_some_and(|wanted| wanted != verdict.predicate_id) {
1218                        continue;
1219                    }
1220                    let said = verdict.probs.iter().max_by_key(|prob| prob.ppm).map_or_else(
1221                        || format!("judge {}", verdict.status),
1222                        |top| format!("judge said {} at {}", top.label, f64::from(top.ppm) / 1e6),
1223                    );
1224                    let example = pv_judge::calibration::LabelledExample {
1225                        predicate_id: verdict.predicate_id.clone(),
1226                        tool: inputs.tool.clone(),
1227                        arguments: arguments.clone(),
1228                        context: inputs.context.clone(),
1229                        label: String::new(),
1230                        source: Some(format!(
1231                            "recorded {} event {}: {said}",
1232                            event
1233                                .get("session_uid")
1234                                .and_then(serde_json::Value::as_str)
1235                                .unwrap_or("-"),
1236                            event
1237                                .pointer("/metadata/uid")
1238                                .and_then(serde_json::Value::as_str)
1239                                .unwrap_or("-"),
1240                        )),
1241                    };
1242                    serde_json::to_writer(&mut out, &example)?;
1243                    out.write_all(b"\n")?;
1244                    exported += 1;
1245                }
1246            }
1247        }
1248    }
1249    out.flush()?;
1250    eprintln!(
1251        "{exported} examples exported from {judged} judged calls ({without_inputs} recorded without \
1252         inputs: set [judge] record_inputs = true to keep them; {unreadable_records} records could \
1253         not be read; {unreadable_arguments} skipped because their arguments are not JSON). Set \
1254         each `label` before `pvctl judge-calibrate` reads the file."
1255    );
1256    Ok(())
1257}

1259fn judge_replay(data_dir: &Path, config: &Path, what_if: Option<&Path>) -> Result<()> {
1260    use pv_judge::replay::{replay, DecisionSummary};
1261    const TOPIC: &str = "agent.tool_call";
1262    const PAGE: usize = 10_000;
1263    let policy = judge_thresholds(config)?;
1264    let other = what_if.map(judge_thresholds).transpose()?;
1265    let manifest_text = read_capped_str(
1266        &data_dir.join("manifest.yaml"),
1267        MAX_CONFIG_BYTES,
1268        "bridge manifest",
1269    )?;
1270    let manifest = BridgeManifest::from_yaml(&manifest_text).map_err(anyhow::Error::new)?;
1271    let partitions = manifest
1272        .topics
1273        .iter()
1274        .find(|topic| topic.name == TOPIC)
1275        .map(|topic| topic.partitions)
1276        .with_context(|| format!("the Bridge has no {TOPIC} topic"))?;
1277
1278    let mut read = 0u64;
1279    let mut judged = 0u64;
1280    let mut mismatches = 0u64;
1281    let mut unreadable = 0u64;
1282    let mut changed = 0u64;
1283    let mut mismatch_list = Vec::new();
1284    let mut unreadable_list = Vec::new();
1285    let mut changed_list = Vec::new();
1286    let mut tools: std::collections::BTreeMap<String, ToolReplay> = std::collections::BTreeMap::new();
1287    for partition in 0..partitions {
1288        let mut offset = 0u64;
1289        loop {
1290            let page = EmbeddedBroker::fetch_read_only(data_dir, TOPIC, partition, offset, PAGE)
1291                .context("read the Bridge")?;
1292            let Some(last) = page.last() else { break };
1293            offset = last.offset.saturating_add(1);
1294            for stored in &page {
1295                read += 1;
1296                let event = &stored.value;
1297                let Some(raw) = event.pointer("/payload/judge") else {
1298                    continue;
1299                };
1300                judged += 1;
1301                let at = serde_json::json!({
1302                    "session_uid": event.get("session_uid"),
1303                    "event_uid": event.pointer("/metadata/uid"),
1304                    "partition": partition,
1305                    "offset": stored.offset,
1306                });
1307                let record = match serde_json::from_value::<pv_judge::JudgeRecord>(raw.clone())
1308                    .map_err(|error| error.to_string())
1309                    .and_then(|record| replay(&record, &policy).map(|decision| (record, decision)))
1310                {
1311                    Ok(pair) => pair,
1312                    Err(error) => {
1313                        unreadable += 1;
1314                        if unreadable_list.len() < MAX_REPLAY_LISTED {
1315                            unreadable_list.push(serde_json::json!({"at": at, "error": error}));
1316                        }
1317                        continue;
1318                    }
1319                };
1320                let (record, decision) = record;
1321                let tool = event
1322                    .pointer("/payload/tool")
1323                    .and_then(serde_json::Value::as_str)
1324                    .unwrap_or("unknown")
1325                    .to_owned();
1326                let entry = tools.entry(tool.clone()).or_default();
1327                entry.judged += 1;
1328                entry.recorded.add(record.decision);
1329                entry.would_block += u64::from(record.would_block);
1330                entry.refused += u64::from(
1331                    event
1332                        .pointer("/payload/allowed")
1333                        .and_then(serde_json::Value::as_bool)
1334                        == Some(false),
1335                );
1336                let recorded = DecisionSummary::recorded(&record);
1337                let replayed = DecisionSummary::of(&decision);
1338                if recorded != replayed {
1339                    mismatches += 1;
1340                    if mismatch_list.len() < MAX_REPLAY_LISTED {
1341                        mismatch_list.push(serde_json::json!({
1342                            "at": at, "tool": tool, "recorded": recorded, "replayed": replayed,
1343                        }));
1344                    }
1345                }
1346                if let Some(other) = &other {
1347                    let tried = replay(&record, other).map_err(anyhow::Error::msg)?;
1348                    entry
1349                        .what_if
1350                        .get_or_insert_with(ActionCounts::default)
1351                        .add(tried.action);
1352                    if tried.action != record.decision {
1353                        changed += 1;
1354                        if changed_list.len() < MAX_REPLAY_LISTED {
1355                            changed_list.push(serde_json::json!({
1356                                "at": at, "tool": tool, "recorded": recorded,
1357                                "what_if": DecisionSummary::of(&tried),
1358                            }));
1359                        }
1360                    }
1361                }
1362            }
1363        }
1364    }
1365    let mut report = serde_json::json!({
1366        "records_read": read,
1367        "judged": judged,
1368        "consistency": {
1369            "config": config.display().to_string(),
1370            "mismatches": mismatches,
1371            "unreadable": unreadable,
1372            "mismatch_examples": mismatch_list,
1373            "unreadable_examples": unreadable_list,
1374        },
1375        "tools": tools,
1376    });
1377    if let (Some(path), Some(object)) = (what_if, report.as_object_mut()) {
1378        object.insert(
1379            "what_if".to_owned(),
1380            serde_json::json!({
1381                "config": path.display().to_string(),
1382                "changed": changed,
1383                "changed_examples": changed_list,
1384            }),
1385        );
1386    }
1387    println!("{}", serde_json::to_string_pretty(&report)?);
1388    if mismatches > 0 || unreadable > 0 {
1389        anyhow::bail!(
1390            "{mismatches} recorded decision(s) do not follow from their verdicts under {} and {unreadable} \
1391             record(s) could not be replayed; check the receipts and the configuration in force at the time",
1392            config.display()
1393        );
1394    }
1395    Ok(())
1396}

1432fn config_validate(path: &Path, structural_only: bool) -> Result<()> {
1433    let text = read_capped_str(path, MAX_CONFIG_BYTES, "config")?;
1434    let config = pv_harness::HarnessConfig::from_toml(&text).map_err(anyhow::Error::msg)?;
1435    // A shape-valid config is useless if it selects
1436    // backends the binary cannot run — previously this reported
1437    // "valid" for `bridge_backend = "kafka"` on a default-features
1438    // build and the daemon then hard-failed at boot. pvctl and
1439    // proofvisord are built from the same workspace feature set in
1440    // every shipped artifact, so pvctl's own features are the best
1441    // available proxy for what the daemon can run.
1442    // `--structural-only` opts out for configs that target a build
1443    // with a different feature set (the full-feature container image);
1444    // `make schema-check` uses it for harness.docker/container.toml.
1445    if !structural_only {
1446        let unsupported = config.unsupported_backend_requirements();
1447        if !unsupported.is_empty() {
1448            anyhow::bail!(
1449                "config is structurally valid but this build cannot run it:\n  {}",
1450                unsupported.join("\n  ")
1451            );
1452        }
1453    }
1454    // `config.listen` is operator-supplied text with no control-byte
1455    // restriction (validate() only shape-checks the port suffix) — the
1456    // same CVE-2003-0063 class as `manifest.name` above; a crafted
1457    // config could otherwise land raw ANSI escapes in the operator
1458    // terminal.
1459    println!(
1460        "valid config_version={} listen={}",
1461        config.config_version,
1462        sanitize_for_terminal(&config.listen)
1463    );
1464    Ok(())
1465}

1467async fn loadgen(
1468    base_url: &str,
1469    connections: usize,
1470    workflow: &str,
1471    token_file: Option<&Path>,
1472) -> Result<()> {
1473    if connections == 0 {
1474        anyhow::bail!("connections must be greater than zero");
1475    }
1476    // Cap the requested concurrency: at multi-hundred-thousand
1477    // connections the loadgen exhausts source ports / FDs / RAM on the
1478    // operator's own host before any latency numbers land, and the load
1479    // *test* becomes the very thing it was supposed to measure. 10k is
1480    // the stated SLA gate — the previous 100k cap
1481    // encouraged runs that a single host cannot sustain: default Linux
1482    // `net.ipv4.ip_local_port_range` (~28k) exhausts before hyper can
1483    // dial, and `RLIMIT_NOFILE` (1024–65536) trips producing thousands
1484    // of `Address not available` failures that look like the harness
1485    // is broken.
1486    const MAX_CONNECTIONS: usize = 10_000;
1487    if connections > MAX_CONNECTIONS {
1488        anyhow::bail!(
1489            "connections must be <= {MAX_CONNECTIONS} — larger values \
1490             exhaust source ports / FDs before producing usable results"
1491        );
1492    }
1493    if pv_harness::session::Workflow::parse(workflow).is_none() {
1494        anyhow::bail!("workflow must be signed or unsigned");
1495    }
1496    // A hung upstream must not turn `loadgen --connections 10000` into an
1497    // unbounded resource consumer that never fails — the load *test* would
1498    // otherwise mask the failure it is supposed to expose. Per-request
1499    // timeout bounds each in-flight probe; connect_timeout catches
1500    // upstream unreachable faster than TCP retransmit does. Idle pool
1501    // capped at 1024 — more than that is never useful and just wastes
1502    // FDs after the burst ends.
1503    let client = reqwest::Client::builder()
1504        .pool_max_idle_per_host(connections.min(1024))
1505        .timeout(std::time::Duration::from_secs(30))
1506        .connect_timeout(std::time::Duration::from_secs(5))
1507        // No redirects: requests may carry bearer_auth (token leak on a
1508        // proxy-injected 3xx), and following one would silently measure
1509        // the redirect target instead of the configured endpoint.
1510        .redirect(reqwest::redirect::Policy::none())
1511        .build()
1512        .context("build loadgen client")?;
1513    let url = format!("{}/v1/chat/completions", base_url.trim_end_matches('/'));
1514    let token = bearer_token(token_file)?.map(Arc::new);
1515    // Unique per run: reusing ids like `load-0` across runs would bind to
1516    // sessions a previous run left behind — mid-close ones refuse reuse
1517    // with 503 "session close is completing" and quarantined ones with a
1518    // 400.
1519    let run_id = pv_core::new_event_uid();
1520    let started = Instant::now();
1521    let results: Vec<Result<u64, String>> = stream::iter(0..connections)
1522        .map(|index| {
1523            let client = client.clone();
1524            let url = url.clone();
1525            let workflow = workflow.to_owned();
1526            let token = token.clone();
1527            let run_id = run_id.clone();
1528            async move {
1529                let request_started = Instant::now();
1530                let mut request = client
1531                    .post(url)
1532                    .header("x-pv-session", format!("load-{run_id}-{index}"))
1533                    .header("x-pv-workflow", workflow)
1534                    .json(&serde_json::json!({
1535                        "model": "loadgen",
1536                        "stream": true,
1537                        "messages": [{"role": "user", "content": "health check"}],
1538                    }));
1539                if let Some(token) = token.as_ref() {
1540                    request = request.bearer_auth(token.as_str());
1541                }
1542                let response = request.send().await.map_err(|error| error.to_string())?;
1543                let status = response.status();
1544                // Cap loadgen response body reads. With
1545                // `stream: true`, an SSE response accumulates whole
1546                // via `.bytes()` — a stalled server could pin each
1547                // of `--connections 10_000` tasks to unbounded RAM,
1548                // turning the load test into the fault it was meant
1549                // to expose. Refuse past 4 MiB per response.
1550                use futures::StreamExt as _;
1551                let mut stream = response.bytes_stream();
1552                let mut total = 0usize;
1553                const LOADGEN_MAX_RESPONSE: usize = 4 * 1024 * 1024;
1554                while let Some(chunk) = stream.next().await {
1555                    let bytes = chunk.map_err(|error| error.to_string())?;
1556                    total = total.saturating_add(bytes.len());
1557                    if total > LOADGEN_MAX_RESPONSE {
1558                        return Err(format!("response exceeded {LOADGEN_MAX_RESPONSE} bytes"));
1559                    }
1560                }
1561                if !status.is_success() {
1562                    return Err(format!("HTTP {status}"));
1563                }
1564                Ok(u64::try_from(request_started.elapsed().as_micros()).unwrap_or(u64::MAX))
1565            }
1566        })
1567        .buffer_unordered(connections)
1568        .collect()
1569        .await;
1570    let mut latencies = Vec::with_capacity(connections);
1571    let mut failures = Vec::new();
1572    for result in results {
1573        match result {
1574            Ok(latency) => latencies.push(latency),
1575            Err(error) => failures.push(error),
1576        }
1577    }
1578    latencies.sort_unstable();
1579    println!(
1580        "connections={} success={} failed={} wall_ms={} p50_us={} p95_us={} p99_us={}",
1581        connections,
1582        latencies.len(),
1583        failures.len(),
1584        started.elapsed().as_millis(),
1585        percentile(&latencies, 50),
1586        percentile(&latencies, 95),
1587        percentile(&latencies, 99),
1588    );
1589    if !failures.is_empty() {
1590        anyhow::bail!(
1591            "{} load requests failed; first: {}",
1592            failures.len(),
1593            failures.first().map(String::as_str).unwrap_or("unknown")
1594        );
1595    }
1596    Ok(())
1597}

1599/// Read the NHI bearer token into a `Zeroizing` buffer. The token is a
1600/// credential at least as sensitive as an API key — every allocation
1601/// that holds it (the raw read and the trimmed copy) is zero-on-drop so
1602/// a post-run core dump or heap scan does not expose it verbatim.
1603/// Mirrors the wizard's `ask_secret_line` discipline.
1604fn bearer_token(path: Option<&Path>) -> Result<Option<zeroize::Zeroizing<String>>> {
1605    let path = path
1606        .map(Path::to_path_buf)
1607        .or_else(|| std::env::var_os("PV_BEARER_TOKEN_FILE").map(PathBuf::from));
1608    path.map(|path| {
1609        // Refuse a bearer token file that any other local
1610        // user can read. On a shared-tenant host, a 0o644 token file
1611        // in ~/.pvctl/ leaks the operator's credentials to every
1612        // process running as another uid. Mirrors the harness's own
1613        // discipline for signing seed / HMAC secret files (see
1614        // `require_owner_only_mode`).
1615        #[cfg(unix)]
1616        {
1617            use std::os::unix::fs::MetadataExt as _;
1618            let metadata = std::fs::symlink_metadata(&path)
1619                .with_context(|| format!("stat bearer token {}", path.display()))?;
1620            if metadata.file_type().is_symlink() {
1621                anyhow::bail!(
1622                    "bearer token file {} is a symbolic link; refuse to follow (planted-symlink hazard)",
1623                    path.display()
1624                );
1625            }
1626            let mode = metadata.mode() & 0o777;
1627            if mode & 0o077 != 0 {
1628                anyhow::bail!(
1629                    "bearer token file {} has mode 0o{mode:03o}; must be 0o600 (chmod 600 {})",
1630                    path.display(),
1631                    path.display()
1632                );
1633            }
1634        }
1635        let raw = zeroize::Zeroizing::new(read_capped_str(&path, MAX_CONFIG_BYTES, "bearer token")?);
1636        let token = zeroize::Zeroizing::new(raw.trim().to_owned());
1637        if token.is_empty() {
1638            anyhow::bail!("bearer token file {} is empty", path.display());
1639        }
1640        Ok(token)
1641    })
1642    .transpose()
1643}

1645fn percentile(sorted: &[u64], percentile: usize) -> u64 {
1646    if sorted.is_empty() {
1647        return 0;
1648    }
1649    let index = sorted.len().saturating_mul(percentile).saturating_add(99) / 100;
1650    sorted.get(index.saturating_sub(1)).copied().unwrap_or(0)
1651}
