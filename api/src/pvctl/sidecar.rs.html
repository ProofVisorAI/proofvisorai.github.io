1//! `pvctl sidecar`: a loopback proxy that gives an unmodified application an
2//! ProofVisor identity.
3//!
4//! The application's OpenAI SDK and MCP client point at this sidecar
5//! (`OPENAI_BASE_URL=http://127.0.0.1:8485/v1`, MCP URL
6//! `http://127.0.0.1:8485/mcp`; the buildpack sets both). For every request
7//! the sidecar:
8//!
9//! - drops the application's own `Authorization` header (SDKs send a
10//!   placeholder key) and attaches a short-lived ProofVisor identity token;
11//! - forwards method, path, query, body, and end-to-end headers (including
12//!   `MCP-Session-Id` and `MCP-Protocol-Version`) to the gateway;
13//! - streams the response back unchanged, so SSE chat streams and MCP
14//!   event streams pass through as they arrive.
15//!
16//! Identity comes from one of two sources:
17//!
18//! - `cf-instance` (Cloud Foundry): the instance identity certificate and
19//!   key named by `CF_INSTANCE_CERT` / `CF_INSTANCE_KEY`. The sidecar signs
20//!   a short RFC 7523 assertion (`RS256`, certificate chain in `x5c`) and
21//!   exchanges it at the gateway's `/v1/token` for an agent token. The
22//!   files are re-read on every refresh because the platform rotates them.
23//! - `token-file:<path>`: a file an external identity agent keeps filled
24//!   with a current ProofVisor token; it is re-read every 30 seconds.
25//!
26//! The sidecar listens on a loopback address only: any other address is
27//! refused, so no other host can borrow the application's identity.
28

39/// RFC 7523 grant type accepted by the gateway's `/v1/token`.
40const JWT_BEARER_GRANT_TYPE: &str = "urn:ietf:params:oauth:grant-type:jwt-bearer";
41
42/// Lifetime of each assertion the sidecar signs (the gateway accepts ≤ 300 s).
43const ASSERTION_LIFETIME_S: u64 = 120;

45/// How often a token file is re-read.
46const TOKEN_FILE_REFRESH: Duration = Duration::from_secs(30);

48/// Largest request body forwarded (the gateway's own limit is lower by
49/// default).
50const MAX_REQUEST_BYTES: usize = 32 * 1024 * 1024;

52/// Where the sidecar gets the identity it attaches.
53#[derive(Debug, Clone, PartialEq, Eq)]
54pub enum Credential {
55    /// Cloud Foundry instance identity certificate and key.
56    CfInstance {
57        /// PEM file with the instance certificate followed by its chain.
58        cert: PathBuf,
59        /// PEM file with the instance RSA private key.
60        key: PathBuf,
61        /// Instance GUID (the certificate CN; `CF_INSTANCE_GUID`).
62        instance_guid: String,
63    },
64    /// A file holding a current ProofVisor token.
65    TokenFile(PathBuf),
66}
67
68impl Credential {
69    /// Parse `cf-instance` or `token-file:<path>`. `cf-instance` reads
70    /// `CF_INSTANCE_CERT`, `CF_INSTANCE_KEY`, and `CF_INSTANCE_GUID`.
71    pub fn parse(value: &str) -> Result<Self> {
72        if let Some(path) = value.strip_prefix("token-file:") {
73            if path.is_empty() {
74                bail!("token-file: needs a path");
75            }
76            return Ok(Self::TokenFile(PathBuf::from(path)));
77        }
78        if value == "cf-instance" {
79            let var = |name: &str| {
80                std::env::var(name)
81                    .ok()
82                    .filter(|value| !value.trim().is_empty())
83                    .with_context(|| format!("{name} is not set; is this a Cloud Foundry app instance?"))
84            };
85            return Ok(Self::CfInstance {
86                cert: PathBuf::from(var("CF_INSTANCE_CERT")?),
87                key: PathBuf::from(var("CF_INSTANCE_KEY")?),
88                instance_guid: var("CF_INSTANCE_GUID")?,
89            });
90        }
91        bail!("unknown credential {value:?}: use cf-instance or token-file:<path>")
92    }
93}

140/// Refuse any listen address that is not loopback.
141pub fn check_listen(address: &SocketAddr) -> Result<()> {
142    if !address.ip().is_loopback() {
143        bail!(
144            "the sidecar must listen on a loopback address (127.0.0.1 or ::1), not {address}: \
145             any other address would lend this application's identity to other hosts"
146        );
147    }
148    Ok(())
149}

151/// Serve on an already-bound loopback listener until `shutdown` resolves.
152pub async fn serve(
153    listener: tokio::net::TcpListener,
154    options: SidecarOptions,
155    shutdown: impl std::future::Future<Output = ()> + Send + 'static,
156) -> Result<()> {
157    check_listen(&listener.local_addr()?)?;
158    let gateway = reqwest::Url::parse(&options.gateway).context("gateway URL")?;
159    if !matches!(gateway.scheme(), "http" | "https") || gateway.host_str().is_none() {
160        bail!("gateway URL must be http:// or https:// with a host");
161    }
162    let client = reqwest::Client::builder()
163        .redirect(reqwest::redirect::Policy::none())
164        .connect_timeout(Duration::from_secs(10))
165        .build()?;
166    let context = Arc::new(Context {
167        options,
168        client,
169        token: tokio::sync::Mutex::new(None),
170    });
171    let app = axum::Router::new()
172        .fallback(proxy)
173        .with_state(context)
174        .layer(axum::extract::DefaultBodyLimit::max(MAX_REQUEST_BYTES));
175    axum::serve(listener, app)
176        .with_graceful_shutdown(shutdown)
177        .await
178        .context("sidecar server")
179}

181/// Bind `listen` (loopback only) and serve until Ctrl-C or SIGTERM.
182pub async fn run(listen: SocketAddr, options: SidecarOptions) -> Result<()> {
183    check_listen(&listen)?;
184    let listener = tokio::net::TcpListener::bind(listen)
185        .await
186        .with_context(|| format!("bind {listen}"))?;
187    eprintln!(
188        "proofvisor sidecar listening on http://{} -> {}",
189        listener.local_addr()?,
190        options.gateway
191    );
192    serve(listener, options, async {
193        #[cfg(unix)]
194        {
195            let mut terminate = match tokio::signal::unix::signal(tokio::signal::unix::SignalKind::terminate()) {
196                Ok(signal) => signal,
197                Err(_) => {
198                    let _ = tokio::signal::ctrl_c().await;
199                    return;
200                }
201            };
202            tokio::select! {
203                _ = tokio::signal::ctrl_c() => {}
204                _ = terminate.recv() => {}
205            }
206        }
207        #[cfg(not(unix))]
208        {
209            let _ = tokio::signal::ctrl_c().await;
210        }
211    })
212    .await
213}

215/// Hop-by-hop headers (RFC 9110 §7.6.1) plus the ones the sidecar owns.
216fn forwardable(name: &HeaderName) -> bool {
217    !matches!(
218        name.as_str(),
219        "connection"
220            | "keep-alive"
221            | "proxy-authenticate"
222            | "proxy-authorization"
223            | "proxy-connection"
224            | "te"
225            | "trailer"
226            | "transfer-encoding"
227            | "upgrade"
228            | "host"
229            | "content-length"
230            | "authorization"
231    )
232}

234fn error(status: StatusCode, message: &str) -> Response<Body> {
235    let body = serde_json::json!({"error": {"message": message, "type": "proofvisor_sidecar_error"}});
236    let mut response = Response::new(Body::from(body.to_string()));
237    *response.status_mut() = status;
238    response.headers_mut().insert(
239        axum::http::header::CONTENT_TYPE,
240        HeaderValue::from_static("application/json"),
241    );
242    response
243}

245async fn proxy(
246    axum::extract::State(context): axum::extract::State<Arc<Context>>,
247    request: Request<Body>,
248) -> Response<Body> {
249    let token = match context.token().await {
250        Ok(token) => token,
251        Err(problem) => {
252            eprintln!("proofvisor sidecar: could not obtain an identity token: {problem:#}");
253            return error(
254                StatusCode::SERVICE_UNAVAILABLE,
255                "the ProofVisor sidecar has no identity token",
256            );
257        }
258    };
259    let (parts, body) = request.into_parts();
260    let path = parts.uri.path_and_query().map_or("/", |value| value.as_str());
261    let url = format!("{}{path}", context.options.gateway.trim_end_matches('/'));
262    let Ok(body) = axum::body::to_bytes(body, MAX_REQUEST_BYTES).await else {
263        return error(
264            StatusCode::PAYLOAD_TOO_LARGE,
265            "request body is too large for the sidecar",
266        );
267    };
268    let mut headers = HeaderMap::new();
269    for (name, value) in &parts.headers {
270        if forwardable(name) {
271            headers.append(name.clone(), value.clone());
272        }
273    }
274    let Ok(mut authorization) = HeaderValue::from_str(&format!("Bearer {token}")) else {
275        return error(
276            StatusCode::SERVICE_UNAVAILABLE,
277            "identity token is not a header value",
278        );
279    };
280    authorization.set_sensitive(true);
281    headers.insert(axum::http::header::AUTHORIZATION, authorization);
282    let upstream = match context
283        .client
284        .request(parts.method, url)
285        .headers(headers)
286        .body(body)
287        .send()
288        .await
289    {
290        Ok(upstream) => upstream,
291        Err(problem) => {
292            eprintln!(
293                "proofvisor sidecar: gateway unreachable (connect={}, timeout={})",
294                problem.is_connect(),
295                problem.is_timeout()
296            );
297            return error(StatusCode::BAD_GATEWAY, "the ProofVisor gateway is unreachable");
298        }
299    };
300    let mut response = Response::builder().status(upstream.status());
301    if let Some(target) = response.headers_mut() {
302        for (name, value) in upstream.headers() {
303            if forwardable(name) {
304                target.append(name.clone(), value.clone());
305            }
306        }
307    }
308    let stream = upstream
309        .bytes_stream()
310        .map(|chunk| chunk.map_err(std::io::Error::other));
311    response
312        .body(Body::from_stream(stream))
313        .unwrap_or_else(|_| error(StatusCode::BAD_GATEWAY, "invalid gateway response"))
314}

316impl Context {
317    /// The current token, refreshed when due. Concurrent requests share
318    /// one refresh.
319    async fn token(&self) -> Result<String> {
320        let mut cached = self.token.lock().await;
321        if let Some(current) = cached
322            .as_ref()
323            .filter(|current| current.refresh_at > Instant::now())
324        {
325            return Ok(current.token.clone());
326        }
327        let fresh = match &self.options.credential {
328            Credential::TokenFile(path) => {
329                let token = tokio::fs::read_to_string(path)
330                    .await
331                    .with_context(|| format!("read token file {}", path.display()))?
332                    .trim()
333                    .to_owned();
334                if token.is_empty() {
335                    bail!("token file {} is empty", path.display());
336                }
337                CachedToken {
338                    token,
339                    refresh_at: Instant::now() + TOKEN_FILE_REFRESH,
340                }
341            }
342            Credential::CfInstance {
343                cert,
344                key,
345                instance_guid,
346            } => {
347                let cert = tokio::fs::read(cert)
348                    .await
349                    .with_context(|| format!("read {}", cert.display()))?;
350                let key = tokio::fs::read(key)
351                    .await
352                    .with_context(|| format!("read {}", key.display()))?;
353                let assertion = assertion(&cert, &key, instance_guid, &self.options.audience)?;
354                self.exchange(&assertion).await?
355            }
356        };
357        let token = fresh.token.clone();
358        *cached = Some(fresh);
359        Ok(token)
360    }
361
362    async fn exchange(&self, assertion: &str) -> Result<CachedToken> {
363        let url = format!("{}/v1/token", self.options.gateway.trim_end_matches('/'));
364        let form = format!(
365            "grant_type={}&assertion={}",
366            percent_encode(JWT_BEARER_GRANT_TYPE),
367            percent_encode(assertion)
368        );
369        let response = self
370            .client
371            .post(url)
372            .timeout(Duration::from_secs(15))
373            .header(
374                axum::http::header::CONTENT_TYPE,
375                "application/x-www-form-urlencoded",
376            )
377            .body(form)
378            .send()
379            .await
380            .context("reach the gateway token endpoint")?;
381        let status = response.status();
382        let body: serde_json::Value = response.json().await.context("token response is not JSON")?;
383        if !status.is_success() {
384            bail!(
385                "gateway refused the workload assertion ({status}): {}",
386                body.get("error_description")
387                    .and_then(serde_json::Value::as_str)
388                    .unwrap_or("no description")
389            );
390        }
391        let token = body
392            .get("access_token")
393            .and_then(serde_json::Value::as_str)
394            .filter(|token| !token.is_empty())
395            .context("token response has no access_token")?
396            .to_owned();
397        let expires_in = body
398            .get("expires_in")
399            .and_then(serde_json::Value::as_u64)
400            .unwrap_or(60);
401        // Refresh at 80% of the lifetime, never sooner than 5 seconds.
402        let refresh_in = Duration::from_secs((expires_in.saturating_mul(4) / 5).max(5));
403        Ok(CachedToken {
404            token,
405            refresh_at: Instant::now() + refresh_in,
406        })
407    }
408}

410/// Every `CERTIFICATE` block of a PEM file, as DER.
411fn certificates(pem: &[u8]) -> Result<Vec<Vec<u8>>> {
412    let text = std::str::from_utf8(pem).context("certificate file is not UTF-8")?;
413    let mut chain = Vec::new();
414    let mut rest = text;
415    while let Some(start) = rest.find("-----BEGIN CERTIFICATE-----") {
416        let after = rest
417            .get(start.saturating_add("-----BEGIN CERTIFICATE-----".len())..)
418            .unwrap_or_default();
419        let end = after
420            .find("-----END CERTIFICATE-----")
421            .context("unterminated CERTIFICATE block")?;
422        let body: String = after
423            .get(..end)
424            .unwrap_or_default()
425            .chars()
426            .filter(|c| !c.is_whitespace())
427            .collect();
428        chain.push(
429            base64::engine::general_purpose::STANDARD
430                .decode(body)
431                .context("CERTIFICATE block is not base64")?,
432        );
433        rest = after
434            .get(end.saturating_add("-----END CERTIFICATE-----".len())..)
435            .unwrap_or_default();
436    }
437    if chain.is_empty() {
438        bail!("certificate file contains no certificate");
439    }
440    Ok(chain)
441}

443/// Sign an RFC 7523 assertion with the instance key, carrying the
444/// certificate chain in `x5c`.
445pub fn assertion(cert_pem: &[u8], key_pem: &[u8], instance_guid: &str, audience: &str) -> Result<String> {
446    let chain = certificates(cert_pem)?;
447    let key =
448        jsonwebtoken::EncodingKey::from_rsa_pem(key_pem).context("instance key is not an RSA PEM key")?;
449    let mut header = jsonwebtoken::Header::new(jsonwebtoken::Algorithm::RS256);
450    header.x5c = Some(
451        chain
452            .iter()
453            .map(|der| base64::engine::general_purpose::STANDARD.encode(der))
454            .collect(),
455    );
456    let now = pv_core::time::now_ms() / 1000;
457    let claims = serde_json::json!({
458        "iss": instance_guid,
459        "sub": instance_guid,
460        "aud": audience,
461        "iat": now,
462        "exp": now.saturating_add(ASSERTION_LIFETIME_S),
463        "jti": pv_core::new_event_uid(),
464    });
465    jsonwebtoken::encode(&header, &claims, &key).context("sign the workload assertion")
466}

468/// `application/x-www-form-urlencoded` value encoding.
469fn percent_encode(value: &str) -> String {
470    let mut encoded = String::with_capacity(value.len());
471    for byte in value.bytes() {
472        if byte.is_ascii_alphanumeric() || matches!(byte, b'-' | b'_' | b'.' | b'~') {
473            encoded.push(char::from(byte));
474        } else {
475            encoded.push_str(&format!("%{byte:02X}"));
476        }
477    }
478    encoded
479}
