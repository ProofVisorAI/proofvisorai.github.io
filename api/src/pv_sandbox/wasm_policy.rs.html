1//! WebAssembly policy modules via wasmtime (the sandboxing engine).
2//!
3//! ABI version 1 (documented for policy authors):
4//! - export `memory` (linear memory) and `alloc(len: i32) -> ptr: i32`;
5//! - export `evaluate(ptr: i32, len: i32) -> code: i32`;
6//! - the host writes the UTF-8 JSON `{"tool": …, "arguments": …}` at `ptr`;
7//! - return `0` to allow, any other code to deny.
8//!
9//! ABI version 2 (opt in by exporting `evaluate_v2` instead of, or as
10//! well as, `evaluate`; version 1 policies are unchanged):
11//! - export `evaluate_v2(ptr: i32, len: i32) -> code: i32`;
12//! - the host writes `{"tool": …, "arguments": …, "judge": [ … ]}`, where
13//!   `judge` holds one semantic-judge outcome per evaluated predicate
14//!   (empty when the tool is not judged): either
15//!   `{"status": "scored", "predicate_id": …, "probs": [{"label": …, "ppm": …}], "model": {…}, "latency_us": …}`
16//!   or `{"status": "failed", "predicate_id": …, "kind": "timeout" | "saturated" | "unavailable" | "malformed" | "model_mismatch", …}`.
17//!   Probabilities are integer parts per million;
18//! - return `0` to allow, `1` to deny, `2` to ask (hold for human
19//!   approval); any other code denies.
20//!
21//! Containment: every evaluation runs in a fresh `Store` with a fuel budget
22//! and a linear-memory cap. Traps, missing exports, fuel exhaustion, and
23//! memory overruns all fail **closed** (deny) — a hostile or buggy policy can
24//! neither hang the pipeline nor allow by accident.
25

33/// Fuel budget per evaluation (~millions of instructions; a policy should use
34/// a tiny fraction of this).
35const FUEL_PER_CALL: u64 = 50_000_000;
36
37/// Linear memory cap per evaluation.
38const MAX_MEMORY_BYTES: usize = 16 * 1024 * 1024;
39
40/// Host-memory DoS caps for guest-visible
41/// resource growth. `StoreLimitsBuilder::memory_size` bounds one
42/// `Memory`, but a hostile policy could:
43///
44/// * declare many `(memory ...)` sections and reach ~10k×16 MiB of
45///   allocation before fuel meaningfully activates;
46/// * `table.grow` a table by millions of function-reference slots
47///   (each `Option<Func>` ≈ 16 bytes on 64-bit), forcing a huge host
48///   allocation on the growth step;
49/// * declare many tables per module to multiply the above.
50///
51/// Fuel/epoch help only against runtime-bounded exploits — allocation
52/// pressure lands at instantiation or on a single `memory.grow` /
53/// `table.grow` instruction. Cap every dimension explicitly.
54const MAX_MEMORIES: usize = 1;
55const MAX_TABLES: usize = 4;
56const MAX_TABLE_ELEMENTS: usize = 65_536;
57const MAX_INSTANCES: usize = 4;
58
59/// Maximum wall time is approximately this many 1 ms epoch ticks.
60///
61/// This was 25 — a WALL-CLOCK deadline on a
62/// work-bounded computation. On a busy 2-core box, 16 concurrent
63/// evaluations were descheduled past 25 ms and legitimately-tiny
64/// policies failed closed (`valid_policy_does_not_false_trip_under_
65/// parallel_load` failed 10/10 with 4 busy-loop competitors). Fuel
66/// (`FUEL_PER_CALL`) is the deterministic work bound and remains
67/// authoritative; the epoch deadline exists only to interrupt fuel
68/// escapes (host-call stalls, wasmtime bugs), so it can be generous —
69/// 2 s of wall time never fires on a healthy evaluation regardless of
70/// scheduler pressure, while still bounding a stuck guest to seconds
71/// instead of forever.
72const EPOCH_DEADLINE_TICKS: u64 = 2_000;

74/// A compiled WASM policy.
75pub struct WasmPolicy {
76    name: String,
77    engine: Engine,
78    module: Module,
79    epoch_stop: Arc<AtomicBool>,
80    abi_v2: bool,
81}

83impl WasmPolicy {
84    /// Compile a policy from `.wasm` bytes or WAT text.
85    pub fn from_bytes(name: impl Into<String>, bytes: &[u8]) -> Result<Self, String> {
86        let mut config = Config::new();
87        config.consume_fuel(true);
88        config.epoch_interruption(true);
89        let engine = Engine::new(&config).map_err(|e| e.to_string())?;
90        let module = Module::new(&engine, bytes).map_err(|e| e.to_string())?;
91        // Validate the ABI statically at LOAD time. `run()` re-checks at
92        // instantiation, but that check only fires on the first tool
93        // call — a policy missing `evaluate` (typo'd export name, wrong
94        // build target) booted fine and then silently denied every tool
95        // call as "failed closed: missing evaluate". Correct posture,
96        // terrible operability: the operator sees a deny-storm instead
97        // of a boot refusal naming the broken file. Fail fast like the
98        // empty-wasm_policy_paths boot guard in main.rs.
99        let mut has_memory = false;
100        let mut alloc_ok = false;
101        let mut evaluate_ok = false;
102        let mut evaluate_v2_ok = false;
103        // Check TYPES, not just arity: the runtime calls
104        // `get_typed_func::<(i32, i32), i32>` (and `<i32, i32>` for
105        // alloc), so a module exporting the right arity with i64/f32
106        // params passed this load-time gate and then failed EVERY
107        // evaluation — the startup-success/runtime-denial split this
108        // validation exists to prevent.
109        let all_i32 = |f: &wasmtime::FuncType| {
110            f.params().all(|p| matches!(p, wasmtime::ValType::I32))
111                && f.results().all(|r| matches!(r, wasmtime::ValType::I32))
112        };
113        for export in module.exports() {
114            match (export.name(), export.ty()) {
115                ("memory", wasmtime::ExternType::Memory(_)) => has_memory = true,
116                ("alloc", wasmtime::ExternType::Func(f)) => {
117                    alloc_ok = f.params().len() == 1 && f.results().len() == 1 && all_i32(&f);
118                }
119                ("evaluate", wasmtime::ExternType::Func(f)) => {
120                    evaluate_ok = f.params().len() == 2 && f.results().len() == 1 && all_i32(&f);
121                }
122                ("evaluate_v2", wasmtime::ExternType::Func(f)) => {
123                    evaluate_v2_ok = f.params().len() == 2 && f.results().len() == 1 && all_i32(&f);
124                }
125                _ => {}
126            }
127        }
128        if !has_memory || !alloc_ok || !(evaluate_ok || evaluate_v2_ok) {
129            let mut missing = Vec::new();
130            if !has_memory {
131                missing.push("`memory` (exported linear memory)");
132            }
133            if !alloc_ok {
134                missing.push("`alloc(len: i32) -> i32`");
135            }
136            if !(evaluate_ok || evaluate_v2_ok) {
137                missing.push("`evaluate(ptr: i32, len: i32) -> i32` (or `evaluate_v2`)");
138            }
139            return Err(format!(
140                "policy does not export the required ABI: missing {} — see crates/pv-sandbox/src/wasm_policy.rs for the policy-author contract",
141                missing.join(", ")
142            ));
143        }
144        let epoch_stop = Arc::new(AtomicBool::new(false));
145        let ticker_stop = Arc::clone(&epoch_stop);
146        let ticker_engine = engine.clone();
147        std::thread::spawn(move || {
148            while !ticker_stop.load(Ordering::Acquire) {
149                std::thread::sleep(std::time::Duration::from_millis(1));
150                ticker_engine.increment_epoch();
151            }
152        });
153        Ok(Self {
154            name: name.into(),
155            engine,
156            module,
157            epoch_stop,
158            abi_v2: evaluate_v2_ok,
159        })
160    }
161
162    /// True when the policy speaks ABI version 2 (sees judge outcomes
163    /// and can return ask).
164    pub fn is_abi_v2(&self) -> bool {
165        self.abi_v2
166    }
167
168    fn run(&self, payload: &[u8]) -> Result<i32, String> {
169        let limits = StoreLimitsBuilder::new()
170            .memory_size(MAX_MEMORY_BYTES)
171            .memories(MAX_MEMORIES)
172            .tables(MAX_TABLES)
173            .table_elements(MAX_TABLE_ELEMENTS)
174            .instances(MAX_INSTANCES)
175            .build();
176        let mut store: Store<StoreLimits> = Store::new(&self.engine, limits);
177        store.limiter(|l| l);
178        store.set_fuel(FUEL_PER_CALL).map_err(|e| e.to_string())?;
179        store.set_epoch_deadline(EPOCH_DEADLINE_TICKS);
180
181        let instance =
182            Instance::new(&mut store, &self.module, &[]).map_err(|e| format!("instantiate: {e}"))?;
183        let memory = instance
184            .get_memory(&mut store, "memory")
185            .ok_or_else(|| "policy exports no `memory`".to_owned())?;
186        let alloc = instance
187            .get_typed_func::<i32, i32>(&mut store, "alloc")
188            .map_err(|e| format!("missing alloc: {e}"))?;
189        let entry = if self.abi_v2 { "evaluate_v2" } else { "evaluate" };
190        let evaluate = instance
191            .get_typed_func::<(i32, i32), i32>(&mut store, entry)
192            .map_err(|e| format!("missing {entry}: {e}"))?;
193
194        let len = i32::try_from(payload.len()).map_err(|_| "payload too large".to_owned())?;
195        let ptr = alloc
196            .call(&mut store, len)
197            .map_err(|e| format!("alloc trapped: {e}"))?;
198        let ptr_usize = usize::try_from(ptr).map_err(|_| "alloc returned negative ptr".to_owned())?;
199        memory
200            .write(&mut store, ptr_usize, payload)
201            .map_err(|e| format!("payload write out of bounds: {e}"))?;
202        evaluate
203            .call(&mut store, (ptr, len))
204            .map_err(|e| format!("evaluate trapped/exhausted: {e}"))
205    }
206}

208impl Drop for WasmPolicy {
209    fn drop(&mut self) {
210        self.epoch_stop.store(true, Ordering::Release);
211    }

214impl PolicyEngine for WasmPolicy {
215    fn name(&self) -> &str {
216        &self.name
217    }
218
219    fn evaluate(&self, tool: &str, arguments: &Value) -> PolicyDecision {
220        self.evaluate_with_judge(tool, arguments, &[])
221    }
222
223    fn evaluate_with_judge(&self, tool: &str, arguments: &Value, judge: &[JudgeOutcome]) -> PolicyDecision {
224        let payload = if self.abi_v2 {
225            serde_json::json!({ "tool": tool, "arguments": arguments, "judge": judge })
226        } else {
227            serde_json::json!({ "tool": tool, "arguments": arguments })
228        };
229        let bytes = match serde_json::to_vec(&payload) {
230            Ok(b) => b,
231            Err(e) => {
232                return PolicyDecision::Deny {
233                    reason: format!("policy input serialization failed: {e}"),
234                }
235            }
236        };
237        match self.run(&bytes) {
238            Ok(0) => PolicyDecision::Allow,
239            Ok(2) if self.abi_v2 => PolicyDecision::Ask {
240                reason: format!("wasm policy {:?} asks for human approval", self.name),
241            },
242            Ok(code) => PolicyDecision::Deny {
243                reason: format!("wasm policy {:?} denied (code {code})", self.name),
244            },
245            // Fail closed: any trap/fuel/memory/ABI failure is a deny.
246            Err(e) => PolicyDecision::Deny {
247                reason: format!("wasm policy {:?} failed closed: {e}", self.name),
248            },
249        }
250    }
251
252    fn sees_judge(&self) -> bool {
253        self.abi_v2
254    }
255}

345        let started = std::time::Instant::now();
346        let d = p.evaluate("anything", &json!({}));

351        // fuel under coverage instrumentation or noisy neighbours can
352        // legitimately exceed 100 ms. Ten seconds distinguishes

362            PolicyDecision::Allow => panic!("hostile policy allowed"),

517            (drop (memory.grow (i32.const 4096)))
518            (i32.const 0))
519          (func (export "evaluate") (param i32 i32) (result i32)
520            (i32.const 0)))
521        "#;
522        let p = WasmPolicy::from_bytes("memory-bomb", MEMORY_BOMB.as_bytes()).unwrap();
523        // Ensure the module loads (parse succeeds) but the evaluation fails
524        // closed once memory.grow is denied.
525        let decision = p.evaluate("anything", &json!({}));
526        match decision {
527            PolicyDecision::Allow => {
528                // The memory.grow can legally return -1 (denied) without
529                // trapping. That's fine — the module correctly declined to
530                // exceed the limit. The proof that the guest can NOT
531                // actually use memory past the cap is the companion test
532                // `memory_grow_past_cap_leaves_grown_region_inaccessible`,
533                // which stores at a beyond-cap address and must fail closed.
534            }
535            PolicyDecision::Deny { reason } => {
536                assert!(
537                    reason.contains("out of bounds")
538                        || reason.contains("trapped")
539                        || reason.contains("failed closed"),
540                    "expected memory-cap failure, got {reason}"
541                );
542            }
543            PolicyDecision::Ask { reason } => panic!("a version 1 policy cannot ask: {reason}"),
544        }
545    }

547    /// The companion proof the memory-bomb
548    /// test's Allow arm defers to. After a beyond-cap `memory.grow`, the
549    /// guest tries to STORE at an address inside the would-be grown
550    /// region (page 512 = 32 MiB, past the 16 MiB cap). StoreLimits must
551    /// have refused the growth, so the store is out of bounds and the
552    /// evaluation must fail closed (Deny) — the grown region is
553    /// inaccessible, not silently usable.
554    #[test]
555    fn memory_grow_past_cap_leaves_grown_region_inaccessible() {
556        const GROW_THEN_TOUCH: &str = r#"
557        (module
558          (memory (export "memory") 1)
559          (func (export "alloc") (param i32) (result i32)
560            (i32.const 0))
561          (func (export "evaluate") (param i32 i32) (result i32)
562            (drop (memory.grow (i32.const 4096)))
563            ;; store at 32 MiB — inside the grown region iff growth succeeded
564            (i32.store (i32.const 33554432) (i32.const 1))
565            (i32.const 0)))
566        "#;
567        let p = WasmPolicy::from_bytes("grow-then-touch", GROW_THEN_TOUCH.as_bytes()).unwrap();
568        match p.evaluate("anything", &json!({})) {
569            PolicyDecision::Deny { reason } => {
570                assert!(
571                    reason.contains("out of bounds")
572                        || reason.contains("trapped")
573                        || reason.contains("failed closed"),
574                    "beyond-cap store must fail closed with a containment reason, got: {reason}"
575                );
576            }
577            PolicyDecision::Allow => {
578                panic!("store at 32 MiB succeeded — the memory cap did not hold")
579            }
580            PolicyDecision::Ask { reason } => panic!("a version 1 policy cannot ask: {reason}"),
581        }
582    }

601            );
602        }

605    /// ABI v2: deny (1) when the payload mentions `"kind":"timeout"`,
606    /// ask (2) when it mentions `"ppm":9`, allow otherwise. Scans bytes
607    /// for the 9-byte needle starting at the given address.
608    const V2_POLICY: &str = r#"
609    (module
610      (memory (export "memory") 4)
611      (data (i32.const 0) "\"timeout\"")
612      (data (i32.const 16) "\"ppm\":9")
613      (global $next (mut i32) (i32.const 4096))
614      (func (export "alloc") (param $len i32) (result i32)
615        (local $ptr i32)
616        (local.set $ptr (global.get $next))
617        (global.set $next (i32.add (global.get $next) (local.get $len)))
618        (local.get $ptr))
619      (func $contains (param $ptr i32) (param $len i32) (param $needle i32) (param $nlen i32) (result i32)
620        (local $i i32) (local $j i32) (local $end i32)
621        (local.set $end (i32.sub (i32.add (local.get $ptr) (local.get $len)) (local.get $nlen)))
622        (local.set $i (local.get $ptr))
623        (block $done
624          (loop $scan
625            (br_if $done (i32.gt_s (local.get $i) (local.get $end)))
626            (local.set $j (i32.const 0))
627            (block $miss
628              (loop $cmp
629                (br_if $miss (i32.ne
630                  (i32.load8_u (i32.add (local.get $i) (local.get $j)))
631                  (i32.load8_u (i32.add (local.get $needle) (local.get $j)))))
632                (local.set $j (i32.add (local.get $j) (i32.const 1)))
633                (if (i32.eq (local.get $j) (local.get $nlen)) (then (return (i32.const 1))))
634                (br $cmp)))
635            (local.set $i (i32.add (local.get $i) (i32.const 1)))
636            (br $scan)))
637        (i32.const 0))
638      (func (export "evaluate_v2") (param $ptr i32) (param $len i32) (result i32)
639        (if (call $contains (local.get $ptr) (local.get $len) (i32.const 0) (i32.const 9))
640          (then (return (i32.const 1))))
641        (if (call $contains (local.get $ptr) (local.get $len) (i32.const 16) (i32.const 7))
642          (then (return (i32.const 2))))
643        (i32.const 0)))
644    "#;

646    fn outcome(json: Value) -> JudgeOutcome {
647        serde_json::from_value(json).unwrap()
648    }

650    #[test]
651    fn abi_v2_policies_see_judge_outcomes_and_can_ask() {
652        let p = WasmPolicy::from_bytes("v2", V2_POLICY.as_bytes()).unwrap();
653        assert!(p.is_abi_v2());
654        assert!(p.sees_judge());
655        let model = json!({"name": "m", "version": "1", "weights_sha256": "a".repeat(64)});
656        let scored = outcome(json!({
657            "status": "scored", "predicate_id": "arg_matches_task@1",
658            "probs": [{"label": "match", "ppm": 100000}, {"label": "mismatch", "ppm": 900000}],
659            "model": model, "latency_us": 3
660        }));
661        assert!(matches!(
662            p.evaluate_with_judge("t", &json!({}), std::slice::from_ref(&scored)),
663            PolicyDecision::Ask { .. }
664        ));
665        let timeout = outcome(json!({
666            "status": "failed", "predicate_id": "arg_matches_task@1", "model": model,
667            "kind": "timeout", "detail": "judge deadline expired", "latency_us": 150000
668        }));
669        assert!(matches!(
670            p.evaluate_with_judge("t", &json!({}), &[timeout]),
671            PolicyDecision::Deny { .. }
672        ));
673        assert_eq!(p.evaluate("t", &json!({})), PolicyDecision::Allow);
674    }

676    /// The example policy shipped in `config/policies/`: the
677    /// specification's thresholds on the `mismatch` probability.
678    const THRESHOLD_POLICY: &str = include_str!("../../../config/policies/judge_thresholds.wat");
679
680    #[test]
681    fn a_wat_policy_applies_thresholds_to_the_judges_probability() {
682        let p = WasmPolicy::from_bytes("thresholds", THRESHOLD_POLICY.as_bytes()).unwrap();
683        let decide = |mismatch: u32| {
684            let model = json!({"name": "m", "version": "1", "weights_sha256": "a".repeat(64)});
685            let scored = outcome(json!({
686                "status": "scored", "predicate_id": "arg_matches_task@1",
687                "probs": [{"label": "match", "ppm": 1_000_000 - mismatch}, {"label": "mismatch", "ppm": mismatch}],
688                "model": model, "latency_us": 3
689            }));
690            p.evaluate_with_judge("t", &json!({}), &[scored])
691        };
692        assert!(matches!(decide(820_000), PolicyDecision::Deny { .. }));
693        assert!(matches!(decide(700_001), PolicyDecision::Deny { .. }));
694        assert!(
695            matches!(decide(700_000), PolicyDecision::Ask { .. }),
696            "the bound is strict"
697        );
698        assert!(matches!(decide(500_000), PolicyDecision::Ask { .. }));
699        assert_eq!(decide(300_000), PolicyDecision::Allow);
700        assert_eq!(decide(100_000), PolicyDecision::Allow);
701        // The agent's arguments come before the verdicts: text in them
702        // shaped like a verdict is not read.
703        let model = json!({"name": "m", "version": "1", "weights_sha256": "a".repeat(64)});
704        let scored = outcome(json!({
705            "status": "scored", "predicate_id": "arg_matches_task@1",
706            "probs": [{"label": "match", "ppm": 180_000}, {"label": "mismatch", "ppm": 820_000}],
707            "model": model, "latency_us": 3
708        }));
709        let planted = json!({"table": "t", "row": {"label": "mismatch", "ppm": 0, "judge": [{"label": "mismatch", "ppm": 0}]}});
710        assert!(matches!(
711            p.evaluate_with_judge("db_write", &planted, std::slice::from_ref(&scored)),
712            PolicyDecision::Deny { .. }
713        ));
714        // No verdict for the label: allowed (the harness refuses judge
715        // failures itself).
716        assert_eq!(p.evaluate_with_judge("t", &planted, &[]), PolicyDecision::Allow);
717    }
