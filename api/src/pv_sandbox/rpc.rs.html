1//! JSON-RPC 2.0 / MCP `tools/call` parsing. Total: never panics on arbitrary
2//! bytes (property-tested), returns typed errors for every malformed shape.

7/// A parsed MCP tool-call request.
8#[derive(Debug, Clone, PartialEq)]
9pub struct ToolCallRequest {
10    /// JSON-RPC id (absent for notifications).
11    pub id: Option<Value>,
12    /// Tool name (`params.name`).
13    pub tool: String,
14    /// Tool arguments (`params.arguments`, defaults to `{}`).
15    pub arguments: Value,
16}
17
18/// Parse failures.
19#[derive(Debug, thiserror::Error, PartialEq, Eq)]
20#[non_exhaustive]
21pub enum RpcError {
22    /// Not valid JSON at all.
23    #[error("invalid JSON: {0}")]
24    Json(String),
25    /// Valid JSON but not a JSON-RPC 2.0 object.
26    #[error("not a JSON-RPC 2.0 request: {0}")]
27    NotJsonRpc(String),
28    /// A method other than `tools/call` (passthrough, not an error verdict —
29    /// the caller decides what to do with non-tool traffic).
30    #[error("method {0:?} is not tools/call")]
31    NotToolCall(String),
32    /// `tools/call` missing/invalid params.
33    #[error("invalid tools/call params: {0}")]
34    BadParams(String),
35    /// Payload exceeds the configured size bound (DoS guard).
36    #[error("payload of {0} bytes exceeds the {1}-byte bound")]
37    TooLarge(usize, usize),
38}
39
40/// Hard byte bound applied before parsing (attacker-controlled input).
41pub const MAX_PAYLOAD_BYTES: usize = 4 * 1024 * 1024;
42
43/// Depth bound applied post-parse (deeply nested JSON as a DoS vector).
44pub const MAX_JSON_DEPTH: usize = 64;
45
46/// Parse raw bytes as an MCP `tools/call`.
47pub fn parse_tool_call(raw: &[u8]) -> Result<ToolCallRequest, RpcError> {
48    if raw.len() > MAX_PAYLOAD_BYTES {
49        return Err(RpcError::TooLarge(raw.len(), MAX_PAYLOAD_BYTES));
50    }
51    // Duplicate-key rejection: `serde_json` silently keeps the LAST value
52    // on duplicate keys, but the raw body is forwarded unchanged to the
53    // configured tool upstream. If the upstream's JSON decoder is
54    // first-wins (jackson, some Go decoders) or exposes both values, a
55    // request like
56    //   {"params":{"name":"safe_read","name":"db_write","arguments":{}}}
57    // parses here as `tool = "db_write"` (last-wins), passes the
58    // safe_read policy/budget gate, then executes `db_write` upstream —
59    // a permissions-model split with full audit-trail mismatch. Reject
60    // duplicates so gate and upstream see the same request or neither
61    // does. RFC 8259 §4 makes duplicate-name handling "implementation-
62    // defined", so the trust-boundary policy is: refuse ambiguity.
63    reject_duplicate_keys(raw)?;
64    let v: Value = serde_json::from_slice(raw).map_err(|e| RpcError::Json(e.to_string()))?;
65    if depth_of(&v, 0) > MAX_JSON_DEPTH {
66        return Err(RpcError::NotJsonRpc("nesting exceeds depth bound".into()));
67    }
68    let obj = v
69        .as_object()
70        .ok_or_else(|| RpcError::NotJsonRpc("root is not an object".into()))?;
71    match obj.get("jsonrpc").and_then(Value::as_str) {
72        Some("2.0") => {}
73        other => {
74            return Err(RpcError::NotJsonRpc(format!(
75                "jsonrpc field is {other:?}, need \"2.0\""
76            )))
77        }
78    }
79    let method = obj
80        .get("method")
81        .and_then(Value::as_str)
82        .ok_or_else(|| RpcError::NotJsonRpc("method missing or not a string".into()))?;
83    if method != "tools/call" {
84        return Err(RpcError::NotToolCall(method.to_owned()));
85    }
86    let params = obj
87        .get("params")
88        .and_then(Value::as_object)
89        .ok_or_else(|| RpcError::BadParams("params missing or not an object".into()))?;
90    let tool = params
91        .get("name")
92        .and_then(Value::as_str)
93        .filter(|s| !s.is_empty())
94        .ok_or_else(|| RpcError::BadParams("params.name missing or empty".into()))?;
95    if pv_core::text::contains_bidi_or_zero_width(tool) {
96        return Err(RpcError::BadParams(
97            "params.name carries a bidi/zero-width spoofing character".into(),
98        ));
99    }
100    // Reject ASCII control characters and inner whitespace: without this a
101    // request for `"db_write\n"` would slip past exact-string matchers for
102    // per-tool schemas, per-tool policy deny-lists, per-tool budgets, and
103    // the harness consequential-tools workflow veto, then be normalized by
104    // most downstream MCP servers to `"db_write"`. Also cover the same
105    // trailing/leading whitespace shape.
106    if tool.chars().any(|c| c.is_control() || c.is_whitespace()) {
107        return Err(RpcError::BadParams(
108            "params.name contains a control character or whitespace".into(),
109        ));
110    }
111    // Refuse any non-ASCII byte in the tool name. Every downstream
112    // comparison — per-tool schema lookup, deny-list matching, per-tool
113    // budget key — is raw byte-equality. A caller supplying
114    // `de\u{0301}lete` (decomposed) or `𝐝𝐛_𝐰𝐫𝐢𝐭𝐞` (mathematical
115    // bold, NFKC-folds to `db_write`) sees a different byte string
116    // than a policy that lists `delete` / `db_write`, but most MCP
117    // servers apply NFC/NFKC before dispatch — so the visually
118    // "different" name resolves to the same tool downstream.
119    // Restricting to ASCII collapses the Unicode-normalization attack
120    // surface without adding a runtime `unicode-normalization`
121    // dependency to the parse gate.
122    if !tool.is_ascii() {
123        return Err(RpcError::BadParams(
124            "params.name must be ASCII: non-ASCII tool names introduce a Unicode-normalization mismatch \
125             between this proxy's exact-byte matching and downstream MCP servers that fold NFC/NFKC"
126                .into(),
127        ));
128    }
129    // Refuse any uppercase byte. Downstream tool-name comparisons are
130    // exact-byte, but most MCP servers apply ASCII case folding. A
131    // policy that denies `db_write` sees `DB_write` slip through the
132    // deny-list byte-eq check while the server executes the same
133    // tool. Requiring the caller to normalize to lowercase up front
134    // eliminates the class.
135    if tool.chars().any(|c| c.is_ascii_uppercase()) {
136        return Err(RpcError::BadParams(
137            "params.name must be lowercase: mixed-case tool names bypass policy deny-lists that use \
138             exact-byte matching while downstream MCP servers apply ASCII case folding"
139                .into(),
140        ));
141    }
142    // JSON-RPC 2.0 §4 requires `id` to be a string, number, or null: reject
143    // objects/arrays/bool up front so downstream correlation tables that key
144    // on id-as-string cannot be confused by structured ids.
145    let id = obj.get("id").cloned();
146    match id.as_ref() {
147        None => {}
148        Some(Value::String(_) | Value::Number(_) | Value::Null) => {}
149        Some(_) => {
150            return Err(RpcError::BadParams(
151                "id must be a string, number, or null per JSON-RPC 2.0 §4".into(),
152            ));
153        }
154    }
155    // Per JSON-RPC 2.0 §4.1 a request with no `id` is a notification: a
156    // notification MUST NOT elicit a response. Refusing them here keeps
157    // attackers from fire-and-forgetting `tools/call` to drain budget
158    // (max_total_tool_calls, payout) without needing to consume responses.
159    if id.is_none() {
160        return Err(RpcError::BadParams(
161            "tools/call requires an id; JSON-RPC notifications are not accepted".into(),
162        ));
163    }
164    let arguments = params
165        .get("arguments")
166        .cloned()
167        .unwrap_or_else(|| Value::Object(Default::default()));
168    if !arguments.is_object() {
169        return Err(RpcError::BadParams("params.arguments must be an object".into()));
170    }
171    Ok(ToolCallRequest {
172        id,
173        tool: tool.to_owned(),
174        arguments,
175    })
176}

178/// Compute the maximum nesting depth of a JSON value. Exposed so
179/// callers outside this crate (chat-completions ingress and any
180/// admin body parser) can apply the same trust-boundary
181/// depth cap without duplicating logic.
182pub fn depth_of(v: &Value, current: usize) -> usize {
183    if current > MAX_JSON_DEPTH {
184        return current; // early-out: no need to recurse further
185    }
186    match v {
187        Value::Array(items) => items
188            .iter()
189            .map(|i| depth_of(i, current + 1))
190            .max()
191            .unwrap_or(current + 1),
192        Value::Object(map) => map
193            .values()
194            .map(|i| depth_of(i, current + 1))
195            .max()
196            .unwrap_or(current + 1),
197        _ => current,
198    }
199}

201/// The general-purpose duplicate-key
202/// rejection primitive, exposed for callers outside this crate (chat
203/// completions ingress, admin body parsers) that want the same
204/// trust-boundary "refuse ambiguous JSON" semantics without pulling in
205/// `RpcError`.
206///
207/// Returns `Ok(())` if `raw` contains no duplicate keys at any nesting
208/// level; otherwise a description of the offending key.
209pub fn refuse_duplicate_json_keys(raw: &[u8]) -> Result<(), String> {
210    reject_duplicate_keys(raw).map_err(|error| error.to_string())
211}

213/// Refuse any JSON object with duplicate keys anywhere in the payload.
214/// `serde_json` silently keeps the last value; downstream MCP servers may
215/// keep the first, expose both, or interpret the collapse differently.
216/// The trust-boundary policy is "refuse ambiguity" — see the call-site
217/// comment in `parse_tool_call`.
218///
219/// Distinguish scanner-created "duplicate key"
220/// errors from underlying `serde_json` parse errors (EOF, unbalanced
221/// braces, invalid escape, recursion limit). The prior blanket
222/// `.map_err(|e| RpcError::Json(format!("duplicate JSON key rejected:
223/// {e}")))` gave EVERY scanner error the same misleading
224/// "duplicate JSON key rejected" prefix, so operator triage on
225/// dup-key alerts fired on any malformed JSON. Mirrors the
226/// sentinel fix in `pv_receipts::check_no_duplicate_keys`.
227///
228/// Also refuse trailing garbage AFTER a
229/// valid JSON value. `deserialize_any` returns after the first
230/// complete value, so an input like `{"ok":1}garbage` was silently
231/// accepted. `Deserializer::end()` returns Err if any non-whitespace
232/// bytes remain, closing the class of "smuggled second document"
233/// attacks (JSON body-splitting through a proxy that only inspected
234/// the first value).
235fn reject_duplicate_keys(raw: &[u8]) -> Result<(), RpcError> {
236    use serde::de::{Deserializer, MapAccess, SeqAccess, Visitor};
237    use std::fmt;
238
239    const DUP_KEY_SENTINEL: &str = "__av_sandbox_dup:";
240
241    struct NoDupKeys;
242
243    impl<'de> Visitor<'de> for NoDupKeys {
244        type Value = ();
245
246        fn expecting(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
247            f.write_str("any JSON value without duplicate object keys")
248        }
249
250        fn visit_bool<E>(self, _: bool) -> Result<(), E> {
251            Ok(())
252        }
253        fn visit_i64<E>(self, _: i64) -> Result<(), E> {
254            Ok(())
255        }
256        fn visit_u64<E>(self, _: u64) -> Result<(), E> {
257            Ok(())
258        }
259        fn visit_f64<E>(self, _: f64) -> Result<(), E> {
260            Ok(())
261        }
262        fn visit_str<E>(self, _: &str) -> Result<(), E> {
263            Ok(())
264        }
265        fn visit_string<E>(self, _: String) -> Result<(), E> {
266            Ok(())
267        }
268        fn visit_none<E>(self) -> Result<(), E> {
269            Ok(())
270        }
271        fn visit_unit<E>(self) -> Result<(), E> {
272            Ok(())
273        }
274
275        fn visit_some<D>(self, deserializer: D) -> Result<(), D::Error>
276        where
277            D: Deserializer<'de>,
278        {
279            deserializer.deserialize_any(NoDupKeys)
280        }
281
282        fn visit_seq<A>(self, mut seq: A) -> Result<(), A::Error>
283        where
284            A: SeqAccess<'de>,
285        {
286            while seq.next_element::<NoDupWrap>()?.is_some() {}
287            Ok(())
288        }
289
290        fn visit_map<A>(self, mut map: A) -> Result<(), A::Error>
291        where
292            A: MapAccess<'de>,
293        {
294            let mut seen: std::collections::HashSet<String> = std::collections::HashSet::new();
295            while let Some(key) = map.next_key::<String>()? {
296                if !seen.insert(key.clone()) {
297                    return Err(serde::de::Error::custom(format!(
298                        "{DUP_KEY_SENTINEL}duplicate object key `{}`",
299                        key.escape_debug()
300                    )));
301                }
302                let _: NoDupWrap = map.next_value()?;
303            }
304            Ok(())
305        }
306    }
307
308    // Wrapper so nested structures use the same visitor.
309    struct NoDupWrap;
310    impl<'de> serde::Deserialize<'de> for NoDupWrap {
311        fn deserialize<D>(deserializer: D) -> Result<Self, D::Error>
312        where
313            D: Deserializer<'de>,
314        {
315            deserializer.deserialize_any(NoDupKeys)?;
316            Ok(NoDupWrap)
317        }
318    }
319
320    let mut de = serde_json::Deserializer::from_slice(raw);
321    if let Err(error) = de.deserialize_any(NoDupKeys) {
322        let msg = error.to_string();
323        return if let Some(rest) = msg.strip_prefix(DUP_KEY_SENTINEL) {
324            Err(RpcError::Json(format!("duplicate JSON key rejected: {rest}")))
325        } else {
326            // Not a duplicate-key error — the underlying serde_json
327            // error (EOF, unbalanced braces, invalid escape,
328            // recursion limit) surfaces with its own prefix so
329            // triage doesn't misattribute the class.
330            Err(RpcError::Json(msg))
331        };
332    }
333    // Refuse trailing content after the first complete
334    // JSON value. `de.end()` returns Err on any non-whitespace
335    // trailing bytes.
336    de.end()
337        .map_err(|error| RpcError::Json(format!("trailing content after JSON value: {error}")))?;
338    Ok(())
339}

370/// [`denial_error`] for a refusal that involved the semantic judge:
371/// `error.data.predicate_id` names the predicate behind it, so an agent
372/// (or the person it asks) can see which question failed.
373pub fn judge_denial_error(
374    id: Option<&Value>,
375    code: DenialCode,
376    reason: &str,
377    predicate_id: Option<&str>,
378) -> Value {
379    let mut response = denial_error(id, code, reason);
380    if let (Some(predicate_id), Some(data)) = (predicate_id, response.pointer_mut("/error/data")) {
381        if let Some(map) = data.as_object_mut() {
382            map.insert("predicate_id".to_owned(), Value::String(predicate_id.to_owned()));
383        }
384    }
385    response
386}
387
388/// [`denial_error`] for a refusal by a policy that read the semantic
389/// judge's verdicts (WASM interface version 2): `error.data.predicate_ids`
390/// names the predicates the policy saw, and `error.data.predicate_id` the
391/// predicate when there was only one.
392pub fn judged_policy_error(
393    id: Option<&Value>,
394    code: DenialCode,
395    reason: &str,
396    predicate_ids: &[&str],
397) -> Value {
398    let only = match predicate_ids {
399        [only] => Some(*only),
400        _ => None,
401    };
402    let mut response = judge_denial_error(id, code, reason, only);
403    if let Some(map) = response.pointer_mut("/error/data").and_then(Value::as_object_mut) {
404        map.insert(
405            "predicate_ids".to_owned(),
406            Value::Array(
407                predicate_ids
408                    .iter()
409                    .map(|id| Value::String((*id).to_owned()))
410                    .collect(),
411            ),
412        );
413    }
414    response
415}

417/// Recover the request `id` for an error response, per JSON-RPC 2.0 §5:
418/// the response `id` MUST echo the request's when it is detectable, and
419/// be `null` only "if there was an error in detecting the id" (parse
420/// error / invalid request). [`RpcError::NotToolCall`] and
421/// [`RpcError::BadParams`] both mean the envelope parsed as a
422/// well-formed JSON-RPC 2.0 object, so the id IS detectable — replying
423/// `null` broke client correlation (a pending `initialize` call hung
424/// until timeout because its error response carried `id: null`).
425/// Parse-class failures (invalid JSON, duplicate keys, oversize,
426/// non-2.0 envelope) keep `null`: the id is absent, ambiguous, or
427/// untrustworthy there.
428#[must_use]
429pub fn detectable_error_id(raw: &[u8], error: &RpcError) -> Option<Value> {
430    if !matches!(error, RpcError::NotToolCall(_) | RpcError::BadParams(_)) {
431        return None;
432    }
433    // Both variants are only produced after the size, duplicate-key and
434    // depth gates passed, so this bounded re-parse cannot do more work
435    // than the original.
436    let value: Value = serde_json::from_slice(raw).ok()?;
437    let id = value.as_object()?.get("id")?;
438    // §4: id must be a string, number, or null — echo nothing else.
439    matches!(id, Value::String(_) | Value::Number(_) | Value::Null).then(|| id.clone())
440}

442/// JSON-RPC 2.0 reserves -32700 (parse
443/// error), -32600 (invalid request), and -32602 (invalid params) for
444/// requests that never reached application logic. Reporting those as
445/// -32001 "blocked by policy" (with HTTP 403) falsely told clients an
446/// authorization decision was made about a request that never parsed.
447pub fn protocol_error(id: Option<&Value>, error: &RpcError) -> Value {
448    let (code, message) = match error {
449        RpcError::Json(_) => (-32700, "parse error"),
450        RpcError::NotJsonRpc(_) | RpcError::TooLarge(..) => (-32600, "invalid request"),
451        RpcError::BadParams(_) => (-32602, "invalid params"),
452        // Passthrough refusal is an application-level policy choice.
453        RpcError::NotToolCall(_) => {
454            return denial_error(id, DenialCode::PassthroughRefused, &error.to_string());
455        }
456    };
457    serde_json::json!({
458        "jsonrpc": "2.0",
459        "id": id.cloned().unwrap_or(Value::Null),
460        "error": {
461            "code": code,
462            "message": message,
463            "data": { "reason": error.to_string(), "code": DenialCode::ParseError.as_str() }
464        }
465    })
466}

492    /// — the exact gate the harness chat ingress calls before serde ever sees
493    /// a request body — had no direct test, so a mutant replacing its body

498        assert!(refuse_duplicate_json_keys(br#"{"a":1,"a":2}"#).is_err());
499        assert!(refuse_duplicate_json_keys(br#"{"outer":{"a":1,"a":2}}"#).is_err());
500        assert!(refuse_duplicate_json_keys(br#"{"arr":[{"a":1,"a":2}]}"#).is_err());
501        assert!(refuse_duplicate_json_keys(br#"{"ok":1}garbage"#).is_err());
502        assert!(refuse_duplicate_json_keys(br#"{"a":1,"b":{"a":1},"c":[1,"a"]}"#).is_ok());
503        assert!(refuse_duplicate_json_keys(b"[1,2,3]").is_ok());
504    }

522        }

527    /// JSON-RPC 2.0 §5: an error response MUST echo the request `id`
528    /// whenever it is detectable, and use `null` only when detection
529    /// failed. Pre-fix every parse-gate refusal replied `id: null`,
530    /// including well-formed envelopes (`initialize` passthrough, bad
531    /// params) whose id was plainly readable — SDK correlation tables
532    /// never matched the response, so the pending call hung to timeout.
533    #[test]
534    fn error_responses_echo_a_detectable_request_id() {
535        // Well-formed envelope, non-tools/call method: id 42 detectable.
536        let raw = br#"{"jsonrpc":"2.0","id":42,"method":"initialize","params":{}}"#;
537        let error = parse_tool_call(raw).unwrap_err();
538        assert!(matches!(error, RpcError::NotToolCall(_)));
539        let id = detectable_error_id(raw, &error);
540        assert_eq!(id, Some(json!(42)));
541        let response = protocol_error(id.as_ref(), &error);
542        assert_eq!(response["id"], json!(42));
543
544        // Well-formed envelope, bad params: string id detectable.
545        let raw = br#"{"jsonrpc":"2.0","id":"req-1","method":"tools/call","params":{}}"#;
546        let error = parse_tool_call(raw).unwrap_err();
547        assert!(matches!(error, RpcError::BadParams(_)));
548        assert_eq!(detectable_error_id(raw, &error), Some(json!("req-1")));
549
550        // Structured (spec-invalid) id: refused as BadParams, and the
551        // echo must NOT propagate the invalid shape — null stands.
552        let raw = br#"{"jsonrpc":"2.0","id":{"nested":1},"method":"tools/call","params":{"name":"t"}}"#;
553        let error = parse_tool_call(raw).unwrap_err();
554        assert!(matches!(error, RpcError::BadParams(_)));
555        assert_eq!(detectable_error_id(raw, &error), None);
556
557        // Parse-class failures: id undetectable per spec — null.
558        let raw = b"not json";

888    /// `reject_duplicate_keys`
889    /// used to wrap EVERY scanner error with the "duplicate JSON key
890    /// rejected: ..." prefix — misleading operator triage on
891    /// dup-key alerts because malformed JSON, EOF, and recursion
892    /// limits were all reported as if they were duplicate-key
893    /// rejections. Mirror the sentinel fix from
894    /// `pv_receipts`: real duplicate-key errors still get the
895    /// dup-key prefix; parse errors surface their own class.
896    #[test]
897    fn duplicate_key_class_distinguished_from_generic_parse_error() {
898        // Duplicate top-level key → "duplicate JSON key rejected: ..."
899        let real_dup = serde_json::to_vec(&json!({
900            "jsonrpc": "2.0",
901            "id": "dup",
902            "method": "tools/call",
903            "params": {"name": "read", "arguments": {}}
904        }))
905        .unwrap();
906        // Inject a duplicate by hand — serde_json's default parser
907        // would take last-wins.
908        let dup_raw: Vec<u8> = {
909            let mut v = real_dup.clone();
910            // Insert a second `"jsonrpc": "2.0"` right after the
911            // opening brace.
912            let insertion = br#""jsonrpc":"2.0","#;
913            v.splice(1..1, insertion.iter().copied());
914            v
915        };
916        let err = parse_tool_call(&dup_raw).unwrap_err();
917        assert!(
918            matches!(&err, RpcError::Json(msg) if msg.contains("duplicate JSON key rejected")),
919            "real dup-key must carry the dup-key prefix, got {err:?}"
920        );
921
922        // Malformed JSON (unbalanced brace) → NO dup-key prefix.
923        let malformed: &[u8] = b"{\"jsonrpc\":\"2.0\",\"id\":1,";
924        let err = parse_tool_call(malformed).unwrap_err();
925        match &err {
926            RpcError::Json(msg) => assert!(
927                !msg.contains("duplicate JSON key rejected"),
928                "malformed JSON must not be labeled as duplicate-key, got {msg:?}"
929            ),
930            other => panic!("expected RpcError::Json for malformed input, got {other:?}"),
931        }
932
933        // Empty input → NO dup-key prefix.
934        let err = parse_tool_call(b"").unwrap_err();
935        if let RpcError::Json(msg) = &err {
936            assert!(
937                !msg.contains("duplicate JSON key rejected"),
938                "empty input must not be labeled as duplicate-key, got {msg:?}"
939            );
940        }
941    }

984    /// Mutation-run hardening: the depth cap was only proven far
985    /// past the limit, so boundary mutants (`current + 1` -> `current * 1`,
986    /// deleting the Array arm) survived. Pin the exact boundary through the
987    /// public parser for BOTH container kinds. Depth accounting: the
988    /// envelope contributes 3 levels (root -> params -> arguments), and each
989    /// wrapper inside `arguments.k` adds one, so 61 wrappers sit exactly at
990    /// MAX_JSON_DEPTH = 64 and 62 exceed it.
991    #[test]
992    fn json_depth_boundary_is_exact_for_arrays_and_objects() {
993        let build = |inner: String| {
994            format!(
995                r#"{{"jsonrpc":"2.0","id":1,"method":"tools/call","params":{{"name":"t","arguments":{{"k":{inner}}}}}}}"#
996            )
997        };
998        let arrays = |n: usize| format!("{}0{}", "[".repeat(n), "]".repeat(n));
999        let objects = |n: usize| format!("{}0{}", "{\"x\":".repeat(n), "}".repeat(n));
1000        for shape in [&arrays as &dyn Fn(usize) -> String, &objects] {
1001            let at_limit = build(shape(MAX_JSON_DEPTH - 3));
1002            assert!(
1003                parse_tool_call(at_limit.as_bytes()).is_ok(),
1004                "depth exactly at MAX_JSON_DEPTH must parse"
1005            );
1006            let past = build(shape(MAX_JSON_DEPTH - 2));
1007            let outcome = parse_tool_call(past.as_bytes());
1008            assert!(
1009                matches!(outcome, Err(RpcError::NotJsonRpc(ref m)) if m.contains("depth")),
1010                "one past MAX_JSON_DEPTH must be refused, got {outcome:?}"
1011            );
1012        }
1013        // Empty containers exercise the `unwrap_or(current + 1)` fallback
1014        // arms — an empty object/array is still one level deep, so one
1015        // sitting past the cap must be refused too.
1016        for leaf in ["{}", "[]"] {
1017            let wrappers = "{\"x\":".repeat(MAX_JSON_DEPTH - 3);
1018            let closers = "}".repeat(MAX_JSON_DEPTH - 3);
1019            let past = build(format!("{wrappers}{leaf}{closers}"));
1020            let outcome = parse_tool_call(past.as_bytes());
1021            assert!(
1022                matches!(outcome, Err(RpcError::NotJsonRpc(ref m)) if m.contains("depth")),
1023                "empty {leaf} leaf past MAX_JSON_DEPTH must be refused, got {outcome:?}"
1024            );
1025        }
1026    }
