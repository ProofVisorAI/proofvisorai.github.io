1//! A deterministic judge for tests, shadow replay, and calibration
2//! harness checks. It is not selectable from a production config.
3
4use crate::adapter::{AdapterKind, BoxFuture, JudgeAdapter, JudgeError};
5use crate::model::{JudgeRequest, ModelPin, RawScores};
6use crate::predicate::PredicateDef;
7use parking_lot::Mutex;
8use std::collections::BTreeMap;
9use std::sync::atomic::{AtomicU64, Ordering};
10use std::time::Duration;
11
12/// What the mock does for a call.
13#[derive(Debug, Clone)]
14pub enum MockBehavior {
15    /// Answer with these probabilities, as `model` (the pin when `None`).
16    Answer {
17        /// `(label, probability)` pairs.
18        probs: Vec<(String, f64)>,
19        /// Model to report; `None` reports the pinned model.
20        model: Option<ModelPin>,
21    },
22    /// Fail with this error.
23    Fail(JudgeError),
24    /// Wait, then behave as the inner behaviour.
25    Delay(Duration, Box<MockBehavior>),
26    /// Panic inside the scoring task (exercises crash handling).
27    Panic,
28}
29
30impl MockBehavior {
31    /// Answer with these probabilities as the pinned model.
32    pub fn scores(probs: &[(&str, f64)]) -> Self {
33        Self::Answer {
34            probs: probs.iter().map(|(l, p)| ((*l).to_owned(), *p)).collect(),
35            model: None,
36        }
37    }
38
39    /// Wait `delay`, then behave as `then`.
40    pub fn delayed(delay: Duration, then: Self) -> Self {
41        Self::Delay(delay, Box::new(then))
42    }
43}

45/// The mock judge. Behaviour is chosen by `(predicate, tool)`, then by
46/// predicate, then the default; a scorer function can replace all of it.
47pub struct MockJudge {
48    pin: ModelPin,
49    default: Mutex<MockBehavior>,
50    by_predicate: Mutex<BTreeMap<String, MockBehavior>>,
51    by_tool: Mutex<BTreeMap<(String, String), MockBehavior>>,
52    #[allow(clippy::type_complexity)]
53    scorer: Mutex<Option<Box<dyn Fn(&JudgeRequest) -> MockBehavior + Send + Sync>>>,
54    calls: AtomicU64,
55    last: Mutex<Option<JudgeRequest>>,
56}
57
58impl MockJudge {
59    /// A mock that behaves as `default` for every call.
60    pub fn new(pin: ModelPin, default: MockBehavior) -> Self {
61        Self {
62            pin,
63            default: Mutex::new(default),
64            by_predicate: Mutex::new(BTreeMap::new()),
65            by_tool: Mutex::new(BTreeMap::new()),
66            scorer: Mutex::new(None),
67            calls: AtomicU64::new(0),
68            last: Mutex::new(None),
69        }
70    }
71
72    /// Replace the default behaviour.
73    pub fn set_default(&self, behavior: MockBehavior) {
74        *self.default.lock() = behavior;
75    }
76
77    /// Behaviour for one predicate.
78    pub fn set_for_predicate(&self, predicate_id: &str, behavior: MockBehavior) {
79        self.by_predicate.lock().insert(predicate_id.to_owned(), behavior);
80    }
81
82    /// Behaviour for one predicate on one tool.
83    pub fn set_for_tool(&self, predicate_id: &str, tool: &str, behavior: MockBehavior) {
84        self.by_tool
85            .lock()
86            .insert((predicate_id.to_owned(), tool.to_owned()), behavior);
87    }
88
89    /// Decide every call with a function of the request.
90    pub fn set_scorer(&self, scorer: impl Fn(&JudgeRequest) -> MockBehavior + Send + Sync + 'static) {
91        *self.scorer.lock() = Some(Box::new(scorer));
92    }
93
94    /// Calls made so far.
95    pub fn calls(&self) -> u64 {
96        self.calls.load(Ordering::SeqCst)
97    }
98
99    /// The most recent request.
100    pub fn last_request(&self) -> Option<JudgeRequest> {
101        self.last.lock().clone()
102    }
103
104    fn behavior_for(&self, request: &JudgeRequest) -> MockBehavior {
105        if let Some(scorer) = self.scorer.lock().as_ref() {
106            return scorer(request);
107        }
108        if let Some(behavior) = self
109            .by_tool
110            .lock()
111            .get(&(request.predicate_id.clone(), request.tool.clone()))
112        {
113            return behavior.clone();
114        }
115        if let Some(behavior) = self.by_predicate.lock().get(&request.predicate_id) {
116            return behavior.clone();
117        }
118        self.default.lock().clone()
119    }
120}
121
122impl JudgeAdapter for MockJudge {
123    fn id(&self) -> &ModelPin {
124        &self.pin
125    }
126
127    fn kind(&self) -> AdapterKind {
128        AdapterKind::Mock
129    }
130
131    fn score<'a>(
132        &'a self,
133        request: &'a JudgeRequest,
134        _predicate: &'a PredicateDef,
135    ) -> BoxFuture<'a, Result<RawScores, JudgeError>> {
136        self.calls.fetch_add(1, Ordering::SeqCst);
137        *self.last.lock() = Some(request.clone());
138        let behavior = self.behavior_for(request);
139        Box::pin(async move { run(behavior, &self.pin).await })
140    }
141}
142
143async fn run(mut behavior: MockBehavior, pin: &ModelPin) -> Result<RawScores, JudgeError> {
144    loop {
145        match behavior {
146            MockBehavior::Delay(delay, then) => {
147                tokio::time::sleep(delay).await;
148                behavior = *then;
149            }
150            MockBehavior::Answer { probs, model } => {
151                return Ok(RawScores {
152                    probs,
153                    model: model.unwrap_or_else(|| pin.clone()),
154                });
155            }
156            MockBehavior::Fail(error) => return Err(error),
157            MockBehavior::Panic => crash(),
158        }
159    }
160}
