1//! The judge record written into the tool-call OCSF event payload and
2//! the ATIF tool call's `extra`.
3//!
4//! For signed workflows the session receipt signs the head of the event
5//! chain, which hashes every event in JCS form; for unsigned workflows
6//! a retroactive receipt signs the digest of the ATIF trajectory file.
7//! Writing this record into both places means every verdict is covered
8//! by whichever signature the session ends up with, without changing
9//! the receipt format. Every number in it is an integer.
10
11use crate::adapter::{AdapterKind, JudgeOutcome};
12use crate::model::{LabelProb, ModelPin};
13use crate::thresholds::{DecisionCause, JudgeAction, JudgeDecision};
14use serde::{Deserialize, Serialize};

16/// Longest failure detail kept in a record.
17pub const MAX_DETAIL_CHARS: usize = 256;
18
19/// Whether the judge can block.
20#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, Serialize, Deserialize)]
21#[serde(rename_all = "snake_case")]
22pub enum JudgeMode {
23    /// Score and record; never block.
24    Shadow,
25    /// Score, record, and let policy act on the verdicts.
26    Enforce,
27}
28
29impl JudgeMode {
30    /// Stable wire string.
31    pub fn as_str(self) -> &'static str {
32        match self {
33            Self::Shadow => "shadow",
34            Self::Enforce => "enforce",
35        }
36    }
37}

39/// One predicate's result as recorded.
40#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
41#[serde(deny_unknown_fields)]
42pub struct OutcomeRecord {
43    /// Versioned predicate id.
44    pub predicate_id: String,
45    /// Configured adapter name.
46    pub adapter: String,
47    /// Adapter kind.
48    pub adapter_kind: AdapterKind,
49    /// The pinned model.
50    pub model: ModelPin,
51    /// `scored` or the failure class.
52    pub status: String,
53    /// The full distribution (empty when not scored), ppm.
54    #[serde(default, skip_serializing_if = "Vec::is_empty")]
55    pub probs: Vec<LabelProb>,
56    /// Operator-facing failure detail.
57    #[serde(default, skip_serializing_if = "Option::is_none")]
58    pub detail: Option<String>,
59    /// Scoring latency, microseconds.
60    pub latency_us: u64,
61    /// Present when this verdict was sampled for human review.
62    #[serde(default, skip_serializing_if = "Option::is_none")]
63    pub review_id: Option<String>,
64}
65
66impl OutcomeRecord {
67    /// Record an outcome produced by `adapter`.
68    pub fn from_outcome(outcome: &JudgeOutcome, adapter: &str, adapter_kind: AdapterKind) -> Self {
69        let (probs, detail) = match outcome {
70            JudgeOutcome::Scored(verdict) => (verdict.probs.clone(), None),
71            JudgeOutcome::Failed(failure) => (
72                Vec::new(),
73                Some(truncate_chars(&failure.detail, MAX_DETAIL_CHARS)),
74            ),
75        };
76        Self {
77            predicate_id: outcome.predicate_id().to_owned(),
78            adapter: adapter.to_owned(),
79            adapter_kind,
80            model: outcome.model().clone(),
81            status: outcome.status_str().to_owned(),
82            probs,
83            detail,
84            latency_us: outcome.latency_us(),
85            review_id: None,
86        }
87    }
88}
89/// The request a judged call put to the judge, kept when the operator
90/// records it. The arguments are kept as JSON text: the event chain is
91/// hashed in canonical JSON, where floating-point numbers are a known
92/// source of disagreement.
93#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
94#[serde(deny_unknown_fields)]
95pub struct JudgeInputs {
96    /// The tool called.
97    pub tool: String,
98    /// The validated arguments, as JSON text.
99    pub arguments_json: String,
100    /// The context the judge saw.
101    pub context: crate::model::SessionContext,
102}

104/// Everything an auditor needs to replay a judged decision.
105#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
106#[serde(deny_unknown_fields)]
107pub struct JudgeRecord {
108    /// Shadow or enforce.
109    pub mode: JudgeMode,
110    /// Wall time spent waiting for the judge, microseconds. Kept apart
111    /// from the sandbox's own decision latency.
112    pub judge_us: u64,
113    /// One entry per evaluated predicate.
114    pub verdicts: Vec<OutcomeRecord>,
115    /// What the configured thresholds decided over the verdicts (the
116    /// replayable part: probability, threshold, action). In shadow mode,
117    /// or after the pilot cap, this is what they WOULD have done.
118    pub decision: JudgeAction,
119    /// True when judge evidence was applied to the call.
120    pub enforced: bool,
121    /// True when judge-aware policy (the thresholds and any policy that
122    /// reads verdicts) blocks a call the rest of policy allows. In
123    /// shadow mode this is the "would have blocked" signal to compare
124    /// against what actually happened.
125    pub would_block: bool,
126    /// What the whole policy stage decided with the verdicts in hand:
127    /// the thresholds, every policy that reads verdicts (WASM interface
128    /// version 2), and the rules that ignore them. In shadow mode this is
129    /// what it would have decided.
130    pub policy_decision: JudgeAction,
131    /// The policy that blocked or held the call, when `policy_decision`
132    /// is not `allow`.
133    #[serde(default, skip_serializing_if = "Option::is_none")]
134    pub policy: Option<String>,
135    /// The predicate that drove the decision.
136    #[serde(default, skip_serializing_if = "Option::is_none")]
137    pub decision_predicate: Option<String>,
138    /// Why.
139    pub cause: DecisionCause,
140    /// True when the pilot's daily block cap turned an enforced block
141    /// into a recorded allow.
142    #[serde(default, skip_serializing_if = "std::ops::Not::not")]
143    pub pilot_cap_reached: bool,
144    /// Predicates in ask-by-default mode after a calibration drift check.
145    #[serde(default, skip_serializing_if = "Vec::is_empty")]
146    pub drift_tripped: Vec<String>,
147    /// What the judge was asked, when the operator records it
148    /// (`[judge] record_inputs`): the material for a labelled set.
149    #[serde(default, skip_serializing_if = "Option::is_none")]
150    pub inputs: Option<JudgeInputs>,
151}

153impl JudgeRecord {
154    /// Build a record from a decision.
155    pub fn new(
156        mode: JudgeMode,
157        judge_us: u64,
158        verdicts: Vec<OutcomeRecord>,
159        decision: &JudgeDecision,
160        enforced: bool,
161    ) -> Self {
162        Self {
163            mode,
164            judge_us,
165            verdicts,
166            decision: decision.action,
167            enforced,
168            would_block: false,
169            policy_decision: decision.action,
170            policy: None,
171            decision_predicate: decision.predicate_id.clone(),
172            cause: decision.cause.clone(),
173            pilot_cap_reached: false,
174            drift_tripped: Vec::new(),
175            inputs: None,
176        }
177    }
178}

180fn truncate_chars(text: &str, max: usize) -> String {
181    match text.char_indices().nth(max) {
182        Some((cut, _)) => text.get(..cut).unwrap_or(text).to_owned(),
183        None => text.to_owned(),
184    }
185}

199    #[test]
200    fn record_round_trips_and_carries_no_floats() {
201        let outcome = JudgeOutcome::Scored(Verdict {
202            predicate_id: "arg_matches_task@1".into(),
203            probs: vec![
204                LabelProb {
205                    label: "match".into(),
206                    ppm: 180_000,
207                },
208                LabelProb {
209                    label: "mismatch".into(),
210                    ppm: 820_000,
211                },
212            ],
213            model: ModelPin::new("judge", "7", "d".repeat(64)).unwrap(),
214            latency_us: 1_234,
215        });
216        let decision = JudgeDecision {
217            action: JudgeAction::Deny,
218            predicate_id: Some("arg_matches_task@1".into()),
219            cause: DecisionCause::Threshold {
220                label: "mismatch".into(),
221                ppm: 820_000,
222                threshold_ppm: 700_000,
223            },
224            reason: "r".into(),
225        };
226        let record = JudgeRecord::new(
227            JudgeMode::Enforce,
228            1_500,
229            vec![OutcomeRecord::from_outcome(&outcome, "local", AdapterKind::Local)],
230            &decision,
231            true,
232        );
233        let text = serde_json::to_string(&record).unwrap();
234        assert!(!text.contains('.'), "{text}");
235        let back: JudgeRecord = serde_json::from_str(&text).unwrap();
236        assert_eq!(back, record);
237    }

239    /// The record's fields, as the configuration reference documents them.
240    /// The receipt signs the record through the event chain, so this list
241    /// is the record's own whitelist: a field added, renamed or removed
242    /// fails this test until the list and the documentation are updated.
243    const RECORD_FIELDS: &[&str] = &[
244        "mode",
245        "judge_us",
246        "verdicts",
247        "decision",
248        "enforced",
249        "would_block",
250        "policy_decision",
251        "policy",
252        "decision_predicate",
253        "cause",
254        "pilot_cap_reached",
255        "drift_tripped",
256        "inputs",
257    ];
258    const VERDICT_FIELDS: &[&str] = &[
259        "predicate_id",
260        "adapter",
261        "adapter_kind",
262        "model",
263        "status",
264        "probs",
265        "detail",
266        "latency_us",
267        "review_id",
268    ];

270    #[test]
271    fn record_fields_are_exactly_the_documented_ones() {
272        let scored = JudgeOutcome::Scored(Verdict {
273            predicate_id: "arg_matches_task@1".into(),
274            probs: vec![LabelProb {
275                label: "mismatch".into(),
276                ppm: 820_000,
277            }],
278            model: ModelPin::new("judge", "7", "d".repeat(64)).unwrap(),
279            latency_us: 1,
280        });
281        let failed = JudgeOutcome::Failed(crate::JudgeFailure {
282            predicate_id: "arg_matches_task@1".into(),
283            model: ModelPin::new("judge", "7", "d".repeat(64)).unwrap(),
284            kind: crate::FailureKind::Timeout,
285            detail: "late".into(),
286            latency_us: 1,
287        });
288        let decision = JudgeDecision {
289            action: JudgeAction::Deny,
290            predicate_id: Some("arg_matches_task@1".into()),
291            cause: DecisionCause::Threshold {
292                label: "mismatch".into(),
293                ppm: 820_000,
294                threshold_ppm: 700_000,
295            },
296            reason: "r".into(),
297        };
298        let mut verdicts = vec![
299            OutcomeRecord::from_outcome(&scored, "local", AdapterKind::Local),
300            OutcomeRecord::from_outcome(&failed, "local", AdapterKind::Local),
301        ];
302        if let Some(first) = verdicts.first_mut() {
303            first.review_id = Some("r-1".into());
304        }
305        let mut record = JudgeRecord::new(JudgeMode::Enforce, 1, verdicts, &decision, true);
306        record.policy = Some("p".into());
307        record.pilot_cap_reached = true;
308        record.drift_tripped = vec!["arg_matches_task@1".into()];
309        record.inputs = Some(JudgeInputs {
310            tool: "t".into(),
311            arguments_json: "{}".into(),
312            context: crate::model::SessionContext::default(),
313        });
314        let value = serde_json::to_value(&record).unwrap();
315        let keys = |object: &serde_json::Value| -> std::collections::BTreeSet<String> {
316            object.as_object().unwrap().keys().cloned().collect()
317        };
318        let expected = |fields: &[&str]| fields.iter().map(|field| (*field).to_owned()).collect();
319        assert_eq!(keys(&value), expected(RECORD_FIELDS));
320        let mut seen = std::collections::BTreeSet::new();
321        for verdict in value
322            .get("verdicts")
323            .and_then(serde_json::Value::as_array)
324            .unwrap()
325        {
326            seen.extend(keys(verdict));
327        }
328        assert_eq!(seen, expected(VERDICT_FIELDS));
329        let reference = include_str!("../../../docs/reference/CONFIGURATION.md");
330        for field in RECORD_FIELDS {
331            assert!(
332                reference.contains(&format!("`{field}`")),
333                "CONFIGURATION.md does not document `{field}`"
334            );
335        }
336    }

338    #[test]
339    fn failure_detail_is_bounded() {
340        let outcome = JudgeOutcome::Failed(JudgeFailure {
341            predicate_id: "p@1".into(),
342            model: pin(),
343            kind: FailureKind::Unavailable,
344            detail: "é".repeat(1_000),
345            latency_us: 3,
346        });
347        let record = OutcomeRecord::from_outcome(&outcome, "cloud", AdapterKind::Cloud);
348        assert_eq!(record.detail.unwrap().chars().count(), MAX_DETAIL_CHARS);
349        assert_eq!(record.status, "unavailable");
350        assert!(record.probs.is_empty());
351    }
