1//! Named, versioned predicates with fixed label sets.
2//!
3//! A predicate is the question the judge answers ("do these arguments
4//! match the task?"). Its id carries a version (`name@N`) and its label
5//! set is fixed for that version, so a policy written against
6//! `arg_matches_task@1` and its `mismatch` label keeps meaning the same
7//! thing. Changing the labels means a new version.
8
9use serde::{Deserialize, Serialize};
10use std::collections::BTreeMap;
11
12/// Built-in predicate: do the validated arguments fit the task, intent
13/// and mission the operator configured for this agent?
14pub const ARG_MATCHES_TASK: &str = "arg_matches_task@1";
15/// Built-in predicate: is a delegated (child) agent using its parent's
16/// entitlement to do what the parent was refused?
17pub const CHILD_USES_PARENT_ENTITLEMENT: &str = "child_uses_parent_entitlement@1";
18/// Built-in predicate for governed memory writeback: is this write to a
19/// memory or knowledge store safe to persist?
20pub const MEMORY_WRITE_SAFE: &str = "memory_write_safe@1";
21
22/// Built-in predicate for trace judging: does a recorded tool or model
23/// output carry instructions aimed at the agent, or data the task does
24/// not need?
25pub const OUTPUT_SAFE: &str = "output_safe@1";
26/// Built-in predicate for the loop breaker's second opinion: is the agent
27/// still making progress, or repeating itself?
28pub const LOOP_PROGRESS: &str = "loop_progress@1";
29
30/// Maximum labels per predicate.
31pub const MAX_LABELS: usize = 16;
32
33/// A predicate definition.
34#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
35#[serde(deny_unknown_fields)]
36pub struct PredicateDef {
37    /// Versioned id, `name@N`.
38    pub id: String,
39    /// The fixed label set, in canonical order.
40    pub labels: Vec<String>,
41    /// The label that marks the risky outcome. Calibration measures how
42    /// well the model separates it from the others.
43    pub positive_label: String,
44    /// What the predicate asks, for operators and model builders.
45    pub description: String,
46}
47
48impl PredicateDef {
49    /// Validate the id, labels, and positive label.
50    pub fn validate(&self) -> Result<(), String> {
51        validate_predicate_id(&self.id)?;
52        if self.labels.len() < 2 || self.labels.len() > MAX_LABELS {
53            return Err(format!(
54                "predicate {:?} must have between 2 and {MAX_LABELS} labels",
55                self.id
56            ));
57        }
58        for (index, label) in self.labels.iter().enumerate() {
59            if !is_identifier(label) {
60                return Err(format!(
61                    "predicate {:?} label {label:?} must be 1..=64 characters of [a-z0-9_]",
62                    self.id
63                ));
64            }
65            if self.labels.iter().take(index).any(|earlier| earlier == label) {
66                return Err(format!("predicate {:?} repeats label {label:?}", self.id));
67            }
68        }
69        if !self.labels.contains(&self.positive_label) {
70            return Err(format!(
71                "predicate {:?} positive_label {:?} is not one of its labels",
72                self.id, self.positive_label
73            ));
74        }
75        if self.description.trim().is_empty() {
76            return Err(format!("predicate {:?} needs a description", self.id));
77        }
78        Ok(())
79    }
80
81    /// True when `label` belongs to this predicate.
82    pub fn has_label(&self, label: &str) -> bool {
83        self.labels.iter().any(|l| l == label)
84    }
85}
86
87/// Check the `name@N` form: name of `[a-z0-9_]`, N a positive integer.
88pub fn validate_predicate_id(id: &str) -> Result<(), String> {
89    let Some((name, version)) = id.split_once('@') else {
90        return Err(format!("predicate id {id:?} must have the form name@version"));
91    };
92    if !is_identifier(name) {
93        return Err(format!(
94            "predicate id {id:?}: name must be 1..=64 characters of [a-z0-9_]"
95        ));
96    }
97    let valid_version = !version.is_empty()
98        && version.len() <= 9
99        && version.bytes().all(|b| b.is_ascii_digit())
100        && !version.starts_with('0');
101    if !valid_version {
102        return Err(format!(
103            "predicate id {id:?}: version must be a positive integer without leading zeros"
104        ));
105    }
106    Ok(())
107}
108
109fn is_identifier(value: &str) -> bool {
110    !value.is_empty()
111        && value.len() <= 64
112        && value
113            .bytes()
114            .all(|b| b.is_ascii_lowercase() || b.is_ascii_digit() || b == b'_')
115}

117/// The predicates a deployment knows: the built-ins plus operator-defined ones.
118#[derive(Debug, Clone, PartialEq, Eq)]
119pub struct PredicateRegistry {
120    defs: BTreeMap<String, PredicateDef>,
121}
122
123impl PredicateRegistry {
124    /// The built-in predicates only.
125    pub fn builtin() -> Self {
126        let defs = builtin_defs()
127            .into_iter()
128            .map(|def| (def.id.clone(), def))
129            .collect();
130        Self { defs }
131    }
132
133    /// Built-ins plus `custom`. A custom predicate may not reuse a
134    /// built-in id or another custom id.
135    pub fn with_custom(custom: &[PredicateDef]) -> Result<Self, String> {
136        let mut registry = Self::builtin();
137        for def in custom {
138            def.validate()?;
139            if registry.defs.contains_key(&def.id) {
140                return Err(format!("predicate {:?} is defined more than once", def.id));
141            }
142            registry.defs.insert(def.id.clone(), def.clone());
143        }
144        Ok(registry)
145    }
146
147    /// Look up a predicate.
148    pub fn get(&self, id: &str) -> Option<&PredicateDef> {
149        self.defs.get(id)
150    }
151
152    /// Every predicate, ordered by id.
153    pub fn iter(&self) -> impl Iterator<Item = &PredicateDef> {
154        self.defs.values()
155    }
156}

203#[cfg(test)]
204mod tests {
205    #![allow(clippy::unwrap_used)]
206
207    use super::*;
208
209    fn def(id: &str, labels: &[&str], positive: &str) -> PredicateDef {
210        PredicateDef {
211            id: id.to_owned(),
212            labels: labels.iter().map(|l| (*l).to_owned()).collect(),
213            positive_label: positive.to_owned(),
214            description: "test".to_owned(),
215        }
216    }
217
218    #[test]
219    fn builtins_are_valid() {
220        let registry = PredicateRegistry::builtin();
221        for def in registry.iter() {
222            def.validate().unwrap();
223        }
224        assert!(registry.get(ARG_MATCHES_TASK).is_some());
225        assert!(registry.get(CHILD_USES_PARENT_ENTITLEMENT).is_some());
226        assert!(registry.get(MEMORY_WRITE_SAFE).is_some());
227        assert!(registry.get(OUTPUT_SAFE).is_some());
228        assert!(registry.get(LOOP_PROGRESS).is_some());
229    }
230
231    #[test]
232    fn ids_need_a_positive_version() {
233        assert!(validate_predicate_id("refund_ok@1").is_ok());
234        assert!(validate_predicate_id("refund_ok@12").is_ok());
235        for bad in [
236            "refund_ok",
237            "refund_ok@",
238            "refund_ok@0",
239            "refund_ok@01",
240            "Refund@1",
241            "a b@1",
242            "@1",
243        ] {
244            assert!(validate_predicate_id(bad).is_err(), "{bad}");
245        }
246    }
247
248    #[test]
249    fn custom_predicates_are_validated_and_unique() {
250        let ok = def("refund_ok@1", &["within", "outside"], "outside");
251        let registry = PredicateRegistry::with_custom(std::slice::from_ref(&ok)).unwrap();
252        assert!(registry.get("refund_ok@1").is_some());
253        assert!(PredicateRegistry::with_custom(&[ok.clone(), ok]).is_err());
254        assert!(PredicateRegistry::with_custom(&[def(ARG_MATCHES_TASK, &["a", "b"], "a")]).is_err());
255        assert!(PredicateRegistry::with_custom(&[def("p@1", &["a"], "a")]).is_err());
256        assert!(PredicateRegistry::with_custom(&[def("p@1", &["a", "a"], "a")]).is_err());
257        assert!(PredicateRegistry::with_custom(&[def("p@1", &["a", "b"], "c")]).is_err());
258        assert!(PredicateRegistry::with_custom(&[def("p@1", &["a", "B"], "a")]).is_err());
259    }
