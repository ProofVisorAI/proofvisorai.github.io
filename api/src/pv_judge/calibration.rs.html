1//! Build-time calibration: score a labelled set and fail when the
2//! model's measured separation drops below a recorded floor.
3//!
4//! The labelled set is JSON Lines, one example per line:
5//!
6//! ```json
7//! {"predicate_id":"arg_matches_task@1","tool":"db_write","arguments":{"table":"orders"},
8//!  "context":{"charter":"billing-agent","delegation_depth":0,"task":"Reconcile March invoices"},
9//!  "label":"match","source":"shadow-2026-11"}
10//! ```
11//!
12//! Metrics per predicate, all integers (ppm):
13//! * accuracy of the most probable label;
14//! * AUROC of the positive label's probability (Mann–Whitney, ties
15//!   counted as one half);
16//! * separation: mean positive-label probability on positive examples
17//!   minus the same mean on negative examples;
18//! * accuracy by confidence band, and whether it rises with confidence.
19

28/// One labelled example.
29#[derive(Debug, Clone, PartialEq, Serialize, Deserialize)]
30#[serde(deny_unknown_fields)]
31pub struct LabelledExample {
32    /// Versioned predicate id.
33    pub predicate_id: String,
34    /// Tool name.
35    pub tool: String,
36    /// Validated arguments.
37    pub arguments: Value,
38    /// Context the judge sees.
39    #[serde(default)]
40    pub context: SessionContext,
41    /// The correct label.
42    pub label: String,
43    /// Where the example came from (shadow traces, a design-partner
44    /// call, a synthetic fixture).
45    #[serde(default, skip_serializing_if = "Option::is_none")]
46    pub source: Option<String>,
47}
48
49/// Parse a JSON Lines labelled set. Blank lines are skipped; errors
50/// name the line.
51pub fn load_jsonl(text: &str) -> Result<Vec<LabelledExample>, String> {
52    let mut examples = Vec::new();
53    for (index, line) in text.lines().enumerate() {
54        if line.trim().is_empty() {
55            continue;
56        }
57        let example: LabelledExample = serde_json::from_str(line)
58            .map_err(|error| format!("labelled set line {}: {error}", index + 1))?;
59        examples.push(example);
60    }
61    Ok(examples)
62}

64/// Measured quality for one predicate.
65#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
66pub struct PredicateReport {
67    /// Versioned predicate id.
68    pub predicate_id: String,
69    /// Examples for this predicate.
70    pub examples: u64,
71    /// Examples whose correct label is the positive label.
72    pub positives: u64,
73    /// Examples the judge could not score.
74    pub failures: u64,
75    /// Accuracy of the most probable label over scored examples, ppm.
76    pub accuracy_ppm: u32,
77    /// AUROC of the positive label's probability, ppm (500_000 = chance).
78    pub auroc_ppm: u32,
79    /// Mean positive-label ppm on positives minus on negatives.
80    pub separation_ppm: i64,
81    /// Accuracy by confidence band.
82    pub bands: Vec<BandStats>,
83}

85/// The whole run.
86#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
87pub struct CalibrationReport {
88    /// Adapter name.
89    pub adapter: String,
90    /// The pinned model that was measured.
91    pub model: crate::ModelPin,
92    /// One report per predicate in the set.
93    pub predicates: Vec<PredicateReport>,
94}

96/// The minimum a model must reach. Record the measured values (with the
97/// date and the set they were measured on) next to the floor, the way
98/// `pv-loopdetect/tests/calibrate.rs` does for the breaker.
99#[derive(Debug, Clone, Copy, PartialEq, Eq, Serialize, Deserialize)]
100#[serde(deny_unknown_fields)]
101pub struct CalibrationFloor {
102    /// Fewest examples per predicate.
103    pub min_examples: u64,
104    /// Lowest acceptable AUROC, ppm.
105    pub min_auroc_ppm: u32,
106    /// Lowest acceptable separation, ppm.
107    pub min_separation_ppm: i64,
108    /// Most failures tolerated, ppm of examples.
109    pub max_failure_ppm: u32,
110    /// Require accuracy to rise with confidence.
111    pub require_monotone: bool,
112    /// Reviews a band needs to count in the monotonicity check.
113    pub min_per_band: u64,
114    /// Allowed accuracy drop between bands, ppm.
115    pub tolerance_ppm: u32,
116}
117
118impl PredicateReport {
119    /// Every way this report falls short of `floor`.
120    pub fn shortfalls(&self, floor: &CalibrationFloor) -> Vec<String> {
121        let mut problems = Vec::new();
122        if self.examples < floor.min_examples {
123            problems.push(format!(
124                "{}: {} examples, floor {}",
125                self.predicate_id, self.examples, floor.min_examples
126            ));
127        }
128        if self.auroc_ppm < floor.min_auroc_ppm {
129            problems.push(format!(
130                "{}: AUROC {} ppm, floor {}",
131                self.predicate_id, self.auroc_ppm, floor.min_auroc_ppm
132            ));
133        }
134        if self.separation_ppm < floor.min_separation_ppm {
135            problems.push(format!(
136                "{}: separation {} ppm, floor {}",
137                self.predicate_id, self.separation_ppm, floor.min_separation_ppm
138            ));
139        }
140        let failure_ppm = ratio_ppm(self.failures, self.examples);
141        if failure_ppm > floor.max_failure_ppm {
142            problems.push(format!(
143                "{}: {} of {} examples failed to score",
144                self.predicate_id, self.failures, self.examples
145            ));
146        }
147        if floor.require_monotone && !is_monotone(&self.bands, floor.min_per_band, floor.tolerance_ppm) {
148            problems.push(format!(
149                "{}: accuracy does not rise with confidence",
150                self.predicate_id
151            ));
152        }
153        problems
154    }
155}

157impl CalibrationReport {
158    /// Every shortfall across predicates; empty when the floor holds.
159    pub fn shortfalls(&self, floor: &CalibrationFloor) -> Vec<String> {
160        self.predicates
161            .iter()
162            .flat_map(|report| report.shortfalls(floor))
163            .collect()
164    }
165}

167/// Score every example with `adapter` and measure.
168pub async fn calibrate(
169    runner: &JudgeRunner,
170    adapter: &str,
171    examples: &[LabelledExample],
172) -> Result<CalibrationReport, String> {
173    // An empty set measures nothing, so no floor could fail: refuse it
174    // rather than report a pass.
175    if examples.is_empty() {
176        return Err("the labelled set has no examples".to_owned());
177    }
178    let model = runner
179        .adapter(adapter)
180        .ok_or_else(|| format!("adapter {adapter:?} is not configured"))?
181        .adapter
182        .id()
183        .clone();
184    let mut grouped: BTreeMap<String, Vec<(&LabelledExample, JudgeOutcome)>> = BTreeMap::new();
185    for example in examples {
186        let predicate = runner
187            .registry()
188            .get(&example.predicate_id)
189            .ok_or_else(|| format!("labelled set uses unknown predicate {:?}", example.predicate_id))?;
190        if !predicate.has_label(&example.label) {
191            return Err(format!(
192                "labelled set uses label {:?}, which {} does not have",
193                example.label, example.predicate_id
194            ));
195        }
196        let task = JudgeTask {
197            predicate_id: example.predicate_id.clone(),
198            adapter: adapter.to_owned(),
199        };
200        let outcome = runner
201            .evaluate(
202                std::slice::from_ref(&task),
203                &example.tool,
204                &example.arguments,
205                &example.context,
206            )
207            .await
208            .into_iter()
209            .next()
210            .ok_or_else(|| "runner returned no outcome".to_owned())?;
211        grouped
212            .entry(example.predicate_id.clone())
213            .or_default()
214            .push((example, outcome));
215    }
216    let mut predicates = Vec::new();
217    for (predicate_id, results) in grouped {
218        let positive = runner
219            .registry()
220            .get(&predicate_id)
221            .map(|def| def.positive_label.clone())
222            .unwrap_or_default();
223        predicates.push(measure(&predicate_id, &positive, &results));
224    }
225    Ok(CalibrationReport {
226        adapter: adapter.to_owned(),
227        model,
228        predicates,
229    })
230}

232fn measure(
233    predicate_id: &str,
234    positive: &str,
235    results: &[(&LabelledExample, JudgeOutcome)],
236) -> PredicateReport {
237    let mut bands: Vec<BandStats> = BAND_LOWER_PPM
238        .iter()
239        .map(|lower| BandStats {
240            lower_ppm: *lower,
241            ..BandStats::default()
242        })
243        .collect();
244    let mut failures = 0u64;
245    let mut correct = 0u64;
246    let mut scored: Vec<(u32, bool)> = Vec::new();
247    let mut positives = 0u64;
248    for (example, outcome) in results {
249        let is_positive = example.label == positive;
250        if is_positive {
251            positives += 1;
252        }
253        let JudgeOutcome::Scored(verdict) = outcome else {
254            failures += 1;
255            continue;
256        };
257        let right = top_label(verdict) == Some(example.label.as_str());
258        if right {
259            correct += 1;
260        }
261        if let Some(top) = verdict.top() {
262            if let Some(band) = bands.get_mut(band_of(top.ppm)) {
263                band.reviewed += 1;
264                if right {
265                    band.correct += 1;
266                }
267            }
268        }
269        scored.push((verdict.ppm(positive).unwrap_or(0), is_positive));
270    }
271    let examples = u64::try_from(results.len()).unwrap_or(u64::MAX);
272    PredicateReport {
273        predicate_id: predicate_id.to_owned(),
274        examples,
275        positives,
276        failures,
277        accuracy_ppm: ratio_ppm(correct, u64::try_from(scored.len()).unwrap_or(u64::MAX)),
278        auroc_ppm: auroc_ppm(&scored),
279        separation_ppm: separation_ppm(&scored),
280        bands,
281    }
282}

284fn top_label(verdict: &Verdict) -> Option<&str> {
285    verdict.top().map(|top| top.label.as_str())
286}

288fn ratio_ppm(part: u64, whole: u64) -> u32 {
289    if whole == 0 {
290        return 0;
291    }
292    let ppm = u128::from(part) * u128::from(crate::PPM_ONE) / u128::from(whole);
293    u32::try_from(ppm).unwrap_or(crate::PPM_ONE)
294}

296/// Mann–Whitney AUROC over `(score, is_positive)`; 500_000 when either
297/// class is empty.
298pub fn auroc_ppm(scored: &[(u32, bool)]) -> u32 {
299    let positives = scored.iter().filter(|(_, p)| *p).count();
300    let negatives = scored.len() - positives;
301    if positives == 0 || negatives == 0 {
302        return crate::PPM_ONE / 2;
303    }
304    let mut sorted: Vec<(u32, bool)> = scored.to_vec();
305    sorted.sort_by_key(|(score, _)| *score);
306    // Sum of (doubled) ranks of positives, with ties at their average rank.
307    let mut rank_sum_doubled: u128 = 0;
308    let mut start = 0usize;
309    while start < sorted.len() {
310        let score = sorted.get(start).map_or(0, |(s, _)| *s);
311        let mut end = start;
312        while sorted.get(end).is_some_and(|(s, _)| *s == score) {
313            end += 1;
314        }
315        // Ranks start + 1 ..= end; doubled average = start + 1 + end.
316        let doubled_average = (start + 1 + end) as u128;
317        let tied_positives = sorted
318            .get(start..end)
319            .map_or(0, |slice| slice.iter().filter(|(_, p)| *p).count());
320        rank_sum_doubled += doubled_average * tied_positives as u128;
321        start = end;
322    }
323    let p = positives as u128;
324    let n = negatives as u128;
325    // U = R - P(P+1)/2 ; AUROC = U / (P N). All doubled to stay integral.
326    let u_doubled = rank_sum_doubled.saturating_sub(p * (p + 1));
327    let ppm = u_doubled * u128::from(crate::PPM_ONE) / (2 * p * n);
328    u32::try_from(ppm).unwrap_or(crate::PPM_ONE)
329}

331fn separation_ppm(scored: &[(u32, bool)]) -> i64 {
332    let mean = |positive: bool| -> Option<i64> {
333        let values: Vec<i64> = scored
334            .iter()
335            .filter(|(_, p)| *p == positive)
336            .map(|(s, _)| i64::from(*s))
337            .collect();
338        let count = i64::try_from(values.len()).ok().filter(|count| *count > 0)?;
339        Some(values.iter().sum::<i64>() / count)
340    };
341    match (mean(true), mean(false)) {
342        (Some(pos), Some(neg)) => pos - neg,
343        _ => 0,
344    }
345}

353    #[test]
354    fn auroc_matches_hand_computed_values() {
355        assert_eq!(auroc_ppm(&[(900_000, true), (100_000, false)]), 1_000_000);
356        assert_eq!(auroc_ppm(&[(100_000, true), (900_000, false)]), 0);
357        assert_eq!(auroc_ppm(&[(500_000, true), (500_000, false)]), 500_000);
358        // P = {0.8, 0.4}, N = {0.6, 0.2}: pairs won 3 of 4.
359        assert_eq!(
360            auroc_ppm(&[
361                (800_000, true),
362                (400_000, true),
363                (600_000, false),
364                (200_000, false)
365            ]),
366            750_000
367        );
368        assert_eq!(auroc_ppm(&[(800_000, true)]), 500_000);
369    }
370
371    #[test]
372    fn separation_is_the_difference_of_means() {
373        assert_eq!(
374            separation_ppm(&[(900_000, true), (700_000, true), (100_000, false)]),
375            700_000
376        );
377        assert_eq!(separation_ppm(&[(900_000, true)]), 0);
378    }
