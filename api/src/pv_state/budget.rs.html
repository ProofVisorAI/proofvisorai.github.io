1//! Action budgets (Module B): stateful tool limits — per-tool caps via
2//! `max_tool_calls` (e.g. `db_write: 3`), payout ceilings via
3//! `max_payout_usd_micros` — plus per-session token ceilings.
4//!
5//! Money is tracked in integer micro-USD; a payout of $12.34 spends
6//! 12_340_000. Fractional-cent dust can therefore never accumulate invisibly.
7
8use crate::store::{Refund, Spend, StateError, StateStore, TrySpendOutcome};
9use serde::{Deserialize, Serialize};
10use std::collections::BTreeMap;
11
12/// Declarative budget for one session/agent (config-file surface).
13///
14/// Unknown keys are rejected so `[budget]` typos fail loudly at startup.
15#[derive(Debug, Clone, Default, Serialize, Deserialize, PartialEq, Eq)]
16#[serde(deny_unknown_fields)]
17pub struct BudgetSpec {
18    /// Max total tokens (prompt+completion) per session. `None` = unlimited.
19    #[serde(default, skip_serializing_if = "Option::is_none")]
20    pub max_tokens: Option<u64>,
21    /// Max cumulative payout in micro-USD.
22    #[serde(default, skip_serializing_if = "Option::is_none")]
23    pub max_payout_usd_micros: Option<u64>,
24    /// Per-tool invocation caps, e.g. `db_write: 3`.
25    #[serde(default, skip_serializing_if = "BTreeMap::is_empty")]
26    pub max_tool_calls: BTreeMap<String, u64>,
27    /// Cap on *all* tool calls combined.
28    #[serde(default, skip_serializing_if = "Option::is_none")]
29    pub max_total_tool_calls: Option<u64>,
30}
31
32impl BudgetSpec {
33    /// True when at least one dimension is configured — i.e. this spec
34    /// actually constrains something. A default/omitted `[budget]`
35    /// block parses to all-`None`, which enforces NOTHING: every
36    /// session runs with unlimited tokens, payout, and tool calls.
37    /// Diagnostics (`pvctl doctor`) use this to distinguish "budget
38    /// configured" from the silent-unbounded footgun.
39    pub fn binds_anything(&self) -> bool {
40        self.max_tokens.is_some()
41            || self.max_payout_usd_micros.is_some()
42            || self.max_total_tool_calls.is_some()
43            || !self.max_tool_calls.is_empty()
44    }
45
46    /// The stricter of two specs on every dimension: a limit set in
47    /// either spec binds, and where both set one the smaller wins.
48    /// Console-pushed policy is layered on the local config with this,
49    /// so a pushed bundle can only ever tighten what the operator
50    /// configured on the host, never loosen it.
51    pub fn tightened_by(&self, other: &BudgetSpec) -> BudgetSpec {
52        fn min_opt(a: Option<u64>, b: Option<u64>) -> Option<u64> {
53            match (a, b) {
54                (Some(a), Some(b)) => Some(a.min(b)),
55                (a, None) => a,
56                (None, b) => b,
57            }
58        }
59        let mut max_tool_calls = self.max_tool_calls.clone();
60        for (tool, cap) in &other.max_tool_calls {
61            max_tool_calls
62                .entry(tool.clone())
63                .and_modify(|existing| *existing = (*existing).min(*cap))
64                .or_insert(*cap);
65        }
66        BudgetSpec {
67            max_tokens: min_opt(self.max_tokens, other.max_tokens),
68            max_payout_usd_micros: min_opt(self.max_payout_usd_micros, other.max_payout_usd_micros),
69            max_tool_calls,
70            max_total_tool_calls: min_opt(self.max_total_tool_calls, other.max_total_tool_calls),
71        }
72    }
73}

75/// Outcome of a budget check.
76#[derive(Debug, Clone, PartialEq, Eq, Serialize, Deserialize)]
77pub enum BudgetDecision {
78    /// Spend recorded; remaining headroom for the tightest matching limit.
79    Allowed {
80        /// Remaining amount under the binding limit (min across dimensions).
81        remaining: u64,
82    },
83    /// Refused: which limit would be exceeded.
84    Refused {
85        /// Human-readable limit name (`max_tool_calls.db_write`, …).
86        limit: String,
87        /// The configured cap.
88        cap: u64,
89    },
90}
91
92impl BudgetDecision {
93    /// True when the action was allowed.
94    pub fn is_allowed(&self) -> bool {
95        matches!(self, Self::Allowed { .. })
96    }
97}

99/// Budget enforcement bound to an accounting scope (a session id, or a
100/// stable principal id) and a state store.
101///
102/// Historically this type was hard-coded to a session-scoped view, but a
103/// session-only budget can be
104/// bypassed by any caller that rotates the caller-chosen `X-PV-Session`
105/// header — every request would land on a virgin counter. Callers that
106/// want to catch that vector layer a principal-scoped `ActionBudget` on
107/// top of the session-scoped one; the anti-rotation property lives in the
108/// key prefix, not in the check logic.
109pub struct ActionBudget<'a> {
110    store: &'a dyn StateStore,
111    scope: Scope<'a>,
112    spec: &'a BudgetSpec,
113}
114
115/// Which accounting bucket a budget check writes to.
116///
117/// Two identical `try_spend_many` calls made under different scopes hit
118/// entirely disjoint keys in the state store; a session budget cannot
119/// exhaust a principal budget or vice versa.
120#[derive(Debug, Clone, Copy)]
121enum Scope<'a> {
122    Session(&'a str),
123    Principal(&'a str),
124}
125
126impl<'a> ActionBudget<'a> {
127    /// Bind a spec to a session id.
128    pub fn new(store: &'a dyn StateStore, session: &'a str, spec: &'a BudgetSpec) -> Self {
129        Self {
130            store,
131            scope: Scope::Session(session),
132            spec,
133        }
134    }
135
136    /// Bind a spec to a principal id (JWT `sub`/`instance_uid`, HMAC-KID,
137    /// or the sentinel `"anonymous"` when `require_identity = false`).
138    /// The resulting counters aggregate across every session belonging to
139    /// the same principal, which is what makes a header-rotation attack
140    /// visible — the token debit persists even after the client mints a
141    /// fresh `X-PV-Session`.
142    pub fn for_principal(store: &'a dyn StateStore, principal: &'a str, spec: &'a BudgetSpec) -> Self {
143        Self {
144            store,
145            scope: Scope::Principal(principal),
146            spec,
147        }
148    }
149
150    fn key(&self, dim: &str) -> String {
151        let prefix = match self.scope {
152            Scope::Session(id) => Self::session_prefix(id),
153            Scope::Principal(id) => Self::principal_prefix(id),
154        };
155        format!("{prefix}{dim}")
156    }
157
158    /// Common key prefix for every budget counter of `session`. Callers use
159    /// this with [`StateStore::remove_prefix`] to drop a finalized session's
160    /// counters (per-tool keys are dynamic, so single-key removal cannot
161    /// enumerate them).
162    pub fn session_prefix(session: &str) -> String {
163        let digest = pv_core::digest::sha256_hex(session.as_bytes());
164        // 32 hex chars = 128 bits: collision-safe well beyond realistic session counts.
165        format!("budget:{{{}}}:", digest.get(..32).unwrap_or(&digest))
166    }
167
168    /// Common key prefix for every principal-scoped budget counter. Distinct
169    /// namespace from [`Self::session_prefix`] so a session cleanup on close does
170    /// not touch the principal ledger — the principal ledger persists across
171    /// session boundaries by design.
172    pub fn principal_prefix(principal: &str) -> String {
173        let digest = pv_core::digest::sha256_hex(principal.as_bytes());
174        format!("budget:principal:{{{}}}:", digest.get(..32).unwrap_or(&digest))
175    }
176
177    /// Check-and-spend one invocation of `tool`, with an optional payout
178    /// amount in micro-USD carried by this call.
179    ///
180    /// A non-zero payout with NO configured `max_payout_usd_micros`
181    /// refuses first (fail-closed: an unbounded payout dimension must
182    /// never spend), citing `max_payout_usd_micros(unset)` — even when
183    /// another dimension is also at cap. The remaining dimensions are
184    /// then checked in a fixed order (total calls → per-tool → payout)
185    /// and committed atomically via `try_spend_many`: every dimension
186    /// is validated first and either all spends commit or none do, so
187    /// a refused call consumes nothing.
188    pub fn try_tool_call(&self, tool: &str, payout_usd_micros: u64) -> Result<BudgetDecision, StateError> {
189        // Parallel arrays keep spends and limit-names together without paying
190        // for a joined-tuple clone before hitting the state store. There are
191        // at most three dimensions (total, per-tool, payout).
192        let mut spends: Vec<Spend> = Vec::with_capacity(3);
193        let mut limit_names: Vec<String> = Vec::with_capacity(3);
194
195        if let Some(cap) = self.spec.max_total_tool_calls {
196            spends.push(Spend {
197                key: self.key("total_calls"),
198                amount: 1,
199                limit: cap,
200            });
201            limit_names.push("max_total_tool_calls".into());
202        }
203        if let Some(cap) = self.spec.max_tool_calls.get(tool).copied() {
204            spends.push(Spend {
205                key: self.key(&format!("tool:{tool}")),
206                amount: 1,
207                limit: cap,
208            });
209            limit_names.push(format!("max_tool_calls.{tool}"));
210        }
211        if payout_usd_micros > 0 {
212            match self.spec.max_payout_usd_micros {
213                Some(cap) => {
214                    spends.push(Spend {
215                        key: self.key("payout"),
216                        amount: payout_usd_micros,
217                        limit: cap,
218                    });
219                    limit_names.push("max_payout_usd_micros".into());
220                }
221                None => {
222                    return Ok(BudgetDecision::Refused {
223                        limit: "max_payout_usd_micros(unset)".into(),
224                        cap: 0,
225                    });
226                }
227            }
228        }
229
230        // R66 F3: previously called `try_spend_many` for the atomic
231        // commit and then `remaining_min` for the reported headroom
232        // — three additional GET round-trips that could race with a
233        // concurrent `remove_prefix` DEL and inflate the reported
234        // `remaining` to the full cap (a debit that WAS committed).
235        // Now the outcome carries the post-commit min-headroom
236        // computed inside the SAME atomic section that committed the
237        // spend, so `BudgetDecision::Allowed { remaining }` can be
238        // taken directly without a follow-up read.
239        match self.store.try_spend_many(&spends)? {
240            TrySpendOutcome::Refused { index } => {
241                let spend = spends
242                    .get(index)
243                    .ok_or_else(|| StateError::Backend(format!("invalid failed spend index {index}")))?;
244                let limit = limit_names
245                    .get(index)
246                    .ok_or_else(|| StateError::Backend(format!("invalid failed spend index {index}")))?;
247                Ok(BudgetDecision::Refused {
248                    limit: limit.clone(),
249                    cap: spend.limit,
250                })
251            }
252            TrySpendOutcome::Committed {
253                post_commit_min_remaining,
254            } => Ok(BudgetDecision::Allowed {
255                remaining: post_commit_min_remaining,
256            }),
257        }
258    }
259
260    /// Compensating refund for a previously-successful
261    /// [`Self::try_tool_call`]. Reverses the spend on exactly the same
262    /// dimensions that were debited (total_calls, per-tool, payout) so
263    /// a lost-race path in the caller (concurrent identical MCP
264    /// request loses `execution.claim()` after the sandbox gate has
265    /// already spent) does not double-charge the session budget.
266    /// Best-effort: any backend error is silently absorbed by the
267    /// underlying [`StateStore::refund`] contract — a Redis blip on
268    /// the compensation path must never turn a lost-race response
269    /// into a 5xx.
270    pub fn refund_tool_call(&self, tool: &str, payout_usd_micros: u64) {
271        // R66 F2: single atomic refund_many call so all three
272        // dimensions (total_calls / tool:{X} / payout) are refunded
273        // as one transaction under InMemoryStore's `transaction_lock`
274        // and one Lua EVAL under RedisStore. Prior shape was 3
275        // independent `store.refund` calls — a crash / connection
276        // drop between steps left partial refund state, and a
277        // concurrent `try_tool_call` on InMemoryStore could observe
278        // the mid-batch state (fresh `total_calls` headroom but
279        // still-charged per-tool cap), an ordering impossible on
280        // RedisStore's single-Lua path.
281        let mut refunds: Vec<Refund> = Vec::with_capacity(3);
282        if self.spec.max_total_tool_calls.is_some() {
283            refunds.push(Refund {
284                key: self.key("total_calls"),
285                amount: 1,
286            });
287        }
288        if self.spec.max_tool_calls.contains_key(tool) {
289            refunds.push(Refund {
290                key: self.key(&format!("tool:{tool}")),
291                amount: 1,
292            });
293        }
294        if payout_usd_micros > 0 && self.spec.max_payout_usd_micros.is_some() {
295            refunds.push(Refund {
296                key: self.key("payout"),
297                amount: payout_usd_micros,
298            });
299        }
300        if !refunds.is_empty() {
301            self.store.refund_many(&refunds);
302        }
303    }
304
305    /// Compensating refund for a previously-successful [`Self::try_tokens`]
306    /// on a request that provably never reached the provider (sanitize
307    /// block, capture failure, in-flight-marker write failure, connect
308    /// failure). Same rationale and best-effort contract as
309    /// [`Self::refund_tool_call`]: without it, each client retry against
310    /// a flaky-but-unreachable upstream re-debits `max_tokens` with zero
311    /// LLM work performed, draining the session budget.
312    pub fn refund_tokens(&self, tokens: u64) {
313        if tokens > 0 && self.spec.max_tokens.is_some() {
314            self.store.refund(&self.key("tokens"), tokens);
315        }
316    }
317
318    /// Check-and-spend `tokens` against `max_tokens`.
319    pub fn try_tokens(&self, tokens: u64) -> Result<BudgetDecision, StateError> {
320        match self.spec.max_tokens {
321            Some(cap) => {
322                // R66 F3: route through `try_spend_many` so the
323                // reported `remaining` comes from the same atomic
324                // section that committed the spend. The previous
325                // shape did `try_spend` + `get` — a 2-RTT window
326                // in which a concurrent `remove_prefix` could DEL
327                // the counter and make `get` return 0, inflating
328                // `remaining` to `cap`. See `TrySpendOutcome`.
329                match self.store.try_spend_many(&[Spend {
330                    key: self.key("tokens"),
331                    amount: tokens,
332                    limit: cap,
333                }])? {
334                    TrySpendOutcome::Committed {
335                        post_commit_min_remaining,
336                    } => Ok(BudgetDecision::Allowed {
337                        remaining: post_commit_min_remaining,
338                    }),
339                    TrySpendOutcome::Refused { .. } => Ok(BudgetDecision::Refused {
340                        limit: "max_tokens".into(),
341                        cap,
342                    }),
343                }
344            }
345            None => Ok(BudgetDecision::Allowed { remaining: u64::MAX }),
346        }
347    }
348}

366    /// Mutation-run hardening: nothing previously pinned the ledger
367    /// prefix shapes, so a mutant replacing `principal_prefix` with an
368    /// arbitrary string survived the suite. The shapes are load-bearing
369    /// twice over: the `{hash-tag}` keeps every key of one ledger in a
370    /// single Redis Cluster slot (remove_prefix routes SCAN/DEL to the
371    /// slot derived from the prefix), and the distinct `budget:` vs
372    /// `budget:principal:` namespaces are what keep a session cleanup on
373    /// close from touching the principal ledger.
374    #[test]
375    fn ledger_prefixes_are_namespaced_and_cluster_hash_tagged() {
376        let session = ActionBudget::session_prefix("same-id");
377        let principal = ActionBudget::principal_prefix("same-id");
378        assert!(session.starts_with("budget:{"), "{session}");
379        assert!(principal.starts_with("budget:principal:{"), "{principal}");
380        for prefix in [&session, &principal] {
381            let open = prefix.find('{').unwrap();
382            let close = prefix.find('}').unwrap();
383            assert!(close > open + 1, "empty hash tag in {prefix}");
384            assert!(prefix.ends_with("}:"), "{prefix}");
385        }
386        // The SAME id in both scopes must land in disjoint key spaces —
387        // neither prefix may be a prefix of the other, or remove_prefix
388        // on one ledger would sweep the other's counters.
389        assert!(!session.starts_with(&principal) && !principal.starts_with(&session));
390    }

564        // The same call now succeeds — proving all three dimensions
565        // were compensated.
566        assert!(b.try_tool_call("db_write", 500_000).unwrap().is_allowed());
567    }

596    /// Refund saturates at 0 under a concurrent clear.
597    /// The compensating refund must never leave a negative "budget
598    /// spent" counter — that would give the next legit call a free
599    /// ride relative to its cap.
600    #[test]
601    fn refund_is_saturating() {
602        let store = InMemoryStore::new();
603        let s = BudgetSpec {
604            max_total_tool_calls: Some(10),
605            ..BudgetSpec::default()
606        };
607        let b = ActionBudget::new(&store, "sess-sat", &s);
608        assert!(b.try_tool_call("t", 0).unwrap().is_allowed());
609        // Two refunds in a row: the second must clamp at 0, not
610        // underflow the counter.
611        b.refund_tool_call("t", 0);
612        b.refund_tool_call("t", 0);
613        // 10 successful calls remain — the counter is 0 (clamped).
614        for _ in 0..10 {
615            assert!(b.try_tool_call("t", 0).unwrap().is_allowed());
616        }
617        assert!(matches!(
618            b.try_tool_call("t", 0).unwrap(),
619            BudgetDecision::Refused { .. }
620        ));
621    }

646    #[test]
647    fn tightened_by_keeps_the_stricter_limit_on_every_dimension() {
648        let mut local = BudgetSpec {
649            max_tokens: Some(1_000),
650            max_payout_usd_micros: None,
651            max_total_tool_calls: Some(10),
652            ..BudgetSpec::default()
653        };
654        local.max_tool_calls.insert("db_write".to_owned(), 5);
655        local.max_tool_calls.insert("search".to_owned(), 50);
656        let mut pushed = BudgetSpec {
657            max_tokens: Some(5_000),
658            max_payout_usd_micros: Some(2_000_000),
659            max_total_tool_calls: Some(3),
660            ..BudgetSpec::default()
661        };
662        pushed.max_tool_calls.insert("db_write".to_owned(), 2);
663        pushed.max_tool_calls.insert("create_payment".to_owned(), 1);
664        let merged = local.tightened_by(&pushed);
665        assert_eq!(
666            merged.max_tokens,
667            Some(1_000),
668            "a looser pushed cap never loosens the local one"
669        );
670        assert_eq!(
671            merged.max_payout_usd_micros,
672            Some(2_000_000),
673            "a pushed cap binds where local had none"
674        );
675        assert_eq!(merged.max_total_tool_calls, Some(3));
676        assert_eq!(merged.max_tool_calls.get("db_write"), Some(&2));
677        assert_eq!(merged.max_tool_calls.get("search"), Some(&50));
678        assert_eq!(merged.max_tool_calls.get("create_payment"), Some(&1));
679        assert_eq!(
680            local.tightened_by(&BudgetSpec::default()),
681            local,
682            "an empty overlay changes nothing"
683        );
684    }
