1//! Redis-backed `StateStore` — the distributed budget/state layer for
2//! multi-replica deployments.
3//!
4//! Check-and-spend runs as a server-side Lua script so it is atomic across
5//! distributed clients — same contract as `InMemoryStore::try_spend`.
6//! Contract tests live in `tests/redis_contract.rs`, gated on `PV_REDIS_URL`
7//! (skipped loudly when unset — a skipped gate prints, never silently passes).
8
9use crate::store::{Spend, StateError, StateStore};
10use redis::Commands;

12/// TTL (seconds) applied to every counter key this store touches (the
13/// spend/add INCRBY paths and the refund clamp). 24 h: counters are
14/// session-scoped and the TTL is the backstop that keeps abandoned
15/// sessions from leaking Redis memory forever. NOTE the prod/dev
16/// divergence this creates: `InMemoryStore` never expires, so a session
17/// idle longer than this window has its budget counters silently reset
18/// against Redis only. Callers must keep session lifetimes within this
19/// window (or persist budgets elsewhere) — see the `StateStore` trait
20/// docs on counter lifetime.
21const BUDGET_COUNTER_TTL_SECS: u64 = 86_400;
22/// One-shot refund replay markers (see `REFUND_LUA`) only need to
23/// outlive the cluster driver's transparent retry window (seconds);
24/// 5 minutes is generous without accumulating a day of marker keys.
25const REFUND_MARKER_TTL_SECS: u64 = 300;
26
27/// Atomic check-and-spend using subtraction so `current + amount` never rounds.
28static TRY_SPEND_LUA: std::sync::LazyLock<String> = std::sync::LazyLock::new(|| {
29    format!(
30        r"
31for i, key in ipairs(KEYS) do
32    local current = tonumber(redis.call('GET', key) or '0')
33    local amount = tonumber(ARGV[(i - 1) * 2 + 1])
34    local limit = tonumber(ARGV[(i - 1) * 2 + 2])
35    if current > limit or amount > limit - current then
36        -- Refresh TTL on every EXISTING key
37        -- in this invocation before returning the refusal. Without
38        -- this, a session that hit its cap and keeps retrying (every
39        -- attempt refused, commit loop never reached) has its counter
40        -- expire {BUDGET_COUNTER_TTL_SECS}s after the last SUCCESSFUL
41        -- spend — the next attempt then sees current=0 and the cap
42        -- silently RESETS mid-session, doubling the budget. The TTL's
43        -- purpose is reclaiming ABANDONED sessions; an actively
44        -- retrying session is not abandoned.
45        for j, k in ipairs(KEYS) do
46            if redis.call('EXISTS', k) == 1 then
47                redis.call('EXPIRE', k, {BUDGET_COUNTER_TTL_SECS})
48            end
49        end
50        -- Refusal shape: {{-1, refusal_index_1_based}}.
51        -- The leading -1 discriminates from the commit shape
52        -- ({{post_commit_min_remaining, 0}}). See TrySpendOutcome
53        -- in store.rs for rationale.
54        return {{-1, i}}
55    end
56end
57-- R66 F3: compute post-commit min-headroom inside the SAME atomic
58-- EVAL that commits the spends, so `BudgetDecision::Allowed
59-- {{ remaining }}` cannot race with a concurrent
60-- remove_prefix/spend/refund on the same keys. Every key shares
61-- the caller's session hash tag, so the whole script runs on ONE
62-- server-side slot.
63local min_remaining = -1
64for i, key in ipairs(KEYS) do
65    local amount = tonumber(ARGV[(i - 1) * 2 + 1])
66    local limit = tonumber(ARGV[(i - 1) * 2 + 2])
67    local new = redis.call('INCRBY', key, amount)
68    redis.call('EXPIRE', key, {BUDGET_COUNTER_TTL_SECS})
69    local remaining = limit - new
70    if remaining < 0 then
71        remaining = 0
72    end
73    if min_remaining == -1 or remaining < min_remaining then
74        min_remaining = remaining
75    end
76end
77if min_remaining == -1 then
78    -- Empty KEYS: sentinel meaning `u64::MAX` on the client side.
79    min_remaining = -1
80end
81-- Commit shape: {{min_remaining, 0}}. See TRY_SPEND_LUA_REFUSED_SENTINEL
82-- in redis_store.rs.
83return {{min_remaining, 0}}
84"
85    )
86});

88static ADD_LUA: std::sync::LazyLock<String> = std::sync::LazyLock::new(|| {
89    format!(
90        r"
91local current = tonumber(redis.call('GET', KEYS[1]) or '0')
92local amount = tonumber(ARGV[1])
93local limit = tonumber(ARGV[2])
94if current > limit or amount > limit - current then
95    -- Keep an actively-refused counter
96    -- alive — see the twin comment in TRY_SPEND_LUA.
97    if redis.call('EXISTS', KEYS[1]) == 1 then
98        redis.call('EXPIRE', KEYS[1], {BUDGET_COUNTER_TTL_SECS})
99    end
100    return -1
101end
102local result = redis.call('INCRBY', KEYS[1], ARGV[1])
103-- Match TRY_SPEND_LUA's TTL. Without this, any counter touched
104-- only through `add()` (bookkeeping, telemetry, non-budget spending)
105-- persists forever in Redis; over a long-running deployment that
106-- silently leaks memory until Redis OOMs. The two APIs must be
107-- interchangeable from the persistence perspective.
108redis.call('EXPIRE', KEYS[1], {BUDGET_COUNTER_TTL_SECS})
109return result
110"
111    )
112});

114/// Push to the front of a list, keep its newest `ARGV[2]` entries, and
115/// expire it like the counters. A script rather than MULTI/EXEC: a
116/// Redis Cluster connection does not run transactions, and one key
117/// keeps the script in one slot.
118static PUSH_RECENT_LUA: std::sync::LazyLock<String> = std::sync::LazyLock::new(|| {
119    format!(
120        r"
121redis.call('LPUSH', KEYS[1], ARGV[1])
122redis.call('LTRIM', KEYS[1], 0, tonumber(ARGV[2]) - 1)
123redis.call('EXPIRE', KEYS[1], {BUDGET_COUNTER_TTL_SECS})
124return 1
125"
126    )
127});

129static FETCH_MAX_LUA: std::sync::LazyLock<String> = std::sync::LazyLock::new(|| {
130    format!(
131        r"
132local current = tonumber(redis.call('GET', KEYS[1]) or '0')
133local value = tonumber(ARGV[1])
134local limit = tonumber(ARGV[2])
135if not current or current < 0 or current > limit then return -1 end
136if value > current or redis.call('EXISTS', KEYS[1]) == 0 then
137    redis.call('SET', KEYS[1], ARGV[1])
138    current = value
139end
140redis.call('EXPIRE', KEYS[1], {BUDGET_COUNTER_TTL_SECS})
141return current
142"
143    )
144});

146/// Saturating single-key refund. `EXISTS`-gated to never resurrect
147/// a `remove_prefix`-cleared cell; `SET key 0 EX {BUDGET_COUNTER_TTL_SECS}`
148/// on underflow keeps the counter TTL-aligned; `EXPIRE` on the
149/// non-underflow path matches the same discipline. Extracted to a
150/// static (R69) so `every_counter_script_applies_the_shared_ttl`
151/// enrolls it in the TTL-drift regression test (R67 review L3).
152///
153/// KEYS[2] is a one-shot idempotency marker (same hash tag as the
154/// counter — a suffix on KEYS[1] — so cluster mode keeps the script
155/// single-slot). The cluster driver transparently RETRIES commands on
156/// connection loss (redis-rs `DEFAULT_RETRIES`), and a refund replayed
157/// after its first execution committed-but-lost-the-reply would credit
158/// the budget twice — refunding tokens that were only debited once,
159/// i.e. spendable budget minted out of a network flap. `SET NX` on the
160/// marker makes the replay a no-op. Spend-side scripts deliberately
161/// have no marker: their replay double-DEBITS, which fails
162/// conservative (the documented EVAL-uncertainty posture), never
163/// budget-inflating.
164static REFUND_LUA: std::sync::LazyLock<String> = std::sync::LazyLock::new(|| {
165    format!(
166        r"
167if redis.call('SET', KEYS[2], '1', 'NX', 'EX', {REFUND_MARKER_TTL_SECS}) == false then
168    return -1
169end
170if redis.call('EXISTS', KEYS[1]) == 0 then
171    return 0
172end
173local new = redis.call('DECRBY', KEYS[1], ARGV[1])
174if new < 0 then
175    redis.call('SET', KEYS[1], 0, 'EX', {BUDGET_COUNTER_TTL_SECS})
176    return 0
177end
178redis.call('EXPIRE', KEYS[1], {BUDGET_COUNTER_TTL_SECS})
179return new
180"
181    )
182});

184/// Atomic multi-key refund. Per-key semantics match `REFUND_LUA`
185/// exactly (EXISTS gate + DECRBY + clamp-at-zero via SET with
186/// fresh TTL / EXPIRE refresh on non-underflow). Extracted to a
187/// static (R69) so the same TTL-drift test enrolls it.
188///
189/// The trailing key (`KEYS[#KEYS]`) is the replay marker — see
190/// `REFUND_LUA`. The decrement passes `ARGV[i]` to DECRBY as the raw
191/// integer STRING: the previous `tonumber(ARGV[i])` converted through
192/// a Lua double, so an `i64::MAX`-clamped amount rounded to 2^63,
193/// DECRBY rejected it mid-loop, and earlier keys in the same batch
194/// stayed decremented while later ones were skipped — a partial
195/// commit the in-memory backend (which clamps every entry) never
196/// produces. Redis parses the string as an exact i64.
197static REFUND_MANY_LUA: std::sync::LazyLock<String> = std::sync::LazyLock::new(|| {
198    format!(
199        r"
200if redis.call('SET', KEYS[#KEYS], '1', 'NX', 'EX', {REFUND_MARKER_TTL_SECS}) == false then
201    return -1
202end
203for i = 1, #KEYS - 1 do
204    local key = KEYS[i]
205    if redis.call('EXISTS', key) == 1 then
206        local new = redis.call('DECRBY', key, ARGV[i])
207        if new < 0 then
208            redis.call('SET', key, 0, 'EX', {BUDGET_COUNTER_TTL_SECS})
209        else
210            redis.call('EXPIRE', key, {BUDGET_COUNTER_TTL_SECS})
211        end
212    end
213end
214return 0
215"
216    )
217});

219/// Redis-backed store. Connections are pooled internally (r2d2 for both
220/// single-node and cluster; the redis crate implements
221/// `r2d2::ManageConnection` for `ClusterClient` directly).
222///
223/// **EVAL uncertainty on network drop.** `try_spend_many` / `add` invoke
224/// atomic Lua scripts server-side; if the connection breaks between the
225/// server's INCRBY commit and the client's response read, the caller
226/// sees `StateError::Backend` and cannot distinguish "commit succeeded,
227/// response lost" from "commit never ran". Sandbox path treats
228/// backend-error as blocked (fails closed for the request) — but a
229/// subsequent client retry that lands with the same intent then hits a
230/// pre-debited counter and cumulates the two spends into one intended
231/// tool call, overcharging the budget by the retry's amount. The
232/// 24 h counter TTL bounds unbounded growth, and refund-on-loss-race
233/// (`refund_tool_call`) recovers the common cases, but the strictly
234/// idempotent shape would require a client-supplied request nonce
235/// stored briefly in Redis — deferred as a design change.
236pub struct RedisStore {
237    backend: RedisBackend,
238}
239
240enum RedisBackend {
241    Single(r2d2::Pool<RedisConnectionManager>),
242    Cluster(r2d2::Pool<redis::cluster::ClusterClient>),
243}
244
245struct RedisConnectionManager {
246    client: redis::Client,
247    timeout: std::time::Duration,
248}
249
250impl r2d2::ManageConnection for RedisConnectionManager {
251    type Connection = redis::Connection;
252    type Error = redis::RedisError;
253
254    fn connect(&self) -> Result<Self::Connection, Self::Error> {
255        let connection = self.client.get_connection_with_timeout(self.timeout)?;
256        connection.set_read_timeout(Some(self.timeout))?;
257        connection.set_write_timeout(Some(self.timeout))?;
258        Ok(connection)
259    }
260
261    fn is_valid(&self, connection: &mut Self::Connection) -> Result<(), Self::Error> {
262        redis::cmd("PING").query(connection)
263    }
264
265    fn has_broken(&self, _connection: &mut Self::Connection) -> bool {
266        false
267    }
268}

270/// Verify one connection during startup, then grow on demand. r2d2's
271/// default minimum equals the maximum; requiring 32 complete Cluster
272/// connections can exhaust the startup deadline during concurrent TLS setup.
273fn redis_pool_builder<M: r2d2::ManageConnection>() -> r2d2::Builder<M> {
274    r2d2::Pool::builder()
275        .max_size(32)
276        .min_idle(Some(1))
277        .connection_timeout(std::time::Duration::from_secs(2))
278}

280impl RedisStore {
281    /// Connect to `url` (e.g. `rediss://redis.example.com:6379`).
282    /// TLS verifies the server name and the native certificate trust store;
283    /// `SSL_CERT_FILE` can supply a PEM trust bundle for a private CA.
284    /// Comma-separated URLs select Redis Cluster mode. All seeds must use
285    /// the same transport, and disabling TLS verification is never allowed.
286    pub fn connect(url: &str) -> Result<Self, StateError> {
287        // Other connectors can enable a second rustls crypto backend through
288        // Cargo feature unification. Select one explicitly before redis-rs
289        // creates its ClientConfig; preserve a provider installed by the host.
290        let _ = rustls_tls::crypto::ring::default_provider().install_default();
291        let nodes: Vec<String> = url
292            .split(',')
293            .map(str::trim)
294            .filter(|node| !node.is_empty())
295            .map(str::to_owned)
296            .collect();
297        validate_transports(&nodes)?;
298        if nodes.len() > 1 {
299            // Previously one `ClusterConnection` behind a mutex —
300            // every quota/budget operation across all sessions serialized on
301            // a single socket, which under 10k-connection load stalled
302            // admission long enough to blow upstream timeouts (observed as
303            // 502 "upstream timed out" in the 10k SLA gate). Pool cluster
304            // connections exactly like the single-node path.
305            let client = redis::cluster::ClusterClientBuilder::new(nodes)
306                .connection_timeout(std::time::Duration::from_secs(2))
307                .response_timeout(std::time::Duration::from_secs(2))
308                .build()
309                .map_err(|e| StateError::Backend(e.to_string()))?;
310            let pool = redis_pool_builder()
311                .build(client)
312                .map_err(|e| StateError::Backend(e.to_string()))?;
313            return Ok(Self {
314                backend: RedisBackend::Cluster(pool),
315            });
316        }
317        let client = redis::Client::open(url).map_err(|e| StateError::Backend(e.to_string()))?;
318        let manager = RedisConnectionManager {
319            client,
320            timeout: std::time::Duration::from_secs(2),
321        };
322        let pool = redis_pool_builder()
323            .build(manager)
324            .map_err(|e| StateError::Backend(e.to_string()))?;
325        Ok(Self {
326            backend: RedisBackend::Single(pool),
327        })
328    }
329}

331fn validate_transports(nodes: &[String]) -> Result<(), StateError> {
332    use redis::IntoConnectionInfo as _;
333    let mut previous_tls = None;
334    for node in nodes {
335        let info = node
336            .as_str()
337            .into_connection_info()
338            .map_err(|error| StateError::Backend(error.to_string()))?;
339        let tls = match info.addr {
340            redis::ConnectionAddr::TcpTls { insecure: true, .. } => {
341                return Err(StateError::Backend(
342                    "Redis TLS certificate verification cannot be disabled".into(),
343                ));
344            }
345            redis::ConnectionAddr::TcpTls { .. } => true,
346            _ => false,
347        };
348        if previous_tls.is_some_and(|previous| previous != tls) {
349            return Err(StateError::Backend(
350                "Redis Cluster seeds must all use the same transport".into(),
351            ));
352        }
353        previous_tls = Some(tls);
354    }
355    Ok(())
356}

358fn add_on<C: redis::ConnectionLike>(conn: &mut C, key: &str, delta: u64) -> Result<u64, StateError> {
359    if delta > pv_core::error::JCS_SAFE_MAX {
360        return Err(StateError::Overflow(key.to_owned()));
361    }
362    let value: i64 = redis::Script::new(&ADD_LUA)
363        .key(key)
364        .arg(delta)
365        .arg(pv_core::error::JCS_SAFE_MAX)
366        .invoke(conn)
367        .map_err(|e| StateError::Backend(e.to_string()))?;
368    if value < 0 {
369        return Err(StateError::Overflow(key.to_owned()));
370    }
371    u64::try_from(value).map_err(|_| StateError::Overflow(key.to_owned()))
372}
373
374fn get_on<C: redis::ConnectionLike>(conn: &mut C, key: &str) -> Result<u64, StateError> {
375    let value: Option<i64> = conn.get(key).map_err(|e| StateError::Backend(e.to_string()))?;
376    let value = match value {
377        None => 0,
378        Some(v) if v < 0 => return Err(StateError::Overflow(key.to_owned())),
379        Some(v) => u64::try_from(v).map_err(|_| StateError::Overflow(key.to_owned()))?,
380    };
381    if value > pv_core::error::JCS_SAFE_MAX {
382        return Err(StateError::Overflow(key.to_owned()));
383    }
384    Ok(value)
385}
386
387fn fetch_max_on<C: redis::ConnectionLike>(conn: &mut C, key: &str, value: u64) -> Result<u64, StateError> {
388    if value > pv_core::error::JCS_SAFE_MAX {
389        return Err(StateError::Overflow(key.to_owned()));
390    }
391    let result: i64 = redis::Script::new(&FETCH_MAX_LUA)
392        .key(key)
393        .arg(value)
394        .arg(pv_core::error::JCS_SAFE_MAX)
395        .invoke(conn)
396        .map_err(|e| StateError::Backend(e.to_string()))?;
397    u64::try_from(result).map_err(|_| StateError::Overflow(key.to_owned()))
398}

400fn spend_many_on<C: redis::ConnectionLike>(
401    conn: &mut C,
402    spends: &[Spend],
403) -> Result<crate::TrySpendOutcome, StateError> {
404    // Shared duplicate-key guard: the Lua script reads
405    // GET(key) once per iteration in the check phase, so two spends on
406    // the same key each see the pre-commit value — same hazard as the
407    // in-memory backend, one implementation.
408    crate::store::refuse_duplicate_spend_keys(spends)?;
409    for spend in spends {
410        if spend.amount > pv_core::error::JCS_SAFE_MAX || spend.limit > pv_core::error::JCS_SAFE_MAX {
411            return Err(StateError::Overflow(spend.key.clone()));
412        }
413    }
414    if spends.is_empty() {
415        // Match the sentinel InMemoryStore returns for the empty-
416        // slice case. No EVAL round-trip; nothing to spend.
417        return Ok(crate::TrySpendOutcome::Committed {
418            post_commit_min_remaining: u64::MAX,
419        });
420    }
421    let script = redis::Script::new(&TRY_SPEND_LUA);
422    let mut invocation = script.prepare_invoke();
423    for spend in spends {
424        invocation.key(&spend.key).arg(spend.amount).arg(spend.limit);
425    }
426    // Return shape: [i64, i64]. Two disjoint carriers:
427    //   Commit:  [post_commit_min_remaining, 0]  (min_remaining
428    //            may be -1 to sentinel u64::MAX for empty KEYS;
429    //            spends.is_empty() short-circuits above so we
430    //            should not see -1 here).
431    //   Refusal: [-1, refusal_index_1_based]
432    let outcome: (i64, i64) = invocation
433        .invoke(conn)
434        .map_err(|e| StateError::Backend(e.to_string()))?;
435    match outcome {
436        (-1, refused) => {
437            let refused = usize::try_from(refused - 1)
438                .map_err(|_| StateError::Backend(format!("invalid Lua failure index {refused}")))?;
439            Ok(crate::TrySpendOutcome::Refused { index: refused })
440        }
441        (min_remaining, 0) => {
442            // -1 is the empty-KEYS sentinel (see the Lua); in practice
443            // unreachable here (guarded above) but honour it.
444            let remaining = if min_remaining < 0 {
445                u64::MAX
446            } else {
447                u64::try_from(min_remaining).unwrap_or(0)
448            };
449            Ok(crate::TrySpendOutcome::Committed {
450                post_commit_min_remaining: remaining,
451            })
452        }
453        (a, b) => Err(StateError::Backend(format!(
454            "TRY_SPEND_LUA returned unrecognised shape [{a}, {b}]"
455        ))),
456    }
457}

459impl StateStore for RedisStore {
460    fn fetch_max(&self, key: &str, value: u64) -> Result<u64, StateError> {
461        match &self.backend {
462            RedisBackend::Single(pool) => fetch_max_on(
463                &mut pool.get().map_err(|e| StateError::Backend(e.to_string()))?,
464                key,
465                value,
466            ),
467            RedisBackend::Cluster(pool) => fetch_max_on(
468                &mut *pool.get().map_err(|e| StateError::Backend(e.to_string()))?,
469                key,
470                value,
471            ),
472        }
473    }
474    fn counter_ttl_secs(&self) -> Option<u64> {
475        Some(BUDGET_COUNTER_TTL_SECS)
476    }
477
478    fn add(&self, key: &str, delta: u64) -> Result<u64, StateError> {
479        match &self.backend {
480            RedisBackend::Single(pool) => add_on(
481                &mut pool.get().map_err(|e| StateError::Backend(e.to_string()))?,
482                key,
483                delta,
484            ),
485            RedisBackend::Cluster(pool) => add_on(
486                &mut *pool.get().map_err(|e| StateError::Backend(e.to_string()))?,
487                key,
488                delta,
489            ),
490        }
491    }
492
493    fn get(&self, key: &str) -> Result<u64, StateError> {
494        match &self.backend {
495            RedisBackend::Single(pool) => get_on(
496                &mut pool.get().map_err(|e| StateError::Backend(e.to_string()))?,
497                key,
498            ),
499            RedisBackend::Cluster(pool) => get_on(
500                &mut *pool.get().map_err(|e| StateError::Backend(e.to_string()))?,
501                key,
502            ),
503        }
504    }
505
506    // `try_spend` uses the default trait impl (routes through
507    // `try_spend_many` with a 1-element slice). Removed the standalone
508    // wrapper in R68 — the default is exactly the same shape.
509
510    fn try_spend_many(&self, spends: &[Spend]) -> Result<crate::TrySpendOutcome, StateError> {
511        match &self.backend {
512            RedisBackend::Single(pool) => spend_many_on(
513                &mut pool.get().map_err(|e| StateError::Backend(e.to_string()))?,
514                spends,
515            ),
516            RedisBackend::Cluster(pool) => spend_many_on(
517                &mut *pool.get().map_err(|e| StateError::Backend(e.to_string()))?,
518                spends,
519            ),
520        }
521    }
522
523    fn remove(&self, key: &str) {
524        // Mirror the `refund` treatment
525        // — best-effort by contract, but a completely-silent Redis
526        // failure on the cleanup path leaves an operator with no
527        // signal that a session's stale key wasn't removed. The next
528        // recycled-session-id can then inherit the counter. Warn on
529        // both pool-get and DEL failure with structured fields.
530        let outcome: Result<(), redis::RedisError> = match &self.backend {
531            RedisBackend::Single(pool) => match pool.get() {
532                Ok(mut connection) => connection.del(key),
533                Err(error) => Err(redis::RedisError::from((
534                    redis::ErrorKind::IoError,
535                    "connection pool exhausted",
536                    error.to_string(),
537                ))),
538            },
539            RedisBackend::Cluster(pool) => match pool.get() {
540                Ok(mut connection) => connection.del(key),
541                Err(error) => Err(redis::RedisError::from((
542                    redis::ErrorKind::IoError,
543                    "connection pool exhausted",
544                    error.to_string(),
545                ))),
546            },
547        };
548        if let Err(error) = outcome {
549            let error_kind = error.kind();
550            tracing::warn!(
551                target: "pv_state::redis",
552                kind = ?error_kind,
553                detail = %error,
554                "Redis remove failed silently; per-key cleanup was not applied — \
555                 a subsequent session recycling this key may inherit its counter \
556                 until the 24 h TTL expires"
557            );
558        }
559    }

561    /// Saturating refund via `DECRBY` + a MAX(0) clamp.
562    /// Best-effort — errors are silently swallowed so a Redis blip on
563    /// the compensation path can never turn a lost-race response into
564    /// a 5xx.
565    ///
566    /// NEVER resurrect a key that was already `DEL`'d
567    /// by a concurrent `remove_prefix`. The prior implementation
568    /// called `DECRBY` on the raw key: Redis initialises a missing
569    /// key to 0 first, so `DECRBY` returned `-amount` and the
570    /// `MAX(0)` clamp branch did `SET key 0` (no `EX`), producing
571    /// a permanent TTL-less key. Under the lost-claim-
572    /// plus-idle-close ordering (mcp_call sandbox-gate debit
573    /// races with the reconciler's clear_budget_state), the
574    /// refund path leaked one-to-three TTL-less keys per session
575    /// — attacker-choosable memory growth against the exact class
576    /// the `clear_budget_state` doc in reconciler.rs documents as
577    /// impossible. Fix: gate
578    /// the whole DECRBY on `EXISTS`. If the session was cleared,
579    /// the budget is already gone and there is nothing to
580    /// compensate; the refund is a silent no-op. If the session
581    /// is alive, we DECRBY-clamp AND refresh the counter TTL to
582    /// match `TRY_SPEND_LUA` (a plain DECRBY does not refresh).
583    fn refund(&self, key: &str, amount: u64) {
584        // Redis DECRBY takes i64. Cap at i64::MAX so a caller passing
585        // u64::MAX cannot silently wrap into a negative value.
586        let amount = i64::try_from(amount).unwrap_or(i64::MAX);
587        // Fresh replay marker per logical refund: the same-slot suffix
588        // keeps cluster mode single-slot; a driver-level retry reuses
589        // the SAME marker and no-ops (see REFUND_LUA).
590        let marker = format!("{key}:r:{}", pv_core::new_event_uid());
591        // Refund is best-effort by the trait contract, but
592        // its silent-swallow used to be COMPLETELY invisible — no log,
593        // no metric — so an operator seeing budget depletion during a
594        // Redis outage had no signal that compensation had been
595        // failing. Log the failed refund with structured fields so
596        // downstream aggregators (Vector → OTLP → SIEM) can alert on
597        // it. The response path still stays 200 OK so the caller
598        // never learns a compensation failure as a 5xx.
599        let outcome: Result<i64, redis::RedisError> = match &self.backend {
600            RedisBackend::Single(pool) => match pool.get() {
601                Ok(mut connection) => redis::Script::new(&REFUND_LUA)
602                    .key(key)
603                    .key(&marker)
604                    .arg(amount)
605                    .invoke(&mut *connection),
606                Err(error) => Err(redis::RedisError::from((
607                    redis::ErrorKind::IoError,
608                    "connection pool exhausted",
609                    error.to_string(),
610                ))),
611            },
612            RedisBackend::Cluster(pool) => match pool.get() {
613                Ok(mut connection) => redis::Script::new(&REFUND_LUA)
614                    .key(key)
615                    .key(&marker)
616                    .arg(amount)
617                    .invoke(&mut *connection),
618                Err(error) => Err(redis::RedisError::from((
619                    redis::ErrorKind::IoError,
620                    "connection pool exhausted",
621                    error.to_string(),
622                ))),
623            },
624        };
625        if let Err(error) = outcome {
626            let error_kind = error.kind();
627            tracing::warn!(
628                target: "pv_state::redis",
629                kind = ?error_kind,
630                detail = %error,
631                "Redis refund failed silently; per-key compensation was not applied — \
632                 budget counter may remain over-charged until the 24 h TTL expires or a \
633                 successful spend/refund lands"
634            );
635        }
636    }

638    /// R66 F2: atomic multi-key refund via a single Lua script.
639    /// All keys share the caller's Redis Cluster hash tag (guaranteed
640    /// by `ActionBudget::session_prefix` which wraps its digest in
641    /// `{...}`), so cross-slot dispatch is impossible and the script
642    /// runs on ONE server-side EVAL. Per-key semantics match
643    /// `refund` exactly: EXISTS gate (never resurrect a
644    /// remove_prefix-cleared cell), DECRBY, clamp-at-zero via SET
645    /// with fresh TTL on underflow, EXPIRE refresh on non-underflow.
646    /// Best-effort like `refund` — Redis errors log-warn but never
647    /// propagate.
648    fn refund_many(&self, refunds: &[crate::Refund]) {
649        let Some(first) = refunds.first() else {
650            return;
651        };
652        // Replay marker key (see REFUND_LUA): derived from the first
653        // refund key so it shares the batch's hash tag (all keys carry
654        // the same `{...}` tag per the invariant above) — single-slot
655        // in cluster mode.
656        let marker = format!("{}:r:{}", first.key, pv_core::new_event_uid());
657        let outcome: Result<i64, redis::RedisError> = match &self.backend {
658            RedisBackend::Single(pool) => match pool.get() {
659                Ok(mut connection) => {
660                    let script = redis::Script::new(&REFUND_MANY_LUA);
661                    let mut invocation = script.prepare_invoke();
662                    for r in refunds {
663                        invocation.key(&r.key);
664                        // Cap at i64::MAX so a caller passing u64::MAX
665                        // cannot silently wrap into a negative value.
666                        invocation.arg(i64::try_from(r.amount).unwrap_or(i64::MAX));
667                    }
668                    invocation.key(&marker);
669                    invocation.invoke(&mut *connection)
670                }
671                Err(error) => Err(redis::RedisError::from((
672                    redis::ErrorKind::IoError,
673                    "connection pool exhausted",
674                    error.to_string(),
675                ))),
676            },
677            RedisBackend::Cluster(pool) => match pool.get() {
678                Ok(mut connection) => {
679                    let script = redis::Script::new(&REFUND_MANY_LUA);
680                    let mut invocation = script.prepare_invoke();
681                    for r in refunds {
682                        invocation.key(&r.key);
683                        invocation.arg(i64::try_from(r.amount).unwrap_or(i64::MAX));
684                    }
685                    invocation.key(&marker);
686                    invocation.invoke(&mut *connection)
687                }
688                Err(error) => Err(redis::RedisError::from((
689                    redis::ErrorKind::IoError,
690                    "connection pool exhausted",
691                    error.to_string(),
692                ))),
693            },
694        };
695        if let Err(error) = outcome {
696            let error_kind = error.kind();
697            tracing::warn!(
698                target: "pv_state::redis",
699                kind = ?error_kind,
700                detail = %error,
701                refunds = refunds.len(),
702                "Redis refund_many failed silently; per-key compensation was not applied — \
703                 budget counters may remain over-charged until the 24 h TTL expires or a \
704                 successful spend/refund lands"
705            );
706        }
707    }

709    /// Whole-session counter cleanup. Previously left as the
710    /// trait's default no-op on the assumption that the 24 h TTL made it
711    /// pure hygiene — but `SessionRegistry::get_or_open` recycles a
712    /// finalized session id into a fresh open session, which then spends
713    /// against the SAME hash-tagged keys. Against `InMemoryStore` (dev,
714    /// CI, and every budget test) the recycled incarnation starts from
715    /// zero; against Redis it silently inherited up to 24 h of the prior
716    /// incarnation's `tokens`/`total_calls`/`tool:*`/`payout` counters —
717    /// the exact cross-backend divergence class the
718    /// `add`/`try_spend_many` alignment closed.
719    ///
720    /// All of a session's keys share one cluster slot by construction
721    /// (`ActionBudget::session_prefix` wraps the digest in a `{hash-tag}`),
722    /// so a SCAN cursor + bounded DEL batches — routed by the prefix
723    /// pattern's hash-tag to the owning master in cluster mode — reach
724    /// every key on both single-node and cluster backends.
725    ///
726    /// SCAN, not KEYS. The first take used
727    /// `redis.call('KEYS', pattern)` inside a Lua script, which is
728    /// O(entire keyspace per node) AND blocks the Redis event loop
729    /// atomically for the whole scan+delete. On a shared Redis with high
730    /// session churn or attacker-chosen ids, every close stalled all
731    /// other Redis clients (including hot-path `try_spend_many` gates)
732    /// for the duration. SCAN is O(matches) amortized and yields between
733    /// batches, so per-session cleanup no longer blocks concurrent
734    /// traffic.
735    ///
736    /// `route_command` with an explicit slot for cluster
737    /// mode. An intermediate revision used
738    /// `Commands::scan_match` on the cluster connection, but bare
739    /// `SCAN` has no key argument — `RoutingInfo::for_routable` returns
740    /// `None`, and `ClusterConnection::request` treats that as an
741    /// immediate `UNROUTABLE_ERROR`. The iterator failed on
742    /// construction, the `Err(_) => return` arm swallowed it, and
743    /// cleanup silently did nothing on every cluster deployment.
744    /// Best-effort like `remove`/`refund`: any Redis blip on the
745    /// cleanup path must not fail the close.
746    fn remove_prefix(&self, prefix: &str) {
747        // `MATCH` glob treats `* ? [ ] \` as metacharacters. The prefix
748        // is `budget:{<32 hex>}:` today (no metacharacters), but escape
749        // defensively so a future prefix shape cannot over-match.
750        let mut pattern = String::with_capacity(prefix.len() + 1);
751        for c in prefix.chars() {
752            if matches!(c, '*' | '?' | '[' | ']' | '\\') {
753                pattern.push('\\');
754            }
755            pattern.push(c);
756        }
757        pattern.push('*');
758        match &self.backend {
759            RedisBackend::Single(pool) => {
760                if let Ok(mut connection) = pool.get() {
761                    scan_and_delete_single(&mut connection, &pattern);
762                }
763            }
764            RedisBackend::Cluster(pool) => {
765                if let Ok(mut connection) = pool.get() {
766                    // Route by the ORIGINAL (unescaped) prefix so the
767                    // hash-tag slot matches the one Redis derives from
768                    // the on-cluster keys. If the future adds a prefix
769                    // shape that contains any glob metachar, computing
770                    // the slot from the ESCAPED pattern would embed
771                    // `\` characters that never appear in the on-cluster
772                    // keys — SCAN would route to a slot no key lives on
773                    // and `remove_prefix` would silently no-op, letting
774                    // budget counters survive up to the 24 h TTL. Today's
775                    // prefix is metachar-free so escaping is a no-op and
776                    // this bug is latent; the fix protects against the
777                    // future shape change.
778                    scan_and_delete_cluster(&mut connection, &pattern, prefix);
779                }
780            }
781        }
782    }

847/// Bounded SCAN+DEL loop against a single-node connection. Each SCAN
848/// call is O(COUNT) at the server; the loop terminates when the cursor
849/// returns to 0. DEL is batched so one round-trip retires up to
850/// `SCAN_COUNT` keys.
851fn scan_and_delete_single(conn: &mut redis::Connection, pattern: &str) {
852    const SCAN_COUNT: usize = 500;
853    let mut cursor: u64 = 0;
854    loop {
855        let scan: Result<(u64, Vec<String>), _> = redis::cmd("SCAN")
856            .arg(cursor)
857            .arg("MATCH")
858            .arg(pattern)
859            .arg("COUNT")
860            .arg(SCAN_COUNT)
861            .query(conn);
862        let (next, batch) = match scan {
863            Ok(pair) => pair,
864            Err(error) => {
865                // SCAN failing means we bail
866                // out mid-cleanup — the remaining keys survive with
867                // their TTLs, and a future session recycling this id
868                // (within 24 h) would inherit the leftover counters
869                // silently. Log with structured fields so an operator
870                // aggregating warns can spot chronic backend outages.
871                let error_kind = error.kind();
872                tracing::warn!(
873                    target: "pv_state::redis",
874                    kind = ?error_kind,
875                    detail = %error,
876                    "Redis SCAN failed during remove_prefix; partial cleanup — surviving \
877                     keys will expire at the 24 h TTL or on the next successful \
878                     remove_prefix for the same prefix"
879                );
880                return;
881            }
882        };
883        if !batch.is_empty() {
884            let mut del = redis::cmd("DEL");
885            for key in &batch {
886                del.arg(key);
887            }
888            let outcome: Result<i64, _> = del.query(conn);
889            if let Err(error) = outcome {
890                let error_kind = error.kind();
891                tracing::warn!(
892                    target: "pv_state::redis",
893                    kind = ?error_kind,
894                    detail = %error,
895                    batch_len = batch.len(),
896                    "Redis DEL batch failed during remove_prefix; these keys survive \
897                     until the 24 h TTL expires"
898                );
899            }
900        }
901        if next == 0 {
902            return;
903        }
904        cursor = next;
905    }
906}

908/// The Redis Cluster hash slot of `key`: CRC16-XMODEM of the key (or of
909/// the first non-empty `{hash-tag}` when the key has one) modulo 16384,
910/// exactly as `CLUSTER KEYSLOT` computes it. The `redis` crate stopped
911/// exporting its own copy in 0.32, so the daemon keeps this one and pins
912/// it with the reference values from the Redis documentation.
913fn cluster_key_slot(key: &[u8]) -> u16 {
914    const SLOTS: u16 = 16_384;
915    let hashed = key
916        .iter()
917        .position(|b| *b == b'{')
918        .and_then(|open| key.get(open + 1..))
919        .and_then(|after_open| {
920            let len = after_open.iter().position(|b| *b == b'}')?;
921            (len > 0).then(|| after_open.get(..len)).flatten()
922        })
923        .unwrap_or(key);
924    let mut crc: u16 = 0;
925    for byte in hashed {
926        crc ^= u16::from(*byte) << 8;
927        for _ in 0..8 {
928            crc = if crc & 0x8000 != 0 {
929                (crc << 1) ^ 0x1021
930            } else {
931                crc << 1
932            };
933        }
934    }
935    crc % SLOTS
936}

938/// Cluster variant: bare `SCAN` has no key argument so
939/// `redis::cluster_routing::RoutingInfo::for_routable` returns `None`
940/// on it (the sync ClusterConnection then fails with
941/// `UNROUTABLE_ERROR`, silently deleting nothing — the
942/// regression this replaces). All matches share one hash-slot by
943/// construction (`ActionBudget::session_prefix` wraps the digest in
944/// `{hash-tag}`), so compute the slot from the prefix and route both
945/// SCAN and DEL through `route_command` to that specific master.
946///
947/// `pattern` carries the ESCAPED glob (metachars prefixed with `\`)
948/// for the SCAN MATCH argument. `route_prefix` carries the ORIGINAL
949/// unescaped prefix used SOLELY for slot computation — the hash-tag
950/// content Redis extracts from an on-cluster key is the RAW bytes,
951/// so a slot computed from an escaped pattern would miss on any
952/// prefix shape containing a glob metachar. Today's prefix is
953/// metachar-free so escape is a no-op; the separation protects
954/// against a future prefix shape change.
955fn scan_and_delete_cluster(conn: &mut redis::cluster::ClusterConnection, pattern: &str, route_prefix: &str) {
956    use redis::cluster_routing::{Route, RoutingInfo, SingleNodeRoutingInfo, SlotAddr};
957    const SCAN_COUNT: usize = 500;
958    let route_key = route_prefix.trim_end_matches('*');
959    let slot = cluster_key_slot(route_key.as_bytes());
960    let routing = RoutingInfo::SingleNode(SingleNodeRoutingInfo::SpecificNode(Route::new(
961        slot,
962        SlotAddr::Master,
963    )));
964    let mut cursor: u64 = 0;
965    loop {
966        let scan_cmd = {
967            let mut cmd = redis::cmd("SCAN");
968            cmd.arg(cursor)
969                .arg("MATCH")
970                .arg(pattern)
971                .arg("COUNT")
972                .arg(SCAN_COUNT);
973            cmd
974        };
975        let value = match conn.route_command(&scan_cmd, routing.clone()) {
976            Ok(value) => value,
977            Err(error) => {
978                // Parity with
979                // `scan_and_delete_single`. Cluster-mode SCAN failure
980                // was completely silent, so an operator watching
981                // `pv_state::redis::warn` for cleanup problems saw
982                // nothing on cluster deployments even under Redis
983                // slowdown.
984                let error_kind = error.kind();
985                tracing::warn!(
986                    target: "pv_state::redis",
987                    kind = ?error_kind,
988                    detail = %error,
989                    slot,
990                    "Redis cluster SCAN failed during remove_prefix; partial cleanup — \
991                     surviving keys will expire at the 24 h TTL or on the next successful \
992                     remove_prefix for the same prefix"
993                );
994                return;
995            }
996        };
997        let parsed: Result<(u64, Vec<String>), _> = redis::FromRedisValue::from_redis_value(&value);
998        let (next, batch) = match parsed {
999            Ok(pair) => pair,
1000            Err(error) => {
1001                let error_kind = error.kind();
1002                tracing::warn!(
1003                    target: "pv_state::redis",
1004                    kind = ?error_kind,
1005                    detail = %error,
1006                    "Redis cluster SCAN returned an unparsable response during remove_prefix; \
1007                     aborting further cleanup for this prefix"
1008                );
1009                return;
1010            }
1011        };
1012        if !batch.is_empty() {
1013            let mut del = redis::cmd("DEL");
1014            for key in &batch {
1015                del.arg(key);
1016            }
1017            // Multi-key DEL is safe here because every match shares the
1018            // slot we already computed above — route it explicitly so
1019            // the driver does not need to inspect the keys again.
1020            if let Err(error) = conn.route_command(&del, routing.clone()) {
1021                let error_kind = error.kind();
1022                tracing::warn!(
1023                    target: "pv_state::redis",
1024                    kind = ?error_kind,
1025                    detail = %error,
1026                    slot,
1027                    batch_len = batch.len(),
1028                    "Redis cluster DEL batch failed during remove_prefix; these keys survive \
1029                     until the 24 h TTL expires"
1030                );
1031            }
1032        }
1033        if next == 0 {
1034            return;
1035        }
1036        cursor = next;
1037    }
1038}

1137            ("REFUND_LUA", REFUND_LUA.as_str()),
1138            ("REFUND_MANY_LUA", REFUND_MANY_LUA.as_str()),
1139        ] {
1140            assert!(
1141                script.contains(&format!("EXPIRE', KEYS[1], {ttl}"))
1142                    || script.contains(&format!("EXPIRE', key, {ttl}")),
1143                "{name} must EXPIRE with BUDGET_COUNTER_TTL_SECS: {script}"
1144            );
1145            // R111 F1: the R69 review-L3 guard was `!script.contains('{')`
1146            // — designed to catch unexpanded `{BUDGET_COUNTER_TTL_SECS}`
1147            // format placeholders left over from the format!() template.
1148            // But TRY_SPEND_LUA legitimately contains literal `{`
1149            // characters: the raw-string template writes `{{-1, i}}`
1150            // and `{{min_remaining, 0}}` for Lua-return-table syntax
1151            // (double-braces escape in format!()), producing literal
1152            // `{...}` in the compiled script. The blanket contains-'{'
1153            // scan panicked on every run — the guard has been silently
1154            // broken since it landed. Fix: check for the ONLY
1155            // placeholder name we actually use (BUDGET_COUNTER_TTL_SECS)
1156            // — any surviving occurrence of the identifier means the
1157            // format!() expansion didn't happen.
1158            assert!(
1159                !script.contains("BUDGET_COUNTER_TTL_SECS"),
1160                "{name} has an unexpanded BUDGET_COUNTER_TTL_SECS placeholder"
1161            );
1162        }
1163        // Refund replay/precision pins (pass-7 findings):
1164        // 1. Both refund scripts open with the SET-NX replay marker —
1165        //    the cluster driver transparently retries scripts on
1166        //    connection loss, and an unmarked refund replay CREDITS the
1167        //    budget twice (spendable budget minted from a network flap).
1168        // 2. REFUND_MANY passes ARGV[i] to DECRBY as the raw integer
1169        //    string — `tonumber` converted through a Lua double, so an
1170        //    i64::MAX amount rounded to 2^63, DECRBY rejected mid-loop,
1171        //    and the batch PARTIALLY committed (in-memory clamps every
1172        //    entry; the backends must agree).
1173        for (name, script) in [
1174            ("REFUND_LUA", REFUND_LUA.as_str()),
1175            ("REFUND_MANY_LUA", REFUND_MANY_LUA.as_str()),
1176        ] {
1177            assert!(
1178                script.contains("'NX', 'EX'"),
1179                "{name} must guard against driver-level replay with a SET-NX marker: {script}"
1180            );
1181        }
1182        assert!(
1183            !REFUND_MANY_LUA.contains("tonumber(ARGV"),
1184            "REFUND_MANY_LUA must pass ARGV to DECRBY as an exact integer string, \
1185             not through a Lua double: {}",
1186            REFUND_MANY_LUA.as_str()
1187        );
1188    }
