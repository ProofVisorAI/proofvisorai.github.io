1//! The Receipt: an offline-verifiable, Ed25519-signed record of a session.
2//!
3//! Payload per the brief Module G: session id, agent identity block
4//! (version/charter/instance_uid), tool-call summary, cost, stop reason,
5//! event-chain hash, signature, signer public-key reference. Subjects are an
6//! enum so the same envelope covers signed-workflow chains and retroactive
7//! ATIF promotions (Module H reconciliation).
8//!
9//! Money is carried as integer micro-USD — floats never touch a signed field.
10

16/// Version of the receipt format itself (evolution surface).
17///
18/// Version history:
19/// * **1** — the signature covers the bare JCS canonicalization of the
20///   body. Legacy; still verifiable (see [`signing_message`]) but never
21///   issued anymore.
22/// * **2** — the signature covers a domain-separated message:
23///   [`RECEIPT_DOMAIN_TAG_V2`] + the u64 big-endian byte length of the
24///   JCS canonical body + the JCS canonical body bytes. This closes a
25///   cross-protocol confusion risk: the same Ed25519 key also
26///   feeds the journal-MAC KDF (`journal.rs`), and a signature over
27///   un-tagged bytes could in principle be replayed as a signature over
28///   some other protocol's message. Mirrors the framing style of
29///   `journal.rs` (`b"agentvisor-journal-v1\0"` + BE length prefixes).
30pub const RECEIPT_VERSION: u32 = 2;
31
32/// Domain-separation tag prepended to every version-2 receipt signing
33/// message. The trailing NUL makes the tag self-delimiting so no JCS
34/// body could extend it into a different tag.
35///
36/// Wire format: every v2 receipt ever issued is signed over these bytes.
37/// They predate the ProofVisor name and must never be renamed.
38pub const RECEIPT_DOMAIN_TAG_V2: &[u8] = b"agentvisor-receipt-v2\0";
39
40/// Build the exact byte message the receipt signature covers for a
41/// given `receipt_version`, from the JCS canonical form of the body.
42///
43/// * version 1: the bare canonical bytes (legacy wire format);
44/// * version 2: `RECEIPT_DOMAIN_TAG_V2 || len(canonical) as u64 BE ||
45///   canonical` — domain-separated and length-framed, mirroring
46///   `journal.rs`;
47/// * any other version: refused with
48///   [`ReceiptError::SemanticInvariant`] (there is no defined framing,
49///   so neither signing nor verifying may proceed).
50///
51/// Public so external tooling and tests can reproduce the signing
52/// message byte-for-byte.
53pub fn signing_message(receipt_version: u32, canonical_body: &str) -> Result<Vec<u8>, ReceiptError> {
54    match receipt_version {
55        1 => Ok(canonical_body.as_bytes().to_vec()),
56        2 => {
57            let mut message = Vec::with_capacity(RECEIPT_DOMAIN_TAG_V2.len() + 8 + canonical_body.len());
58            message.extend_from_slice(RECEIPT_DOMAIN_TAG_V2);
59            message.extend_from_slice(&(canonical_body.len() as u64).to_be_bytes());
60            message.extend_from_slice(canonical_body.as_bytes());
61            Ok(message)
62        }
63        other => Err(ReceiptError::SemanticInvariant(format!(
64            "receipt_version {other} has no defined signature framing (supported: 1, 2)"
65        ))),
66    }
67}

150/// A complete receipt: body + detached signature over the version's
151/// signing message (see [`signing_message`] — bare `JCS(body)` for v1,
152/// domain-tagged and length-framed `JCS(body)` for v2).
153///
154/// The wire shape is intentionally flat (13 body fields + `signature_b64`
155/// at the top level) — the schemas at `schemas/receipt-v1.schema.json`
156/// and `schemas/receipt-v2.schema.json` commit to it (they differ only
157/// in the pinned `receipt_version`) and downstream verifiers
158/// (`pvctl receipt-verify`, external tools) consume it that way.
159///
160/// `#[serde(flatten)]` normally silently disables `deny_unknown_fields`
161/// on the inner struct, which would let a hostile issuer add extra
162/// top-level fields ("claim_extra": "grant admin") that the human
163/// reviewer sees but `verify()` accepts (the field never survives the
164/// round-trip to `ReceiptBody` used inside `canonicalize`). To close
165/// that gap without breaking the wire shape, the [`Deserialize`] impl
166/// is written by hand and rejects any top-level key outside the
167/// declared whitelist. See the
168/// `allowed_receipt_top_level_keys_cover_body_fields_exactly`
169/// test for the compile-time-ish drift guard.
170#[derive(Debug, Clone, PartialEq, Serialize)]
171pub struct Receipt {
172    /// Signed body.
173    #[serde(flatten)]
174    pub body: ReceiptBody,
175    /// Ed25519 signature, base64. Covers the version's signing message:
176    /// bare `JCS(body)` for v1, [`RECEIPT_DOMAIN_TAG_V2`] + u64-BE
177    /// length + `JCS(body)` for v2.
178    pub signature_b64: String,
179}

205impl<'de> Deserialize<'de> for Receipt {
206    fn deserialize<D: serde::Deserializer<'de>>(deserializer: D) -> Result<Self, D::Error> {
207        use serde::de::{DeserializeSeed, Error as _, MapAccess, SeqAccess, Visitor};
208
209        /// Recursively-checked JSON value: refuses duplicate keys and
210        /// explicit `null`s at EVERY depth while parsing. The top-level
211        /// duplicate guard below is not enough — `next_value::<Value>()`
212        /// collapses NESTED duplicates last-wins (serde_json), so a
213        /// hostile issuer could ship `"prompt_tokens":0,
214        /// "prompt_tokens":100` inside `cost`: the signature verifies
215        /// (canonicalization sees the collapsed value) while a
216        /// first-wins parser shows an auditor the other number — the
217        /// exact equivocation the top-level guard exists to refuse.
218        /// Explicit nulls are refused for the same display-vs-canonical
219        /// reason: every optional field is written with
220        /// `skip_serializing_if`, so a null can only be a planted
221        /// artifact that vanishes on reserialization.
222        /// (`Receipt::from_json_slice` enforces both at the bytes level;
223        /// this closes the plain `serde_json::from_*::<Receipt>` path.)
224        struct CheckedValue;
225        struct CheckedValueVisitor;
226        impl<'de> Visitor<'de> for CheckedValueVisitor {
227            type Value = serde_json::Value;
228            fn expecting(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
229                f.write_str("a JSON value without duplicate keys or nulls")
230            }
231            fn visit_bool<E: serde::de::Error>(self, v: bool) -> Result<Self::Value, E> {
232                Ok(serde_json::Value::Bool(v))
233            }
234            fn visit_i64<E: serde::de::Error>(self, v: i64) -> Result<Self::Value, E> {
235                Ok(serde_json::Value::from(v))
236            }
237            fn visit_u64<E: serde::de::Error>(self, v: u64) -> Result<Self::Value, E> {
238                Ok(serde_json::Value::from(v))
239            }
240            fn visit_f64<E: serde::de::Error>(self, v: f64) -> Result<Self::Value, E> {
241                Ok(serde_json::Value::from(v))
242            }
243            fn visit_str<E: serde::de::Error>(self, v: &str) -> Result<Self::Value, E> {
244                Ok(serde_json::Value::String(v.to_owned()))
245            }
246            fn visit_string<E: serde::de::Error>(self, v: String) -> Result<Self::Value, E> {
247                Ok(serde_json::Value::String(v))
248            }
249            fn visit_unit<E: serde::de::Error>(self) -> Result<Self::Value, E> {
250                Err(E::custom(
251                    "explicit null in Receipt; optional fields are omitted when absent, \
252                     so a null is a planted artifact that would vanish on reserialization",
253                ))
254            }
255            fn visit_seq<A: SeqAccess<'de>>(self, mut seq: A) -> Result<Self::Value, A::Error> {
256                let mut items = Vec::new();
257                while let Some(item) = seq.next_element_seed(CheckedValue)? {
258                    items.push(item);
259                }
260                Ok(serde_json::Value::Array(items))
261            }
262            fn visit_map<A: MapAccess<'de>>(self, mut map: A) -> Result<Self::Value, A::Error> {
263                let mut object = serde_json::Map::new();
264                while let Some(key) = map.next_key::<String>()? {
265                    if object.contains_key(&key) {
266                        return Err(A::Error::custom(format!(
267                            "duplicate field `{key}` nested in Receipt; parsers disagree on \
268                             duplicate-key semantics, so accepting one would let a hostile \
269                             issuer sign one interpretation while auditors read the other"
270                        )));
271                    }
272                    let value = map.next_value_seed(CheckedValue)?;
273                    object.insert(key, value);
274                }
275                Ok(serde_json::Value::Object(object))
276            }
277        }
278        impl<'de> DeserializeSeed<'de> for CheckedValue {
279            type Value = serde_json::Value;
280            fn deserialize<D2: serde::Deserializer<'de>>(
281                self,
282                deserializer: D2,
283            ) -> Result<Self::Value, D2::Error> {
284                deserializer.deserialize_any(CheckedValueVisitor)
285            }
286        }
287
288        // Custom visitor so we can reject duplicate keys explicitly.
289        // `serde_json::Map::deserialize` silently collapses duplicate
290        // keys (last-wins) — RFC 8259 leaves the behaviour undefined
291        // and other JSON parsers (jq's `--sort-keys`, some strict
292        // Python configs, several audit tools) pick first-wins.
293        // A hostile issuer could sign under the last-wins interpretation
294        // while an auditor reading with a first-wins parser saw the
295        // friendly value; the signature would verify but the audit
296        // would show a different receipt.
297        struct ReceiptVisitor;
298        impl<'de> Visitor<'de> for ReceiptVisitor {
299            type Value = Receipt;
300            fn expecting(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
301                f.write_str("an ProofVisor Receipt JSON object")
302            }
303            fn visit_map<M: MapAccess<'de>>(self, mut map: M) -> Result<Receipt, M::Error> {
304                let mut raw = serde_json::Map::new();
305                while let Some(key) = map.next_key::<String>()? {
306                    if !ALLOWED_RECEIPT_TOP_LEVEL_KEYS.contains(&key.as_str()) {
307                        return Err(M::Error::custom(format!(
308                            "unknown field `{key}` in Receipt; a signed receipt \
309                             may only carry the fields declared in ReceiptBody + \
310                             signature_b64. Extra fields would be visible to a raw \
311                             JSON reader but not covered by the signature — treat \
312                             as tampering."
313                        )));
314                    }
315                    if raw.contains_key(&key) {
316                        return Err(M::Error::custom(format!(
317                            "duplicate field `{key}` in Receipt; JSON parsers \
318                             disagree on duplicate-key semantics (last-wins vs \
319                             first-wins), so accepting duplicates would let a \
320                             hostile issuer sign under one interpretation while an \
321                             auditor's parser reports the other"
322                        )));
323                    }
324                    let value = map.next_value_seed(CheckedValue)?;
325                    raw.insert(key, value);
326                }
327                let signature_value = raw
328                    .remove("signature_b64")
329                    .ok_or_else(|| M::Error::missing_field("signature_b64"))?;
330                let signature_b64: String =
331                    serde_json::from_value(signature_value).map_err(M::Error::custom)?;
332                let body: ReceiptBody =
333                    serde_json::from_value(serde_json::Value::Object(raw)).map_err(M::Error::custom)?;
334                Ok(Receipt { body, signature_b64 })
335            }
336        }
337        deserializer.deserialize_map(ReceiptVisitor)
338    }
339}

341/// Receipt errors.
342#[derive(Debug, thiserror::Error)]
343#[non_exhaustive]
344pub enum ReceiptError {
345    /// Canonicalization failed (unsafe numbers in body).
346    #[error("canonicalization: {0}")]
347    Jcs(#[from] crate::jcs::JcsError),
348    /// Serialization failed.
349    #[error("serialization: {0}")]
350    Serde(#[from] serde_json::Error),
351    /// Key/signature failure.
352    #[error("key: {0}")]
353    Key(#[from] KeyError),
354    /// Signature or public key not valid base64.
355    #[error("invalid base64 in receipt")]
356    Base64,
357    /// The receipt's embedded public key does not match the keyring entry for
358    /// its key id (substitution attempt).
359    #[error("embedded public key mismatches keyring entry for {0:?}")]
360    KeyMismatch(String),
361    /// A JSON object at some nesting level carried a
362    /// duplicate key. Rejecting these closes the last-wins vs
363    /// first-wins auditor split-brain that would otherwise let a
364    /// hostile issuer sign under one interpretation while an
365    /// external audit tool displayed the other.
366    ///
367    /// The payload is a complete scanner message (duplicate key,
368    /// nesting-depth, or explicit-null refusal), so the template
369    /// must not re-wrap it in a `key`-shaped slot — a
370    /// "duplicate JSON key `{0}` at nesting" template produced garbled
371    /// double-wrapped diagnostics like ``duplicate JSON key
372    /// `duplicate key `x` in JSON object …` at nesting``.
373    #[error("{0}; strict receipt parsers refuse this")]
374    DuplicateKey(String),
375    /// A receipt carries a semantically-invalid field.
376    /// Enforced by `verify_semantic_invariants`: supported
377    /// `receipt_version` ∈ {1, 2}, `ai_agent.charter.type_id == 1`
378    /// (OCSF Regular File), non-empty required string fields
379    /// (`receipt_id`, `session_id`, `ai_agent.charter.name`,
380    /// `ai_agent.charter.instance_uid`), 64-hex `chain_head` and
381    /// per-step `trajectory_digest`, `issued_at_iso ==
382    /// iso8601_ms(issued_at)`, stop-reason ↔ stop-reason-id
383    /// cross-wiring, and `AtifTrajectory.retroactive == true`.
384    /// Distinct from `Serde` and `DuplicateKey` so ops triage
385    /// can name the class.
386    #[error("semantic invariant violated: {0}")]
387    SemanticInvariant(String),
388}

390impl Receipt {
391    /// Deserialize a receipt from wire bytes, rejecting any duplicate
392    /// key at ANY nesting level.
393    ///
394    /// The custom `Deserialize` impl for `Receipt` (below) already
395    /// rejects duplicate top-level keys, but nested structs
396    /// (`ai_agent`, `tool_calls`, `cost`, `subject`) went through
397    /// serde's default derive which relies on `serde_json::Map` /
398    /// `IndexMap` — both silently collapse duplicate keys with
399    /// last-wins semantics. A hostile issuer could then sign a
400    /// receipt whose `ai_agent.instance_uid` appeared twice
401    /// (`"a"` then `"b"`); JCS canonicalisation saw only `"b"` so
402    /// the signature verified, but a first-wins auditor tool (jq's
403    /// default, some Python configs) displayed `"a"` — the exact
404    /// split-brain the top-level guard was written to prevent, one
405    /// level deeper.
406    ///
407    /// Pre-scans the JSON with a strict duplicate-key
408    /// checker (`check_no_duplicate_keys`) before deserialising into
409    /// `Receipt`. Callers verifying a receipt off the wire should
410    /// prefer this over `serde_json::from_slice::<Receipt>`.
411    pub fn from_json_slice(bytes: &[u8]) -> Result<Self, ReceiptError> {
412        check_no_duplicate_keys(bytes)?;
413        serde_json::from_slice(bytes).map_err(ReceiptError::Serde)
414    }
415
416    /// Same as [`Receipt::from_json_slice`] for owned/borrowed strings.
417    pub fn from_json_str(s: &str) -> Result<Self, ReceiptError> {
418        Self::from_json_slice(s.as_bytes())
419    }
420
421    /// Issue (sign) a receipt over `body` with `signer`.
422    ///
423    /// The `key_id` and `public_key_b64` fields of the body are overwritten
424    /// from the signer — a caller can never claim someone else's key identity.
425    /// `receipt_version` is likewise overwritten to [`RECEIPT_VERSION`]:
426    /// every freshly issued receipt uses the current (domain-separated)
427    /// signing format, so no issuance path can produce a legacy v1
428    /// signature.
429    pub fn issue(mut body: ReceiptBody, signer: &dyn Signer) -> Result<Self, ReceiptError> {
430        body.receipt_version = RECEIPT_VERSION;
431        body.key_id = signer.key_id().to_owned();
432        body.public_key_b64 = base64::engine::general_purpose::STANDARD.encode(signer.public_key_bytes());
433        let canon = canonicalize(&serde_json::to_value(&body)?)?;
434        let message = signing_message(body.receipt_version, &canon)?;
435        let sig = signer.sign(&message);
436        let receipt = Self {
437            body,
438            signature_b64: base64::engine::general_purpose::STANDARD.encode(sig),
439        };
440        // Never mint a receipt this crate's own verifier refuses:
441        // issuance used to sign ANY canonicalizable body (empty
442        // session_id, retroactive:false trajectory subjects,
443        // inconsistent timestamps, allowed+blocked > total) and the
444        // dead receipt only surfaced when a verifier finally ran —
445        // long after the session evidence that could have re-minted
446        // it was gone. verify_embedded runs the full semantic
447        // invariant set plus the signature round-trip (~one Ed25519
448        // verify per session close — noise next to the sign itself).
449        receipt.verify_embedded()?;
450        Ok(receipt)
451    }

454    /// that `verify` refuses hostile-keyholder receipts that `issue`

456    #[cfg(test)]
457    pub(crate) fn issue_unchecked(mut body: ReceiptBody, signer: &dyn Signer) -> Result<Self, ReceiptError> {

470    /// Verify offline against a keyring. Checks, in execution order:
471    /// 1. `AtifTrajectory.retroactive == true` (
472    ///    `retroactive: false` on an ATIF-promoted receipt is
473    ///    semantically nonsense; the schema pins it to `const: true`
474    ///    but the Rust type still accepts `false`; refuse it here so
475    ///    the two agree);
476    /// 2. the embedded public key matches the ring's key for `key_id`
477    ///    (anti-substitution);
478    /// 3. the signature verifies over the version's signing message
479    ///    (bare `JCS(body)` for v1; domain-tagged, length-framed
480    ///    `JCS(body)` for v2 — see [`signing_message`]).
481    pub fn verify(&self, ring: &Keyring) -> Result<(), ReceiptError> {
482        self.verify_semantic_invariants()?;
483        let embedded = base64::engine::general_purpose::STANDARD
484            .decode(&self.body.public_key_b64)
485            .map_err(|_| ReceiptError::Base64)?;
486        let embedded: [u8; 32] = embedded.try_into().map_err(|_| ReceiptError::Base64)?;
487        // Re-derive the ring id for the embedded key; it must equal the stated
488        // key id AND exist in the ring with the same bytes.
489        let mut probe = Keyring::new();
490        let derived_id = probe.add_key_bytes(&embedded)?;
491        if derived_id != self.body.key_id {
492            return Err(ReceiptError::KeyMismatch(self.body.key_id.clone()));
493        }
494        let canon = canonicalize(&serde_json::to_value(&self.body)?)?;
495        let message = signing_message(self.body.receipt_version, &canon)?;
496        let sig = base64::engine::general_purpose::STANDARD
497            .decode(&self.signature_b64)
498            .map_err(|_| ReceiptError::Base64)?;
499        ring.verify(&self.body.key_id, &message, &sig)?;
500        Ok(())
501    }

503    /// **SECURITY WARNING — proves internal consistency, NOT authenticity.**
504    ///
505    /// Verifies the signature under the receipt's OWN embedded
506    /// `public_key_b64`. An attacker who generates a fresh Ed25519 keypair
507    /// and signs an arbitrary body with it will pass this check. The
508    /// result is equivalent to "this JSON blob is syntactically a
509    /// well-formed receipt whose signature is consistent with the key it
510    /// embeds"; it says nothing about whether the KEY is one the verifier
511    /// should trust.
512    ///
513    /// This method is safe ONLY when:
514    ///   1. The receipt was obtained over an authenticated channel that
515    ///      binds it to a trusted origin (e.g. a mutually-authenticated
516    ///      TLS connection to a known signer), **and/or**
517    ///   2. The caller separately pins `receipt.body.key_id` against a
518    ///      known-good allowlist and rejects any mismatch BEFORE reading
519    ///      any receipt content.
520    ///
521    /// For general receipt verification, use [`Receipt::verify`] with a
522    /// [`Keyring`] pre-populated with trusted keys. The browser `/verify`
523    /// page enforces the same discipline via `TRUSTED_RECEIPT_KEYS`
524    /// (docs/verify/verify.js) and shows an "internally consistent — trust
525    /// anchor NOT verified" warning without a pinned key; downstream Rust
526    /// callers must maintain the same distinction.
527    pub fn verify_embedded(&self) -> Result<(), ReceiptError> {
528        self.verify_semantic_invariants()?;
529        let embedded = base64::engine::general_purpose::STANDARD
530            .decode(&self.body.public_key_b64)
531            .map_err(|_| ReceiptError::Base64)?;
532        let embedded: [u8; 32] = embedded.try_into().map_err(|_| ReceiptError::Base64)?;
533        let mut ring = Keyring::new();
534        let id = ring.add_key_bytes(&embedded)?;
535        if id != self.body.key_id {
536            return Err(ReceiptError::KeyMismatch(self.body.key_id.clone()));
537        }
538        let canon = canonicalize(&serde_json::to_value(&self.body)?)?;
539        let message = signing_message(self.body.receipt_version, &canon)?;
540        let sig = base64::engine::general_purpose::STANDARD
541            .decode(&self.signature_b64)
542            .map_err(|_| ReceiptError::Base64)?;
543        ring.verify(&id, &message, &sig)?;
544        Ok(())
545    }
546
547    /// Check semantic invariants that the type system
548    /// cannot express. Currently enforces: supported
549    /// `receipt_version` ∈ {1, 2}, `ai_agent.charter.type_id == 1`
550    /// (OCSF Regular File), non-empty required string fields
551    /// (`receipt_id`, `session_id`, `ai_agent.charter.name`,
552    /// `ai_agent.charter.instance_uid`), 64-hex `chain_head` and
553    /// per-step `trajectory_digest`, `issued_at_iso ==
554    /// iso8601_ms(issued_at)`, stop-reason ↔ stop-reason-id
555    /// cross-wiring, and `AtifTrajectory.retroactive == true`
556    /// (schema pins it via `const: true`; the Rust type is `bool`
557    /// for auditability). Called from both `verify` and
558    /// `verify_embedded` so no verification path can skip it.
559    fn verify_semantic_invariants(&self) -> Result<(), ReceiptError> {
560        // Schema parity: `receipt_version` is pinned per-version in the
561        // shipped schemas (`const: 1` in receipt-v1, `const: 2` in
562        // receipt-v2) and the subject digests are pinned
563        // `^[0-9a-f]{64}$`, but verification never checked either — a
564        // keyholder could issue `receipt_version: 999` or free-text
565        // digests that `pvctl receipt-verify` accepted while
566        // schema-based external verifiers rejected (split
567        // offline-verification verdicts). Versions 1 (bare-JCS
568        // signature) and 2 (domain-separated signature) are the only
569        // ones with a defined signing message; anything else is
570        // refused before a signature check could misinterpret it.
571        if !matches!(self.body.receipt_version, 1 | 2) {
572            return Err(ReceiptError::SemanticInvariant(format!(
573                "receipt_version {} is not a supported version (1 or 2)",
574                self.body.receipt_version
575            )));
576        }
577        // Schema parity: the receipt schema (and the events validator)
578        // pin `ai_agent.charter.type_id` to the OCSF Regular File
579        // const 1; in-tree issuers always set it, so this only rejects
580        // inbound documents that external schema verifiers already
581        // refuse — keeping the two verification paths in agreement.
582        if self.body.ai_agent.charter.type_id != 1 {
583            return Err(ReceiptError::SemanticInvariant(format!(
584                "ai_agent.charter.type_id {} is not Regular File (1)",
585                self.body.ai_agent.charter.type_id
586            )));
587        }
588        // Schema parity: the shipped schema pins `minLength: 1` on these
589        // string fields. In-tree issuers always populate them, so this
590        // only rejects documents that external schema-based verifiers
591        // already refuse — without it, a keyholder could sign a receipt
592        // with an empty session_id/instance_uid that `pvctl
593        // receipt-verify` accepted while schema verifiers rejected
594        // (split offline-verification verdicts, same class as the
595        // receipt_version / issued_at_iso checks around this one).
596        for (field, value) in [
597            ("receipt_id", self.body.receipt_id.as_str()),
598            ("session_id", self.body.session_id.as_str()),
599            ("ai_agent.version", self.body.ai_agent.version.as_str()),
600            ("ai_agent.charter.name", self.body.ai_agent.charter.name.as_str()),
601            ("ai_agent.instance_uid", self.body.ai_agent.instance_uid.as_str()),
602        ] {
603            if value.is_empty() {
604                return Err(ReceiptError::SemanticInvariant(format!(
605                    "{field} is empty (schema requires minLength 1)"
606                )));
607            }
608        }
609        // Schema parity, upper bound: the shipped schemas pin
610        // `maxLength: 128` (code points) on the identifier fields, and
611        // RFC 3339 `date-time` cannot express years past 9999. A
612        // keyholder could otherwise sign a 10-KB session_id or a
613        // year-10000 timestamp that this verifier accepted while every
614        // schema-based verifier refused — the same split-verdict class
615        // as the emptiness checks above, in the other direction.
616        for (field, value) in [
617            ("receipt_id", self.body.receipt_id.as_str()),
618            ("session_id", self.body.session_id.as_str()),
619            ("ai_agent.version", self.body.ai_agent.version.as_str()),
620            ("ai_agent.charter.name", self.body.ai_agent.charter.name.as_str()),
621            ("ai_agent.instance_uid", self.body.ai_agent.instance_uid.as_str()),
622        ] {
623            if value.chars().count() > 128 {
624                return Err(ReceiptError::SemanticInvariant(format!(
625                    "{field} exceeds 128 characters (schema maxLength)"
626                )));
627            }
628        }
629        // 9999-12-31T23:59:59.999Z in epoch ms — the last instant RFC
630        // 3339 date-time (and iso8601_ms's 24-char shape) can carry.
631        const MAX_ISSUED_AT_MS: u64 = 253_402_300_799_999;
632        if self.body.issued_at > MAX_ISSUED_AT_MS {
633            return Err(ReceiptError::SemanticInvariant(format!(
634                "issued_at {} is beyond year 9999 (RFC 3339 date-time cannot express it)",
635                self.body.issued_at
636            )));
637        }
638        let digest_ok = |digest: &str| {
639            digest.len() == 64
640                && digest
641                    .bytes()
642                    .all(|b| b.is_ascii_digit() || (b'a'..=b'f').contains(&b))
643        };
644        match &self.body.subject {
645            ReceiptSubject::EventChain { chain_head, .. } => {
646                if !digest_ok(chain_head) {
647                    return Err(ReceiptError::SemanticInvariant(
648                        "chain_head is not a lowercase 64-hex digest".to_owned(),
649                    ));
650                }
651            }
652            ReceiptSubject::AtifTrajectory {
653                trajectory_digest, ..
654            } => {
655                if !digest_ok(trajectory_digest) {
656                    return Err(ReceiptError::SemanticInvariant(
657                        "trajectory_digest is not a lowercase 64-hex digest".to_owned(),
658                    ));
659                }
660            }
661        }
662        if let ReceiptSubject::AtifTrajectory { retroactive, .. } = &self.body.subject {
663            if !retroactive {
664                return Err(ReceiptError::SemanticInvariant(
665                    "AtifTrajectory.retroactive must be true (this receipt attests a retroactive promotion)"
666                        .to_owned(),
667                ));
668            }
669        }
670        // Cross-check `issued_at_iso` against the
671        // authoritative epoch-ms `issued_at`. Checking
672        // only shapes plus the retroactive invariant would let a
673        // hostile-but-keyholding issuer sign a receipt whose ISO
674        // caption and epoch-ms disagreed — auditors reading ISO and
675        // tooling reading ms would see different facts from the same
676        // verified receipt. Derive the ISO deterministically and
677        // require equality.
678        let expected_iso = pv_core::time::iso8601_ms(self.body.issued_at);
679        if self.body.issued_at_iso != expected_iso {
680            return Err(ReceiptError::SemanticInvariant(format!(
681                "issued_at_iso {:?} does not match issued_at ({}); expected {:?}",
682                self.body.issued_at_iso, self.body.issued_at, expected_iso
683            )));
684        }
685        // Cross-check the stop-reason dual representation — the same
686        // split-brain class as issued_at/issued_at_iso above, but with
687        // the same tolerance the events validator uses:
688        // `stop_reason` may legitimately carry
689        // a provider-native or free-text caption, so equality with the
690        // canonical caption cannot be required. Refuse only the
691        // unambiguous cross-wiring where the caption IS the canonical
692        // caption of a DIFFERENT known variant than `stop_reason_id`
693        // names (e.g. id 92 "Budget Exceeded" paired with caption
694        // "Stop") — the two signed fields then assert contradictory
695        // outcomes to auditors reading the caption vs tooling reading
696        // the id.
697        let expected_caption = pv_events::StopReason::from_id(self.body.stop_reason_id).caption();
698        let id_is_known = matches!(self.body.stop_reason_id, 0..=4 | 90..=94 | 99);
699        let caption_is_canonical_of_other_variant = id_is_known
700            && self.body.stop_reason != expected_caption
701            && [
702                pv_events::StopReason::Unknown,
703                pv_events::StopReason::Stop,
704                pv_events::StopReason::MaxTokens,
705                pv_events::StopReason::ToolUse,
706                pv_events::StopReason::SessionClosed,
707                pv_events::StopReason::ContentFilter,
708                pv_events::StopReason::LoopDetected,
709                pv_events::StopReason::BudgetExceeded,
710                pv_events::StopReason::PolicyBlocked,
711                pv_events::StopReason::IdentityRejected,
712                pv_events::StopReason::Other,
713            ]
714            .iter()
715            .any(|variant| variant.caption() == self.body.stop_reason);
716        if caption_is_canonical_of_other_variant {
717            return Err(ReceiptError::SemanticInvariant(format!(
718                "stop_reason {:?} is the canonical caption of a different variant than stop_reason_id ({}); expected {:?} or a non-canonical native caption",
719                self.body.stop_reason, self.body.stop_reason_id, expected_caption
720            )));
721        }
722        // Tool-accounting invariant: classified outcomes can never
723        // exceed observed calls. The issuer maintains this by
724        // construction and journal recovery hard-fails on its violation
725        // (`RecoveredTotals::validate_tool_accounting`), but
726        // verification never checked it — a keyholding issuer could
727        // sign `total: 1, allowed: 1_000_000` and every offline
728        // verifier would attest arithmetic the pipeline itself refuses
729        // to recover. Same cross-field class as issued_at/issued_at_iso
730        // above (the JSON Schema cannot express cross-field sums).
731        let classified = self
732            .body
733            .tool_calls
734            .allowed
735            .checked_add(self.body.tool_calls.blocked);
736        if classified.is_none_or(|classified| classified > self.body.tool_calls.total) {
737            return Err(ReceiptError::SemanticInvariant(format!(
738                "tool_calls allowed ({}) + blocked ({}) exceeds total ({})",
739                self.body.tool_calls.allowed, self.body.tool_calls.blocked, self.body.tool_calls.total
740            )));
741        }
742        Ok(())
743    }
744}

746/// Walk a JSON document and reject any object whose keys contain a
747/// duplicate at ANY nesting level.
748///
749/// Used by [`Receipt::from_json_slice`] as a strict pre-parse gate.
750/// serde_json's default `Value` / `Map` uses `IndexMap` (with
751/// `preserve_order`), which silently keeps only the LAST value for a
752/// duplicate key at deserialize time — collapsing evidence that a
753/// hostile issuer signed a document with two spellings of the same
754/// key and a first-wins auditor would see the earlier one. This
755/// walk runs upfront so the collapse never happens.
756///
757/// Also enforce a hard depth cap so a hostile issuer
758/// cannot use a deeply-nested `[[[[…]]]]` payload to blow the walker's
759/// stack. serde_json's own limit is 128 by default, but the cost of
760/// a per-level check is a single integer increment.
761const MAX_NESTED_DEPTH: usize = 128;
762
763/// Sentinel prefix that lets `check_no_duplicate_keys` distinguish a
764/// duplicate-key rejection (mapped to `ReceiptError::DuplicateKey`)
765/// from a general parse failure (mapped to `ReceiptError::Serde`).
766/// A blanket `.map_err(DuplicateKey)` would fold
767/// serde_json's own EOF / recursion / expected-value errors into the
768/// DuplicateKey variant, actively misleading ops.
769const DUP_KEY_SENTINEL: &str = "__av_dup:";
770
771fn check_no_duplicate_keys(bytes: &[u8]) -> Result<(), ReceiptError> {
772    use serde::Deserializer as _;
773    let mut deser = serde_json::Deserializer::from_slice(bytes);
774    match deser.deserialize_any(NoDupVisitor { depth: 0 }) {
775        Ok(()) => Ok(()),
776        Err(error) => {
777            let msg = error.to_string();
778            if let Some(rest) = msg.strip_prefix(DUP_KEY_SENTINEL) {
779                Err(ReceiptError::DuplicateKey(rest.to_owned()))
780            } else if msg.contains("recursion limit exceeded") {
781                // Preserve the intent of the depth check even when
782                // serde_json fires first, so callers still see a
783                // DuplicateKey-family error class for hostile
784                // deeply-nested input.
785                Err(ReceiptError::DuplicateKey(msg))
786            } else {
787                // Genuine parse error (EOF, expected value, invalid
788                // number). Bubble the underlying serde_json::Error
789                // via Serde so tracing sites don't misattribute.
790                Err(ReceiptError::Serde(error))
791            }
792        }
793    }
794}

797    depth: usize,
798}

800impl<'de> serde::de::Visitor<'de> for NoDupVisitor {
801    type Value = ();
802    fn expecting(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
803        f.write_str("any JSON value (strict-mode duplicate-key check)")
804    }
805    fn visit_bool<E>(self, _: bool) -> Result<(), E> {
806        Ok(())
807    }
808    fn visit_i64<E>(self, _: i64) -> Result<(), E> {
809        Ok(())
810    }
811    fn visit_u64<E>(self, _: u64) -> Result<(), E> {
812        Ok(())
813    }
814    fn visit_f64<E>(self, _: f64) -> Result<(), E> {
815        Ok(())
816    }
817    fn visit_i128<E>(self, _: i128) -> Result<(), E> {
818        Ok(())
819    }
820    fn visit_u128<E>(self, _: u128) -> Result<(), E> {
821        Ok(())
822    }
823    fn visit_str<E>(self, _: &str) -> Result<(), E> {
824        Ok(())
825    }
826    fn visit_string<E>(self, _: String) -> Result<(), E> {
827        Ok(())
828    }
829    /// Serde may invoke `visit_none` for `Option` layers that are
830    /// already Nothing; `deserialize_any` on JSON never routes there
831    /// (JSON null goes through `visit_unit`), so this remains a
832    /// no-op fallback that only fires if a future caller layers a
833    /// typed deserializer on top.
834    fn visit_none<E>(self) -> Result<(), E> {
835        Ok(())
836    }
837
838    /// Reject explicit JSON `null` anywhere
839    /// in the receipt wire form. Every optional field on `ReceiptBody`
840    /// / `AgentIdentity` / `EventMetrics` / … carries
841    /// `#[serde(skip_serializing_if = "Option::is_none")]`, so
842    /// `Option::None` is emitted as *field absence*. But
843    /// `serde_json::from_slice` also accepts `"field": null` for those
844    /// same Options — and `Receipt::verify` re-canonicalises via
845    /// `serde_json::to_value(&self.body)`, which drops the `None`
846    /// again. The consequence: an intermediary can flip `"…": null`
847    /// on/off in the raw bytes without invalidating the signature,
848    /// violating the "same signature ⇔ same bytes" auditor
849    /// invariant. Refuse explicit null across the whole payload
850    /// (receipts contain no legitimate JSON null anywhere — top level
851    /// is an object; every field is either a value type, a container
852    /// with omit-on-none, or a bounded set-of-strings enum).
853    fn visit_unit<E: serde::de::Error>(self) -> Result<(), E> {
854        Err(E::custom(format!(
855            "{DUP_KEY_SENTINEL}explicit JSON null is not permitted (a signed absent-Option \
856             cannot be re-serialised as null without breaking canonicalisation invariance)"
857        )))
858    }
859    fn visit_seq<S: serde::de::SeqAccess<'de>>(self, mut seq: S) -> Result<(), S::Error> {
860        use serde::de::Error as _;
861        let next_depth = self.depth.saturating_add(1);
862        if next_depth > MAX_NESTED_DEPTH {
863            return Err(S::Error::custom(format!(
864                "{DUP_KEY_SENTINEL}JSON nesting exceeds {MAX_NESTED_DEPTH}"
865            )));
866        }
867        while seq.next_element_seed(NoDupSeed { depth: next_depth })?.is_some() {}
868        Ok(())
869    }
870    fn visit_map<M: serde::de::MapAccess<'de>>(self, mut map: M) -> Result<(), M::Error> {
871        use serde::de::Error as _;
872        let next_depth = self.depth.saturating_add(1);
873        if next_depth > MAX_NESTED_DEPTH {
874            return Err(M::Error::custom(format!(
875                "{DUP_KEY_SENTINEL}JSON nesting exceeds {MAX_NESTED_DEPTH}"
876            )));
877        }
878        let mut seen: std::collections::HashSet<String> = std::collections::HashSet::new();
879        while let Some(key) = map.next_key::<String>()? {
880            if !seen.insert(key.clone()) {
881                // Escape the key so a hostile spelling
882                // containing `\n` or ANSI escapes cannot inject a
883                // fake extra log line at every tracing site that
884                // logs the resulting error via `%error`.
885                return Err(M::Error::custom(format!(
886                    "{DUP_KEY_SENTINEL}duplicate key `{}` in JSON object",
887                    key.escape_debug()
888                )));
889            }
890            map.next_value_seed(NoDupSeed { depth: next_depth })?;
891        }
892        Ok(())
893    }
894}
